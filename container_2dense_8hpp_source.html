26/// @brief Rank-1 matrix update \f$A \leftarrow A + \alpha \mathbf{x} \mathbf{y}^T\f$.
27///
28/// Row-tiled rather than routed through `dispatch::parallel_apply`: see
29/// `omp/matrix_ops.hpp` for why row/matrix operations use their own tiling
30/// instead of the vector-element-tuned block helper.
31inline void ger(real alpha, const vec &x, const vec &y, mat &A) {
32 constexpr idx block_size = 64;
33 const idx m = x.size();
34 const idx n = y.size();
35 const real *xd = x.data();
36 const real *yd = y.data();
37 real *ad = A.data();
38#pragma omp parallel for schedule(static)
39 for (idx ii = 0; ii < m; ii += block_size) {
40 const idx rows = std::min(block_size, m - ii);
41 kernel::ger(ad + (ii * n), n, xd + ii, yd, alpha, rows, n);
42 }
43}

58/// @brief Forward substitution solving lower triangular system \f$L \mathbf{x} = \mathbf{b}\f$.
59inline void trsv_lower(const mat &L, const vec &b, vec &x) {
60 const idx n = L.rows();
61 if (L.cols() != n || b.size() != n) {
62 throw std::invalid_argument("trsv_lower: dimension mismatch");
63 }
64 if (x.size() != n) {
65 x = vec(n);
66 }
67 kernel::trsv_lower(kernel::contract::alias_safe, x.data(), L.data(), b.data(), n);
68}
