60inline hessenberg_decomposition::hessenberg_decomposition(const mat &A, bool use_lapack)
61 : H_(A), Q_(A.rows(), A.cols(), 0.0) {
62 debug::check_dim(A.rows(), A.cols(), "hessenberg_decomposition matrix must be square");
63 debug::check_non_empty(A.rows(), "hessenberg_decomposition matrix");
64
65 const idx n = A.rows();
66 // Initialize Q as the identity matrix
67 for (idx i = 0; i < n; ++i) {
68 Q_(i, i) = 1.0;
69 }
70
71 if (n <= 2) {
72 return;
73 }
74
75#if defined(NUMERICS_HAS_LAPACK)
76 if (use_lapack) {
77 // LAPACK dgehrd and dorghr assume column-major layout.
78 // A (row-major) corresponds to A^T (column-major).
79 // Transpose A to column-major buffer:
80 array<double> a_col(n * n);
81 kernel::transpose(a_col.data(), A.data(), n, n);
82
83 array<double> tau(n - 1, 0.0);
84 lapack_int lapack_n = static_cast<lapack_int>(n);
85 lapack_int ilo = 1;
86 lapack_int ihi = lapack_n;
87
88 int info = LAPACKE_dgehrd(LAPACK_COL_MAJOR, lapack_n, ilo, ihi, a_col.data(), lapack_n,
89 tau.data());
90 if (info != 0) {
91 throw std::runtime_error("dgehrd failed with info=" + std::to_string(info));
92 }
93
94 // Copy out upper Hessenberg matrix H (from column-major a_col to row-major H_)
95 for (idx i = 0; i < n; ++i) {
96 for (idx j = 0; j < n; ++j) {
97 if (i > j + 1) {
98 H_(i, j) = 0.0;
99 } else {
100 H_(i, j) = a_col[(j * n) + i];
101 }
102 }
103 }
104
105 // Generate orthogonal matrix Q via dorghr
106 info = LAPACKE_dorghr(LAPACK_COL_MAJOR, lapack_n, ilo, ihi, a_col.data(), lapack_n,
107 tau.data());
108 if (info != 0) {
109 throw std::runtime_error("dorghr failed with info=" + std::to_string(info));
110 }
111
112 // Copy Q from column-major a_col to row-major Q_
113 kernel::transpose(Q_.data(), a_col.data(), n, n);
114 return;
115 }
116#endif
117
118 // High-performance vectorized sequential Householder elimination
119 array<double> col_k(n);
120 array<double> v(n);
121 array<double> w(n);
122 double *H_raw = H_.data();
123 double *Q_raw = Q_.data();
124
125 // Eliminate below subdiagonal column by column
126 for (idx k = 0; k < n - 2; ++k) {
127 const idx m = n - 1 - k; // length of subvector to reflect
128
129 for (idx i = 0; i < m; ++i) {
130 col_k[i] = H_raw[((k + 1 + i) * n) + k];
131 }
132
133 double beta = 0.0;
134 kernel::householder_vector(v.data(), beta, col_k.data(), m);
135 if (beta == 0.0) {
136 continue;
137 }
138
139 // 1. Left multiplication: H(k+1:n, k:n) <- (I - beta * v * v^T) * H(k+1:n, k:n)
140 kernel::householder_left(&H_raw[((k + 1) * n) + k], n, v.data(), beta, m, n - k,
141 w.data());
142
143 // 2. Right multiplication: H(0:n, k+1:n) <- H(0:n, k+1:n) * (I - beta * v * v^T)
144 kernel::householder_right(&H_raw[k + 1], n, v.data(), beta, n, m);
145
146 // 3. Accumulate into Q: Q(0:n, k+1:n) <- Q(0:n, k+1:n) * (I - beta * v * v^T)
147 kernel::householder_right(&Q_raw[k + 1], n, v.data(), beta, n, m);
148
149 // Set strictly zero entries below subdiagonal
150 for (idx i = k + 2; i < n; ++i) {
151 H_raw[(i * n) + k] = 0.0;
152 }
153 }
154}

157/// @brief Solve the shifted Hessenberg system \f$(sI - H)y = \tilde b\f$ in \f$O(n^2)\f$.
158///
159/// Gaussian elimination with partial pivoting on an upper Hessenberg matrix touches
160/// one subdiagonal per column, so the factorization is \f$O(n^2)\f$ rather than
161/// \f$O(n^3)\f$. This is what makes a resolvent \f$(sI-A)^{-1}b\f$ cheap once
162/// \f$A\f$ has been reduced once: every subsequent shift reuses the same \f$H\f$.
163///
164/// @param H Upper Hessenberg matrix, n*n.
165/// @param shift The scalar s.
166/// @param b_tilde Right-hand side, length n.
167/// @param y Output, resized to n.
168/// @param M_buf Scratch, grown to n*n and reusable across calls.
169/// @param pivots Scratch, grown to n and reusable across calls.
170inline void hessenberg_shifted_solve(const mat &H, cplx shift, const array<cplx> &b_tilde,
171 array<cplx> &y, array<cplx> &M_buf,
172 array<idx> &pivots) {
173 const idx n = H.rows();
174 if (b_tilde.size() != n) {
175 throw std::invalid_argument("hessenberg_shifted_solve: dimension mismatch");
176 }
177 if (M_buf.size() < n * n) {
178 M_buf.resize(n * n);
179 }
180 if (pivots.size() < n) {
181 pivots.resize(n);
182 }
183 if (y.size() != n) {
184 y.resize(n);
185 }
186 kernel::hessenberg_shifted_solve(y.data(), H.data(), shift, b_tilde.data(), n,
187 M_buf.data(), pivots.data());
188}

190/// @brief Project a right-hand side onto the Hessenberg basis: \f$\tilde b = Q^T b\f$.
191///
192/// Accepts a real or complex right-hand side. The result is complex either way,
193/// since the shift generally is.
194template <class Rhs>
195inline array<cplx> hessenberg_project(const mat &Q, const Rhs &b) {
196 const idx n = Q.rows();
197 array<cplx> b_tilde(n);
198 kernel::matvec_transpose_into_complex(b_tilde.data(), Q.data(), b.data(), n, n);
199 return b_tilde;
200}

202/// @brief Carry a solution back to the original basis: \f$x = Q y\f$.
203inline void hessenberg_back_project(const mat &Q, const array<cplx> &y,
204 array<cplx> &x) {
205 const idx n = Q.rows();
206 if (x.size() != n) {
207 x.resize(n);
208 }
209 kernel::matvec_real_complex(x.data(), Q.data(), y.data(), n, Q.cols());
210}
