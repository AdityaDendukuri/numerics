179/// Write values at the supplied indices, optionally accumulating into output.
180void scatter(view<const T> values, view<const idx> indices, view<T> output,
181 bool add = false) {
182 if (values.size() != indices.size()) {
183 throw std::invalid_argument("scatter: values and indices must have the same size");
184 }
185 for (idx position = 0; position < indices.size(); ++position) {
186 if (indices[position] >= output.size()) {
187 throw std::out_of_range("scatter: index out of range");
188 }
189 if (add) {
190 output[indices[position]] += values[position];
191 } else {
192 output[indices[position]] = values[position];
193 }
194 }
195}

198/// Copy the indexed input entries into a compact vector.
199[[nodiscard]] array<T> gather(view<const T> input, view<const idx> indices) {
200 array<T> result;
201 result.reserve(indices.size());
202 for (idx index : indices) {
203 if (index >= input.size()) {
204 throw std::out_of_range("gather: index out of range");
205 }
206 result.push_back(input[index]);
207 }
