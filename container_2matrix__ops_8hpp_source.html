1/// @file container/matrix_ops.hpp
2/// @brief Untagged Level-2/3 dense matrix operations: resolve through `num::accel`.
3///
4/// A caller who wants a specific backend calls it by name — `num::omp::matmul`,
5/// `num::blas::matvec` — directly. These untagged overloads exist only for call
6/// sites that do not care which backend runs.
7///
8/// There is one `matmul` and no tuning knobs beside it: `kernel::gemm` blocks
9/// itself for the register file and the cache, so the `matmul_blocked` /
10/// `matmul_register_blocked` / `matmul_simd` variants that used to sit here are
11/// gone. Each was slower than the kernel now is.

72template <std::floating_point T>
73template <class X, class Y>
74inline void basic_mat<T>::apply(const X &x, Y &y) const {
75 if constexpr (std::is_same_v<T, real> && std::is_same_v<X, vec> && std::is_same_v<Y, vec>) {
76 matvec(*this, x, y);
77 } else {
78 for (idx i = 0; i < rows_; ++i) {
79 T sum = T(0);
80 for (idx j = 0; j < cols_; ++j) {
81 sum += (*this)(i, j) * x[j];
82 }
83 y[i] = sum;
84 }
85 }
86}
