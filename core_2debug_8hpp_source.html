50// -----------------------------------------------------------------------------
51// The compile-time ceiling
52// -----------------------------------------------------------------------------
53//
54// `NUMERICS_DIAGNOSTICS` bounds what this build can do; the runtime preset selects from
55// what is left. The two answer different questions. The ceiling decides what code exists,
56// the preset decides whether it runs.
57//
58// 0 nothing. Every check and probe is discarded at compile time.
59// 1 shape checks. Dimensions, emptiness, finiteness. O(1) or O(n), and worth keeping.
60// 2 property sampling as well. Symmetry, definiteness and linearity probes over random
61// vectors, which cost O(n^2) in the operator's size.
62//
63// The default is 2 without NDEBUG and 1 with it. Level 2 is not a Release default because
64// it is not a small cost: attaching an SPD claim to a 1024x1024 operator sampled at about
65// 15 ms against a 1.5 ms conjugate-gradient solve of the same system. A default that
66// makes the assertion ten times more expensive than the work it guards is one that gets
67// measured and blamed on the library.
68//
69// Override it explicitly. `-DNUMERICS_DIAGNOSTICS=2` keeps full sampling in an optimized
70// build, which is what a test suite or a numerically suspicious run wants.
71#if !defined(NUMERICS_DIAGNOSTICS)
72#if defined(NDEBUG)
73#define NUMERICS_DIAGNOSTICS 1
74#else
75#define NUMERICS_DIAGNOSTICS 2
76#endif
77#endif
78
79#if NUMERICS_DIAGNOSTICS < 0 || NUMERICS_DIAGNOSTICS > 2
80#error "NUMERICS_DIAGNOSTICS must be 0 (off), 1 (shape checks), or 2 (property sampling)"
81#endif

101constexpr diagnostic_level level_for(diagnostic_preset p) noexcept {
102 switch (p) {
103 case diagnostic_preset::strict:
104 return diagnostic_level::full;
105 case diagnostic_preset::balanced:
106 return diagnostic_level::basic;
107 case diagnostic_preset::unsafe:
108 case diagnostic_preset::production:
109 break;
110 }
111 return diagnostic_level::off;
112}
113
114constexpr diagnostic_level clamp_to_ceiling(diagnostic_level lvl) noexcept {
115 return static_cast<int>(lvl) > NUMERICS_DIAGNOSTICS ? compiled_level : lvl;
116}
117
118constexpr diagnostic_preset default_preset() noexcept {
119 return compiled_level == diagnostic_level::full ? diagnostic_preset::strict
120 : compiled_level == diagnostic_level::basic ? diagnostic_preset::balanced
121 : diagnostic_preset::production;
122}
123
124// Read on every check and written rarely, so a relaxed atomic is a plain load on any real
125// target. It is atomic rather than a bare global because `scoped_preset` makes it
126// reachable from more than one thread, and a torn read there would be undefined behaviour
127// for the sake of nothing.
128inline std::atomic<diagnostic_level> g_level{clamp_to_ceiling(level_for(default_preset()))};
129inline std::atomic<diagnostic_preset> g_preset{default_preset()};
130
131} // namespace detail

141/// @brief Configure the global execution preset.
142///
143/// The request is clamped to the compile-time ceiling. Asking for `strict` in a build
144/// configured at level 1 leaves sampling off, because that code was never emitted.
145/// `get_preset` still reports what was asked for; `preset_fully_applied` reports whether
146/// the request was met.
147inline void set_preset(diagnostic_preset p) noexcept {
148 detail::g_preset.store(p, std::memory_order_relaxed);
149 detail::g_level.store(detail::clamp_to_ceiling(detail::level_for(p)),
150 std::memory_order_relaxed);
151}
152
153/// @brief The preset most recently requested.
154///
155/// Stored rather than derived from the level. Deriving it could not tell `unsafe` from
156/// `production`, which share a level, so `set_preset(production)` used to report back
157/// `unsafe`.
158[[nodiscard]] inline diagnostic_preset get_preset() noexcept {
159 return detail::g_preset.load(std::memory_order_relaxed);
160}
161
162/// @brief Whether the preset in force is also reachable in this build.
163///
164/// False when a caller asked for more than the ceiling allows. Worth reporting once at
165/// startup rather than silently running fewer checks than the caller believes.
166[[nodiscard]] inline bool preset_fully_applied() noexcept {
167 return detail::level_for(get_preset()) == get_level();
168}

170/// @brief RAII guard to temporarily apply an execution preset within a scope.
171class scoped_preset {
172 public:
173 explicit scoped_preset(diagnostic_preset temp_preset) noexcept
174 : previous_preset_(get_preset()) {
175 set_preset(temp_preset);
176 }
177
178 ~scoped_preset() noexcept { set_preset(previous_preset_); }

182 scoped_preset(scoped_preset &&) = delete;
183 scoped_preset &operator=(scoped_preset &&) = delete;
184
185 private:
186 diagnostic_preset previous_preset_;
187};

198/// @brief Verify dimension equality (e.g. A.rows() == b.size())
199inline void check_dim(idx expected, idx actual, std::string_view label,
200 std::source_location loc = std::source_location::current()) {
201 if constexpr (checks_compiled_in) {
202 if (get_level() != diagnostic_level::off && expected != actual) {
203 panic("DimensionError",
204 "Dimension mismatch for " + std::string(label) + ": expected " +
205 std::to_string(expected) + ", got " + std::to_string(actual),
206 loc);
207 }
208 }
209}

211/// @brief Verify a container is non-empty
212inline void check_non_empty(idx size, std::string_view label,
213 std::source_location loc = std::source_location::current()) {
214 if constexpr (checks_compiled_in) {

216 panic("ValueError", std::string(label) + " cannot be empty (size is 0)", loc);
217 }

221/// @brief Verify all values in array are finite (not NaN or Inf)
222template <typename T>
223inline void check_finite(const T *data, idx n, std::string_view label,
224 std::source_location loc = std::source_location::current()) {
225 if constexpr (checks_compiled_in) {
226 if (get_level() == diagnostic_level::off) {
227 return;
228 }
229 for (idx i = 0; i < n; ++i) {
230 if (!std::isfinite(data[i])) {
231 panic("ValueError",
232 std::string(label) + " contains non-finite value (NaN or Inf) at index " +
233 std::to_string(i),
234 loc);
235 }
236 }
237 }
