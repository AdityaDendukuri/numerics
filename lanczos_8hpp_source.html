49template <class Op>
50requires linear_operator<Op, vec, vec> lanczos_result
51lanczos_operator_impl(const Op &A, idx k, real tol, idx max_steps) {
52 const idx n = A.rows();
53 if (A.cols() != n) {
54 throw std::invalid_argument("lanczos: operator must be square");
55 }
56 if (k == 0 || k > n) {
57 throw std::invalid_argument("lanczos: k must satisfy 0 < k <= n");
58 }
59
60 if (max_steps == 0) {
61 max_steps = std::min(3 * k, n);
62 }
63 max_steps = std::min(max_steps, n);
64
65 mat V(n, max_steps, 0.0);
66 vec alpha(max_steps, 0.0);
67 vec beta(max_steps, 0.0);
68
69 // v_0 <- e_0
70 V(0, 0) = 1.0;
71
72 idx steps = 0;
73
74 for (idx j = 0; j < max_steps; ++j) {
75 vec vj(n);
76 // v_j <- V[:,j]
77 kernel::copy_strided(vj.data(), 1, &V(0, j), V.cols(), n);
78
79 vec w(n, 0.0);
80 A.apply(vj, w);
81
82 const real a = dot(vj, w);
83 alpha[j] = a;
84
85 axpy(-a, vj, w);
86 if (j > 0) {
87 // w <- w - beta_{j-1}*v_{j-1}
88 kernel::axpy_strided(w.data(), 1, &V(0, j - 1), V.cols(), -beta[j - 1], n);
89 }
90
91 const real b = dispatch::subspace::mgs_orthogonalize(V, j + 1, w);
92 ++steps;
93
94 if (b < real(1e-12)) {
95 break;
96 }
97
98 beta[j] = b;
99
100 if (j + 1 < max_steps) {
101 // V[:,j+1] <- w/beta_j
102 kernel::scale_copy_strided(&V(0, j + 1), V.cols(), w.data(), 1, real(1) / b, n);
103 }
104 }
105
106 const idx m = steps;
107 mat T(m, m, 0.0);
108 for (idx j = 0; j < m; ++j) {
109 T(j, j) = alpha[j];
110 if (j + 1 < m) {
111 T(j, j + 1) = beta[j];
112 T(j + 1, j) = beta[j];
113 }
114 }
115
116 // T is the Lanczos tridiagonal, symmetric by construction: it is filled from a
117 // single alpha/beta recurrence with T(j,j+1) and T(j+1,j) written from the same
118 // beta. The invariant is established here rather than assumed downstream.
119 eigen_result teig = eig_sym(linear::sym_mat<mat>(T), tol * real(1e-2));
120 const idx nret = std::min(k, m);
121
122 mat ritz_vecs(n, nret, 0.0);
123 // U_Ritz <- V_m*Z_selected
124 kernel::gemm(ritz_vecs.data(), ritz_vecs.cols(), V.data(), V.cols(),
125 &teig.vectors(0, m - nret), teig.vectors.cols(), real(1), real(0), n, nret,
126 m);
127
128 vec ritz_vals(nret);
129 // lambda_Ritz <- lambda(T_m)_selected
130 kernel::copy(ritz_vals.data(), teig.values.data() + (m - nret), nret);
131
132 bool all_converged = true;
133 for (idx i = 0; i < nret; ++i) {
134 vec u(n);
135 // u_i <- U_Ritz[:,i]
136 kernel::copy_strided(u.data(), 1, &ritz_vecs(0, i), ritz_vecs.cols(), n);
137
138 vec Au(n, 0.0);
139 A.apply(u, Au);
140
141 const real lam = ritz_vals[i];
142 // res^2 <- ||A*u_i - lambda_i*u_i||_2^2
143 const real res =
144 kernel::linear_combination_norm_sq(Au.data(), real(1), u.data(), -lam, n);
145 if (std::sqrt(res) > tol) {
146 all_converged = false;
147 break;
148 }
149 }
150
151 return {ritz_vals, ritz_vecs, steps, all_converged};
152}

156/// @brief Compute the largest \f$k\f$ extremal Ritz pairs of a symmetric / self-adjoint operator using Lanczos iteration.
157///
158/// Builds an orthonormal Krylov basis with modified Gram-Schmidt reorthogonalization, generates
159/// a symmetric tridiagonal projection \f$T_m\f$, and extracts Ritz values and Ritz vectors.
160///
161/// @tparam Op Linear operator type satisfying `self_adjoint_operator<Op, vec, vec>`.
162/// @param A Self-adjoint linear operator (matrix-free callable, sparse, or dense wrapper).
163/// @param k Number of extremal eigenpairs to compute (\f$0 < k \le n\f$).
164/// @param tol Residual tolerance \f$\|A v - \lambda v\|_2\f$ for declaring convergence (default: 1e-10).
165/// @param max_steps Maximum Lanczos steps to perform (defaults to \f$\min(3k, n)\f$).
166/// @return `lanczos_result` containing Ritz values, column Ritz vectors, steps performed, and convergence flag.
167/// @throws std::invalid_argument If `k` is invalid or operator is not square.
168/// @see eig_sym, power_iteration
169template <class Op>
170requires self_adjoint_operator<Op, vec, vec> lanczos_result
171lanczos(const Op &A, idx k, real tol = 1e-10, idx max_steps = 0) {
172 return detail::lanczos_operator_impl(A, k, tol, max_steps);
173}

188/// @brief Compute largest \f$k\f$ Ritz pairs of a matrix carrying certified symmetry evidence.
189///
190/// @param A Symmetric matrix carrying symmetry evidence (e.g. `num::assume_symmetric(A)`).
191/// @param k Number of extremal eigenpairs to compute.
192/// @param tol Residual tolerance (default: 1e-10).
193/// @param max_steps Maximum Lanczos steps (default: \f$\min(3k, n)\f$).
194/// @return `lanczos_result` with Ritz pairs and convergence metadata.
195inline lanczos_result lanczos(const linear::sym_mat<mat> &A, idx k, real tol = 1e-10,
196 idx max_steps = 0) {
197 return unsafe::lanczos(A.base(), k, tol, max_steps);
198}
