58[[nodiscard]] inline vec inverse_sqrt_from_projection(const mat &basis, const vec &alpha,
59 const vec &beta, idx steps,
60 real right_hand_side_norm) {
61 mat tridiagonal(steps, steps, 0.0);
62 for (idx j = 0; j < steps; ++j) {
63 tridiagonal(j, j) = alpha[j];
64 if (j + 1 < steps) {
65 tridiagonal(j, j + 1) = beta[j];
66 tridiagonal(j + 1, j) = beta[j];
67 }
68 }
69
70 const eigen_result eig = eig_sym(linear::sym_mat<mat>(tridiagonal), 1e-13);
71 vec coefficients(steps, 0.0);
72 for (idx eigenvector = 0; eigenvector < steps; ++eigenvector) {
73 if (!(eig.values[eigenvector] > 0.0)) {
74 throw std::runtime_error(
75 "inverse_sqrt_lanczos: projected operator is not positive definite");
76 }
77 const real weight = eig.vectors(0, eigenvector) / std::sqrt(eig.values[eigenvector]);
78 for (idx j = 0; j < steps; ++j) {
79 coefficients[j] += eig.vectors(j, eigenvector) * weight;
80 }
81 }
82
83 vec value(basis.rows(), 0.0);
84 for (idx row = 0; row < basis.rows(); ++row) {
85 for (idx j = 0; j < steps; ++j) {
86 value[row] += right_hand_side_norm * basis(row, j) * coefficients[j];
87 }
88 }
89 return value;
90}

92[[nodiscard]] inline vec sqrt_from_projection(const mat &basis, const vec &alpha, const vec &beta,
93 idx steps, real right_hand_side_norm) {
94 mat tridiagonal(steps, steps, 0.0);
95 for (idx j = 0; j < steps; ++j) {
96 tridiagonal(j, j) = alpha[j];
97 if (j + 1 < steps) {
98 tridiagonal(j, j + 1) = beta[j];
99 tridiagonal(j + 1, j) = beta[j];
100 }
101 }
102
103 const eigen_result eig = eig_sym(linear::sym_mat<mat>(tridiagonal), 1e-13);
104 vec coefficients(steps, 0.0);
105 for (idx eigenvector = 0; eigenvector < steps; ++eigenvector) {
106 if (!(eig.values[eigenvector] > 0.0)) {
107 throw std::runtime_error("sqrt_lanczos: projected operator is not positive definite");
108 }
109 const real weight = eig.vectors(0, eigenvector) * std::sqrt(eig.values[eigenvector]);
110 for (idx j = 0; j < steps; ++j) {
111 coefficients[j] += eig.vectors(j, eigenvector) * weight;
112 }
113 }
114
115 vec value(basis.rows(), 0.0);
116 for (idx row = 0; row < basis.rows(); ++row) {
117 for (idx j = 0; j < steps; ++j) {
118 value[row] += right_hand_side_norm * basis(row, j) * coefficients[j];
119 }
120 }

124template <class Op>
125requires linear_operator<Op, vec, vec> lanczos_inverse_sqrt_result
126inverse_sqrt_lanczos_impl(const Op &A, const vec &right_hand_side, real tolerance, idx max_steps) {
127 const idx n = A.rows();
128 if (A.cols() != n || right_hand_side.size() != n)
129 throw std::invalid_argument("inverse_sqrt_lanczos: dimension mismatch");
130 if (!(tolerance > 0.0))
131 throw std::invalid_argument("inverse_sqrt_lanczos: tolerance must be positive");
132 if (max_steps == 0)
133 max_steps = std::min<idx>(64, n);
134 max_steps = std::min(max_steps, n);
135
136 const real right_hand_side_norm = norm(right_hand_side);
137 if (right_hand_side_norm == 0.0)
138 return {vec(n, 0.0), 0, 0.0, true};
139
140 mat basis(n, max_steps, 0.0);
141 for (idx row = 0; row < n; ++row)
142 basis(row, 0) = right_hand_side[row] / right_hand_side_norm;
143 vec alpha(max_steps, 0.0);
144 vec beta(max_steps, 0.0);
145 vec previous;
146
147 constexpr idx minimum_steps_for_convergence_check = 4;
148 for (idx j = 0; j < max_steps; ++j) {
149 vec direction(n);
150 kernel::copy_strided(direction.data(), 1, &basis(0, j), basis.cols(), n);
151 vec image(n, 0.0);
152 A.apply(direction, image);
153
154 alpha[j] = dot(direction, image);
155 axpy(-alpha[j], direction, image);
156 if (j > 0)
157 kernel::axpy_strided(image.data(), 1, &basis(0, j - 1), basis.cols(), -beta[j - 1], n);
158
159 const real next_beta = dispatch::subspace::mgs_orthogonalize(basis, j + 1, image);
160 const idx steps = j + 1;
161 const bool breakdown = next_beta <= real(1e-14);
162 const bool geometric_checkpoint =
163 steps >= minimum_steps_for_convergence_check && (steps & (steps - 1)) == 0;
164 const bool check = geometric_checkpoint || breakdown || steps == max_steps;
165 real relative_change = 0.0;
166 if (check) {
167 const vec approximation =
168 inverse_sqrt_from_projection(basis, alpha, beta, steps, right_hand_side_norm);
169 if (previous.size() != 0) {
170 vec difference = approximation;
171 axpy(-1.0, previous, difference);
172 relative_change = norm(difference) / std::max(norm(approximation), real(1e-30));
173 if (relative_change <= tolerance) {
174 return {approximation, steps, relative_change, true};
175 }
176 }
177 if (breakdown)
178 return {approximation, steps, relative_change, true};
179 if (steps == max_steps)
180 return {approximation, steps, relative_change, false};
181 previous = approximation;
182 }
183
184 beta[j] = next_beta;
185 for (idx row = 0; row < n; ++row)
186 basis(row, j + 1) = image[row] / next_beta;
187 }
188 return {};
189}

258template <class Op>
259requires linear_operator<Op, vec, vec>
260 lanczos_result lanczos_operator_impl(const Op &A, idx k, real tol, idx max_steps) {
261 const idx n = A.rows();
262 if (A.cols() != n) {
263 throw std::invalid_argument("lanczos: operator must be square");
264 }
265 if (k == 0 || k > n) {
266 throw std::invalid_argument("lanczos: k must satisfy 0 < k <= n");
267 }
268
269 if (max_steps == 0) {
270 max_steps = std::min(3 * k, n);
271 }
272 max_steps = std::min(max_steps, n);
273
274 mat V(n, max_steps, 0.0);
275 vec alpha(max_steps, 0.0);
276 vec beta(max_steps, 0.0);
277
278 // v_0 <- e_0
279 V(0, 0) = 1.0;
280
281 idx steps = 0;
282
283 for (idx j = 0; j < max_steps; ++j) {
284 vec vj(n);
285 // v_j <- V[:,j]
286 kernel::copy_strided(vj.data(), 1, &V(0, j), V.cols(), n);
287
288 vec w(n, 0.0);
289 A.apply(vj, w);
290
291 const real a = dot(vj, w);
292 alpha[j] = a;
293
294 axpy(-a, vj, w);
295 if (j > 0) {
296 // w <- w - beta_{j-1}*v_{j-1}
297 kernel::axpy_strided(w.data(), 1, &V(0, j - 1), V.cols(), -beta[j - 1], n);
298 }
299
300 const real b = dispatch::subspace::mgs_orthogonalize(V, j + 1, w);
301 ++steps;
302
303 if (b < real(1e-12)) {
304 break;
305 }
306
307 beta[j] = b;
308
309 if (j + 1 < max_steps) {
310 // V[:,j+1] <- w/beta_j
311 kernel::scale_copy_strided(&V(0, j + 1), V.cols(), w.data(), 1, real(1) / b, n);
312 }
313 }
314
315 const idx m = steps;
316 mat T(m, m, 0.0);
317 for (idx j = 0; j < m; ++j) {
318 T(j, j) = alpha[j];
319 if (j + 1 < m) {
320 T(j, j + 1) = beta[j];
321 T(j + 1, j) = beta[j];
322 }
323 }
324
325 // T is the Lanczos tridiagonal, symmetric by construction: it is filled from a
326 // single alpha/beta recurrence with T(j,j+1) and T(j+1,j) written from the same
327 // beta. The invariant is established here rather than assumed downstream.
328 eigen_result teig = eig_sym(linear::sym_mat<mat>(T), tol * real(1e-2));
329 const idx nret = std::min(k, m);
330
331 mat ritz_vecs(n, nret, 0.0);
332 // U_Ritz <- V_m*Z_selected
333 kernel::gemm(ritz_vecs.data(), ritz_vecs.cols(), V.data(), V.cols(), &teig.vectors(0, m - nret),
334 teig.vectors.cols(), real(1), real(0), n, nret, m);
335
336 vec ritz_vals(nret);
337 // lambda_Ritz <- lambda(T_m)_selected
338 kernel::copy(ritz_vals.data(), teig.values.data() + (m - nret), nret);
339
340 bool all_converged = true;
341 for (idx i = 0; i < nret; ++i) {
342 vec u(n);
343 // u_i <- U_Ritz[:,i]
344 kernel::copy_strided(u.data(), 1, &ritz_vecs(0, i), ritz_vecs.cols(), n);
345
346 vec Au(n, 0.0);
347 A.apply(u, Au);
348
349 const real lam = ritz_vals[i];
350 // res^2 <- ||A*u_i - lambda_i*u_i||_2^2
351 const real res = kernel::linear_combination_norm_sq(Au.data(), real(1), u.data(), -lam, n);
352 if (std::sqrt(res) > tol) {
353 all_converged = false;
354 break;
355 }
356 }
357
358 return {ritz_vals, ritz_vecs, steps, all_converged};
359}

363/// Approximate \f$A^{-1/2}b\f$ for a symmetric positive-definite operator.
364///
365/// At step \f$m\f$, the returned vector is
366/// \f$\|b\|_2 Q_m T_m^{-1/2}e_1\f$. Iteration stops when consecutive
367/// projected actions differ by at most `tolerance` in relative Euclidean norm,
368/// or when `max_steps` is reached.
369template <class Op>
370requires spd_operator<Op, vec, vec> [[nodiscard]] lanczos_inverse_sqrt_result
371inverse_sqrt_lanczos(const Op &A, const vec &right_hand_side, real tolerance = 1e-8,
372 idx max_steps = 0) {
373 return detail::inverse_sqrt_lanczos_impl(A, right_hand_side, tolerance, max_steps);
374}

383/// @brief Compute the largest \f$k\f$ extremal Ritz pairs of a symmetric / self-adjoint operator
384/// using Lanczos iteration.
385///
386/// Builds an orthonormal Krylov basis with modified Gram-Schmidt reorthogonalization, generates
387/// a symmetric tridiagonal projection \f$T_m\f$, and extracts Ritz values and Ritz vectors.
388///
389/// @tparam Op Linear operator type satisfying `self_adjoint_operator<Op, vec, vec>`.
390/// @param A Self-adjoint linear operator (matrix-free callable, sparse, or dense wrapper).
391/// @param k Number of extremal eigenpairs to compute (\f$0 < k \le n\f$).
392/// @param tol Residual tolerance \f$\|A v - \lambda v\|_2\f$ for declaring convergence (default:
393/// 1e-10).
394/// @param max_steps Maximum Lanczos steps to perform (defaults to \f$\min(3k, n)\f$).
395/// @return `lanczos_result` containing Ritz values, column Ritz vectors, steps performed, and
396/// convergence flag.
397/// @throws std::invalid_argument If `k` is invalid or operator is not square.
398/// @see eig_sym, power_iteration
399template <class Op>
400requires self_adjoint_operator<Op, vec, vec>
401 lanczos_result lanczos(const Op &A, idx k, real tol = 1e-10, idx max_steps = 0) {
402 return detail::lanczos_operator_impl(A, k, tol, max_steps);
403}
