60inline solver_result field_solver::solve_poisson(scalar_field_3d &phi, const scalar_field_3d &source, double tol,
61 int max_iter) {
62 const int nx = phi.nx(), ny = phi.ny(), nz = phi.nz();
63 const double inv_dx2 = 1.0 / (phi.dx() * phi.dx());
64 const idx N = phi.size();
65
66 // RHS = -source on the interior; Dirichlet boundary rows stay 0.
67 vec b(N, 0.0);
68 for (int k = 1; k < nz - 1; ++k) {
69 for (int j = 1; j < ny - 1; ++j) {
70 for (int i = 1; i < nx - 1; ++i) {
71 b[phi.grid().flat(i, j, k)] = -source(i, j, k);
72 }
73 }
74 }
75
76 // A = -Laplacian (SPD). phi's own storage is the solution vector, so CG
77 // solves in place -- no copy in or out.
78 auto A = operators::assume_spd(operators::make_op(
79 [&](const vec &v, vec &Av) { neg_laplacian_3d(v, Av, nx, ny, nz, inv_dx2); }, N));
80 return num::cg(A, b, phi.as_vec(), tol, static_cast<idx>(max_iter));
81}

83inline solver_result field_solver::solve_var_poisson(scalar_field_3d &phi, const scalar_field_3d &coeff,
84 const array<dirichlet_bc> &bcs, double tol,
85 int max_iter) {
86 const int nx = phi.nx(), ny = phi.ny(), nz = phi.nz();
87 const idx N = phi.size();
88 const double inv_dx2 = 1.0 / (phi.dx() * phi.dx());
89
90 auto flat = [&](int i, int j, int k) -> idx {
91 return static_cast<idx>((k * ny * nx) + (j * nx) + i);
92 };
93
94 table<int, double> bc_map;
95 bc_map.reserve(bcs.size());
96 for (const auto &e : bcs) {
97 bc_map[e.flat_idx] = e.value;
98 }
99
100 constexpr int DI[6] = {1, -1, 0, 0, 0, 0};
101 constexpr int DJ[6] = {0, 0, 1, -1, 0, 0};
102 constexpr int DK[6] = {0, 0, 0, 0, 1, -1};
103 constexpr double penalty = 1e10;
104
105 // Symmetric penalty elimination: fold each Dirichlet value into the RHS of
106 // its free neighbours so the operator stays SPD.
107 vec b(N, 0.0);
108 for (const auto &e : bcs) {
109 b[e.flat_idx] = penalty * e.value;
110 const int ei = e.flat_idx % nx;
111 const int ej = (e.flat_idx / nx) % ny;
112 const int ek = e.flat_idx / (nx * ny);
113 for (int d = 0; d < 6; ++d) {
114 int ni = ei + DI[d], nj = ej + DJ[d], nk = ek + DK[d];
115 if (ni < 0 || ni >= nx || nj < 0 || nj >= ny || nk < 0 || nk >= nz) {
116 continue;
117 }
118 int nidx = flat(ni, nj, nk);
119 if (bc_map.count(nidx)) {
120 continue;
121 }
122 double sigma_face = 0.5 * (coeff(ei, ej, ek) + coeff(ni, nj, nk));
123 b[nidx] += sigma_face * inv_dx2 * e.value;
124 }
125 }
126
127 auto matvec = [&](const vec &v, vec &Av) {
128 for (int k = 0; k < nz; ++k) {
129 for (int j = 0; j < ny; ++j) {
130 for (int i = 0; i < nx; ++i) {
131 int id = flat(i, j, k);
132 if (bc_map.count(id)) {
133 Av[id] = penalty * v[id];
134 continue;
135 }
136 double Av_ijk = 0.0;
137 for (int d = 0; d < 6; ++d) {
138 int ni = std::max(0, std::min(i + DI[d], nx - 1));
139 int nj = std::max(0, std::min(j + DJ[d], ny - 1));
140 int nk = std::max(0, std::min(k + DK[d], nz - 1));
141 int nidx = flat(ni, nj, nk);
142 double c_face = 0.5 * (coeff(i, j, k) + coeff(ni, nj, nk));
143 double v_nb = bc_map.count(nidx) ? 0.0 : v[nidx];
144 Av_ijk += c_face * inv_dx2 * (v[id] - v_nb);
145 }
146 Av[id] = Av_ijk;
147 }
148 }
149 }
150 };
151
152 // Solve A phi = b in place; phi's storage is the solution vector.
153 auto A = operators::assume_spd(operators::make_op(matvec, N));
154 return num::cg(A, b, phi.as_vec(), tol, static_cast<idx>(max_iter));
155}

177inline vector_field_3d magnetic_solver::current_density(const scalar_field_3d &sigma,
178 const scalar_field_3d &phi) {
179 vector_field_3d J = field_solver::gradient(phi);
180 const int nx = sigma.nx(), ny = sigma.ny(), nz = sigma.nz();
181 for (int k = 0; k < nz; ++k) {
182 for (int j = 0; j < ny; ++j) {
183 for (int i = 0; i < nx; ++i) {
184 const double neg_s = -sigma(i, j, k);
185 J.x(i, j, k) *= neg_s;
186 J.y(i, j, k) *= neg_s;
187 J.z(i, j, k) *= neg_s;
188 }
189 }
190 }
191 return J;
192}

194inline vector_field_3d magnetic_solver::solve_magnetic_field(const vector_field_3d &J, double tol,
195 int max_iter) {
196 const int nx = J.x.nx(), ny = J.x.ny(), nz = J.x.nz();
197 const float dx = J.x.dx(), ox = J.x.ox(), oy = J.x.oy(), oz = J.x.oz();
198
199 auto make_source = [&](const scalar_field_3d &Jc) {
200 scalar_field_3d src(nx, ny, nz, dx, ox, oy, oz);
201 for (int k = 0; k < nz; ++k) {
202 for (int j = 0; j < ny; ++j) {
203 for (int i = 0; i < nx; ++i) {
204 src(i, j, k) = -MU0 * Jc(i, j, k);
205 }
206 }
207 }
208 return src;
209 };
210
211 scalar_field_3d Ax(nx, ny, nz, dx, ox, oy, oz);
212 scalar_field_3d Ay(nx, ny, nz, dx, ox, oy, oz);
213 scalar_field_3d Az(nx, ny, nz, dx, ox, oy, oz);
214
215 field_solver::solve_poisson(Ax, make_source(J.x), tol, max_iter);
216 field_solver::solve_poisson(Ay, make_source(J.y), tol, max_iter);
217 field_solver::solve_poisson(Az, make_source(J.z), tol, max_iter);
218
219 vector_field_3d A(nx, ny, nz, dx, ox, oy, oz);
220 A.x = Ax;
221 A.y = Ay;
222 A.z = Az;
223 return field_solver::curl(A);
224}
