2/// @brief Matrices produced from a graph.

4/// A graph in `structures` carries no algebraic operations. The adjacency matrix,

21/// @brief graph from which a Laplacian can be assembled.
22///
23/// The Laplacian is symmetric positive semi-definite with \f$L\mathbf{1} = 0\f$,
24/// and its null space has one dimension per connected component. Algorithms rely
25/// on that, which is why the contract names the matrix rather than a method.
26template <typename G>
27concept laplacian_graph = concepts::incidence_structure<G> && requires(const G &g) {
28 {laplacian(g)};
29};

31template <typename Weight = real, typename Index = idx>
32[[nodiscard]] inline spmat to_sparse_adjacency(const basic_graph<Weight, Index> &g) {
33 array<idx> row_ptr(g.n_vertices() + 1, 0);
34 array<idx> col_idx;
35 array<double> values;
36
37 for (Index u = 0; u < g.n_vertices(); ++u) {
38 for (const auto &e : g.neighbors(u)) {
39 col_idx.push_back(static_cast<idx>(e.to));
40 values.push_back(static_cast<double>(e.weight));
41 }
42 row_ptr[u + 1] = col_idx.size();
43 }
44 return spmat(static_cast<idx>(g.n_vertices()), static_cast<idx>(g.n_vertices()),
45 std::move(values), std::move(col_idx), std::move(row_ptr));
46}

49[[nodiscard]] inline mat to_dense_adjacency(const basic_graph<Weight, Index> &g) {
50 mat A(static_cast<idx>(g.n_vertices()), static_cast<idx>(g.n_vertices()), 0.0);
51 for (Index u = 0; u < g.n_vertices(); ++u) {
52 for (const auto &e : g.neighbors(u)) {
53 A(static_cast<idx>(u), static_cast<idx>(e.to)) += static_cast<double>(e.weight);

59template <typename Weight = real, typename Index = idx>
60[[nodiscard]] inline spmat laplacian(const basic_graph<Weight, Index> &g) {
61 array<idx> rows;
62 array<idx> cols;
63 array<double> vals;
64
65 for (Index u = 0; u < g.n_vertices(); ++u) {
66 double deg = 0.0;
67 for (const auto &e : g.neighbors(u)) {
68 if (e.to != u) {
69 rows.push_back(static_cast<idx>(u));
70 cols.push_back(static_cast<idx>(e.to));
71 vals.push_back(-static_cast<double>(e.weight));
72 deg += static_cast<double>(e.weight);
73 }
74 }
75 // Diagonal entry L(u, u) = sum_{v != u} w(u, v)
76 rows.push_back(static_cast<idx>(u));
77 cols.push_back(static_cast<idx>(u));
78 vals.push_back(deg);
79 }
80 return spmat::from_triplets(static_cast<idx>(g.n_vertices()),
81 static_cast<idx>(g.n_vertices()), rows, cols, vals);
82}

84template <typename Weight = real, typename Index = idx>
85[[nodiscard]] inline mat dense_laplacian(const basic_graph<Weight, Index> &g) {
86 mat L(static_cast<idx>(g.n_vertices()), static_cast<idx>(g.n_vertices()), 0.0);
87 for (Index u = 0; u < g.n_vertices(); ++u) {
88 double deg = 0.0;
89 for (const auto &e : g.neighbors(u)) {
90 if (e.to != u) {
91 L(static_cast<idx>(u), static_cast<idx>(e.to)) -= static_cast<double>(e.weight);
92 deg += static_cast<double>(e.weight);
93 }
94 }
95 L(static_cast<idx>(u), static_cast<idx>(u)) += deg;
96 }
97 return L;
98}

100template <typename Weight = real, typename Index = idx>
101[[nodiscard]] inline spmat markov_generator(const basic_graph<Weight, Index> &g,
102 bool column_oriented = true) {
103 array<idx> rows;
104 array<idx> cols;
105 array<double> vals;
106
107 for (Index u = 0; u < g.n_vertices(); ++u) {
108 double deg = 0.0;
109 for (const auto &e : g.neighbors(u)) {
110 if (e.to != u) {
111 if (column_oriented) {
112 rows.push_back(static_cast<idx>(e.to));
113 cols.push_back(static_cast<idx>(u));
114 } else {
115 rows.push_back(static_cast<idx>(u));
116 cols.push_back(static_cast<idx>(e.to));
117 }
118 vals.push_back(static_cast<double>(e.weight));
119 deg += static_cast<double>(e.weight);
120 }
121 }
122 rows.push_back(static_cast<idx>(u));
123 cols.push_back(static_cast<idx>(u));
124 vals.push_back(-deg);
125 }
126 return spmat::from_triplets(static_cast<idx>(g.n_vertices()),
127 static_cast<idx>(g.n_vertices()), rows, cols, vals);
128}

130template <typename Weight = real, typename Index = idx>
131[[nodiscard]] inline mat dense_markov_generator(const basic_graph<Weight, Index> &g,
132 bool column_oriented = true) {
133 mat Q(static_cast<idx>(g.n_vertices()), static_cast<idx>(g.n_vertices()), 0.0);
134 auto L = dense_laplacian(g);
135 for (idx i = 0; i < static_cast<idx>(g.n_vertices()); ++i) {
136 for (idx j = 0; j < static_cast<idx>(g.n_vertices()); ++j) {
137 Q(i, j) = column_oriented ? -L(i, j) : -L(j, i);
138 }
139 }
140 return Q;
141}

143/// @brief Laplacian of a multigraph, summing parallel edge multiplicities.
144[[nodiscard]] inline spmat laplacian(const structures::multigraph &g) {
145 const idx n = g.n_vertices();
146 array<idx> rows, cols;
147 array<real> vals;
148
149 for (idx u = 0; u < n; ++u) {
150 real diag = 0.0;
151 for (const auto &e : g.adjacency()[u]) {
152 if (e.to != u) {
153 rows.push_back(static_cast<idx>(u));
154 cols.push_back(static_cast<idx>(e.to));
155 vals.push_back(-static_cast<real>(e.weight));
156 diag += static_cast<real>(e.weight);
157 }
158 }
159 rows.push_back(static_cast<idx>(u));
160 cols.push_back(static_cast<idx>(u));
161 vals.push_back(diag);
162 }
163 return spmat::from_triplets(static_cast<idx>(n), static_cast<idx>(n), rows, cols, vals);
164}

166/// @brief Recover a multigraph from a Laplacian.
167[[nodiscard]] inline structures::multigraph to_multigraph(const spmat &L) {
168 const idx n = L.n_rows();
169 if (L.n_cols() != n) {
170 throw std::invalid_argument("to_multigraph: matrix must be square");
171 }
172 multigraph mg(n);
173 for (idx i = 0; i < n; ++i) {
174 const idx row_start = L.row_ptr()[i];
175 const idx row_end = L.row_ptr()[i + 1];
176 for (idx p = row_start; p < row_end; ++p) {
177 const idx j = L.col_idx()[p];
178 const real val = L.values()[p];
179 if (j > i && val < 0.0) {
180 mg.add_edge(i, j, -val, 1);
181 }
182 }
183 }
184 return mg;
185}
