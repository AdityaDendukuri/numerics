17/// @brief Callable returning the position of particle \f$i\f$ as \f$(x, y)\f$.
18///
19/// The acceleration structures never own coordinates. A caller supplies an
20/// accessor, so positions may live in a struct of arrays, an array of structs,

27/// @brief Structure answering which particles lie near a point.
28///
29/// The contract is a visitor rather than a returned container: `query` calls
30/// `f(j)` for each candidate, so nothing allocates in the inner loop.
31template <class L, class Scalar = real>
32concept neighbor_query_2d = scalars::field<Scalar> &&
33 requires(const L &list, Scalar px, Scalar py) {
34 list.query(px, py, [](int) {});
35};

37/// @brief Radially symmetric smoothing kernel \f$W(r, h)\f$ with its radial derivative.
38///
39/// An SPH kernel is normalized so that \f$\int W \, dV = 1\f$ over its support and
40/// vanishes beyond \f$r = 2h\f$. Both are properties of the values rather than the
41/// type, and `num::spatial::debug::verify_kernel_normalization` samples them.
42template <class K, class Scalar = float>
43concept smoothing_kernel = scalars::field<Scalar> && requires(Scalar r, Scalar h) {
44 { K::W(r, h) } -> std::convertible_to<Scalar>;
45 { K::dW_dr(r, h) } -> std::convertible_to<Scalar>;
46};

48/// @brief Site-indexed lattice supplying periodic nearest neighbours.
49///
50/// Each site \f$i\f$ on an \f$N \times N\f$ lattice has four neighbours obtained by
51/// stepping one row or column with wraparound. The tables are precomputed so a
52/// sweep never evaluates a modulus.
53template <class P>
54concept periodic_lattice_2d = periodic_neighbourhood_2d<P>;
