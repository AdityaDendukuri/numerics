30[[nodiscard]] inline real l1_norm(const vec &x) {
31#if defined(NUMERICS_HAS_OMP)
32 const idx n = x.size();
33 if (n < parallel_threshold) {
34 return seq::l1_norm(x);
35 }
36 const real *xd = x.data();
37 const idx blocks = (n + parallel_block - 1) / parallel_block;
38 real s = real(0);
39#pragma omp parallel for reduction(+ : s) schedule(static)
40 for (idx b = 0; b < blocks; ++b) {
41 const idx lo = b * parallel_block;
42 s += kernel::l1_norm(xd + lo, std::min(parallel_block, n - lo));
43 }
44 return s;
45#else
46 return seq::l1_norm(x);
47#endif
48}

50[[nodiscard]] inline real linf_norm(const vec &x) {
51#if defined(NUMERICS_HAS_OMP)
52 const idx n = x.size();
53 if (n < parallel_threshold) {
54 return seq::linf_norm(x);
55 }
56 const real *xd = x.data();
57 const idx blocks = (n + parallel_block - 1) / parallel_block;
58 real mx = real(0);
59#pragma omp parallel for reduction(max : mx) schedule(static)
60 for (idx b = 0; b < blocks; ++b) {
61 const idx lo = b * parallel_block;
62 const real v = kernel::linf_norm(xd + lo, std::min(parallel_block, n - lo));
63 if (v > mx) {
64 mx = v;
65 }
66 }
67 return mx;
68#else
69 return seq::linf_norm(x);
70#endif
71}
