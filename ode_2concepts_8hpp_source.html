11/// @brief State exposing its underlying vector space to implicit integrators.
12///
13/// Implicit steppers solve a linear system in the state space, so they need the
14/// state as a vector rather than as a field, grid or particle set. The space is a
15/// parameter refining `vector_space`, not the concrete `vec`: a state over complex
16/// amplitudes or single precision exposes its own space just as well, and the earlier
17/// form — which named `vec` outright — excluded both.
18template <class T, class V = vec>
19concept vec_field = vector_space<V> && requires(T &field) {
20 { field.as_vec() } -> std::same_as<V &>;
21};
22
23/// @brief Initial value problem \f$\dot{y} = f(t,y)\f$ on \f$[t_0, t_f]\f$ with \f$y(t_0) = u_0\f$.
24///
25/// The mathematical content is a time-dependent vector field \f$f\f$ on a state
26/// space, an initial state, and an interval. The state is required only to be a
27/// `num::vector_space`, so a problem posed over complex amplitudes or over
28/// single precision is as admissible as one over `double` — the earlier form fixed
29/// the state to `vec` and excluded both.
30template <typename Problem, class State = vec>
31concept is_ode_problem = vector_space<State> &&
32 requires(const Problem &problem, real time, const State &state, State &derivative) {
33 problem.f(time, state, derivative);
34 { problem.u0 } -> std::convertible_to<const State &>;
35 { problem.t0 } -> std::convertible_to<real>;
36 { problem.tf } -> std::convertible_to<real>;
37};
38
39/// @brief Separable Hamiltonian system \f$\dot q = f_{\mathrm{pos}}(t,p)\f$, \f$\dot p = f_{\mathrm{mom}}(t,q)\f$.
40///
41/// Separability is what symplectic integrators exploit: because each half-step
42/// depends only on the *other* coordinate, the update is explicit and exactly
43/// volume-preserving in phase space. A general Hamiltonian admits no such split,
44/// which is why this is a distinct concept rather than a refinement of
45/// `is_ode_problem`.
46template <typename Problem, class State = vec>
47concept is_symplectic_ode_problem = vector_space<State> &&
48 requires(const Problem &problem, real time, const State &q, const State &p, State &dq,
49 State &dp) {
50 problem.f_pos(time, p, dq);
51 problem.f_mom(time, q, dp);
52 { problem.q0 } -> std::convertible_to<const State &>;
53 { problem.p0 } -> std::convertible_to<const State &>;
54 { problem.t0 } -> std::convertible_to<real>;
55 { problem.tf } -> std::convertible_to<real>;
56};
