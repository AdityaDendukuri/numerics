1/// @file operator/sum.hpp
2/// @brief The sum of two operators, carrying whatever law both operands carry.
3///
4/// \f$(A + B)x = Ax + Bx\f$, evaluated without forming a matrix. The point of the type is
5/// not the arithmetic, which is two `apply` calls; it is that the sum keeps a law the
6/// caller already established.
7///
8/// The rule is a theorem, and it is exactly the meet of the two operands' laws. If \f$A\f$
9/// and \f$B\f$ are both positive definite then so is \f$A + B\f$, because
10/// \f$\langle x, (A+B)x \rangle = \langle x, Ax \rangle + \langle x, Bx \rangle > 0\f$;
11/// the same argument gives semidefiniteness, and self-adjointness follows from linearity
12/// of the adjoint. Where the operands disagree the sum keeps only what both have: an SPD
13/// operator plus a unitary one is normal and nothing stronger.
14///
15/// Without this the sum would claim nothing, and a caller wanting to solve with it would
16/// have to re-assert definiteness -- paying an O(n^2) probe to re-establish a fact already
17/// in hand, or asserting it unverified.

27/// @brief Non-owning \f$A + B\f$ for two operators over the same spaces.
28template <class Lhs, class Rhs>
29requires math::linear_operator<Lhs> && math::linear_operator<Rhs> &&
30 std::same_as<math::domain_t<Lhs>, math::domain_t<Rhs>> &&
31 std::same_as<math::codomain_t<Lhs>, math::codomain_t<Rhs>>
32class sum_op final {
33 public:
34 using domain_type = math::domain_t<Lhs>;
35 using codomain_type = math::codomain_t<Lhs>;
36
37 sum_op(const Lhs &lhs, const Rhs &rhs) : lhs_(&lhs), rhs_(&rhs) {}
38
39 void apply(const domain_type &x, codomain_type &y) const {
40 math::apply(*lhs_, x, y);
41 codomain_type t = math::zero_like(y);
42 math::apply(*rhs_, x, t);
43 math::axpy(scalar_t<codomain_type>(1), t, y);
44 }
45
46 [[nodiscard]] auto rows() const { return lhs_->rows(); }
47 [[nodiscard]] auto cols() const { return lhs_->cols(); }
48 [[nodiscard]] const Lhs &left() const noexcept { return *lhs_; }
49 [[nodiscard]] const Rhs &right() const noexcept { return *rhs_; }
50
51 private:
52 const Lhs *lhs_;
53 const Rhs *rhs_;
54};

56/// @brief Form \f$A + B\f$. Neither operand is copied, so both must outlive the result.
57template <class Lhs, class Rhs>
58[[nodiscard]] auto sum(const Lhs &lhs, const Rhs &rhs) {
59 return sum_op<Lhs, Rhs>(lhs, rhs);
60}

62// Binding to a temporary would leave the sum holding a dangling operand at the end of the
63// full expression, so rvalues are rejected rather than silently accepted.
64template <class Lhs, class Rhs>
65requires(!std::is_lvalue_reference_v<Lhs> || !std::is_lvalue_reference_v<Rhs>)
66auto sum(Lhs &&, Rhs &&) = delete;
