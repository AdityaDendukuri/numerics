1/// @file linear/graph/randommat/preconditioner.hpp
2/// @brief Adapting the ApproxChol factorization to numerics' operator vocabulary.
3///
4/// The factorization is in `approxchol.hpp` and depends on nothing above `num::structures`,
5/// so it can be lifted out of this project on its own. This header is the part that cannot:
6/// it converts a `num::spmat` Laplacian into the graph the algorithm wants, wraps the
7/// resulting factor in something satisfying `num::preconditioner`, and records the law that
8/// wrapper claims.
9///
10/// The split is what lets one implementation serve both audiences. A caller who wants the
11/// algorithm takes `approxchol.hpp` and the `structures/` tier; a caller inside numerics
12/// takes this and gets a preconditioner that drops into `num::pcg`.

29/// @brief preconditioner adapter backed by Randomized Approximate Cholesky (ApproxChol).
30/// Satisfies the num::preconditioner concept for use with num::pcg and Krylov solvers.
31class approx_chol_preconditioner final {
32 public:
33 using domain_type = vec;
34 using codomain_type = vec;
35 // A graph-Laplacian factor is singular on the constant-vector nullspace.
36 // Callers using PCG on a compatible subspace must attach that stronger,
37 // problem-specific evidence explicitly.
38 using math_laws = math::type_list<law::psd>;
39
40 /// Construct from an existing factor.
41 explicit approx_chol_preconditioner(cholesky_factor<real, idx> factor)
42 : factor_(std::move(factor)), n_(factor_.order.size()), scratch_(n_, 0.0) {}
43
44 /// Number of rows.
45 [[nodiscard]] idx rows() const noexcept { return n_; }
46
47 /// Number of columns.
48 [[nodiscard]] idx cols() const noexcept { return n_; }
49
50 /// Apply preconditioner z = M^-1 r via forward and backward substitution.
51 void apply(const vec &r, vec &z) const {
52 if (r.size() != n_) {
53 throw std::invalid_argument("approx_chol_preconditioner: dimension mismatch");
54 }
55 if (z.size() != n_) {
56 z = vec(n_, 0.0);
57 }
58 randommat::solve(factor_, r.data(), z.data(), scratch_);
59 }
60
61 /// Access underlying factor.
62 [[nodiscard]] const cholesky_factor<real, idx> &factor() const noexcept { return factor_; }
63
64 private:
65 cholesky_factor<real, idx> factor_;
66 idx n_ = 0;
67 mutable std::vector<real> scratch_;
68};

70/// Sparse approximate factor C of a grounded SDDM matrix, stored through the
71/// Cholesky factor of its one-vertex Laplacian extension. The ground vertex is
72/// pinned last and omitted from all vector actions below.
73class grounded_approx_chol_factor final {
74 public:
75 explicit grounded_approx_chol_factor(cholesky_factor<real, idx> factor)
76 : factor_(std::move(factor)),
77 n_(factor_.order.empty() ? 0 : static_cast<idx>(factor_.order.size() - 1)) {
78 if (factor_.order.empty() || factor_.order.back() != n_) {
79 throw std::invalid_argument(
80 "grounded_approx_chol_factor: ground vertex must be ordered last");
81 }
82 }
83
84 [[nodiscard]] idx rows() const noexcept { return n_; }
85 [[nodiscard]] idx cols() const noexcept { return n_; }
86
87 /// Apply C, where the approximate grounded matrix is C C^T.
88 [[nodiscard]] vec apply_lower(const vec &x) const {
89 check_dimension(x);
90 const vec permuted = permute(x);
91 vec product(n_, 0.0);
92 for (idx column = 0; column < n_; ++column) {
93 const auto &entries = retained_column(column);
94 product[column] += entries[0].value * permuted[column];
95 for (std::size_t entry = 1; entry < entries.size(); ++entry)
96 if (entries[entry].row < n_)
97 product[entries[entry].row] += entries[entry].value * permuted[column];
98 }
99 return inverse_permute(product);
100 }
101
102 /// Apply C^{-1} by forward substitution.
103 [[nodiscard]] vec solve_lower(const vec &b) const {
104 check_dimension(b);
105 vec solution = permute(b);
106 for (idx column = 0; column < n_; ++column) {
107 const auto &entries = retained_column(column);
108 solution[column] /= entries[0].value;
109 const real value = solution[column];
110 for (std::size_t entry = 1; entry < entries.size(); ++entry)
111 if (entries[entry].row < n_)
112 solution[entries[entry].row] -= entries[entry].value * value;
113 }
114 return inverse_permute(solution);
115 }
116
117 /// Apply C^{-T} by backward substitution.
118 [[nodiscard]] vec solve_upper(const vec &b) const {
119 check_dimension(b);
120 vec solution = permute(b);
121 for (idx step = 0; step < n_; ++step) {
122 const idx column = n_ - 1 - step;
123 const auto &entries = retained_column(column);
124 real value = solution[column];
125 for (std::size_t entry = 1; entry < entries.size(); ++entry)
126 if (entries[entry].row < n_)
127 value -= entries[entry].value * solution[entries[entry].row];
128 solution[column] = value / entries[0].value;
129 }
130 return inverse_permute(solution);
131 }
132
133 [[nodiscard]] const cholesky_factor<real, idx> &factor() const noexcept { return factor_; }
134
135 private:
136 void check_dimension(const vec &x) const {
137 if (x.size() != n_)
138 throw std::invalid_argument("grounded_approx_chol_factor: dimension mismatch");
139 }
140
141 [[nodiscard]] const std::vector<factor_entry<real, idx>> &retained_column(idx column) const {
142 const auto &entries = factor_.columns[column].entries;
143 if (entries.empty())
144 throw std::runtime_error("grounded_approx_chol_factor: singular retained factor");
145 return entries;
146 }
147
148 [[nodiscard]] vec permute(const vec &x) const {
149 vec result(n_, 0.0);
150 for (idx position = 0; position < n_; ++position)
151 result[position] = x[factor_.order[position]];
152 return result;
153 }
154
155 [[nodiscard]] vec inverse_permute(const vec &x) const {
156 vec result(n_, 0.0);
157 for (idx position = 0; position < n_; ++position)
158 result[factor_.order[position]] = x[position];
159 return result;
160 }
161
162 cholesky_factor<real, idx> factor_;
163 idx n_ = 0;
164};

221/// Construct a sparse approximate factor of a symmetric diagonally dominant
222/// M-matrix by adding one ground vertex for the diagonal excess.
223[[nodiscard]] inline grounded_approx_chol_factor
224grounded_approxchol_factor(const spmat &A, idx samples = 1, std::uint64_t seed = 42) {
225 const idx n = A.n_rows();
226 if (A.n_cols() != n)
227 throw std::invalid_argument("grounded_approxchol_factor: matrix must be square");
228
229 structures::multigraph graph_with_ground(n + 1);
230 bool has_ground_edge = false;
231 for (idx row = 0; row < n; ++row) {
232 real diagonal = 0.0;
233 real off_diagonal_sum = 0.0;
234 for (idx entry = A.row_ptr()[row]; entry < A.row_ptr()[row + 1]; ++entry) {
235 const idx column = A.col_idx()[entry];
236 const real value = A.values()[entry];
237 if (column == row) {
238 diagonal += value;
239 } else {
240 if (value > 1e-12)
241 throw std::invalid_argument(
242 "grounded_approxchol_factor: positive off-diagonal entry");
243 off_diagonal_sum -= value;
244 if (column > row && value < 0.0)
245 graph_with_ground.add_edge(row, column, -value, 1);
246 }
247 }
248 real ground_weight = diagonal - off_diagonal_sum;
249 const real tolerance = 1e-12 * std::max(real{1}, std::abs(diagonal));
250 if (ground_weight < -tolerance)
251 throw std::invalid_argument(
252 "grounded_approxchol_factor: matrix is not diagonally dominant");
253 if (ground_weight > tolerance) {
254 graph_with_ground.add_edge(row, n, ground_weight, 1);
255 has_ground_edge = true;
256 }
257 }
258 if (!has_ground_edge)
259 throw std::invalid_argument(
260 "grounded_approxchol_factor: matrix has no positive grounding term");
261
262 rng64 rng(seed);
263 auto factor = factorize<real, idx>(graph_with_ground.adjacency(), samples, false, &rng,
264 clique_sampler::independent, n);
265 return grounded_approx_chol_factor(std::move(factor));
266}
