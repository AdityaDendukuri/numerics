26/// @brief Validate structural invariants of CSR sparse storage.
27template <class SparseType>
28inline void verify_sparse_structure(const SparseType &A,
29 std::source_location loc = std::source_location::current()) {
30 if (get_level() == diagnostic_level::off) {
31 return;
32 }
33 const idx nrows = A.n_rows();
34 const idx ncols = A.n_cols();
35 const idx *row_ptr = A.row_ptr();
36 const idx *col_idx = A.col_idx();
37 const real *values = A.values();
38
39 if (row_ptr[0] != 0) {
40 panic("SparseStructureError", "row_ptr[0] must be 0", loc);
41 }
42 for (idx i = 0; i < nrows; ++i) {
43 if (row_ptr[i] > row_ptr[i + 1]) {
44 panic("SparseStructureError",
45 "row_ptr is not monotonic at row " + std::to_string(i), loc);
46 }
47 for (idx k = row_ptr[i]; k < row_ptr[i + 1]; ++k) {
48 if (col_idx[k] >= ncols) {
49 panic("SparseStructureError",
50 "col_idx[" + std::to_string(k) + "] = " + std::to_string(col_idx[k]) +
51 " exceeds n_cols (" + std::to_string(ncols) + ")",
52 loc);
53 }
54 if (!std::isfinite(values[k])) {
55 panic("SparseStructureError",
56 "non-finite sparse value at index " + std::to_string(k), loc);
57 }
58 }
59 }
60}

82/// @brief Validate banded matrix bandwidth bounds.
83template <class BandedType>
84inline void verify_banded_structure(const BandedType &B,
85 std::source_location loc = std::source_location::current()) {
86 if (get_level() == diagnostic_level::off) {
87 return;
88 }
89 check_dim(B.rows(), B.cols(), "banded matrix dimension", loc);
90 const num::idx kl = repr::lower_bandwidth_of(B);
91 const num::idx ku = repr::upper_bandwidth_of(B);
92 if (kl >= B.rows() || ku >= B.cols()) {
93 panic("BandedStructureError",
94 "bandwidth exceeds matrix dimension: kl=" + std::to_string(kl) + ", ku=" +
95 std::to_string(ku) + " for a " + std::to_string(B.rows()) + "x" +
96 std::to_string(B.cols()) + " matrix",
97 loc);
98 }
99}

102/// @brief Verify that entries outside the declared band actually vanish.
103///
104/// The dimension check alone only confirms the bandwidths are representable. This
105/// is the substantive claim: a banded solver reads nothing outside the band, so a
106/// nonzero entry there is silently discarded rather than producing a wrong answer
107/// loudly.
108template <class BandedType>
109inline void verify_band_occupancy(const BandedType &B, double tol = 0.0,
110 std::source_location loc = std::source_location::current()) {
111 if (get_level() == diagnostic_level::off) {
112 return;
113 }
114 const num::idx kl = repr::lower_bandwidth_of(B);
115 const num::idx ku = repr::upper_bandwidth_of(B);
116 for (num::idx i = 0; i < B.rows(); ++i) {
117 for (num::idx j = 0; j < B.cols(); ++j) {
118 const bool below = (i > j) && ((i - j) > kl);
119 const bool above = (j > i) && ((j - i) > ku);
120 if (!below && !above) {
121 continue;
122 }
123 if (std::abs(static_cast<double>(B(i, j))) > tol) {
124 panic("BandedStructureError",
125 "entry (" + std::to_string(i) + "," + std::to_string(j) + ") = " +
126 std::to_string(static_cast<double>(B(i, j))) +
127 " lies outside the declared band (kl=" + std::to_string(kl) +
128 ", ku=" + std::to_string(ku) +
129 "), where a banded routine would never read it.",
130 loc);
131 }
132 }
133 }
