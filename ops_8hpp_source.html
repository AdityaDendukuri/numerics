1/// @file algebra/ops.hpp
2/// @brief vec space operations, resolved to whichever form a type provides.
3///
4/// A concept requiring `size()` and `operator[]` describes a *coordinate array*.
5/// It does not describe a vector space, because it says nothing about closure —
6/// whether a sum of two elements is itself an element, or whether a zero exists.
7/// `std::span<const T>` is indexable over a field and is emphatically not a vector
8/// space: there is nowhere to put the sum.
9///
10/// So the structure here is stated in two halves, matching the split in
11/// `algebra/properties.hpp`:
12///
13/// - **Structure**, decided by the compiler: the space supplies the operations
14/// (addition, scalar multiplication, a zero) and is closed under them.
15/// - **Laws**, asserted and sampled: associativity, commutativity, distributivity,
16/// conjugate symmetry of the inner product, the triangle inequality. No type
17/// can prove these; `num::algebra::verify_*` probes them on real instances.
18///
19/// The samplers deliberately exercise *the type's own* operations where it has
20/// them — `num::dot`, `num::norm`, `num::axpy` are found by argument-dependent
21/// lookup — so the laws are checked against the shipped implementation, backend
22/// dispatch included, rather than against a reimplementation written here.

48/// True when V exposes contiguous storage over a real field, so a raw kernel applies.
49template <class V>
50concept raw_reducible = std::floating_point<scalar_t<V>> && requires(const V &v) {
51 { v.data() } -> std::convertible_to<const scalar_t<V> *>;
52 { v.size() };
53};

77/// @brief Inner product \f$\langle x,y \rangle = \sum_i \overline{x_i} y_i\f$.
78///
79/// Uses the type's own `dot` when it has one, so what gets law-checked is the
80/// shipped routine rather than a local reimplementation.
81template <class V>
82[[nodiscard]] inline scalar_t<V> inner(const V &x, const V &y) {
83 using T = scalar_t<V>;
84 if constexpr (detail::has_free_dot<V>) {
85 return static_cast<T>(dot(x, y));
86 } else if constexpr (detail::raw_reducible<V>) {
87 // contiguous real storage: defer to the kernel rather than restating the loop.
88 return kernel::dot(x.data(), y.data(), x.size());
89 } else {
90 T sum = T(0);
91 for (idx i = 0; i < x.size(); ++i) {
92 sum += scalars::conj(x[i]) * y[i];
93 }
94 return sum;
95 }
96}

98/// @brief Norm \f$\|x\|\f$, taken from the type's own `norm` when available.
99template <class V>
100[[nodiscard]] inline scalars::real_t<scalar_t<V>> norm_of(const V &x) {
101 if constexpr (detail::has_free_norm<V>) {
102 return static_cast<scalars::real_t<scalar_t<V>>>(norm(x));
103 } else if constexpr (detail::raw_reducible<V>) {
104 return kernel::norm(x.data(), x.size());
105 } else {
106 return std::sqrt(scalars::re(inner(x, x)));
107 }
108}

110/// @brief In-place update \f$y \leftarrow y + a x\f$.
111template <class V>
112inline void axpy_into(scalar_t<V> a, const V &x, V &y) {
113 if constexpr (detail::has_free_axpy<V, scalar_t<V>>) {
114 axpy(a, x, y);
115 } else {
116 for (idx i = 0; i < y.size(); ++i) {
117 y[i] = y[i] + (a * x[i]);
118 }
119 }
120}

122/// @brief In-place scaling \f$v \leftarrow a v\f$.
123template <class V>
124inline void scale_inplace(V &v, scalar_t<V> a) {
125 if constexpr (detail::has_free_scale<V, scalar_t<V>>) {
126 scale(v, a);
127 } else {
128 for (idx i = 0; i < v.size(); ++i) {
129 v[i] = a * v[i];
130 }
131 }
132}

134/// @brief The zero vector of dimension n; the additive identity.

136[[nodiscard]] inline V zero(idx n) {
137 V z(n);
138 for (idx i = 0; i < n; ++i) {
