55void dense_resolvent_solver::solve(const array<cplx> &rhs, array<cplx> &result) const {
56 const idx n = impl_->decomp.size();
57 if (!impl_->factored) {
58 throw std::invalid_argument("dense_resolvent_solver: factorization required before solve");
59 }
60 debug::check_dim(n, static_cast<idx>(rhs.size()), "dense_resolvent_solver RHS");
61
62 // 1. Project RHS to Hessenberg coordinates: b_tilde = Q^T * rhs (O(n^2))
63 array<cplx> b_tilde(n);
64 kernel::matvec_transpose_into_complex(b_tilde.data(), impl_->decomp.Q().data(),
65 rhs.data(), n, n);
66
67 // 2. Substitute through the cached factorization (O(n^2))
68 array<cplx> y(n);
69 kernel::hessenberg_shifted_substitute(y.data(), impl_->M_buf.data(),
70 impl_->pivots.data(), b_tilde.data(), n);
71
72 // 4. Back-project solution: result = Q * y (O(n^2))
73 hessenberg_back_project(impl_->decomp.Q(), y, result);
74}
