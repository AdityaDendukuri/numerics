16inline void bit_reverse(num::cvec &a) {
17 num::idx n = a.size();
18 for (num::idx i = 1, j = 0; i < n; ++i) {
19 num::idx bit = n >> 1;
20 for (; j & bit; bit >>= 1) {
21 j ^= bit;
22 }
23 j ^= bit;
24 if (i < j) {
25 std::swap(a[i], a[j]);
26 }
27 }
28}

30inline void cooley_tukey(num::cvec &a, bool invert) {
31 num::idx n = a.size();
32 if (n == 0 || (n & (n - 1))) {
33 throw std::invalid_argument("FFT: length must be a power of two");
34 }
35 bit_reverse(a);
36 for (num::idx len = 2; len <= n; len <<= 1) {
37 double ang = TWO_PI / static_cast<double>(len) * (invert ? 1.0 : -1.0);
38 num::cplx wlen{std::cos(ang), std::sin(ang)};
39 for (num::idx i = 0; i < n; i += len) {
40 num::cplx w{1.0, 0.0};
41 for (num::idx j = 0; j < len / 2; ++j) {
42 num::cplx u = a[i + j];
43 num::cplx v = a[i + j + (len / 2)] * w;
44 a[i + j] = u + v;
45 a[i + j + (len / 2)] = u - v;
46 w *= wlen;
47 }
48 }
49 }
50}

78inline void irfft(const num::cvec &in, int n, num::vec &out) {

80 for (num::idx k = 0; k < static_cast<num::idx>((n / 2) + 1); ++k) {

98 fft_plan_impl(int n_, bool inv) : n(n_), invert(inv) {
99 if (n_ == 0 || (n_ & (n_ - 1))) {
100 throw std::invalid_argument("fft_plan: length must be a power of two");
101 }
102 for (int len = 2; len <= n_; len <<= 1) {
103 double ang = TWO_PI / static_cast<double>(len) * (inv ? 1.0 : -1.0);
104 num::cplx wlen{std::cos(ang), std::sin(ang)};
105 std::vector<num::cplx> tw(len / 2);
106 num::cplx w{1.0, 0.0};
107 for (int j = 0; j < len / 2; ++j) {
108 tw[j] = w;
109 w *= wlen;
110 }
111 twiddles.push_back(std::move(tw));
112 }
