19/// Conjugate gradients over any certified real inner-product space.
20///
21/// This is the canonical implementation: mathematical requirements are checked
22/// here and representation-specific operations lower through the shared CPOs.
23template <class Op, class V>
24requires math::inner_product_space<V> &&math::endomorphism_on<Op, V> &&
25 claims<Op, law::spd> &&
26 std::floating_point<math::scalar_t<V>> [[nodiscard]] solver_result
27 cg(const Op &A, const V &b, V &x, cg_options options = {}) {
28 using S = math::scalar_t<V>;
29
30 const auto n = math::dimension(b);
31 if (math::dimension(x) != n || A.rows() != n || A.cols() != n) {
32 throw std::invalid_argument("cg: incompatible operator and vector dimensions");
33 }
34 if (!(options.tolerance > 0.0)) {
35 throw std::invalid_argument("cg: invalid convergence options");
36 }
37
38 V residual = math::zero_like(b);
39 V direction = math::zero_like(b);
40 V applied = math::zero_like(b);
41
42 math::apply(A, x, residual);
43 // r <- b - A*x
44 math::linear_combination(S(1), b, S(-1), residual);
45 direction = residual;
46
47 S residual_square = math::inner(residual, residual);
48 solver_result result{0, static_cast<real>(std::sqrt(residual_square)), false};
49 if (result.residual < options.tolerance) {
50 result.converged = true;
51 return result;
52 }
53
54 for (idx iteration = 0; iteration < options.max_iterations; ++iteration) {
55 result.iterations = iteration + 1;
56 math::apply(A, direction, applied);
57
58 const S curvature = math::inner(direction, applied);
59 if (!(curvature > S(0)) || !std::isfinite(curvature)) {
60 throw std::runtime_error("cg: positive-definite curvature invariant was violated");
61 }
62
63 const S alpha = residual_square / curvature;
64 // x <- x + alpha*p
65 math::axpy(alpha, direction, x);
66
67 // r <- r - alpha*A*p; ||r||_2^2
68 const S next_square = math::axpy_norm_sq(-alpha, applied, residual);
69 if (!(next_square >= S(0)) || !std::isfinite(next_square)) {
70 throw std::runtime_error("cg: inner-product norm invariant was violated");
71 }
72 result.residual = static_cast<real>(std::sqrt(next_square));
73 if (result.residual < options.tolerance) {
74 result.converged = true;
75 break;
76 }
77
78 const S beta = next_square / residual_square;
79 // p <- r + beta*p
80 math::linear_combination(S(1), residual, beta, direction);
81 residual_square = next_square;
82 }
83
84 return result;
85}
