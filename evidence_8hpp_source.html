2/// @brief Immutable evidence that a runtime value satisfies a law.
3///
4/// `claims<T, L>` (models.hpp) asks what a *type* asserts. This header is about what a
5/// particular *value* has been shown to satisfy, and by what means: `assume<L>(x)` records
6/// a caller's promise, `require<L>(x)` records an exhaustive check. Both carry an

111/// Attach caller-supplied evidence. Decidable shape prerequisites remain enforced.
112template <class P, class T>
113requires mathematical_proposition<P> [[nodiscard]] certified_ref<T, P>
114assume(const T &value, std::source_location location = std::source_location::current()) {
115 if constexpr (std::derived_from<P, law::endomorphism>) {
116 if (value.rows() != value.cols()) {
117 throw std::invalid_argument(
118 "cannot certify a self-adjoint property on a non-square value");
119 }
120 }
121 return detail::evidence_access::make<T, P>(
122 value, {evidence_origin::assumed, location, "explicit caller assumption"});
123}

125// certified_ref is non-owning. Binding it to a temporary would leave dangling
126// evidence at the end of the full expression, so rvalues are rejected even
127// though a const reference could otherwise bind to them.
128template <class P, class T>
129requires mathematical_proposition<P> certified_ref<T, P>

132/// Exhaustively validate P using a type-specific validator before attaching evidence.
133template <class P, class T>
