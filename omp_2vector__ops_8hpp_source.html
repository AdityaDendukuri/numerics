1/// @file omp/vector_ops.hpp
2/// @brief OpenMP-accelerated Level-1 vector operations and threaded block reduction.
3///
4/// Plain functions, no tag/enum indirection: a caller who wants this backend
5/// calls `num::omp::scale`/`num::omp::axpy`/... directly, or an algorithm
6/// templated on the backend namespace instantiates with `num::omp`.
7///
8/// Every function here hands each thread a block and calls the matching
9/// `num::kernel` routine on it — never a hand-written per-element loop. OMP's
10/// job is strictly the division of labor across threads; the arithmetic inside
11/// each thread's block is exactly the same inlined, vectorized loop `num::seq`
12/// runs, so there is exactly one implementation of "what scale/axpy/etc. does"
13/// in the whole tree, and OpenMP only decides how it's sliced up.

28inline void scale(vec &v, real alpha) noexcept {
29 real *d = v.data();
30 parallel_apply(v.size(),

34inline void axpy(real alpha, const vec &x, vec &y) noexcept {
35 const real *xd = x.data();
36 real *yd = y.data();
37 parallel_apply(x.size(), [xd, yd, alpha](idx offset, idx length) {
38 kernel::axpy(yd + offset, xd + offset, alpha, length);

42[[nodiscard]] inline real dot(const vec &x, const vec &y) noexcept {
43 const real *xd = x.data();
44 const real *yd = y.data();
45 // A block each, reduced by the raw kernel, so threading composes with the
46 // vector accumulators instead of replacing them with a scalar chain.
47 return parallel_reduce<real>(
48 x.size(), [xd, yd](idx offset, idx length) { return kernel::dot(xd + offset, yd + offset, length); });
49}

51[[nodiscard]] inline real norm(const vec &x) noexcept {
