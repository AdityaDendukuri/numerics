1/// @file lu_no_pivot.hpp
2/// @brief LU factorization without row pivoting, for matrices whose structure
3/// guarantees nonzero pivots (e.g. diagonally dominant M-matrices).
4///
5/// Skipping pivoting removes the row search and swap from `num::lu`'s inner
6/// loop, which matters when the factorization runs once per timestep (as in
7/// ELSE-style implicit solvers) and the matrix's structure already rules out a
8/// zero or tiny pivot. On a matrix that does not have that guarantee, `singular`
9/// reports a zero pivot but the factors are otherwise unchecked — prefer
10/// `num::lu`, which pivots, unless the structural guarantee is known to hold.

21/// @brief Packed factorization \f$A = LU\f$ with an implicit unit-diagonal \f$L\f$
22/// and no row pivoting.
23struct no_pivot_lu {
24 mat packed;
25 bool singular = false;
26
27 [[nodiscard]] idx size() const { return packed.rows(); }

30[[nodiscard]] inline no_pivot_lu factor_no_pivot(const linear::sq_mat<mat> &matrix) {
31 no_pivot_lu factor{matrix.base(), false};
32 factor.singular = !kernel::lu_no_pivot(factor.packed.data(), factor.packed.rows());
33 return factor;
34}
