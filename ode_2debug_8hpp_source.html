1/// @file ode/debug.hpp
2/// @brief Runtime verification of the laws an integrator is supposed to obey.
3///
4/// The diagnostic sibling of ode/concepts.hpp. What distinguishes a correct ODE
5/// integrator from a plausible one is not its interface but two mathematical
6/// properties, and both are measurable at runtime:
7///
8/// - **Order of accuracy.** A method of order \f$p\f$ has local error
9/// \f$O(h^{p+1})\f$, so halving the step must shrink the error by
10/// \f$2^{p+1}\f$. Comparing the observed ratio against the claimed order
11/// catches a mis-specified Butcher tableau, which otherwise merely converges
12/// more slowly than advertised and is easy to miss.
13/// - **Symplecticity.** A symplectic map preserves the 2-form
14/// \f$\omega = dq \wedge dp\f$. On a Hamiltonian system this is what bounds
15/// energy error over long integrations; a method that loses it drifts
16/// secularly no matter how small the step.

32/// @brief Measure the observed order of accuracy of a one-step integrator.
33///
34/// Advances the same initial state over one interval with step \f$h\f$ and again
35/// with \f$h/2\f$, and compares both against a reference taken at \f$h/16\f$. For a
36/// method of order \f$p\f$ the error ratio approaches \f$2^p\f$ over the fixed
37/// interval, so \f$\log_2\f$ of the ratio recovers \f$p\f$.
38///
39/// @param advance Callable `(real t0, real t1, real h, const vec &y0, vec &y1)`.
40/// @param y0 Initial state.
41/// @param t0 Start of the integration interval.
42/// @param t1 End of the integration interval.
43/// @param claimed_order The order the method advertises.
44/// @param slack Tolerated shortfall in the measured order.
45/// @param loc Call site reported in the diagnostic.
46template <class Advance>
47inline void verify_order_of_accuracy(Advance &&advance, const vec &y0, real t0, real t1,
48 real claimed_order, real slack = 0.4,
49 std::source_location loc = std::source_location::current()) {
50 if constexpr (!num::debug::sampling_compiled_in) {
51 return;
52 }
53 if (num::debug::get_level() != num::debug::diagnostic_level::full) {
54 return;
55 }
56 const real span = t1 - t0;
57 if (!(span > real(0)) || y0.size() == 0) {
58 return;
59 }
60
61 const real h = span / real(8);
62 vec coarse(y0.size()), fine(y0.size()), reference(y0.size());
63 advance(t0, t1, h, y0, coarse);
64 advance(t0, t1, h / real(2), y0, fine);
65 advance(t0, t1, h / real(16), y0, reference);
66
67 auto distance = [](const vec &a, const vec &b) {
68 real sum = 0.0;
69 for (idx i = 0; i < a.size(); ++i) {
70 const real d = a[i] - b[i];
71 sum += d * d;
72 }
73 return std::sqrt(sum);
74 };
75
76 const real error_coarse = distance(coarse, reference);
77 const real error_fine = distance(fine, reference);
78
79 // Both errors at round-off means the method is exact on this problem
80 // (a linear system under a high-order method, say); there is no slope to fit.
81 if (error_coarse < 1e-13 || error_fine < 1e-14) {
82 return;
83 }
84
85 const real observed = std::log2(error_coarse / error_fine);
86 if (observed < claimed_order - slack) {
87 panic("PropertyError",
88 "integrator order-of-accuracy check failed: measured order " +
89 std::to_string(observed) + " but the method claims " +
90 std::to_string(claimed_order) +
91 ". Halving the step did not reduce the error as the claimed order requires.",
92 loc);
93 }
94}

96/// @brief Verify that one step preserves the symplectic 2-form \f$\omega = dq \wedge dp\f$.
97///
98/// Propagates two tangent vectors through the discrete flow by finite differences
99/// and checks that their symplectic pairing
100/// \f$\omega(u,v) = \delta q_u \cdot \delta p_v - \delta q_v \cdot \delta p_u\f$
101/// is unchanged. This is the defining property of a symplectic map, and it is
102/// strictly stronger than energy conservation: a method can nearly conserve energy
103/// over a short window while destroying \f$\omega\f$, and will then drift.
104///
105/// @param step Callable `(const vec &q, const vec &p, real h, vec &q1, vec &p1)`.
106/// @param q0 Position at which to test.
107/// @param p0 Momentum at which to test.
108/// @param h Step size.
109/// @param tol Relative tolerance on the preserved form.
110/// @param loc Call site reported in the diagnostic.
111template <class Step>
112inline void verify_symplectic_2form(Step &&step, const vec &q0, const vec &p0, real h,
113 real tol = 1e-6,
114 std::source_location loc = std::source_location::current()) {
115 if constexpr (!num::debug::sampling_compiled_in) {
116 return;
117 }
118 if (num::debug::get_level() != num::debug::diagnostic_level::full) {
119 return;
120 }
121 const idx n = q0.size();
122 if (n == 0 || p0.size() != n) {
123 return;
124 }
125
126 const real eps = 1e-6;
127
128 // Two independent tangent directions in phase space.
129 auto flow = [&](const vec &q, const vec &p, vec &q1, vec &p1) { step(q, p, h, q1, p1); };
130
131 vec base_q(n), base_p(n);
132 flow(q0, p0, base_q, base_p);
133
134 // Tangent 1: perturb q[0]. Tangent 2: perturb p[0].
135 vec qa = q0, pa = p0, qb = q0, pb = p0;
136 qa[0] += eps;
137 pb[0] += eps;
138
139 vec qa1(n), pa1(n), qb1(n), pb1(n);
140 flow(qa, pa, qa1, pa1);
141 flow(qb, pb, qb1, pb1);
142
143 // Finite-difference tangents of the discrete flow.
144 real omega_after = 0.0;
145 for (idx i = 0; i < n; ++i) {
146 const real dq_u = (qa1[i] - base_q[i]) / eps;
147 const real dp_u = (pa1[i] - base_p[i]) / eps;
148 const real dq_v = (qb1[i] - base_q[i]) / eps;
149 const real dp_v = (pb1[i] - base_p[i]) / eps;
150 omega_after += (dq_u * dp_v) - (dq_v * dp_u);
151 }
152
153 // Before the step the tangents are the unit vectors e_{q0} and e_{p0},
154 // whose pairing is exactly 1.
155 const real omega_before = 1.0;
156 const real drift = std::abs(omega_after - omega_before);
157 if (drift > tol) {
158 panic("PropertyError",
159 "symplectic 2-form is not preserved: omega changed by " + std::to_string(drift) +
160 " over one step (dq^dp = " + std::to_string(omega_after) +
161 ", expected 1). The integrator is NOT symplectic and will drift secularly.",
162 loc);
163 }
164}
