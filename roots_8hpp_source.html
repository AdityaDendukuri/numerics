32/// @brief Bisection method for finding a root on a bracketing interval \f$[a, b]\f$ where \f$f(a) f(b) \le 0\f$.
33///
34/// Guaranteed linear convergence halving the interval width at each step: \f$|x_k - x^*| \le \frac{b-a}{2^k}\f$.
35///
36/// @tparam Float scalar floating-point type (`double`, `float`).
37/// @tparam Func Function callable `Float(Float x)`.
38/// @param f Target continuous function.
39/// @param a Left interval endpoint.
40/// @param b Right interval endpoint.
41/// @param tol Error tolerance on interval radius and residual \f$|f(x)|\f$ (default: 1e-10).
42/// @param max_iter Maximum iterations (default: 1000).
43/// @return `basic_root_result<Float>` with root approximation, iteration count, final residual, and convergence status.
44/// @throws std::invalid_argument If \f$f(a)\f$ and \f$f(b)\f$ have the same sign.
45/// @see brent, newton, secant
46template <typename Float = double, std::invocable<Float> Func = scalar_fn>
47inline basic_root_result<Float> bisection(Func &&f, Float a, Float b,
48 Float tol = Float{1e-10}, idx max_iter = 1000) {
49 Float fa = f(a), fb = f(b);
50 if (fa * fb > Float{0}) {
51 throw std::invalid_argument("bisection: f(a) and f(b) must have opposite signs");
52 }
53
54 for (idx i = 0; i < max_iter; ++i) {
55 Float mid = Float{0.5} * (a + b);
56 Float fm = f(mid);
57 if (std::abs(fm) < tol || Float{0.5} * (b - a) < tol) {
58 return {mid, i + 1, std::abs(fm), true};
59 }
60 if (fa * fm < Float{0}) {
61 b = mid;
62 fb = fm;
63 } else {
64 a = mid;
65 fa = fm;
66 }
67 }
68 Float mid = Float{0.5} * (a + b);
69 return {mid, max_iter, std::abs(f(mid)), false};
70}

72/// @brief Newton–Raphson method with quadratic convergence using analytical derivative \f$f'(x)\f$.
73///
74/// Iteratively computes \f$x_{k+1} = x_k - \frac{f(x_k)}{f'(x_k)}\f$. Achieves quadratic convergence \f$\mathcal{O}(\epsilon^2)\f$ near simple roots.
75///
76/// @tparam Float Floating point type.
77/// @tparam Func Target function callable `Float(Float x)`.
78/// @tparam DFunc Derivative function callable `Float(Float x)`.
79/// @param f Target function \f$f(x)\f$.
80/// @param df Derivative function \f$f'(x)\f$.
81/// @param x0 Initial root estimate.
82/// @param tol Residual convergence tolerance \f$|f(x)| < \text{tol}\f$ (default: 1e-10).
83/// @param max_iter Maximum iterations (default: 1000).
84/// @return `basic_root_result<Float>` with root estimate and convergence info.
85/// @see secant, brent, bisection
86template <typename Float = double, std::invocable<Float> Func = scalar_fn,
87 std::invocable<Float> DFunc = scalar_fn>
88inline basic_root_result<Float> newton(Func &&f, DFunc &&df, Float x0,
89 Float tol = Float{1e-10}, idx max_iter = 1000) {
90 Float x = x0;
91 for (idx i = 0; i < max_iter; ++i) {
92 Float fx = f(x);
93 if (std::abs(fx) < tol) {
94 return {x, i + 1, std::abs(fx), true};
95 }
96 Float dfx = df(x);
97 if (std::abs(dfx) < Float{1e-14}) {
98 return {x, i + 1, std::abs(fx), false}; // near-zero derivative
99 }
100 x -= fx / dfx;
101 }
102 return {x, max_iter, std::abs(f(x)), false};
103}

105/// @brief Secant quasi-Newton root-finding method requiring two initial guesses \f$x_0, x_1\f$.
106///
107/// Replaces the exact derivative with a finite difference chord: \f$x_{k+1} = x_k - f(x_k) \frac{x_k - x_{k-1}}{f(x_k) - f(x_{k-1})}\f$. Superlinear convergence order \f$\approx 1.618\f$.
108///
109/// @tparam Float Floating point type.
110/// @tparam Func Target function callable.
111/// @param f Target function.
112/// @param x0 First initial guess.
113/// @param x1 Second initial guess.
114/// @param tol Residual tolerance (default: 1e-10).
115/// @param max_iter Maximum iterations (default: 1000).
116/// @return `basic_root_result<Float>` with root estimate and convergence metadata.
117/// @see newton, brent
118template <typename Float = double, std::invocable<Float> Func = scalar_fn>
119inline basic_root_result<Float> secant(Func &&f, Float x0, Float x1,
120 Float tol = Float{1e-10}, idx max_iter = 1000) {
121 Float f0 = f(x0), f1 = f(x1);
122 for (idx i = 0; i < max_iter; ++i) {
123 if (std::abs(f1) < tol) {
124 return {x1, i + 1, std::abs(f1), true};
125 }
126 Float df = f1 - f0;
127 if (std::abs(df) < Float{1e-14}) {
128 return {x1, i + 1, std::abs(f1), false}; // stagnation
129 }
130 Float x2 = x1 - (f1 * (x1 - x0) / df);
131 x0 = x1;
132 f0 = f1;
133 x1 = x2;
134 f1 = f(x1);
135 }
136 return {x1, max_iter, std::abs(f1), false};
137}

139/// @brief Brent's hybrid root solver (bisection + secant + inverse quadratic interpolation).
140///
141/// Combines the guaranteed convergence of bisection with the superlinear speed of secant and IQI methods.
142/// Standard choice for derivative-free robust 1D root-finding.
143///
144/// @tparam Float Floating point type.
145/// @tparam Func Target function callable.
146/// @param f Target function.
147/// @param a Left interval endpoint.
148/// @param b Right interval endpoint.
149/// @param tol Convergence tolerance on bracket width and residual (default: 1e-10).
150/// @param max_iter Maximum iterations (default: 1000).
151/// @return `basic_root_result<Float>` with root estimate and convergence metadata.
152/// @throws std::invalid_argument If \f$f(a) f(b) > 0\f$ (no sign change).
153/// @see bisection, newton, secant
154template <typename Float = double, std::invocable<Float> Func = scalar_fn>
155inline basic_root_result<Float> brent(Func &&f, Float a, Float b,
156 Float tol = Float{1e-10}, idx max_iter = 1000) {
157 Float fa = f(a), fb = f(b);
158 if (fa * fb > Float{0}) {
159 throw std::invalid_argument("brent: f(a) and f(b) must have opposite signs");
160 }
161
162 Float c = a, fc = fa;
163 Float d = b - a, e = d;
164
165 for (idx i = 0; i < max_iter; ++i) {
166 if (fb * fc > Float{0}) {
167 c = a;
168 fc = fa;
169 d = e = b - a;
170 }
171 if (std::abs(fc) < std::abs(fb)) {
172 a = b;
173 fa = fb;
174 b = c;
175 fb = fc;
176 c = a;
177 fc = fa;
178 }
179
180 Float tol1 = (Float{2.0} * Float{1e-15} * std::abs(b)) + (Float{0.5} * tol);
181 Float mid = Float{0.5} * (c - b);
182
183 if (std::abs(mid) <= tol1 || std::abs(fb) < tol) {
184 return {b, i + 1, std::abs(fb), true};
185 }
186
187 if (std::abs(e) >= tol1 && std::abs(fa) > std::abs(fb)) {
188 Float s = fb / fa;
189 Float p{}, q{};
190 if (a == c) {
191 p = Float{2.0} * mid * s;
192 q = Float{1.0} - s;
193 } else {
194 Float r = fb / fc;
195 Float t = fa / fc;
196 p = s * ((Float{2.0} * mid * t * (t - r)) - ((b - a) * (r - Float{1.0})));
197 q = (t - Float{1.0}) * (r - Float{1.0}) * (s - Float{1.0});
198 }
199 if (p > Float{0}) {
200 q = -q;
201 } else {
202 p = -p;
203 }
204
205 Float e_prev = e;
206 if (Float{2.0} * p < std::min((Float{3.0} * mid * q) - std::abs(tol1 * q), std::abs(e_prev * q))) {
207 e = d;
208 d = p / q;
209 } else {
210 d = mid;
211 e = mid;
212 }
213 } else {
214 d = mid;
215 e = mid;
216 }
217
218 a = b;
219 fa = fb;
220 b += (std::abs(d) > tol1) ? d : (mid > Float{0} ? tol1 : -tol1);
221 fb = f(b);
222 }
223 return {b, max_iter, std::abs(fb), false};
224}
