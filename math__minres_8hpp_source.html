28inline vec minres_projected_solve(const array<real> &alpha, const array<real> &beta,
29 real beta0, idx m) {
30 mat H(m + 1, m, 0.0);
31 for (idx j = 0; j < m; ++j) {
32 H(j, j) = alpha[j];
33 if (j > 0)
34 H(j - 1, j) = beta[j - 1];
35 H(j + 1, j) = beta[j];
36 }
37 vec rhs(m + 1, 0.0);
38 rhs[0] = beta0;
39 const qr_result factor = qr(H);
40 vec y(m, 0.0);
41 qr_solve(factor, rhs, y);
42 return y;
43}

47/// Minimum residual projection for a certified self-adjoint endomorphism.
48template <class Op, class V>
49requires math::inner_product_space<V> &&math::endomorphism_on<Op, V> &&
50 claims<Op, law::self_adjoint> &&
51 std::same_as<math::scalar_t<V>, real> [[nodiscard]] solver_result
52 minres(const Op &A, const V &b, V &x, minres_options options = {}) {
53 const auto n = math::dimension(b);
54 if (math::dimension(x) != n || A.rows() != n || A.cols() != n) {
55 throw std::invalid_argument("minres: incompatible operator and vector dimensions");
56 }
57 if (!(options.tolerance > 0.0)) {
58 throw std::invalid_argument("minres: invalid convergence options");
59 }
60
61 V residual = math::zero_like(b);
62 V applied = math::zero_like(b);
63 math::apply(A, x, residual);
64 // r <- b - A*x
65 math::linear_combination(real(1), b, real(-1), residual);
66
67 const real beta0 = math::norm(residual);
68 solver_result result{0, beta0, beta0 < options.tolerance};
69 if (result.converged || options.max_iterations == 0)
70 return result;
71
72 const idx mmax = std::min<idx>(options.max_iterations, static_cast<idx>(n));
73 array<V> basis;
74 basis.reserve(mmax + 1);
75 basis.push_back(residual);
76 math::scale(real(1) / beta0, basis[0]);
77
78 array<real> alpha;
79 array<real> beta;
80 alpha.reserve(mmax);
81 beta.reserve(mmax);
82 V previous = math::zero_like(b);
83
84 for (idx j = 0; j < mmax; ++j) {
85 result.iterations = j + 1;
86 math::apply(A, basis[j], applied);
87 if (j > 0)
88 math::axpy(-beta[j - 1], previous, applied);
89
90 const real diagonal = math::inner(basis[j], applied);
91 if (!std::isfinite(diagonal)) {
92 throw std::runtime_error("minres: self-adjoint Lanczos invariant was violated");
93 }
94 alpha.push_back(diagonal);
95 math::axpy(-diagonal, basis[j], applied);
96
97 const real next_beta = math::norm(applied);
98 if (!std::isfinite(next_beta)) {
99 throw std::runtime_error("minres: inner-product norm invariant was violated");
100 }
101 beta.push_back(next_beta);
102
103 const vec y = math_krylov_detail::minres_projected_solve(alpha, beta, beta0, j + 1);
104 V candidate = x;
105 for (idx column = 0; column <= j; ++column)
106 math::axpy(y[column], basis[column], candidate);
107
108 math::apply(A, candidate, residual);
109 // r <- b - A*x_candidate
110 math::linear_combination(real(1), b, real(-1), residual);
111 result.residual = math::norm(residual);
112 if (result.residual < options.tolerance) {
113 x = std::move(candidate);
114 result.converged = true;
115 break;
116 }
117 if (next_beta <= real(1e-15)) {
118 x = std::move(candidate);
119 break;
120 }
121
122 previous = basis[j];
123 math::scale(real(1) / next_beta, applied);
124 basis.push_back(applied);
125 if (j + 1 == mmax)
126 x = std::move(candidate);
127 }
128 return result;
129}
