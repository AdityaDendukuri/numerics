155 /// Element count of a rows-by-cols matrix, rejecting a product that would wrap.
156 ///
157 /// The three SuiteSparse bindings already guard their int32 interfaces this
158 /// way; without the check here a wrapped product allocates a short buffer
159 /// that `operator()` then indexes past.
160 static idx checked_size(idx rows, idx cols) {
161 if (rows != 0 && cols > std::numeric_limits<idx>::max() / rows) {
162 throw std::overflow_error("basic_mat: rows * cols exceeds the index range");
163 }
