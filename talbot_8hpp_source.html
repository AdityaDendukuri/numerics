31 [[nodiscard]] array<contour_node> nodes(real t) const {
32 if (!(t > 0.0) || modes < 2) {
33 throw std::invalid_argument("talbot_quadrature: invalid time or mode count");
34 }
35 array<contour_node> result;
36 result.reserve(modes);
37 const real pi = std::acos(-1.0);
38 for (idx k = 0; k < modes; ++k) {
39 const real theta = -pi + ((static_cast<real>(k) + 0.5) * (2.0 * pi / modes));
40 const real a = sigma * theta;
41 const real cot = std::cos(a) / std::sin(a);
42 const real csc2 = 1.0 / (std::sin(a) * std::sin(a));
43 const real re = (mu * theta * cot) - nu;
44 const real dre = mu * (cot - (a * csc2));
45 const real im = eta * theta;
46 const real dim = eta;
47 const real scale = static_cast<real>(modes);
48 const cplx z(scale * re, scale * im);
49 const cplx dz(scale * dre, scale * dim);
50 result.push_back(
51 {z / t, std::exp(z) * dz / (cplx(0.0, 1.0) * static_cast<real>(modes) * t)});
52 }
53 return result;
54 }
55
56 template <typename Accumulate>
57 void accumulate(real t, Accumulate &&accumulate_fn) const {
58 for (const contour_node &node : nodes(t)) {
59 accumulate_fn(node.shift, node.weight);
60 }
61 }
62};
