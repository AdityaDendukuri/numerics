23template <typename T>
24void laplacian_stencil_2d(const basic_vec<T> &x, basic_vec<T> &y, int N) {
25 for (int i = 0; i < N; ++i) {
26 for (int j = 0; j < N; ++j) {
27 int k = (i * N) + j;
28 T val = T(-4) * x[k];
29 if (i > 0) {
30 val += x[k - N];
31 }
32 if (i < N - 1) {
33 val += x[k + N];
34 }
35 if (j > 0) {
36 val += x[k - 1];
37 }
38 if (j < N - 1) {
39 val += x[k + 1];
40 }
41 y[k] = val;
42 }
43 }
44}

46/// @brief Periodic second-order 2D Laplacian stencil.
47template <typename T>
48void laplacian_stencil_2d_periodic(const basic_vec<T> &x, basic_vec<T> &y, int N) {
49 for (int i = 0; i < N; ++i) {
50 int ip = (i + 1) % N, im = (i + N - 1) % N;
51 const T *row = x.data() + (i * N);
52 const T *row_p = x.data() + (ip * N);
53 const T *row_m = x.data() + (im * N);
54 T *d = y.data() + (i * N);
55
56 d[0] = row_p[0] + row_m[0] + row[1] + row[N - 1] - (T(4) * row[0]);
57 for (int j = 1; j < N - 1; ++j) {
58 d[j] = row_p[j] + row_m[j] + row[j + 1] + row[j - 1] - (T(4) * row[j]);
59 }
60 d[N - 1] = row_p[N - 1] + row_m[N - 1] + row[0] + row[N - 2] - (T(4) * row[N - 1]);
61 }
62}

64/// @brief Fourth-order 2D Laplacian cross stencil.
65///
66/// \f[
67/// y_{ij} = \frac{1}{12}\bigl(
68/// -x_{i-2,j} + 16x_{i-1,j} - 30x_{i,j} + 16x_{i+1,j} - x_{i+2,j}
69/// -x_{i,j-2} + 16x_{i,j-1} + 16x_{i,j+1} - x_{i,j+2}
70/// \bigr)
71/// \f]
72template <typename T>
73void laplacian_stencil_2d_4th(const basic_vec<T> &x, basic_vec<T> &y, int N) {
74 for (int i = 0; i < N; ++i) {
75 for (int j = 0; j < N; ++j) {
76 int k = (i * N) + j;
77 T val = T(-30) * x[k];
78 // i-axis: +/-1
79 if (i > 0) {
80 val += T(16) * x[((i - 1) * N) + j];
81 }
82 if (i < N - 1) {
83 val += T(16) * x[((i + 1) * N) + j];
84 }
85 // i-axis: +/-2
86 if (i > 1) {
87 val -= x[((i - 2) * N) + j];
88 }
89 if (i < N - 2) {
90 val -= x[((i + 2) * N) + j];
91 }
92 // j-axis: +/-1
93 if (j > 0) {
94 val += T(16) * x[k - 1];
95 }
96 if (j < N - 1) {
97 val += T(16) * x[k + 1];
98 }
99 // j-axis: +/-2
100 if (j > 1) {
101 val -= x[k - 2];
102 }
103 if (j < N - 2) {
104 val -= x[k + 2];
105 }
106 y[k] = val / T(12);
107 }
108 }
109}

111/// Bilinear interpolation on a periodic NxN grid with configurable stagger
112/// offset.
113///
114/// field[i,j] is defined at physical position ((i + ox/h)*h, (j + oy/h)*h).
115/// Returns the interpolated field value at physical point (px, py).
116///
117/// @param field Flattened grid samples.
118/// @param N Grid width and height.
119/// @param h Grid spacing.
120/// @param px Physical x coordinate.
121/// @param py Physical y coordinate.
122/// @param ox x-axis origin offset in physical units (0 for unstaggered, h/2
123/// for v-face)
124/// @param oy y-axis origin offset in physical units (0 for unstaggered, h/2
125/// for u-face)
126///
127/// MAC grid usage:
128/// \f[
129/// \text{interp\_u}(px,py) = \texttt{sample\_2d\_periodic}(u, N, h,\; px,
130/// py,\; 0,\; h/2)
131/// \f]
132/// \f[
133/// \text{interp\_v}(px,py) = \texttt{sample\_2d\_periodic}(v, N, h,\; px,
134/// py,\; h/2,\; 0)
135/// \f]
136inline real sample_2d_periodic(const vec &field, idx N, real h, real px, real py, real ox,
137 real oy) {
138 real fx = std::fmod((px - ox) / h, static_cast<real>(N));
139 real fy = std::fmod((py - oy) / h, static_cast<real>(N));
140 if (fx < 0.0) {
141 fx += N;
142 }
143 if (fy < 0.0) {
144 fy += N;
145 }
146 idx i0 = static_cast<idx>(fx) % N;
147 idx i1 = (i0 + 1) % N;
148 real fi = fx - std::floor(fx);
149 idx j0 = static_cast<idx>(fy) % N;
150 idx j1 = (j0 + 1) % N;
151 real fj = fy - std::floor(fy);
152 return ((1 - fi) * (1 - fj) * field[(i0 * N) + j0]) + (fi * (1 - fj) * field[(i1 * N) + j0]) +
153 ((1 - fi) * fj * field[(i0 * N) + j1]) + (fi * fj * field[(i1 * N) + j1]);
154}

156/// @brief Apply a mutable 1D operation to each column fiber.
157template <typename T, typename F>
158void col_fiber_sweep(basic_vec<T> &data, int N, F &&f) {
159 array<T> fiber(N);
160 for (int j = 0; j < N; ++j) {
161 for (int i = 0; i < N; ++i) {
162 fiber[i] = data[(i * N) + j];
163 }
164 f(fiber);
165 for (int i = 0; i < N; ++i) {
166 data[(i * N) + j] = fiber[i];
167 }
168 }
169}

215/// @brief Compute \f$-\Delta_h x\f$ on a 3D grid.
216inline void neg_laplacian_3d(const vec &x, vec &y, int nx, int ny, int nz, double inv_dx2) {
217 auto flat = [&](int i, int j, int k) -> idx {
218 return static_cast<idx>((k * ny * nx) + (j * nx) + i);
219 };
220 auto interior_value = [&](int i, int j, int k) {
221 if (i == 0 || i == nx - 1 || j == 0 || j == ny - 1 || k == 0 || k == nz - 1) {
222 return 0.0;
223 }
224 return x[flat(i, j, k)];
225 };
226 for (int k = 0; k < nz; ++k) {
227 for (int j = 0; j < ny; ++j) {
228 for (int i = 0; i < nx; ++i) {
229 idx id = flat(i, j, k);
230 if (i == 0 || i == nx - 1 || j == 0 || j == ny - 1 || k == 0 || k == nz - 1) {
231 y[id] = x[id];
232 } else {
233 // Eliminate zero-Dirichlet boundary unknowns to keep the full operator
234 // symmetric.
235 y[id] = inv_dx2 * ((6.0 * x[id]) - interior_value(i + 1, j, k) -
236 interior_value(i - 1, j, k) - interior_value(i, j + 1, k) -
237 interior_value(i, j - 1, k) - interior_value(i, j, k + 1) -
238 interior_value(i, j, k - 1));
239 }
240 }
241 }
242 }

245/// @brief Compute \f$\nabla\phi\f$ with central differences.
246inline void gradient_3d(const scalar_field_3d &phi, scalar_field_3d &gx, scalar_field_3d &gy,
247 scalar_field_3d &gz) {
248 int nx = phi.nx(), ny = phi.ny(), nz = phi.nz();
249 double inv2dx = 1.0 / (2.0 * phi.dx());
250 for (int k = 0; k < nz; ++k) {
251 for (int j = 0; j < ny; ++j) {
252 for (int i = 0; i < nx; ++i) {
253 int ip = std::min(i + 1, nx - 1), im = std::max(i - 1, 0);
254 int jp = std::min(j + 1, ny - 1), jm = std::max(j - 1, 0);
255 int kp = std::min(k + 1, nz - 1), km = std::max(k - 1, 0);
256 gx(i, j, k) = (phi(ip, j, k) - phi(im, j, k)) * inv2dx;
257 gy(i, j, k) = (phi(i, jp, k) - phi(i, jm, k)) * inv2dx;
258 gz(i, j, k) = (phi(i, j, kp) - phi(i, j, km)) * inv2dx;
259 }
260 }
261 }

264/// @brief Compute \f$\nabla\cdot f\f$ with central differences.
265inline void divergence_3d(const scalar_field_3d &fx, const scalar_field_3d &fy, const scalar_field_3d &fz,
266 scalar_field_3d &out) {
267 int nx = fx.nx(), ny = fx.ny(), nz = fx.nz();
268 double inv2dx = 1.0 / (2.0 * fx.dx());
269 for (int k = 0; k < nz; ++k) {
270 for (int j = 0; j < ny; ++j) {
271 for (int i = 0; i < nx; ++i) {
272 int ip = std::min(i + 1, nx - 1), im = std::max(i - 1, 0);
273 int jp = std::min(j + 1, ny - 1), jm = std::max(j - 1, 0);
274 int kp = std::min(k + 1, nz - 1), km = std::max(k - 1, 0);
275 out(i, j, k) = ((fx(ip, j, k) - fx(im, j, k)) + (fy(i, jp, k) - fy(i, jm, k)) +
276 (fz(i, j, kp) - fz(i, j, km))) *
277 inv2dx;
278 }
279 }
280 }

283/// @brief Compute \f$\nabla\times A\f$ with central differences.
284inline void curl_3d(const scalar_field_3d &ax, const scalar_field_3d &ay, const scalar_field_3d &az,
285 scalar_field_3d &bx, scalar_field_3d &by, scalar_field_3d &bz) {
286 int nx = ax.nx(), ny = ax.ny(), nz = ax.nz();
287 double inv2dx = 1.0 / (2.0 * ax.dx());
288 for (int k = 0; k < nz; ++k) {
289 for (int j = 0; j < ny; ++j) {
290 for (int i = 0; i < nx; ++i) {
291 int ip = std::min(i + 1, nx - 1), im = std::max(i - 1, 0);
292 int jp = std::min(j + 1, ny - 1), jm = std::max(j - 1, 0);
293 int kp = std::min(k + 1, nz - 1), km = std::max(k - 1, 0);
294 bx(i, j, k) = (az(i, jp, k) - az(i, jm, k) - ay(i, j, kp) + ay(i, j, km)) * inv2dx;
295 by(i, j, k) = (ax(i, j, kp) - ax(i, j, km) - az(ip, j, k) + az(im, j, k)) * inv2dx;
296 bz(i, j, k) = (ay(ip, j, k) - ay(im, j, k) - ax(i, jp, k) + ax(i, jm, k)) * inv2dx;
297 }
