20 fft_plan_impl(int n_, bool inv) : n(n_), invert(inv) {
21 if (n_ == 0 || (n_ & (n_ - 1)))
22 throw std::invalid_argument("fft_plan: length must be a power of two");
23 for (int len = 2; len <= n_; len <<= 1) {
24 double ang = TWO_PI / static_cast<double>(len) * (inv ? 1.0 : -1.0);
25 num::cplx wlen{std::cos(ang), std::sin(ang)};
26 std::vector<num::cplx> tw(len / 2);
27 num::cplx w{1.0, 0.0};
28 for (int j = 0; j < len / 2; ++j) {
29 tw[j] = w;
30 w *= wlen;
31 }
32 twiddles.push_back(std::move(tw));
33 }
34 }

36 void execute(const num::cvec &in, num::cvec &out) const override {
37 using vd = stdx::simd<double, stdx::simd_abi::native<double>>;
38 constexpr int W = static_cast<int>(vd::size());
39
40 for (num::idx i = 0; i < static_cast<num::idx>(n); ++i)
41 out[i] = in[i];
42 backends::seq::bit_reverse(out);
43 num::cplx *data = out.data();
44
45 int stage = 0;
46 for (int len = 2; len <= n; len <<= 1, ++stage) {
47 int hlen = len / 2;
48 const num::cplx *tw = twiddles[stage].data();
49
50 for (int i = 0; i < n; i += len) {
51 num::cplx *up = data + i;
52 num::cplx *vp = data + i + hlen;
53
54 int j = 0;
55 for (; j + W <= hlen; j += W) {
56 vd ur([&](int k) -> double { return up[j + k].real(); });
57 vd ui([&](int k) -> double { return up[j + k].imag(); });
58 vd vr([&](int k) -> double { return vp[j + k].real(); });
59 vd vi([&](int k) -> double { return vp[j + k].imag(); });
60 vd wr([&](int k) -> double { return tw[j + k].real(); });
61 vd wi([&](int k) -> double { return tw[j + k].imag(); });
62
63 vd tr = vr * wr - vi * wi;
64 vd ti = vr * wi + vi * wr;
65
66 for (int k = 0; k < W; ++k) {
67 up[j + k] = {ur[k] + tr[k], ui[k] + ti[k]};
68 vp[j + k] = {ur[k] - tr[k], ui[k] - ti[k]};
69 }
70 }
71 for (; j < hlen; ++j) {
72 num::cplx t = vp[j] * tw[j];
73 num::cplx uu = up[j];
74 up[j] = uu + t;
75 vp[j] = uu - t;
76 }
77 }
78 }
79 }
