61/// @brief Compute unnormalized forward 1D Fast Fourier Transform \f$X_k = \sum_{j=0}^{n-1} x_j e^{-2\pi i j k / n}\f$.
62///
63/// Dispatches to FFTW3 when linked or in-tree SIMD / sequential Cooley-Tukey Radix-2 / Bluestein kernel.
64///
65/// @param in Input complex vector of length \f$n\f$.
66/// @param out Output complex spectrum vector (must be sized to \f$n\f$).
67/// @param b FFT backend tag (`fft_backend::fftw`, `fft_backend::simd`, `fft_backend::seq`).
68/// @see ifft, rfft, fft_plan
69void fft(const cvec &in, cvec &out, fft_backend b = default_fft_backend);

71/// @brief Compute unnormalized inverse 1D Fast Fourier Transform \f$x_j = \sum_{k=0}^{n-1} X_k e^{+2\pi i j k / n}\f$.
72///
73/// Note that dividing by \f$n\f$ is required to recover the original amplitude.
74///
75/// @param in Input complex spectrum of length \f$n\f$.
76/// @param out Output reconstructed complex vector (size \f$n\f$).
77/// @param b FFT backend tag.
78/// @see fft, irfft
79void ifft(const cvec &in, cvec &out, fft_backend b = default_fft_backend);

81/// @brief Compute nonredundant half-spectrum of a real input signal: \f$n \to \lfloor n/2 \rfloor + 1\f$ complex coefficients.
82///
83/// Exploits Hermitian conjugate symmetry \f$X_{n-k} = X_k^*\f$ for real inputs.
84///
85/// @param in Real input vector of size \f$n\f$.
86/// @param out Output complex half-spectrum vector of size \f$n/2 + 1\f$.
87/// @param b FFT backend tag.
88/// @see irfft, fft
89void rfft(const vec &in, cvec &out, fft_backend b = default_fft_backend);
