1/// @file omp/matrix_ops.hpp
2/// @brief OpenMP-accelerated Level-2/3 dense matrix operations.
3///
4/// Same rule as `omp/vector_ops.hpp`: OpenMP only decides how the matrix is
5/// sliced into blocks across threads; the arithmetic inside each block is a
6/// `num::kernel` call, not a hand-written loop.
7///
8/// Row-tiled rather than routed through `dispatch::parallel_apply`: that
9/// helper's block size and threshold are tuned for vector *element* counts
10/// (blocks of ~16K elements, threading only above ~262K elements), which are
11/// the wrong units here — a matmul row does O(n*k) work, not O(1), so a matrix
12/// with a few hundred rows would never cross an element-counted threshold and
13/// would silently never thread. Always parallelizing over row-tiles (as the
14/// previous hand-written version did) is the correct granularity for Level-2/3.

25inline void matmul(const mat &A, const mat &B, mat &C) {
26 constexpr idx block_size = 64;
27 const idx m = A.rows(), k = A.cols(), n = B.cols();
28 const real *ad = A.data();
29 const real *bd = B.data();
30 real *cd = C.data();
31#if defined(NUMERICS_HAS_OMP)
32#pragma omp parallel for schedule(dynamic)
33#endif
34 for (idx ii = 0; ii < m; ii += block_size) {
35 const idx rows = std::min(block_size, m - ii);
36 // The row tile is the only decision made here: `kernel::gemm` does its
37 // own register and cache blocking inside each tile.
38 kernel::gemm(cd + (ii * n), ad + (ii * k), bd, real(1), real(0), rows, n, k);
39 }
40}

42inline void matvec(const mat &A, const vec &x, vec &y) {
43 const idx n = A.cols();
44 const real *ad = A.data();
45 const real *xd = x.data();
46 real *yd = y.data();
47#if defined(NUMERICS_HAS_OMP)
48#pragma omp parallel for schedule(static)
49#endif
50 for (idx i = 0; i < A.rows(); ++i) {
51 yd[i] = kernel::dot(ad + (i * n), xd, n);
52 }
53}

55inline void matadd(real alpha, const mat &A, real beta, const mat &B, mat &C) {
56 constexpr idx block_size = idx{1} << 16;
57 const idx total = A.size();
58 const real *ad = A.data();
59 const real *bd = B.data();
60 real *cd = C.data();
61#if defined(NUMERICS_HAS_OMP)
62#pragma omp parallel for schedule(static)
63#endif
64 for (idx offset = 0; offset < total; offset += block_size) {
65 const idx length = std::min(block_size, total - offset);
66 kernel::axpbyz(cd + offset, ad + offset, bd + offset, alpha, beta, length);
67 }
68}
