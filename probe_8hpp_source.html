14/// @brief An \f$n \times p\f$ matrix of independent Rademacher (\f$\pm 1\f$) probes.
15///
16/// Column \f$k\f$ is a probe vector \f$z_k\f$ with \f$E[z_k z_k^T] = I\f$, the
17/// input to Hutchinson-type estimators of \f$\operatorname{tr}(A)\f$ and
18/// \f$\operatorname{diag}(A)\f$.
19template <typename RNG = rng>
20[[nodiscard]] mat rademacher_probe(idx n, idx probes, RNG &generator) {
21 if (probes == 0) {
22 throw std::invalid_argument("rademacher_probe: at least one probe is required");
23 }
24 std::bernoulli_distribution sign(0.5);
25 mat probe(n, probes, 0.0);
26 for (idx j = 0; j < n; ++j) {
27 for (idx p = 0; p < probes; ++p) {
28 probe(j, p) = sign(generator) ? 1.0 : -1.0;
29 }
30 }
31 return probe;
32}
33
34/// @brief The same probe from a fixed seed.
35[[nodiscard]] inline mat rademacher_probe(idx n, idx probes, unsigned seed) {
36 rng generator(seed);
37 return rademacher_probe(n, probes, generator);
38}

40/// @brief Row-wise mean square of probed columns: the Hutchinson estimate.
41///
42/// If column \f$k\f$ of `probed` is \f$B z_k\f$ for Rademacher probes
43/// \f$z_k\f$, entry \f$j\f$ of the result estimates \f$(B B^T)_{jj}\f$.
44[[nodiscard]] inline vec hutchinson_row_mean_square(const mat &probed) {
45 if (probed.cols() == 0) {
46 throw std::invalid_argument("hutchinson_row_mean_square: no probed columns");
47 }
48 vec estimate(probed.rows(), 0.0);
49 for (idx j = 0; j < probed.rows(); ++j) {
50 real total = 0.0;
51 for (idx p = 0; p < probed.cols(); ++p) {
52 total += probed(j, p) * probed(j, p);
53 }
54 estimate[j] = total / static_cast<real>(probed.cols());
55 }
56 return estimate;
57}
