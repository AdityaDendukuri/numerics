51/// A projection carries its operand's law over to the subspace rather than discarding it.
52///
53/// \f$P A\f$ does not satisfy a global law even when \f$A\f$ does: \f$(PA)^* = AP \ne
54/// PA\f$. It satisfies the law *on the subspace*, because \f$PAx = PAPx\f$ for every
55/// \f$x \in S\f$, and \f$PAP\f$ is self-adjoint, semidefinite or definite exactly when
56/// \f$A\f$ is. `restricted_to` performs that translation.
57///
58/// This is what lets `num::pcg` run on a graph Laplacian restricted to the zero-sum
59/// subspace without the caller re-asserting definiteness that was already established.
60template <class Op, class Subspace>
61struct claims_of<operators::projected_op<Op, Subspace>> {
62 private:
63 template <class... Ls>
64 static auto derive(type_list<Ls...>)
65 -> type_list<law::restricted_to_t<Ls, Subspace>...>;
66
67 public:
68 using type = decltype(derive(typename detail::declared_laws<Op>::type{}));
69};
