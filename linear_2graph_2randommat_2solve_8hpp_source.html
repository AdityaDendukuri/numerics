13/// Solve (P A P^T) (P x) = P b using forward and backward substitution with L L^T.
14template <typename Float = double, std::integral Index = num::idx>
15inline void solve(const cholesky_factor<Float, Index> &factor, const Float *b, Float *x,
16 std::vector<Float> &scratch) {
17 const Index n = static_cast<Index>(factor.order.size());
18 if (scratch.size() < n) {
19 scratch.resize(n);
20 }
21
22 // 1. Permute RHS: w = P b
23 for (Index k = 0; k < n; ++k) {
24 scratch[k] = b[factor.order[k]];
25 }
26
27 // 2. Forward solve: L y = w (column-oriented)
28 for (Index j = 0; j < n; ++j) {
29 const auto &entries = factor.columns[j].entries;
30 if (entries.empty()) {
31 continue;
32 }
33 const Float diag = entries[0].value;
34 scratch[j] /= diag;
35 const Float wj = scratch[j];
36 for (std::size_t e = 1; e < entries.size(); ++e) {
37 scratch[entries[e].row] -= entries[e].value * wj;
38 }
39 }
40
41 // 3. Backward solve: L^T v = y (column-oriented)
42 for (Index step = 0; step < n; ++step) {
43 const Index j = n - 1 - step;
44 const auto &entries = factor.columns[j].entries;
45 if (entries.empty()) {
46 continue;
47 }
48 Float sum = scratch[j];
49 for (std::size_t e = 1; e < entries.size(); ++e) {
50 sum -= entries[e].value * scratch[entries[e].row];
51 }
52 scratch[j] = sum / entries[0].value;
53 }
54
55 // 4. Inverse permute: x = P^T v
56 for (Index k = 0; k < n; ++k) {
57 x[factor.order[k]] = scratch[k];
58 }
59}
