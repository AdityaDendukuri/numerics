19/// @brief Verify Parseval's identity for a forward transform.
20///
21/// \f[ \sum_n |x_n|^2 = \frac{1}{N} \sum_k |X_k|^2 \f]
22///
23/// A transform that violates it is not unitary after normalization, and a
24/// spectral method built on it will gain or lose energy every step.
25template <class Plan>
26requires transform_plan<Plan> inline void
27verify_parseval(const Plan &plan, real tol = 1e-10,
28 std::source_location loc = std::source_location::current()) {
29 if constexpr (!num::debug::sampling_compiled_in) {
30 return;
31 }
32 if (num::debug::get_level() != num::debug::diagnostic_level::full) {
33 return;
34 }
35 const int n = plan.size();
36 if (n <= 0) {
37 return;
38 }
39
40 cvec in(static_cast<idx>(n));
41 cvec out(static_cast<idx>(n));
42 std::uint64_t state = 0x9E3779B97F4A7C15ULL;
43 const auto next = [&state]() {
44 state ^= state << 13;
45 state ^= state >> 7;
46 state ^= state << 17;
47 return ((static_cast<double>(state >> 11) / 9007199254740992.0) * 2.0) - 1.0;
48 };
49 for (idx i = 0; i < in.size(); ++i) {
50 in[i] = cplx(next(), next());
51 }
52
53 plan.execute(in, out);
54
55 real energy_time = 0.0;
56 real energy_freq = 0.0;
57 for (idx i = 0; i < in.size(); ++i) {
58 energy_time += std::norm(in[i]);
59 energy_freq += std::norm(out[i]);
60 }
61 energy_freq /= static_cast<real>(n);
62
63 const real scale = std::max(energy_time, energy_freq) + std::numeric_limits<real>::min();
64 if (std::abs(energy_time - energy_freq) / scale > tol) {
65 panic("TransformError",
66 "Parseval's identity does not hold: time-domain energy " +
67 std::to_string(energy_time) + " against frequency-domain energy " +
68 std::to_string(energy_freq) +
69 ". The transform is not unitary after normalization.",
70 loc);
71 }
72}
