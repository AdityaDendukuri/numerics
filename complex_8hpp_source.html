33/// @brief Mixed real-matrix, complex-vector product \f$x = Q y\f$.
34///
35/// Arises when a real orthogonal basis is applied to a complex Krylov coordinate
36/// vector, as in projecting a resolvent solution back from the Krylov subspace.
37/// Kept separate from `matvec` because the scalar types differ on the two sides.
38template <std::floating_point T>
39NUM_K_AINLINE void matvec_real_complex(std::complex<T> *NUM_K_RESTRICT x, const T *Q,
40 const std::complex<T> *y, idx m, idx n) noexcept {
41 for (idx i = 0; i < m; ++i) {
42 const T *row = Q + (i * n);
43 std::complex<T> sum{};
44 for (idx j = 0; j < n; ++j) {
45 sum += row[j] * y[j];
46 }
47 x[i] = sum;
48 }
49}
50
51/// @brief Mixed transpose product \f$x = Q^T y\f$ with a real matrix and complex result.
52///
53/// The transpose companion to `matvec_real_complex`. Projecting a right-hand side
54/// onto a real orthonormal basis produces complex coordinates when the right-hand
55/// side is complex, and real ones widened to complex when it is not, so the input
56/// scalar is a separate parameter.
57///
58/// @param x Output, length n.
59/// @param Q Real matrix, m*n row-major.
60/// @param y Input, length m, real or complex.
61/// @param m Number of rows in Q.
62/// @param n Number of columns in Q.
63template <std::floating_point T, class In>
64NUM_K_AINLINE void matvec_transpose_into_complex(std::complex<T> *NUM_K_RESTRICT x, const T *Q,
65 const In *y, idx m, idx n) noexcept {
66 for (idx i = 0; i < n; ++i) {
67 x[i] = std::complex<T>{};
68 }
69 for (idx j = 0; j < m; ++j) {
70 const T *row = Q + (j * n);
71 const std::complex<T> yj = y[j];
72 for (idx i = 0; i < n; ++i) {
73 x[i] += row[i] * yj;
74 }
75 }
76}

80/// @brief Factor \f$sI - H\f$ in place for an upper Hessenberg \f$H\f$.
81///
82/// Gaussian elimination needs to clear only one subdiagonal entry per column, so
83/// this costs \f$O(n^2)\f$ rather than the \f$O(n^3)\f$ of a general LU. That is
84/// what makes a Krylov resolvent affordable: the Hessenberg form is computed once
85/// and each shift factors cheaply on top of it.
86///
87/// Partial pivoting compares the diagonal against the single subdiagonal entry,
88/// since no other entry in the column can be larger.
89///
90/// @param work In/out, n*n. Receives \f$sI - H\f$ and its factors.
91/// @param H Upper Hessenberg matrix, n*n row-major, real.
92/// @param shift Complex shift \f$s\f$.
93/// @param n Dimension.
94/// @param piv Output pivot record, length n.
95template <std::floating_point T, class Index>
96inline void hessenberg_shifted_factor(std::complex<T> *NUM_K_RESTRICT work, const T *NUM_K_RESTRICT H,
97 std::complex<T> shift, idx n,
98 Index *NUM_K_RESTRICT piv) noexcept {
99 using C = std::complex<T>;
100 const T tiny = T(1e-30);
101
102 for (idx i = 0; i < n; ++i) {
103 const T *h_row = H + (i * n);
104 C *m_row = work + (i * n);
105 for (idx j = 0; j < n; ++j) {
106 m_row[j] = (i == j ? shift : C(0, 0)) - h_row[j];
107 }
108 }
109
110 for (idx i = 0; i + 1 < n; ++i) {
111 C *row_i = work + (i * n);
112 C *row_next = work + ((i + 1) * n);
113
114 if (std::abs(row_next[i]) > std::abs(row_i[i])) {
115 for (idx j = i; j < n; ++j) {
116 std::swap(row_i[j], row_next[j]);
117 }
118 piv[i] = static_cast<Index>(i + 1);
119 } else {
120 piv[i] = static_cast<Index>(i);
121 }
122
123 const C pivot = row_i[i];
124 if (std::abs(pivot) > tiny) {
125 const C mult = row_next[i] / pivot;
126 row_next[i] = mult;
127 for (idx j = i + 1; j < n; ++j) {
128 row_next[j] -= mult * row_i[j];
129 }
130 }
131 }
132}

134/// @brief Substitute a right-hand side through a factored shifted Hessenberg system.
135///
136/// Separate from the factorization so that many right-hand sides share one
137/// factorization at the same shift. `y` and `b` may alias.
138template <std::floating_point T, class Index>
139inline void hessenberg_shifted_substitute(std::complex<T> *y, const std::complex<T> *NUM_K_RESTRICT work,
140 const Index *NUM_K_RESTRICT piv, const std::complex<T> *b,
141 idx n) noexcept {
142 using C = std::complex<T>;
143 const T tiny = T(1e-30);
144
145 for (idx i = 0; i < n; ++i) {
146 y[i] = b[i];
147 }
148 for (idx i = 0; i + 1 < n; ++i) {
149 if (static_cast<idx>(piv[i]) != i) {
150 std::swap(y[i], y[i + 1]);
151 }
152 y[i + 1] -= work[((i + 1) * n) + i] * y[i];
153 }
154 for (idx step = 0; step < n; ++step) {
155 const idx i = n - 1 - step;
156 const C *row_i = work + (i * n);
157 C sum = y[i];
158 for (idx j = i + 1; j < n; ++j) {
159 sum -= row_i[j] * y[j];
160 }
161 const C diag = row_i[i];
162 y[i] = std::abs(diag) < tiny ? C(0, 0) : sum / diag;
163 }
164}

166/// @brief Solve \f$(sI - H)\,y = b\f$ for a single right-hand side.
167template <std::floating_point T, class Index>
168inline void hessenberg_shifted_solve(std::complex<T> *y, const T *H, std::complex<T> shift,
169 const std::complex<T> *b, idx n,
170 std::complex<T> *NUM_K_RESTRICT work,
171 Index *NUM_K_RESTRICT piv) noexcept {
172 hessenberg_shifted_factor(work, H, shift, n, piv);
173 hessenberg_shifted_substitute(y, work, piv, b, n);
174}
