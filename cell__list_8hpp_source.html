27/// Counting-sorted 2D spatial bins for local-neighbor iteration.
28class cell_list_2d {
29 public:
30 /// Cover the bounding box with padded square cells.
31 cell_list_2d(scalar cell_size, scalar xmin, scalar xmax, scalar ymin, scalar ymax)
32 : cs_(cell_size), xmin_(xmin), ymin_(ymin) {
33 // Padding cells avoid boundary checks in the hot query loops.
34 nx_ = static_cast<int>(std::ceil((xmax - xmin) / cs_)) + 2;
35 ny_ = static_cast<int>(std::ceil((ymax - ymin) / cs_)) + 2;
36 const int total = nx_ * ny_;
37 start_.assign(total + 1, 0);
38 count_.assign(total, 0);
39 }

41 /// @brief Rebuild by counting-sort over cell ids.
42 template <position_accessor_2d<scalar> PosAccessor>
43 void build(PosAccessor &&get_pos, int n) {
44 sorted_.resize(n);
45 const int total = nx_ * ny_;
46
47 std::fill(count_.begin(), count_.end(), 0);
48 for (int i = 0; i < n; ++i) {
49 ++count_[cell_id_of(get_pos(i))];
50 }
51
52 start_[0] = 0;
53 for (int c = 0; c < total; ++c) {
54 start_[c + 1] = start_[c] + count_[c];
55 }
56
57 std::fill(count_.begin(), count_.end(), 0);
58 for (int i = 0; i < n; ++i) {
59 const int cid = cell_id_of(get_pos(i));
60 sorted_[start_[cid] + count_[cid]] = i;
61 ++count_[cid];
62 }
63 }

65 /// @brief Call f(j) for candidate particles near (px, py).
66 template <typename F>
67 void query(scalar px, scalar py, F &&f) const {
68 const int cx = cell_x(px);
69 const int cy = cell_y(py);
70 for (int dy = -1; dy <= 1; ++dy) {
71 const int qy = cy + dy;
72 if (qy < 0 || qy >= ny_) {
73 continue;
74 }
75 for (int dx = -1; dx <= 1; ++dx) {
76 const int qx = cx + dx;
77 if (qx < 0 || qx >= nx_) {
78 continue;
79 }
80 const int cid = (qy * nx_) + qx;
81 for (int k = start_[cid]; k < start_[cid + 1]; ++k) {
82 f(sorted_[k]);
83 }
84 }
85 }
86 }

88 /// @brief Visit each candidate pair once.
89 template <typename F>
90 void iterate_pairs(F &&f) const {
91 // Half-shell offsets cover all neighboring cell pairs once.
92 static constexpr int FDX[4] = {+1, 0, +1, -1};
93 static constexpr int FDY[4] = {0, +1, +1, +1};
94
95 for (int cy = 0; cy < ny_; ++cy) {
96 for (int cx = 0; cx < nx_; ++cx) {
97 const int cid = (cy * nx_) + cx;
98 const int beg = start_[cid];
99 const int end = start_[cid + 1];
100 if (beg == end) {
101 continue;
102 }
103
104 for (int a = beg; a < end; ++a) {
105 for (int b = a + 1; b < end; ++b) {
106 f(sorted_[a], sorted_[b]);
107 }
108 }
109
110 for (int d = 0; d < 4; ++d) {
111 const int ncx = cx + FDX[d];
112 const int ncy = cy + FDY[d];
113 if (ncx < 0 || ncx >= nx_ || ncy < 0 || ncy >= ny_) {
114 continue;
115 }
116 const int ncid = (ncy * nx_) + ncx;
117 const int nbeg = start_[ncid];
118 const int nend = start_[ncid + 1];
119 if (nbeg == nend) {
120 continue;
121 }
122 for (int a = beg; a < end; ++a) {
123 for (int b = nbeg; b < nend; ++b) {
124 f(sorted_[a], sorted_[b]);
125 }
126 }
127 }
128 }
129 }
130 }

152 return cx < 0 ? 0 : (cx >= nx_ ? nx_ - 1 : cx);
153 }
