11/// @brief Single Metropolis sweep: n_sites random single-site updates.
12///
13/// The acceptance probability is min(1, exp(-beta * dE)) computed from
14/// the value returned by delta_energy(i).
15///
16/// @tparam DeltaE Callable: idx -> real. Returns dE for proposing a flip at i.
17/// @tparam Apply Callable: idx -> void. Applies the flip at site i.
18/// @tparam RNG Random number generator (e.g., std::mt19937).
19template <typename DeltaE, typename Apply, typename RNG>
20metropolis_stats metropolis_sweep(idx n_sites, DeltaE delta_energy, Apply apply_flip, real beta,
21 RNG &rng) {
22 std::uniform_real_distribution<real> u01(0.0, 1.0);
23 std::uniform_int_distribution<idx> site_dist(0, n_sites - 1);
24
25 metropolis_stats stats;
26 stats.total = n_sites;
27
28 for (idx k = 0; k < n_sites; ++k) {
29 idx i = site_dist(rng);
30 real dE = delta_energy(i);
31 real p = (dE <= 0.0) ? 1.0 : std::exp(-beta * dE);
32 if (p >= 1.0 || u01(rng) < p) {
33 apply_flip(i);
34 ++stats.accepted;
35 }
36 }
37 return stats;
38}

40/// @brief Metropolis sweep with caller-supplied acceptance probabilities.
41///
42/// Use this variant when acceptance probabilities are precomputed (e.g. a
43/// Boltzmann table for a discrete-dE system like the Ising model), avoiding
44/// a runtime exp() call per proposed flip.
45///
46/// @tparam ProbFn Callable: idx -> real. Returns min(1, exp(-beta*dE)) for
47/// site i.
48/// @tparam Apply Callable: idx -> void. Applies the flip at site i.
49/// @tparam RNG Random number generator.
50template <typename ProbFn, typename Apply, typename RNG>
51metropolis_stats metropolis_sweep_prob(idx n_sites, ProbFn acceptance_prob, Apply apply_flip,
52 RNG &rng) {
53 std::uniform_real_distribution<real> u01(0.0, 1.0);
54 std::uniform_int_distribution<idx> site_dist(0, n_sites - 1);
55
56 metropolis_stats stats;
57 stats.total = n_sites;
58
59 for (idx k = 0; k < n_sites; ++k) {
60 idx i = site_dist(rng);
61 real p = acceptance_prob(i);
62 if (p >= 1.0 || u01(rng) < p) {
63 apply_flip(i);
64 ++stats.accepted;
65 }
66 }
67 return stats;
68}

70/// @brief Umbrella-sampled Metropolis sweep (dE variant).
71///
72/// Performs a sweep, measures the order parameter, then restores the saved
73/// state if the order parameter falls outside [window.lo, window.hi].
74///
75/// @tparam DeltaE Callable: idx -> real
76/// @tparam Apply Callable: idx -> void
77/// @tparam Save Callable: () -> void. Saves system state before the sweep.
78/// @tparam Restore Callable: () -> void. Restores saved state.
79/// @tparam Measure Callable: () -> idx. Returns the order parameter.
80/// @tparam RNG Random number generator.
81template <typename DeltaE, typename Apply, typename Save, typename Restore, typename Measure,
82 typename RNG>
83umbrella_stats umbrella_sweep(idx n_sites, DeltaE delta_energy, Apply apply_flip, Save save_state,
84 Restore restore_state, Measure measure_order, umbrella_window window,
85 real beta, RNG &rng) {
86 save_state();
87 metropolis_stats mc = metropolis_sweep(n_sites, delta_energy, apply_flip, beta, rng);
88 idx op = measure_order();
89
90 umbrella_stats stats;
91 stats.mc = mc;
92 stats.order_param = op;
93
94 if (!window.contains(op)) {
95 restore_state();
96 stats.reverted = true;
97 stats.order_param = measure_order();
98 }
99 return stats;
100}
