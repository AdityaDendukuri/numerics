25/// @brief Constructs Givens rotation parameters \f$(c, s)\f$ such that:
26/// \f[
27/// \begin{bmatrix} c & s \\ -s & c \end{bmatrix} \begin{bmatrix} a \\ b \end{bmatrix} =
28/// \begin{bmatrix} r \\ 0 \end{bmatrix}, \qquad c^2 + s^2 = 1
29/// \f]
30template <std::floating_point T>
31NUM_K_AINLINE void rotg(T a, T b, T &c, T &s) noexcept {
32 if (b == T(0)) {
33 c = T(1);
34 s = T(0);
35 } else if (a == T(0)) {
36 c = T(0);
37 s = T(1);
38 } else {
39 T r = std::hypot(a, b);
40 c = a / r;
41 s = b / r;
42 }
43}
44
45/// @brief Applies Givens plane rotation in-place:
46/// \f[
47/// \begin{bmatrix} x_i \\ y_i \end{bmatrix} \leftarrow \begin{bmatrix} c & s \\ -s & c
48/// \end{bmatrix} \begin{bmatrix} x_i \\ y_i \end{bmatrix}
49/// \f]
50template <std::floating_point T>
51NUM_K_AINLINE void rot(T *NUM_K_RESTRICT x, T *NUM_K_RESTRICT y, T c, T s, idx n) noexcept {
52 NUM_K_IVDEP
53 for (idx i = 0; i < n; ++i) {
54 T xi = x[i];
55 T yi = y[i];
56 x[i] = (c * xi) + (s * yi);
57 y[i] = (-s * xi) + (c * yi);
58 }
59}

63/// @brief Computes elementary Householder reflector vector \f$\mathbf{v}\f$ and scalar \f$\beta\f$
64/// such that:
65/// \f[
66/// (I - \beta \mathbf{v} \mathbf{v}^T) \mathbf{x} = \mp \|\mathbf{x}\|_2 \mathbf{e}_1
67/// \f]
68/// \f$\mathbf{v}\f$ is sized \f$m\f$, with \f$v_0 = 1\f$ implicitly assigned.
69template <std::floating_point T>
70NUM_K_AINLINE void householder_vector(T *NUM_K_RESTRICT v, T &beta, const T *NUM_K_RESTRICT x,
71 idx m) noexcept {
72 T sq = T(0);
73 NUM_K_IVDEP
74 for (idx i = 0; i < m; ++i) {
75 sq += x[i] * x[i];
76 }
77 const T norm_x = std::sqrt(sq);
78 if (norm_x < T(1e-15)) {
79 beta = T(0);
80 v[0] = T(1);
81 return;
82 }
83 const T sign = (x[0] >= T(0)) ? T(1) : T(-1);
84 const T mu = x[0] + (sign * norm_x);
85 v[0] = T(1);
86 T v_sq = T(1);
87 NUM_K_IVDEP
88 for (idx i = 1; i < m; ++i) {
89 v[i] = x[i] / mu;
90 v_sq += v[i] * v[i];
91 }
92 beta = T(2) / v_sq;
93}

95/// @brief Householder reflector for a strided column of a matrix.
96template <std::floating_point T>
97NUM_K_AINLINE void householder_vector_strided(T *NUM_K_RESTRICT v, T &beta,
98 const T *NUM_K_RESTRICT A, idx lda, idx offset,
99 idx m) noexcept {
100 for (idx i = 0; i < m; ++i)
101 v[i] = A[((offset + i) * lda) + offset];
102 householder_vector(v, beta, v, m);
103}

109/// @brief Compact Householder QR factorization; reflector tails remain below R's diagonal.
110template <std::floating_point T>
111inline void qr_factor_blocked(T *NUM_K_RESTRICT A, idx lda, idx m, idx n, T *NUM_K_RESTRICT tau,
112 T *NUM_K_RESTRICT v, T *NUM_K_RESTRICT work,
113 idx block_size = 32) noexcept {
114 (void)block_size;
115 const idx r = std::min(m, n);
116 for (idx k = 0; k < r; ++k) {
117 const idx len = m - k;
118 T beta = T(0);
119 householder_vector_strided(v, beta, A, lda, k, len);
120 tau[k] = beta;
121 if (beta == T(0))
122 continue;
123 // R(k:m,k:n) <- H_k R(k:m,k:n).
124 householder_left(A + (k * lda) + k, lda, v, beta, len, n - k, work);
125 for (idx i = 1; i < len; ++i)
126 A[((k + i) * lda) + k] = v[i];
127 }
128}

130/// @brief Applies left Householder transformation \f$A \leftarrow (I - \beta \mathbf{v}
131/// \mathbf{v}^T) A\f$ on an \f$m \times n\f$ block with stride `lda`. `work` is a caller-provided
132/// scratch buffer of length at least \f$n\f$.
133template <std::floating_point T>
134NUM_K_AINLINE void householder_left(T *NUM_K_RESTRICT A, idx lda, const T *NUM_K_RESTRICT v, T beta,
135 idx m, idx n, T *NUM_K_RESTRICT work) noexcept {
136 NUM_K_IVDEP
137 for (idx j = 0; j < n; ++j) {
138 work[j] = T(0);
139 }
140 for (idx i = 0; i < m; ++i) {
141 const T vi = v[i];
142 const T *row = A + (i * lda);
143 NUM_K_IVDEP
144 for (idx j = 0; j < n; ++j) {
145 work[j] += vi * row[j];
146 }
147 }
148 NUM_K_IVDEP
149 for (idx j = 0; j < n; ++j) {
150 work[j] *= beta;
151 }
152 for (idx i = 0; i < m; ++i) {
153 const T vi = v[i];
154 T *row = A + (i * lda);
155 NUM_K_IVDEP
156 for (idx j = 0; j < n; ++j) {
157 row[j] -= vi * work[j];
158 }
159 }
160}

162/// @brief Applies right Householder transformation \f$A \leftarrow A (I - \beta \mathbf{v}
163/// \mathbf{v}^T)\f$ on an \f$m \times n\f$ block with stride `lda`.
164template <std::floating_point T>
165NUM_K_AINLINE void householder_right(T *NUM_K_RESTRICT A, idx lda, const T *NUM_K_RESTRICT v,
166 T beta, idx m, idx n) noexcept {
167 for (idx i = 0; i < m; ++i) {
168 T *row = A + (i * lda);
169 T dot_val = T(0);
170 NUM_K_IVDEP
171 for (idx j = 0; j < n; ++j) {
172 dot_val += row[j] * v[j];
173 }
174 const T factor = beta * dot_val;
175 NUM_K_IVDEP
176 for (idx j = 0; j < n; ++j) {
177 row[j] -= factor * v[j];
178 }
179 }
180}

184/// @brief Computes Jacobi rotation parameters \f$(c, s)\f$ annihilating off-diagonal entry
185/// \f$A_{pq}\f$ in a symmetric \f$2 \times 2\f$ block.
186template <std::floating_point T>
187NUM_K_AINLINE void jacobi_rotation(T app, T aqq, T apq, T &c, T &s) noexcept {
188 if (std::abs(apq) < T(1e-15)) {
189 c = T(1);
190 s = T(0);
191 return;
192 }
193 const T tau = (aqq - app) / (T(2) * apq);
194 const T t = std::copysign(T(1), tau) / (std::abs(tau) + std::sqrt(T(1) + (tau * tau)));
195 c = T(1) / std::sqrt(T(1) + (t * t));
196 s = c * t;
197}
