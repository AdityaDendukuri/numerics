22/// Parameters for fixed-step implicit integration.
23struct implicit_params {
24 int nstep; ///< number of time steps
25 double dt; ///< step size (reported to observer as t)
26};

28/// Advance u by nstep implicit steps using solver.
29/// obs(step, t, u) is called at step 0 (initial) and after each solve.
30template <vec_field field, typename Observer>
31void advance(field &u, const linear_solver &solver, implicit_params p, Observer &&obs) {
32 obs(0, 0.0, u);
33 for (int s = 0; s < p.nstep; ++s) {
34 vec rhs = u.as_vec();
35 solver(rhs, u.as_vec());
36 obs(s + 1, (s + 1) * p.dt, u);
37 }
38}

43 for (int s = 0; s < p.nstep; ++s) {
44 vec rhs = u.as_vec();
