122[[nodiscard]] inline cplx dot(const cvec &x, const cvec &y) noexcept {
123 cplx sum{};
124 const idx n = x.size();
125 for (idx i = 0; i < n; ++i) {
126 sum += std::conj(x[i]) * y[i];
127 }
128 return sum;
129}

131[[nodiscard]] inline real norm(const cvec &x) noexcept {
132 real sum = 0.0;
133 const idx n = x.size();
134 for (idx i = 0; i < n; ++i) {
135 sum += std::norm(x[i]);
136 }
137 return std::sqrt(sum);
