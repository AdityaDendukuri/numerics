29/// @brief `std::vector` of a floating-point type is a Hilbert space, and says so.
30///
31/// The hierarchy asks a type to claim its laws rather than inferring them from syntax, which
32/// prevents `std::string`, which defines `operator+`, from being mistaken for a vector
33/// space. A standard container cannot declare a member typedef, so the claim is attached
34/// from outside here. `std::vector<int>` is not covered. The integers form a ring rather
35/// than a field, so it has no level on this hierarchy.
36template <std::floating_point T>
37struct claims_of<array<T>> {
38 using type = type_list<law::hilbert_space>;
39};
40
41template <std::floating_point T>
42struct claims_of<array<std::complex<T>>> {
43 using type = type_list<law::hilbert_space>;
44};

50/// @brief scalar field \f$\mathbb{K}\f$ over which spaces and containers are defined.
51///
52/// Satisfies the field axioms: closed under `+ - * /` (by non-zero elements), with
53/// additive identity 0, multiplicative identity 1, and a well-defined modulus \f$|x|\f$.
54///
55/// ### Supported
56/// * `double` and `float`, the real field \f$\mathbb{R}\f$
57/// * `std::complex<double>` and `std::complex<float>`, the complex field \f$\mathbb{C}\f$
58///
59/// ### Example
60/// @code
61/// static_assert(num::scalar<double>);
62/// static_assert(num::scalar<std::complex<double>>);
63/// static_assert(!num::scalar<int>); // integers form a ring, not a field
64/// @endcode
65///
66/// @tparam T The candidate scalar type.
67///
68/// A synonym for `num::field`, kept because "scalar" is what the surrounding mathematics
69/// calls it. Both name the one certified definition in `core/math/concepts.hpp`.
70template <typename T>
71concept scalar = field<T>;

73/// @brief Linear map presented as an entrywise-indexable 2D array.
74///
75/// Distinct from `linear_operator`, which can only be applied. A matrix-free operator is a
76/// `linear_operator` and not a `matrix_space`. A dense matrix is both. A routine that reads
77/// individual entries, such as pivoting or banded factorization, requires `matrix_space`.
78///
79/// @tparam A mat or 2D array type.
80/// @tparam T Entry scalar type.
81template <class A, class T = entry_t<A>>
82concept matrix_space = field<T> && requires(const A &a) {
83 { a.rows() } -> std::convertible_to<idx>;
84 { a.cols() } -> std::convertible_to<idx>;
85 { a(idx{0}, idx{0}) } -> std::convertible_to<T>;
86};
87
88/// @brief Entrywise-presented linear map whose entries can be mutated.
89///
90/// @tparam A mat type.
91/// @tparam T Entry scalar type.
92template <class A, class T = entry_t<A>>
93concept mutable_matrix_space = matrix_space<A, T> && requires(A &a, T x) {
94 { a(idx{0}, idx{0}) = x };
95};

97/// @brief scalar function \f$f: \mathbb{K} \to \mathbb{K}\f$ over a field.
98///
99/// Accepts any callable mapping a scalar to a scalar, including lambdas, function
100/// pointers and `std::function`.
101///
102/// @tparam F Callable object type.
103/// @tparam T scalar field type (defaults to `real`).
104template <class F, class T = real>
105concept scalar_function = field<T> && std::invocable<F, T> && requires(F f, T x) {
106 { f(x) } -> std::convertible_to<T>;
107};

109/// @brief scalar function supplied with its exact derivative \f$(f, f')\f$.
110///
111/// Used by Newton–Raphson and Halley's method.
112///
113/// @tparam F Callable function type.
114/// @tparam D Callable derivative type.
115/// @tparam T scalar field type.
116template <class F, class D, class T = real>
117concept differentiable_function = scalar_function<F, T> && scalar_function<D, T>;
