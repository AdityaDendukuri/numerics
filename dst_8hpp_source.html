19/// @brief Reject a length the odd-extension route cannot transform.
20///
21/// The extension has length 2(N+1), and the radix-2 FFT needs that to be a
22/// power of two, so N has to be one less than a power of two.
23inline void dst_check_size(int N) {
24 if (N <= 0 || (N & (N + 1)) != 0) {
25 throw std::invalid_argument(
26 "spectral::dst: N must equal 2^p - 1 (e.g. 7, 15, 31, 63, ...)");
27 }
28}

30/// Unnormalised DST-I of an N-point vector via complex FFT.
31///
32/// X[k] = sum_{j=1}^{N} x[j] * sin(j*k*pi/(N+1)), k = 1..N (stored 0-indexed).
33/// Odd-extension y = [0, x, 0, -rev(x)] has length M = 2(N+1).
34/// FFT(y)[k] = -2i * sum sin(...) => DST(x)[k-1] = -Im(FFT(y)[k]) / 2.
35inline vec dst1(const vec &x) {
36 const int N = static_cast<int>(x.size());
37 dst_check_size(N);
38 const int M = 2 * (N + 1);
39 cvec y(static_cast<std::size_t>(M), cplx{0.0, 0.0});
40 for (int j = 0; j < N; ++j) {
41 const auto sj = static_cast<std::size_t>(j);
42 y[sj + 1] = cplx{x[sj], 0.0};
43 y[static_cast<std::size_t>(M - 1 - j)] = cplx{-x[sj], 0.0};
44 }
45 cvec Y(static_cast<std::size_t>(M));
46 spectral::fft(y, Y);
47 vec out(static_cast<std::size_t>(N));
48 for (int k = 0; k < N; ++k) {
49 out[static_cast<std::size_t>(k)] = -Y[static_cast<std::size_t>(k) + 1].imag() / 2.0;
50 }
51 return out;
52}

54inline void dst_rows(array<double> &A, int N) {
55 vec row(static_cast<std::size_t>(N));
56 for (int i = 0; i < N; ++i) {
57 const std::size_t base = static_cast<std::size_t>(i) * static_cast<std::size_t>(N);
58 for (int j = 0; j < N; ++j) {
59 row[static_cast<std::size_t>(j)] = A[base + j];
60 }
61 row = dst1(row);
62 for (int j = 0; j < N; ++j) {
63 A[base + j] = row[static_cast<std::size_t>(j)];
64 }
65 }
66}
