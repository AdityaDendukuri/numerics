113/// Convert a basic_graph to a basic_multigraph.
114template <typename Weight, std::integral Index>

116to_multigraph(const basic_graph<Weight, Index> &G) {
117 const Index n = G.n_vertices();
118 basic_multigraph<Weight, Index> mg(n);
119 for (Index u = 0; u < n; ++u) {
