18/// @brief mat-free 2D discrete 5-point Laplacian operator with spmat materialization.
19class laplacian_2d final {
20 public:
21 using domain_type = vec;
22 using codomain_type = vec;
23 using math_laws = math::type_list<law::self_adjoint>;
24
25 explicit laplacian_2d(int N) : N_(N) {
26 if (N_ <= 0) {
27 throw std::invalid_argument("laplacian_2d: grid dimension N must be positive");
28 }
29 }
30
31 void apply(const vec &x, vec &y) const {
32 const idx n = rows();
33 if (x.size() != n) {
34 throw std::invalid_argument("laplacian_2d: input dimension mismatch");
35 }
36 if (y.size() != n) {
37 y = vec(n);
38 }
39 for (int i = 0; i < N_; ++i) {
40 for (int j = 0; j < N_; ++j) {
41 const int k = (i * N_) + j;
42 real val = -4.0 * x[k];
43 if (i > 0) {
44 val += x[k - N_];
45 }
46 if (i < N_ - 1) {
47 val += x[k + N_];
48 }
49 if (j > 0) {
50 val += x[k - 1];
51 }
52 if (j < N_ - 1) {
53 val += x[k + 1];
54 }
55 y[k] = val;
56 }
57 }
58 }
59
60 [[nodiscard]] idx rows() const noexcept { return static_cast<idx>(N_) * N_; }
61 [[nodiscard]] idx cols() const noexcept { return static_cast<idx>(N_) * N_; }
62 [[nodiscard]] int N() const noexcept { return N_; }
63
64 /// @brief Materialize the 5-point discrete Laplacian as an assembled CSR spmat.
65 [[nodiscard]] spmat to_sparse() const {
66 const idx n = rows();
67 array<idx> rows_idx, cols_idx;
68 array<real> vals;
69 rows_idx.reserve(5 * n);
70 cols_idx.reserve(5 * n);
71 vals.reserve(5 * n);
72 for (int i = 0; i < N_; ++i) {
73 for (int j = 0; j < N_; ++j) {
74 const int k = (i * N_) + j;
75 rows_idx.push_back(k);
76 cols_idx.push_back(k);
77 vals.push_back(-4.0);
78 if (i > 0) {
79 rows_idx.push_back(k);
80 cols_idx.push_back(((i - 1) * N_) + j);
81 vals.push_back(1.0);
82 }
83 if (i < N_ - 1) {
84 rows_idx.push_back(k);
85 cols_idx.push_back(((i + 1) * N_) + j);
86 vals.push_back(1.0);
87 }
88 if (j > 0) {
89 rows_idx.push_back(k);
90 cols_idx.push_back((i * N_) + (j - 1));
91 vals.push_back(1.0);
92 }
93 if (j < N_ - 1) {
94 rows_idx.push_back(k);
95 cols_idx.push_back((i * N_) + (j + 1));
96 vals.push_back(1.0);
97 }
98 }
99 }
100 return spmat::from_triplets(n, n, rows_idx, cols_idx, vals);
101 }

103 private:
104 int N_;
105};
