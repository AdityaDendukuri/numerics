38/// @brief Compute full singular value decomposition \f$A = U \Sigma V^T\f$.
39///
40/// Dispatches to LAPACK divide-and-conquer (`dgesdd`) when available, or executes
41/// in-tree one-sided Hestenes-Jacobi orthogonalization sweeps with Givens rotations.
42///
43/// @param A Input \f$m \times n\f$ dense matrix.
44/// @param tol Orthogonality tolerance for Jacobi sweeps (default: 1e-12).
45/// @param max_sweeps Maximum one-sided Jacobi sweeps (default: 100).
46/// @return `svd_result` with left singular vectors \f$U\f$, singular values \f$\Sigma\f$, and transposed right vectors \f$V^T\f$.
47///
48/// Picks LAPACK (`dgesdd`) if configured, else the in-tree one-sided
49/// Hestenes-Jacobi sweeps. To force one explicitly, call
50/// `num::lapack::svd`/`num::seq::svd` directly.
51/// @see svd_truncated, eig_sym, qr
52svd_result svd(const mat &A, real tol = 1e-12, idx max_sweeps = 100);

54/// @brief Compute randomized truncated rank-\f$k\f$ SVD approximation \f$A \approx U_k \Sigma_k V_k^T\f$.
55///
56/// Uses Gaussian random test matrices and QR range-finder to project \f$A\f$ into a small
57/// subspace of dimension \f$l = k + \text{oversampling}\f$, achieving near-optimal low-rank reconstruction.
58///
59/// @param A Input \f$m \times n\f$ dense matrix.
60/// @param k Target low-rank approximation dimension (\f$0 < k \le \min(m, n)\f$).
61/// @param oversampling Additional random test vectors for spectral gap safety (default: 10).
62/// @param rng Optional pointer to custom random number generator for reproducible sampling.
63/// @return `svd_result` containing rank-\f$k\f$ truncated factors \f$U_k, \Sigma_k, V_k^T\f$.
64/// @throws std::invalid_argument If \f$k\f$ is out of range.
65/// @see svd, lanczos
66svd_result svd_truncated(const mat &A, idx k, idx oversampling = 10, rng_state *rng = nullptr);

69inline svd_result svd(const mat &A_in, real tol, idx max_sweeps) {
70 constexpr real tiny = 1e-300;
71 idx m = A_in.rows(), n = A_in.cols();
72 idx r = std::min(m, n);
73
74 mat A = A_in;
75 mat V(n, n, 0.0);
76 for (idx i = 0; i < n; ++i) {
77 V(i, i) = 1.0;
78 }
79
80 idx sweeps = 0;
81 bool converged = false;
82
83 for (idx sweep = 0; sweep < max_sweeps; ++sweep) {
84 real max_cos = 0;
85 for (idx p = 0; p < r - 1; ++p) {
86 for (idx q = p + 1; q < r; ++q) {
87 const real alpha = kernel::column_dot(A.data(), n, m, p, p);
88 const real beta = kernel::column_dot(A.data(), n, m, q, q);
89 const real gamma = kernel::column_dot(A.data(), n, m, p, q);
90 if (alpha < tiny || beta < tiny) {
91 continue;
92 }
93
94 real cos_pq = std::abs(gamma) / std::sqrt(alpha * beta);
95 max_cos = std::max(max_cos, cos_pq);
96
97 if (cos_pq < tol) {
98 continue;
99 }
100
101 real zeta = (beta - alpha) / (2.0 * gamma);
102 real t =
103 std::copysign(1.0, zeta) / (std::abs(zeta) + std::sqrt(1.0 + (zeta * zeta)));
104 real c = 1.0 / std::sqrt(1.0 + (t * t));
105 real s = c * t;
106
107 // [A_p A_q] <- [A_p A_q] J(c,s).
108 kernel::rotate_columns(A.data(), n, m, p, q, c, s);
109 kernel::rotate_columns(V.data(), n, n, p, q, c, s);
110 }
111 }
112
113 ++sweeps;
114 if (max_cos < tol) {
115 converged = true;
116 break;
117 }
118 }
119
120 vec S(r);
121 mat U(m, r, 0.0);
122 for (idx j = 0; j < r; ++j) {
123 const real nrm = kernel::norm_sq_strided(A.data() + j, n, m);
124 S[j] = std::sqrt(nrm);
125 if (S[j] > tiny) {
126 // U[:,j] <- A[:,j] / sigma_j.
127 kernel::scale_copy_strided(U.data() + j, r, A.data() + j, n, real(1) / S[j], m);
128 }
129 }
130
131 for (idx i = 0; i < r - 1; ++i) {
132 idx max_j = i;
133 for (idx j = i + 1; j < r; ++j) {
134 if (S[j] > S[max_j]) {
135 max_j = j;
136 }
137 }
138
139 if (max_j != i) {
140 std::swap(S[i], S[max_j]);
141 // [U_i U_j] <- [U_j U_i], [V_i V_j] <- [V_j V_i].
142 kernel::swap_strided(U.data() + i, r, U.data() + max_j, r, m);
143 kernel::swap_strided(V.data() + i, n, V.data() + max_j, n, n);
144 }
145 }
146
147 mat vt(r, n, 0.0);
148 // V^T <- transpose(V[:,0:r]).
149 for (idx i = 0; i < r; ++i) {
150 kernel::copy_strided(vt.data() + (i * n), 1, V.data() + i, n, n);
151 }
152
153 return {U, S, vt, sweeps, converged};
154}

158inline svd_result svd(const mat &A_in) {
159#if defined(NUMERICS_HAS_LAPACK)
160 const idx m = A_in.rows(), n = A_in.cols();
161 const idx r = std::min(m, n);
162 mat Aw = A_in;
163 vec S(r);
164 mat U(m, r);
165 mat Vt(r, n);
166
167 int info =
168 LAPACKE_dgesdd(LAPACK_ROW_MAJOR, 'S', static_cast<lapack_int>(m),
169 static_cast<lapack_int>(n), Aw.data(), static_cast<lapack_int>(n), S.data(),
170 U.data(), static_cast<lapack_int>(r), Vt.data(), static_cast<lapack_int>(n));
171 if (info != 0) {
172 throw std::runtime_error("svd (lapack): dgesdd failed, info=" + std::to_string(info));
173 }
174
175 return {std::move(U), std::move(S), std::move(Vt), 0, true};
176#else
177 return seq::svd(A_in, 1e-12, 100);
178#endif
179}
180} // namespace lapack

190inline svd_result svd_truncated(const mat &A, idx k, idx oversampling, rng_state *rng) {
191 const idx m = A.rows(), n = A.cols();
192 if (k == 0 || k > std::min(m, n)) {
193 throw std::invalid_argument("svd_truncated: k out of range");
194 }
195
196 const idx l = k + oversampling;
197
198 rng_state local_rng;
199 if (!rng) {
200 rng = &local_rng;
201 }
202
203 mat Omega(n, l);
204 for (idx j = 0; j < l; ++j) {
205 for (idx i = 0; i < n; ++i) {
206 Omega(i, j) = rng_normal(rng, 0.0, 1.0);
207 }
208 }
209
210 mat Y(m, l, 0.0);
211 matmul(A, Omega, Y);
212
213 qr_result qr_res = qr(Y);
214 const mat &Q = qr_res.Q;
215
216 mat B(l, n, 0.0);
217 // B <- Q_l^T*A
218 kernel::gemm_transpose_left(B.data(), B.cols(), Q.data(), Q.cols(), A.data(), A.cols(),
219 real(1), real(0), m, l, n);
220
221 svd_result small = svd(B);
222
223 mat U(m, k, 0.0);
224 // U_k <- Q_l*U(B)[:,0:k]
225 kernel::gemm(U.data(), U.cols(), Q.data(), Q.cols(), small.U.data(), small.U.cols(),
226 real(1), real(0), m, k, l);
227
228 vec S(k);
229 // sigma_k <- sigma(B)[0:k]
230 kernel::copy(S.data(), small.S.data(), k);
231
232 mat Vt(k, n, 0.0);
233 // V_k^T <- V(B)^T[0:k,:]
234 kernel::copy(Vt.data(), small.Vt.data(), k * n);
235
236 return {U, S, Vt, 0, true};
237}
