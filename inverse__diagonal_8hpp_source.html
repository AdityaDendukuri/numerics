20/// @brief Perform x += y, track accumulated floating-point error, and verify precision acceptability.
21template <class T>
22inline bool safe_add(T &x, T &err, T y, T tolerance = 1e-6) {
23 x += y;
24 if (x <= 0) {
25 return false;
26 }
27 err += std::numeric_limits<T>::epsilon() * std::abs(x);
28 return (err / x) < tolerance;
29}
