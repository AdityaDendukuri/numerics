18 public:
19 /// Configure the physical cutoff and rebuild skin.
20 verlet_list_2d(scalar cutoff, scalar skin)

23 /// @brief Build the neighbour list using a pre-built cell_list_2d.
24 template <position_accessor_2d<scalar> PosAccessor>
25 void build(PosAccessor &&get_pos, int n, const cell_list_2d<scalar> &cl) {
26 starts_.resize(n + 1);
27 flat_.clear();
28 ref_x_.resize(n);
29 ref_y_.resize(n);
30
31 starts_[0] = 0;
32 for (int i = 0; i < n; ++i) {
33 auto [xi, yi] = get_pos(i);
34 ref_x_[i] = xi;
35 ref_y_[i] = yi;
36
37 cl.query(xi, yi, [&](int j) {
38 if (j == i) {
39 return;
40 }
41 auto [xj, yj] = get_pos(j);
42 const scalar dx = xi - xj, dy = yi - yj;
43 if ((dx * dx) + (dy * dy) < ext_sq_) {
44 flat_.push_back(j);
45 }
46 });
47
48 starts_[i + 1] = static_cast<int>(flat_.size());
49 }
50 }
51
52 /// @brief Return true if a particle moved more than half the skin.
53 template <position_accessor_2d<scalar> PosAccessor>
54 bool needs_rebuild(PosAccessor &&get_pos, int n) const {
55 if (ref_x_.empty()) {
56 return true;
57 }
58 const scalar half_skin_sq = (skin_ * scalar(0.5)) * (skin_ * scalar(0.5));
59 for (int i = 0; i < n; ++i) {
60 auto [xi, yi] = get_pos(i);
61 const scalar dx = xi - ref_x_[i];
62 const scalar dy = yi - ref_y_[i];
63 if ((dx * dx) + (dy * dy) > half_skin_sq) {
64 return true;
65 }
66 }
67 return false;
68 }
69
70 /// @brief Cached neighbors of particle i.
71 [[nodiscard]] integer_range neighbors(int i) const noexcept {
72 return {flat_.data() + starts_[i], flat_.data() + starts_[i + 1]};
73 }

80 scalar ext_cutoff() const noexcept { return cutoff_ + skin_; }

83 }
