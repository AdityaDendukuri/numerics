33/// @brief Conjugate gradients on a stored matrix, without requiring the SPD invariant.
34///
35/// CG minimizes \f$\tfrac12 x^T A x - b^T x\f$, which is bounded below only when
36/// \f$A\f$ is positive definite. On an indefinite matrix the search direction can
37/// have \f$p^T A p \leq 0\f$ and the iteration breaks down; on a non-symmetric one
38/// the Krylov recurrence is invalid from the first step. Neither is reported as an
39/// error, which is why the invariant is normally required.
40///
41/// Runs on the host via `num::accel` (the build's best available backend). For
42/// the GPU path operating on device buffers directly, see `num::unsafe::cuda::cg`.
43/// @return `solver_result`: `.iterations`, `.residual` (final residual norm), `.converged`.
44inline solver_result cg(const mat &A, const vec &b, vec &x, real tol = 1e-10,
45 idx max_iter = 1000) {
46 const idx n = b.size();
47 if (A.rows() != n || A.cols() != n || x.size() != n) {
48 throw std::invalid_argument("Dimension mismatch in cg_method solver");
49 }
50
51 // The shared iteration, with the matrix supplied as a matvec. `num::accel`
52 // selects how that product is formed; the level-1 work is memory bound and
53 // inlines from the kernel.
54 vec work(3 * n);
55 vec in(n);
56 vec out(n);
57 const auto apply = [&A, &in, &out, n](const real *src, real *dst) {
58 std::copy_n(src, n, in.data());
59 matvec(A, in, out);
60 std::copy_n(out.data(), n, dst);
61 };
62
63 const auto r = kernel::cg(apply, x.data(), b.data(), n, work.data(), tol, max_iter);
64 return solver_result{r.iterations, r.residual, r.converged};
65}

70/// @brief Conjugate gradients entirely on the device.
71///
72/// The device path cannot share the host kernel: its vectors live on the
73/// device, so every level-1 operation is a device call rather than a loop over
74/// host memory. It therefore keeps the iteration written out, mirroring
75/// `num::unsafe::cg`'s structure with `num::cuda::*` in place of `num::accel::*`.
76/// @return `solver_result`: `.iterations`, `.residual` (final residual norm), `.converged`.
77inline solver_result cg(const mat &A, const vec &b, vec &x, real tol = 1e-10,
78 idx max_iter = 1000) {
79 const idx n = b.size();
80 if (A.rows() != n || A.cols() != n || x.size() != n) {
81 throw std::invalid_argument("Dimension mismatch in cg_method solver");
82 }
83
84 const_cast<mat &>(A).to_gpu();
85 const_cast<vec &>(b).to_gpu();
86 x.to_gpu();
87
88 vec r(n), p(n), Ap(n);
89 r.to_gpu();
90 p.to_gpu();
91 Ap.to_gpu();
92
93 num::cuda::matvec(A.gpu_data(), x.gpu_data(), r.gpu_data(), A.rows(), A.cols());
94 num::cuda::scale(r.gpu_data(), n, -1.0);
95 num::cuda::axpy(1.0, b.gpu_data(), r.gpu_data(), n);
96 num::cuda::to_device(p.gpu_data(), r.gpu_data(), n);
97
98 real rsold = num::cuda::dot(r.gpu_data(), r.gpu_data(), n);
99 solver_result result{0, std::sqrt(rsold), false};
100
101 for (idx iter = 0; iter < max_iter; ++iter) {
102 result.iterations = iter + 1;
103 num::cuda::matvec(A.gpu_data(), p.gpu_data(), Ap.gpu_data(), A.rows(), A.cols());
104
105 real pAp = num::cuda::dot(p.gpu_data(), Ap.gpu_data(), n);
106 if (!(pAp > 0.0) || !std::isfinite(pAp)) {
107 break;
108 }
109 real alpha = rsold / pAp;
110
111 num::cuda::axpy(alpha, p.gpu_data(), x.gpu_data(), n);
112 num::cuda::axpy(-alpha, Ap.gpu_data(), r.gpu_data(), n);
113
114 real rsnew = num::cuda::dot(r.gpu_data(), r.gpu_data(), n);
115 result.residual = std::sqrt(rsnew);
116
117 if (result.residual < tol) {
118 result.converged = true;
119 break;
120 }
121
122 real beta = rsnew / rsold;
123 num::cuda::scale(p.gpu_data(), n, beta);
124 num::cuda::axpy(1.0, r.gpu_data(), p.gpu_data(), n);
125 rsold = rsnew;
126 }
127 x.to_cpu();
128 return result;
129}

136/// @brief Solve \f$A x = b\f$ using Conjugate Gradients for certified SPD operators.
137///
138/// Iteratively minimizes the quadratic form \f$\phi(x) = \frac{1}{2} x^T A x - b^T x\f$ over
139/// the Krylov subspace \f$\mathcal{K}_k(A, r_0)\f$. The operator `A` must carry
140/// compile-time or runtime positive-definite evidence (`law::spd`).
141///
142/// @tparam Op Linear operator type satisfying `math::endomorphism_on<Op, vec>` and carrying SPD evidence.
143/// @param A Symmetric positive-definite linear operator or matrix wrapper (e.g. `num::assume_spd(A)`).
144/// @param b Right-hand side vector.
145/// @param x Solution vector (serves as initial guess on input, updated in place).
146/// @param tolerance Convergence tolerance on Euclidean residual norm \f$\|b - A x\|_2\f$.
147/// @param max_iterations Maximum number of Krylov iterations before termination.
148/// @return `solver_result` containing iteration count, final residual norm, and convergence boolean.
149/// @throws std::invalid_argument If dimensions of `A`, `b`, and `x` do not match.
150/// @see assume_spd, pcg, minres, gmres
151template <class Op>
152requires math::inner_product_space<vec> && math::endomorphism_on<Op, vec> &&
153 claims<Op, law::spd>
154inline solver_result cg(const Op &A, const vec &b, vec &x, real tolerance,
155 idx max_iterations = 1000) {
156 return cg(A, b, x, cg_options{.tolerance = tolerance, .max_iterations = max_iterations});
157}
