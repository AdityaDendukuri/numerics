22/// @brief Storage-layout predicates describing contiguous memory layouts and sparse formats.
23///
24/// These representation concepts are structural constraints used by low-level hardware kernels,
25/// BLAS/LAPACK wrappers, and memory allocators.
26namespace repr {
27
28/// @brief vec stored in one contiguous block, addressable via a direct raw pointer `.data()`.
29///
30/// Models include `num::vec`, `num::cvec`, and `std::vector<double>`.
31///
32/// @tparam V Container type.
33template <class V>
34concept contiguous = vector_space<V> && requires(V v, const V cv) {
35 { cv.data() } -> std::convertible_to<const scalar_t<V> *>;
36 { v.data() } -> std::convertible_to<scalar_t<V> *>;
37};
38
39/// @brief mat stored contiguously in row-major order with row stride equal to `cols()`.
40///
41/// Allows zero-copy lowering directly to CBLAS row-major routines (`cblas_dgemm`, `cblas_dgemv`).
42/// Models include `num::mat`.
43///
44/// @tparam A mat type.
45template <class A>
46concept dense_row_major = matrix_space<A> && requires(A a, const A ca) {
47 { ca.data() } -> std::convertible_to<const entry_t<A> *>;
48 { a.data() } -> std::convertible_to<entry_t<A> *>;
49};
50
51/// @brief Compressed sparse row (CSR) storage format: row offsets, column indices, and nonzero values.
52///
53/// Enables matrix-vector multiplication in \f$\mathcal{O}(\text{nnz})\f$ time.
54/// Models include `num::spmat`.
55///
56/// @tparam M Sparse matrix type.
57template <class M>
58concept csr = requires(const M &m) {
59 { m.n_rows() } -> std::convertible_to<idx>;
60 { m.n_cols() } -> std::convertible_to<idx>;
61 { m.nnz() } -> std::convertible_to<idx>;
62 { m.row_ptr() } -> std::convertible_to<const idx *>;
63 { m.col_idx() } -> std::convertible_to<const idx *>;
64 { m.values() };
65};
66
67/// @brief banded storage format: entries confined to \f$-k_l \leq j - i \leq k_u\f$.
68///
69/// Models include `num::band_mat`.
70///
71/// @tparam B banded matrix type.
72template <class B>
73concept banded = matrix_space<B> && (requires(const B &b) {
74 { b.kl() } -> std::convertible_to<idx>;
75 { b.ku() } -> std::convertible_to<idx>;
76} || requires(const B &b) {
77 { b.lower_bandwidth() } -> std::convertible_to<idx>;
78 { b.upper_bandwidth() } -> std::convertible_to<idx>;
79});
80
81/// @brief Lower bandwidth \f$k_l\f$ of a banded matrix under either accessor spelling.
82template <class B>
83[[nodiscard]] constexpr idx lower_bandwidth_of(const B &b) {
84 if constexpr (requires { b.kl(); }) {
85 return b.kl();
86 } else {
87 return b.lower_bandwidth();
88 }
89}
90
91/// @brief Upper bandwidth \f$k_u\f$ of a banded matrix under either accessor spelling.
92template <class B>
93[[nodiscard]] constexpr idx upper_bandwidth_of(const B &b) {
94 if constexpr (requires { b.ku(); }) {
95 return b.ku();
96 } else {
97 return b.upper_bandwidth();
98 }
99}

101/// @brief tridiagonal storage format: subdiagonal (`dl`), main diagonal (`d`), and superdiagonal (`du`).
102///
103/// @tparam T tridiagonal matrix type.
104template <class T>
105concept tridiagonal = requires(const T &t) {
106 { t.dl };
107 { t.d };
108 { t.du };
109 { t.size() } -> std::convertible_to<idx>;
110};
