38template <typename Float, typename Func>
39inline Float adaptive_helper(Func &&f, Float a, Float b, Float fa, Float fm, Float fb, Float whole,
40 Float tol, idx depth) {
41 Float mid = Float{0.5} * (a + b);
42 Float fl = f(Float{0.5} * (a + mid));
43 Float fr = f(Float{0.5} * (mid + b));
44
45 Float left = (b - a) / Float{12.0} * (fa + (Float{4.0} * fl) + fm);
46 Float right = (b - a) / Float{12.0} * (fm + (Float{4.0} * fr) + fb);
47 Float delta = left + right - whole;
48
49 if (depth == 0 || std::abs(delta) <= Float{15.0} * tol) {
50 return left + right + (delta / Float{15.0});
51 }
52
53 return adaptive_helper(f, a, mid, fa, fl, fm, left, tol * Float{0.5}, depth - 1) +
54 adaptive_helper(f, mid, b, fm, fr, fb, right, tol * Float{0.5}, depth - 1);
55}

59/// @brief Composite Trapezoidal numerical integration over \f$[a, b]\f$ with \f$n\f$ panels.
60///
61/// Computes \f$\int_a^b f(x)\,\mathrm{d}x \approx h \left( \frac{f(a)+f(b)}{2} + \sum_{i=1}^{n-1} f(a+ih) \right)\f$ with \f$\mathcal{O}(h^2)\f$ error.
62///
63/// @tparam Float Floating point scalar type (`double`, `float`).
64/// @tparam Func Integrand callable `Float(Float x)`.
65/// @param f Integrand function.
66/// @param a Lower integration limit.
67/// @param b Upper integration limit.
68/// @param n Number of uniform subintervals / panels (default: 100).
69/// @tparam Parallel Thread the panel sum with OpenMP (default: sequential).
70/// @return Approximated integral value.
71/// @see simpson, gauss_legendre, romberg
72template <bool Parallel = false, typename Float = double, std::invocable<Float> Func = scalar_fn>
73inline Float trapz(Func &&f, Float a, Float b, idx n = 100) {
74 Float h = (b - a) / static_cast<Float>(n);
75 Float sum = Float{0};
76#if defined(NUMERICS_HAS_OMP)
77 if constexpr (Parallel) {
78#pragma omp parallel for reduction(+ : sum) schedule(static)
79 for (idx i = 1; i < n; ++i) {
80 sum += f(a + (static_cast<Float>(i) * h));
81 }
82 } else
83#endif
84 {
85 for (idx i = 1; i < n; ++i) {
86 sum += f(a + (static_cast<Float>(i) * h));
87 }
88 }
89 return h * ((Float{0.5} * (f(a) + f(b))) + sum);
90}

92/// @brief Composite Simpson's 1/3 numerical integration with \f$n\f$ panels (\f$n\f$ even).
93///
94/// Approximates \f$\int_a^b f(x)\,\mathrm{d}x\f$ using piecewise quadratic polynomials, achieving \f$\mathcal{O}(h^4)\f$ error.
95///
96/// @tparam Float Floating point scalar type.
97/// @tparam Func Integrand callable `Float(Float x)`.
98/// @param f Integrand function.
99/// @param a Lower limit.
100/// @param b Upper limit.
101/// @param n Number of panels (must be even, default: 100).
102/// @tparam Parallel Thread the panel sum with OpenMP (default: sequential).
103/// @return Approximated integral value.
104/// @throws std::invalid_argument If `n` is not even.
105/// @see trapz, adaptive_simpson, gauss_legendre
106template <bool Parallel = false, typename Float = double, std::invocable<Float> Func = scalar_fn>
107inline Float simpson(Func &&f, Float a, Float b, idx n = 100) {
108 if (n % 2 != 0) {
109 throw std::invalid_argument("simpson: n must be even");
110 }
111 Float h = (b - a) / static_cast<Float>(n);
112 Float sum = f(a) + f(b);
113#if defined(NUMERICS_HAS_OMP)
114 if constexpr (Parallel) {
115#pragma omp parallel for reduction(+ : sum) schedule(static)
116 for (idx i = 1; i < n; ++i) {
117 sum += f(a + (static_cast<Float>(i) * h)) * (i % 2 == 0 ? Float{2.0} : Float{4.0});
118 }
119 } else
120#endif
121 {
122 for (idx i = 1; i < n; ++i) {
123 sum += f(a + (static_cast<Float>(i) * h)) * (i % 2 == 0 ? Float{2.0} : Float{4.0});
124 }
125 }
126 return (h / Float{3.0}) * sum;
127}

129/// @brief Gauss–Legendre quadrature with \f$p \in [1, 5]\f$ optimal nodes.
130///
131/// Exact for all polynomials up to degree \f$2p - 1\f$ using precomputed roots and weights of Legendre polynomials.
132///
133/// @tparam Float Floating point scalar type.
134/// @tparam Func Integrand callable.
135/// @param f Integrand function.
136/// @param a Lower integration limit.
137/// @param b Upper integration limit.
138/// @param p Number of quadrature nodes (\f$1 \le p \le 5\f$, default: 5).
139/// @return Exact or high-precision approximation of the integral.
140/// @throws std::invalid_argument If `p` is not in range \f$[1, 5]\f$.
141template <typename Float = double, std::invocable<Float> Func = scalar_fn>
142inline Float gauss_legendre(Func &&f, Float a, Float b, idx p = 5) {
143 if (p < 1 || p > 5) {
144 throw std::invalid_argument("gauss_legendre: p must be 1..5");
145 }
146 Float mid = Float{0.5} * (a + b);
147 Float half = Float{0.5} * (b - a);
148 Float sum = Float{0};
149 for (idx i = 0; i < p; ++i) {
150 sum += static_cast<Float>(detail::GL_WEIGHTS[p - 1][i]) *
151 f(mid + (half * static_cast<Float>(detail::GL_NODES[p - 1][i])));
152 }
153 return half * sum;
154}

156/// @brief Adaptive Simpson quadrature with recursive interval bisection and error control.
157///
158/// Automatically concentrates function evaluations in regions of high curvature.
159///
160/// @tparam Float Floating point scalar type.
161/// @tparam Func Integrand callable.
162/// @param f Integrand function.
163/// @param a Lower limit.
164/// @param b Upper limit.
165/// @param tol Error tolerance for adaptive refinement (default: 1e-8).
166/// @param max_depth Maximum recursion depth before termination (default: 50).
167/// @return High-accuracy integral approximation satisfying error tolerance.
168template <typename Float = double, std::invocable<Float> Func = scalar_fn>
169inline Float adaptive_simpson(Func &&f, Float a, Float b, Float tol = Float{1e-8}, idx max_depth = 50) {
170 Float fa = f(a), fb = f(b), fm = f(Float{0.5} * (a + b));
171 Float est = (b - a) / Float{6.0} * (fa + (Float{4.0} * fm) + fb);
172 return detail::adaptive_helper(f, a, b, fa, fm, fb, est, tol, max_depth);
173}

175/// @brief Romberg numerical integration using Richardson extrapolation on trapezoidal sums.
176///
177/// Generates triangular tableau \f$R(j, k)\f$ with successive error cancellations of order \f$\mathcal{O}(h^{2(k+1)})\f$.
178///
179/// @tparam Float Floating point scalar type.
180/// @tparam Func Integrand callable.
181/// @param f Integrand function.
182/// @param a Lower limit.
183/// @param b Upper limit.
184/// @param tol Convergence tolerance on diagonal tableau entries (default: 1e-10).
185/// @param max_levels Maximum tableau extrapolation levels (default: 12).
186/// @return Extrapolated integral approximation.
187template <typename Float = double, std::invocable<Float> Func = scalar_fn>
188inline Float romberg(Func &&f, Float a, Float b, Float tol = Float{1e-10}, idx max_levels = 12) {
189 array<array<Float>> R(max_levels, array<Float>(max_levels, Float{0}));
190 R[0][0] = Float{0.5} * (b - a) * (f(a) + f(b));
191
192 for (idx i = 1; i < max_levels; ++i) {
193 idx n = idx(1) << i;
194 Float h = (b - a) / static_cast<Float>(n);
195 Float sum = Float{0};
196 for (idx k = 1; k < n; k += 2) {
197 sum += f(a + (static_cast<Float>(k) * h));
198 }
199 R[i][0] = (Float{0.5} * R[i - 1][0]) + (h * sum);
200
201 Float factor = Float{1.0};
202 for (idx j = 1; j <= i; ++j) {
203 factor *= Float{4.0};
204 R[i][j] = R[i][j - 1] + ((R[i][j - 1] - R[i - 1][j - 1]) / (factor - Float{1.0}));
205 }
206
207 if (i > 0 && std::abs(R[i][i] - R[i - 1][i - 1]) < tol) {
208 return R[i][i];
209 }
210 }
211 return R[max_levels - 1][max_levels - 1];
212}
