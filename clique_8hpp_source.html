18inline Weight collect_neighbors(Index v, array<array<multi_edge<Weight, Index>>> &G,
19 const array<std::uint8_t> &done, Queue &q,
20 array<multi_edge<Weight, Index>> &star_buf,
21 array<multi_edge<Weight, Index>> &nbr_out) {
22 star_buf.clear();
23 for (const auto &e : G[v]) {
24 if (done[e.to]) {
25 continue;
26 }
27 star_buf.push_back(e);
28 q.rekey(e.to, q.degree_of(e.to) - e.count);
29 }
30
31 G[v].clear();
32 G[v].shrink_to_fit();
33
34 if (star_buf.empty()) {
35 return static_cast<Weight>(0);
36 }
37
38 std::sort(star_buf.begin(), star_buf.end(),
39 [](const multi_edge<Weight, Index> &a, const multi_edge<Weight, Index> &b) {
40 return a.to < b.to;
41 });
42
43 nbr_out.clear();
44 Weight total_weight = static_cast<Weight>(0);
45
46 for (const auto &e : star_buf) {
47 if (!nbr_out.empty() && nbr_out.back().to == e.to) {
48 nbr_out.back().weight += e.weight;
49 nbr_out.back().count =
50 static_cast<std::uint8_t>(std::min<Index>(255, nbr_out.back().count + e.count));
51 } else {
52 nbr_out.push_back(e);
53 }
54 total_weight += e.weight;
55 }
56
57 return total_weight;
58}

62inline void add_exact_clique(array<array<multi_edge<Weight, Index>>> &G, Queue &q,
63 const array<multi_edge<Weight, Index>> &nbr, Weight total_weight) {
64 for (Index a = 0; a < nbr.size(); ++a) {
65 for (Index b = a + 1; b < nbr.size(); ++b) {
66 Weight w_exact = (nbr[a].weight * nbr[b].weight) / total_weight;
67 Index u = nbr[a].to;
68 Index j = nbr[b].to;
69
70 G[u].push_back({j, w_exact, 1});
71 G[j].push_back({u, w_exact, 1});
72 q.rekey(u, q.degree_of(u) + 1);
73 q.rekey(j, q.degree_of(j) + 1);
74 }
75 }

78template <typename Weight = double, std::integral Index = num::idx,
79 typename Queue = structures::basic_degree_queue<Index>, typename Rng = std::mt19937_64>
80inline void sample_clique(array<array<multi_edge<Weight, Index>>> &G, Queue &q,
81 array<multi_edge<Weight, Index>> &nbr, Weight total_weight,
82 std::type_identity_t<Index> sample_limit, Rng &rng) {
83 std::sort(nbr.begin(), nbr.end(),
84 [](const multi_edge<Weight, Index> &a, const multi_edge<Weight, Index> &b) {
85 return a.weight < b.weight;
86 });
87
88 Weight rest = total_weight;
89 std::uniform_real_distribution<Weight> unit_dist(static_cast<Weight>(0), static_cast<Weight>(1));
90
91 for (Index a = 0; a + 1 < nbr.size(); ++a) {
92 const auto &nbr_a = nbr[a];
93 rest -= nbr_a.weight;
94
95 if (rest <= static_cast<Weight>(0)) {
96 break;
97 }
98
99 const Index draws = std::min<Index>(nbr_a.count, sample_limit);
100 const Weight w_bar = nbr_a.weight / static_cast<Weight>(draws);
101 const Weight w_new = w_bar * (rest / total_weight);
102
103 for (Index s = 0; s < draws; ++s) {
104 Weight target = unit_dist(rng) * rest;
105
106 Index b = a + 1;
107 while (b + 1 < nbr.size() && target >= nbr[b].weight) {
108 target -= nbr[b].weight;
109 ++b;
110 }
111
112 Index u = nbr_a.to;
113 Index j = nbr[b].to;
114
115 G[u].push_back({j, w_new, 1});
116 G[j].push_back({u, w_new, 1});
117 q.rekey(u, q.degree_of(u) + 1);
118 q.rekey(j, q.degree_of(j) + 1);
119 }
120 }

124/// @brief Replace the star-mesh clique by a random spanning tree of it, reweighted
125/// so the elimination stays unbiased in expectation.
126///
127/// Eliminating `v` adds a clique on its neighbours with the Schur-complement
128/// weights \f$w_{ij} = c_i c_j / C\f$, where \f$c_i\f$ is the conductance from
129/// `v` to neighbour `i` and \f$C = \sum_i c_i\f$. `sample_clique` approximates
130/// that clique by independent draws; this samples a *spanning tree* of it
131/// instead, keeping \f$d-1\f$ edges rather than \f$d\f$ and drawing them from a
132/// negatively dependent (strongly Rayleigh) distribution.
133///
134/// ### Why this is unbiased, and why it needs no effective-resistance solve
135///
136/// For a weighted uniform spanning tree, Kirchhoff gives the inclusion
137/// probability \f$p_e = w_e R_{\mathrm{eff}}(e)\f$. Computing
138/// \f$R_{\mathrm{eff}}\f$ in general needs Laplacian solves — the very problem
139/// this factorization exists to precondition — but on *this* clique it is
140/// closed form. The clique Laplacian is a rank-one update of a diagonal,
141/// \f$\operatorname{diag}(c) - cc^{T}/C\f$, and solving it gives
142/// \f[
143/// R_{\mathrm{eff}}(i,j) = \frac{1}{c_i} + \frac{1}{c_j},
144/// \f]
145/// which is just the series path \f$i \to v \to j\f$ that the clique replaced.
146/// Hence
147/// \f[
148/// p_{ij} = \frac{c_i + c_j}{C}, \qquad
149/// \sum_{i<j} p_{ij} = d - 1,
150/// \f]
151/// exactly the edge count of a spanning tree, and the reweighting
152/// \f$\widehat c_{ij} = w_{ij}/p_{ij} = c_i c_j / (c_i + c_j)\f$ is the harmonic
153/// mean — the same series conductance the independent sampler assigns. So
154/// \f$\mathbb{E}[\widetilde L^{(v)}] = \mathrm{Sc}(L)\f$ holds by construction.
155///
156/// ### Aldous--Broder degenerates here
157///
158/// The clique's transition kernel is \f$P(i \to j) = c_j / (C - c_i)\f$, which
159/// does not depend on where the walk currently is. Sampling the tree is
160/// therefore not a graph traversal but a coupon-collector process: draw
161/// i.i.d. from \f$c/C\f$, and record the entering edge on each first visit.
162///
163/// ### Measured behaviour
164///
165/// On 2D grid Laplacians (n up to 40k), one tree per elimination is *worse* than
166/// the independent sampler: 44 PCG iterations against `ac1`'s 41 and `ac2`'s 29,
167/// and on higher-degree graphs with skewed weights the coupon-collector walk
168/// makes setup several times more expensive as well. Unbiasedness alone does not
169/// buy a preconditioner -- a single tree has too much variance.
170///
171/// Averaging `trees` independent trees per elimination does concentrate, as the
172/// strongly-Rayleigh matrix Chernoff theory predicts, and overtakes the existing
173/// samplers on quality (n = 14400, grid): 44 iterations at k=1, 27 at k=2, 20 at
174/// k=4, 15 at k=8, against `ac2`'s 29. Setup cost grows faster than k, because
175/// each tree adds edges that raise the degree of every later elimination:
176/// 11 ms, 82 ms, 616 ms, 2.7 s for the same k. So this pays only when one
177/// factorization is reused across many solves -- roughly 50 solves to break even
178/// at k=2, 600 at k=8 -- and loses outright for a single solve.
179///
180/// A skewed weight distribution also makes the collection slow, so the walk is
181/// capped. On reaching the cap the remaining vertices are attached to the
182/// heaviest neighbour, which keeps the result a spanning tree and the cost
183/// bounded, at the price of exactness in expectation for that elimination.
184///
185/// @param G Adjacency being eliminated; sampled edges are appended.
186/// @param q Degree queue, rekeyed for each endpoint touched.
187/// @param nbr Neighbours of the eliminated vertex, with their conductances.
188/// @param total_weight \f$C\f$, the sum of those conductances.
189/// @param rng Random source.
190template <typename Weight, std::integral Index, typename Queue, typename Rng>
191inline void sample_clique_tree(array<array<multi_edge<Weight, Index>>> &G, Queue &q,
192 const array<multi_edge<Weight, Index>> &nbr,
193 Weight total_weight, Rng &rng,
194 std::size_t trees = 1) {
195 const std::size_t degree = nbr.size();
196 if (degree < 2 || !(total_weight > Weight(0)) || trees == 0) {
197 return;
198 }
199
200 array<Weight> cumulative(degree);
201 Weight running = Weight(0);
202 std::size_t heaviest = 0;
203 for (std::size_t i = 0; i < degree; ++i) {
204 running += nbr[i].weight;
205 cumulative[i] = running;
206 if (nbr[i].weight > nbr[heaviest].weight) {
207 heaviest = i;
208 }
209 }
210 if (!(running > Weight(0))) {
211 return;
212 }
213
214 std::uniform_real_distribution<Weight> unit(Weight(0), Weight(1));
215 const auto draw = [&]() -> std::size_t {
216 const Weight target = unit(rng) * running;
217 std::size_t low = 0;
218 std::size_t high = degree - 1;
219 while (low < high) {
220 const std::size_t mid = low + ((high - low) / 2);
221 if (cumulative[mid] < target) {
222 low = mid + 1;
223 } else {
224 high = mid;
225 }
226 }
227 return low;
228 };
229
230 const auto connect = [&](std::size_t a, std::size_t b) {
231 const Weight wa = nbr[a].weight;
232 const Weight wb = nbr[b].weight;
233 const Weight sum = wa + wb;
234 if (!(sum > Weight(0))) {
235 return;
236 }
237 // Harmonic mean w_ij/p_ij, split across the trees so the average of
238 // `trees` independent samples remains unbiased.
239 const Weight weight = (wa * wb) / (sum * static_cast<Weight>(trees));
240 const Index u = nbr[a].to;
241 const Index v = nbr[b].to;
242 G[u].push_back({v, weight, 1});
243 G[v].push_back({u, weight, 1});
244 q.rekey(u, q.degree_of(u) + 1);
245 q.rekey(v, q.degree_of(v) + 1);
246 };
247
248 for (std::size_t tree = 0; tree < trees; ++tree) {
249 array<char> visited(degree, 0);
250 std::size_t current = draw();
251 visited[current] = 1;
252 std::size_t remaining = degree - 1;
253
254 // Coupon-collector expectation is d*H_d for uniform weights; the cap leaves
255 // generous room for moderate skew before the deterministic fallback.
256 const std::size_t budget = (8 * degree * (1 + degree / 4)) + 64;
257 for (std::size_t step = 0; step < budget && remaining > 0; ++step) {
258 const std::size_t next = draw();
259 if (next == current) {
260 continue;
261 }
262 if (visited[next] == 0) {
263 connect(current, next);
264 visited[next] = 1;
265 --remaining;
266 }
267 current = next;
268 }
269 if (remaining > 0) {
270 for (std::size_t i = 0; i < degree; ++i) {
271 if (visited[i] == 0 && i != heaviest) {
272 connect(heaviest, i);
273 visited[i] = 1;
274 }
275 }
276 }
277 }

281/// @brief Sample a spanning tree of the directed product biclique left by an LU pivot.
282///
283/// Eliminating pivot `v` from a nonsymmetric matrix leaves the rank-one Schur
284/// update \f$xy^{T}/a\f$, with \f$x_i = -A_{iv}\f$, \f$y_j = -A_{vj}\f$ and
285/// \f$a = A_{vv}\f$. That is not an undirected clique: \f$i \to j\f$ and
286/// \f$j \to i\f$ carry genuinely different weights \f$x_iy_j/a\f$ and
287/// \f$x_jy_i/a\f$.
288///
289/// Lifting to an undirected *bipartite* graph on duplicated vertices recovers the
290/// symmetric machinery without losing that asymmetry: put \f$i_L\f$ for each
291/// incoming neighbour, \f$j_R\f$ for each outgoing one, and give edge
292/// \f$i_L - j_R\f$ conductance \f$x_iy_j/a\f$. Opposite directed edges become
293/// \f$i_L-j_R\f$ and \f$j_L-i_R\f$, distinct objects, so nothing is forced
294/// together. A vertex that is both an in- and out-neighbour contributes
295/// \f$i_L-i_R\f$, which maps back to the diagonal fill \f$x_iy_i/a\f$ — the lift
296/// handles diagonal and off-diagonal fill in one object.
297///
298/// ### Closed-form marginals again
299///
300/// Writing \f$X = \sum_i x_i\f$, \f$Y = \sum_j y_j\f$, solving the bipartite
301/// Laplacian gives
302/// \f[
303/// R_{\mathrm{eff}}(i_L, j_R)
304/// = a\Big[\tfrac{1}{x_iY} + \tfrac{1}{y_jX} - \tfrac{1}{XY}\Big],
305/// \f]
306/// so with \f$\alpha_i = x_i/X\f$ and \f$\beta_j = y_j/Y\f$,
307/// \f[
308/// p_{ij} = w_{ij}R_{\mathrm{eff}} = \alpha_i + \beta_j - \alpha_i\beta_j
309/// = 1 - (1-\alpha_i)(1-\beta_j) \in [0,1],
310/// \f]
311/// which sums to \f$m + n - 1\f$: the edge count of a spanning tree on the
312/// \f$m+n\f$ duplicated vertices. Reweighting by \f$w_{ij}/p_{ij}\f$ therefore
313/// leaves \f$\mathbb{E}[\widetilde S_v] = xy^{T}/a\f$, the exact LU Schur update.
314///
315/// The walk simplifies too: \f$P(i_L \to j_R) = y_j/Y\f$ and
316/// \f$P(j_R \to i_L) = x_i/X\f$, neither depending on where it currently is, so
317/// tree sampling is a bipartite coupon collector alternating two fixed
318/// distributions.
319///
320/// ### Restriction
321///
322/// The conductance reading needs \f$x_i, y_j, a > 0\f$ — a nonsymmetric M-matrix
323/// or directed Laplacian. Under general signed LU the "conductances" go negative
324/// and none of the above applies. Callers are responsible for that check; this
325/// routine returns without emitting anything if it is violated.
326///
327/// Note also that unbiasedness is weaker here than in the symmetric case. mat
328/// Chernoff bounds for strongly Rayleigh measures are Hermitian results, so they
329/// say nothing about \f$\|\widetilde S_v - S_v\|\f$ concentrating for a
330/// nonsymmetric update. Whether this makes a good preconditioner is an empirical
331/// question, not one the expectation settles.
332///
333/// @param x Incoming conductances, all strictly positive.
334/// @param y Outgoing conductances, all strictly positive.
335/// @param pivot \f$a = A_{vv}\f$, strictly positive.
336/// @param rng Random source.
337/// @param trees Independent trees to average; each sampled weight is divided by this.
338/// @param emit Callable `void(std::size_t i, std::size_t j, Weight w)` receiving directed fill.
339template <typename Weight, typename Rng, typename Emit>
340inline void sample_biclique_tree(const array<Weight> &x, const array<Weight> &y,
341 Weight pivot, Rng &rng, std::size_t trees, Emit emit) {
342 const std::size_t m = x.size();
343 const std::size_t n = y.size();
344 if (m == 0 || n == 0 || trees == 0 || !(pivot > Weight(0))) {
345 return;
346 }
347
348 Weight total_x = Weight(0);
349 Weight total_y = Weight(0);
350 for (const Weight value : x) {
351 if (!(value > Weight(0))) {
352 return; // signed entries leave the conductance reading
353 }
354 total_x += value;
355 }
356 for (const Weight value : y) {
357 if (!(value > Weight(0))) {
358 return;
359 }
360 total_y += value;
361 }
362 if (!(total_x > Weight(0)) || !(total_y > Weight(0))) {
363 return;
364 }
365
366 array<Weight> cumulative_x(m);
367 array<Weight> cumulative_y(n);
368 Weight running = Weight(0);
369 for (std::size_t i = 0; i < m; ++i) {
370 running += x[i];
371 cumulative_x[i] = running;
372 }
373 running = Weight(0);
374 for (std::size_t j = 0; j < n; ++j) {
375 running += y[j];
376 cumulative_y[j] = running;
377 }
378
379 std::uniform_real_distribution<Weight> unit(Weight(0), Weight(1));
380 const auto pick = [&](const array<Weight> &cumulative, Weight total) -> std::size_t {
381 const Weight target = unit(rng) * total;
382 std::size_t low = 0;
383 std::size_t high = cumulative.size() - 1;
384 while (low < high) {
385 const std::size_t mid = low + ((high - low) / 2);
386 if (cumulative[mid] < target) {
387 low = mid + 1;
388 } else {
389 high = mid;
390 }
391 }
392 return low;
393 };
394
395 const auto record = [&](std::size_t i, std::size_t j) {
396 const Weight alpha = x[i] / total_x;
397 const Weight beta = y[j] / total_y;
398 const Weight inclusion = Weight(1) - ((Weight(1) - alpha) * (Weight(1) - beta));
399 if (!(inclusion > Weight(0))) {
400 return;
401 }
402 const Weight exact = (x[i] * y[j]) / pivot;
403 emit(i, j, exact / (inclusion * static_cast<Weight>(trees)));
404 };
405
406 const std::size_t budget = (8 * (m + n) * (1 + ((m + n) / 4))) + 64;
407 for (std::size_t tree = 0; tree < trees; ++tree) {
408 array<char> seen_left(m, 0);
409 array<char> seen_right(n, 0);
410 std::size_t remaining = m + n - 1;
411
412 std::size_t current = pick(cumulative_x, total_x);
413 bool on_left = true;
414 seen_left[current] = 1;
415
416 for (std::size_t step = 0; step < budget && remaining > 0; ++step) {
417 if (on_left) {
418 const std::size_t next = pick(cumulative_y, total_y);
419 if (seen_right[next] == 0) {
420 record(current, next);
421 seen_right[next] = 1;
422 --remaining;
423 }
424 current = next;
425 } else {
426 const std::size_t next = pick(cumulative_x, total_x);
427 if (seen_left[next] == 0) {
428 record(next, current);
429 seen_left[next] = 1;
430 --remaining;
431 }
432 current = next;
433 }
434 on_left = !on_left;
435 }
436
437 if (remaining > 0) {
438 // Budget exhausted under a very skewed weight distribution. Attach
439 // the stragglers to the heaviest vertex on the opposite side, which
440 // keeps the result a spanning tree at the cost of exactness here.
441 std::size_t heavy_left = 0;
442 std::size_t heavy_right = 0;
443 for (std::size_t i = 1; i < m; ++i) {
444 if (x[i] > x[heavy_left]) {
445 heavy_left = i;
446 }
447 }
448 for (std::size_t j = 1; j < n; ++j) {
449 if (y[j] > y[heavy_right]) {
450 heavy_right = j;
451 }
452 }
453 if (seen_left[heavy_left] == 0) {
454 record(heavy_left, heavy_right);
455 seen_left[heavy_left] = 1;
456 }
457 if (seen_right[heavy_right] == 0) {
458 record(heavy_left, heavy_right);
459 seen_right[heavy_right] = 1;
460 }
461 for (std::size_t i = 0; i < m; ++i) {
462 if (seen_left[i] == 0) {
463 record(i, heavy_right);
464 seen_left[i] = 1;
465 }
466 }
467 for (std::size_t j = 0; j < n; ++j) {
468 if (seen_right[j] == 0) {
469 record(heavy_left, j);
470 seen_right[j] = 1;
471 }
472 }
473 }
474 }
