78inline power_result power_iteration(const mat &A, real tol, idx max_iter) {
79 constexpr real tiny = 1e-300;
80 const idx n = A.rows();
81 if (A.cols() != n) {
82 throw std::invalid_argument("power_iteration: matrix must be square");
83 }
84
85 vec v(n, 0.0);
86 v[0] = 1.0;
87
88 real lambda = 0.0;
89 power_result result{0.0, v, 0, false};
90
91 for (idx iter = 0; iter < max_iter; ++iter) {
92 result.iterations = iter + 1;
93
94 vec w(n);
95 matvec(A, v, w);
96
97 real new_lambda = dot(v, w);
98 detail::normalise(w);
99
100 real delta = std::abs(new_lambda - lambda);
101 lambda = new_lambda;
102 v = w;
103
104 if (delta < tol * (std::abs(lambda) + tiny)) {
105 result.converged = true;
106 break;
107 }
108 }
109
110 result.eigenvalue = lambda;
111 result.eigenvector = v;
112 return result;
113}

115inline power_result inverse_iteration(const mat &A, real sigma, real tol, idx max_iter) {
116 constexpr real tiny = 1e-300;
117 const idx n = A.rows();
118 if (A.cols() != n) {
119 throw std::invalid_argument("inverse_iteration: matrix must be square");
120 }
121
122 // Factorize (A - sigma*I) once
123 mat M = A;
124 for (idx i = 0; i < n; ++i) {
125 M(i, i) -= sigma;
126 }
127 // M is A (rejected above unless square) shifted along its diagonal.
128 lu_result f = lu(assume_square(M));
129
130 vec v(n, 0.0);
131 v[0] = 1.0;
132
133 real lambda = 0.0;
134 power_result result{0.0, v, 0, false};
135
136 for (idx iter = 0; iter < max_iter; ++iter) {
137 result.iterations = iter + 1;
138
139 vec w(n);
140 lu_solve(f, v, w);
141 detail::normalise(w);
142
143 // Rayleigh quotient as eigenvalue estimate
144 vec av(n);
145 matvec(A, w, av);
146 real new_lambda = dot(w, av);
147
148 real delta = std::abs(new_lambda - lambda);
149 lambda = new_lambda;
150 v = w;
151
152 if (delta < tol * (std::abs(lambda) + tiny)) {
153 result.converged = true;
154 break;
155 }
156 }
157
158 result.eigenvalue = lambda;
159 result.eigenvector = v;
160 return result;
161}

163inline power_result rayleigh_iteration(const mat &A, const vec &x0, real tol, idx max_iter) {
164 const idx n = A.rows();
165 if (A.cols() != n) {
166 throw std::invalid_argument("rayleigh_iteration: matrix must be square");
167 }
168 if (x0.size() != n) {
169 throw std::invalid_argument("rayleigh_iteration: x0 size mismatch");
170 }
171
172 vec v = x0;
173 detail::normalise(v);
174
175 // Initial Rayleigh quotient
176 vec av(n);
177 matvec(A, v, av);
178 real sigma = dot(v, av);
179
180 power_result result{sigma, v, 0, false};
181
182 for (idx iter = 0; iter < max_iter; ++iter) {
183 result.iterations = iter + 1;
184
185 // Factorize (A - sigma*I); fresh each iteration (cubic convergence)
186 mat M = A;
187 for (idx i = 0; i < n; ++i) {
188 M(i, i) -= sigma;
189 }
190 // M is A (rejected above unless square) shifted along its diagonal.
191 lu_result f = lu(assume_square(M));
192
193 if (f.singular) {
194 break;
195 }
196
197 vec w(n);
198 lu_solve(f, v, w);
199 detail::normalise(w);
200
201 matvec(A, w, av);
202 real new_sigma = dot(w, av);
203
204 // res <- ||A*w - sigma*w||_2
205 const real res = std::sqrt(
206 kernel::linear_combination_norm_sq(av.data(), real(1), w.data(), -new_sigma, n));
207
208 sigma = new_sigma;
209 v = w;
210
211 if (res < tol) {
212 result.converged = true;
213 break;
214 }
215 }
216
217 result.eigenvalue = sigma;
218 result.eigenvector = v;
219 return result;
220}
