18/// @brief Verify that a smoothing kernel integrates to one over its support.
19///
20/// \f[ \int_0^{2h} W(r,h) \, 2\pi r \, dr = 1 \f]
21///
22/// SPH interpolates a field as \f$\sum_j m_j (\rho_j)^{-1} A_j W_{ij}\f$, which
23/// reproduces a constant only when the kernel is normalized. A kernel that is not
24/// produces densities off by a constant factor, which looks like a physical result
25/// rather than an error.
26template <class K, class Scalar = float>
27requires smoothing_kernel<K, Scalar> inline void
28verify_kernel_normalization(Scalar h, Scalar tol = Scalar(2e-2),
29 std::source_location loc = std::source_location::current()) {
30 if constexpr (!num::debug::sampling_compiled_in) {
31 return;
32 }
33 if (num::debug::get_level() != num::debug::diagnostic_level::full) {
34 return;
35 }
36 const int panels = 4000;
37 const Scalar upper = Scalar(2) * h;
38 const Scalar step = upper / static_cast<Scalar>(panels);
39 Scalar total = Scalar(0);
40 for (int i = 0; i < panels; ++i) {
41 const Scalar r = (static_cast<Scalar>(i) + Scalar(0.5)) * step;
42 total += K::W(r, h) * Scalar(2) * Scalar(3.14159265358979) * r * step;
43 }
44 if (std::abs(total - Scalar(1)) > tol) {
45 panic("KernelError",
46 "smoothing kernel is not normalized: integral over the support is " +
47 std::to_string(static_cast<double>(total)) +
48 " rather than 1, so SPH sums will not reproduce a constant field.",
49 loc);
50 }
51}

53/// @brief Verify that a kernel vanishes beyond its support radius.
54template <class K, class Scalar = float>
55requires smoothing_kernel<K, Scalar> inline void
56verify_kernel_support(Scalar h, std::source_location loc = std::source_location::current()) {
57 if (get_level() == diagnostic_level::off) {
58 return;
59 }
60 const Scalar outside = Scalar(2.001) * h;
61 if (K::W(outside, h) != Scalar(0)) {
62 panic("KernelError",
63 "smoothing kernel is nonzero beyond r = 2h, so a neighbour search "
64 "cut at that radius will miss contributions.",
65 loc);
66 }
67}

69/// @brief Verify that a periodic lattice's neighbour tables are mutually consistent.
70///
71/// Stepping up from a site and then down must return to it, and likewise left and
72/// right. A table built with the wrong modulus satisfies neither.
73template <class P>
74requires periodic_lattice_2d<P> inline void
75verify_lattice_symmetry(const P &lattice,
76 std::source_location loc = std::source_location::current()) {
77 if (get_level() == diagnostic_level::off) {
78 return;
79 }
80 const int sites = lattice.N * lattice.N;
81 for (int i = 0; i < sites; ++i) {
82 if (lattice.dn[lattice.up[i]] != i || lattice.up[lattice.dn[i]] != i ||
83 lattice.rt[lattice.lt[i]] != i || lattice.lt[lattice.rt[i]] != i) {
84 panic("LatticeError",
85 "periodic neighbour tables are not mutually inverse at site " + std::to_string(i),
86 loc);
87 }
88 }
89}
