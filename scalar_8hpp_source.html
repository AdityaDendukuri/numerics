44/// @brief Mathematical field \f$\mathbb{K}\f$: closed under addition, subtraction, multiplication, and division.
45///
46/// Requires standard scalar arithmetic (`+`, `-`, `*`, `/`) over an underlying floating-point base
47/// (`double`, `float`, `std::complex<double>`, `std::complex<float>`).
48///
49/// ### Axioms
50/// 1. **Additive Group:** Closed under `+`, `-`, with additive identity `0`.
51/// 2. **Multiplicative Group:** Closed under `*`, `/`, with multiplicative identity `1` and inverses for non-zero elements.
52/// 3. **Distributivity:** \f$a (b + c) = a b + a c\f$.
53///
54/// @tparam T Candidate scalar type.
55template <class T>
56concept field = std::floating_point<real_t<T>> && requires(T a, T b) {
57 { a + b } -> std::convertible_to<T>;
58 { a - b } -> std::convertible_to<T>;
59 { a *b } -> std::convertible_to<T>;
60 { a / b } -> std::convertible_to<T>;
61 { T(0) };
62 { T(1) };
63};

85/// @brief Modulus \f$|x|\f$, uniform over real and complex fields.
86template <class T>
87[[nodiscard]] inline real_t<T> mag(const T &x) noexcept {
88 if constexpr (is_complex_v<T>) {
89 return std::abs(x);
90 } else {
91 return x < T(0) ? -x : x;
92 }

101/// @brief Default relative tolerance for sampled property tests.
102///
103/// \f$\sqrt{\varepsilon}\f$ is the standard choice for randomized numerical probes:
104/// tight enough to reject genuine violations, loose enough to absorb the
105/// \f$O(n\varepsilon)\f$ round-off of the operator applications used to probe.
106template <class T>
107[[nodiscard]] inline real_t<T> sampling_tol() noexcept {
108 return std::sqrt(eps<T>());
