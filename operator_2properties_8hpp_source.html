1/// @file operator/properties.hpp
2/// @brief Attaching asserted mathematical properties to linear operators.
3///
4/// An axiom such as self-adjointness cannot be read off a type, so it enters the
5/// type system as a claim: `assume_spd(A)` returns an operator that *says* it is
6/// SPD. Two things make that claim trustworthy rather than decorative.
7///
8/// First, the claim is recorded as a position in the `num::law` hierarchy, so implication
9/// is automatic — an operator asserted SPD satisfies every weaker law without restating
10/// them, and `num::claims` is the only concept that has to read it.
11///
12/// Second, attaching the claim runs the sampled runtime test bound to that law, and to
13/// every law it implies. `assume_spd` on a non-symmetric operator is rejected by the
14/// inherited self-adjointness probe before definiteness is ever considered.
15///
16/// Whether those probes exist at all is a compile-time decision. `NUMERICS_DIAGNOSTICS`
17/// defaults to 1 under `NDEBUG`, which leaves the shape checks and discards the sampling,
18/// so an optimized build attaches the claim and runs no probe. Build with
19/// `-DNUMERICS_DIAGNOSTICS=2` to keep sampling in an optimized build. See `core/debug.hpp`.

33/// @brief A linear operator carrying an asserted axiom.
34///
35/// Replaces what was one hand-written wrapper class per property. The axiom is a
36/// template parameter, so the hierarchy supplies the implications and there are no
37/// per-property tag typedefs to keep in sync.
38template <class Op, class Ax>
39requires linear_operator<Op>
40class structured_op final {
41 public:
42 /// @brief The law this operator claims. Implication comes from the hierarchy, so this
43 /// one entry also supplies every weaker law.
44 using math_laws = math::type_list<Ax>;
45 using domain_type = vec;
46 using codomain_type = vec;
47
48 explicit structured_op(Op op,
49 std::source_location location = std::source_location::current())
50 : op_(std::move(op)),
51 provenance_{math::evidence_origin::assumed, location, "sampled assertion"} {}
52
53 template <class X, class Y>
54 void apply(const X &x, Y &y) const {
55 op_.apply(x, y);
56 }
57
58 template <class Y, class X>
59 void apply_adjoint(const Y &y, X &x) const requires adjointable_linear_operator<Op> {
60 op_.apply_adjoint(y, x);
61 }
62
63 [[nodiscard]] idx rows() const noexcept { return op_.rows(); }
64 [[nodiscard]] idx cols() const noexcept { return op_.cols(); }
65
66 /// @brief The underlying operator, stripped of the assertion.
67 [[nodiscard]] const Op &base() const noexcept { return op_; }
68 [[nodiscard]] const math::evidence_provenance &provenance() const noexcept {
69 return provenance_;
70 }
71
72 private:
73 Op op_;
74 math::evidence_provenance provenance_;
75};

77/// @brief Attach axiom Ax to an operator after sampling it and every axiom it implies.
78template <class Ax, class V = vec, class Op>
79requires linear_operator<Op>
80[[nodiscard]] inline structured_op<Op, Ax>
81assume(Op op, std::source_location loc = std::source_location::current()) {
82 verify_property<Ax, V>(op, loc);
83 return structured_op<Op, Ax>(std::move(op), loc);
84}

122/// @brief Assert that `apply_adjoint` really is the adjoint of `apply`.
123///
124/// Verified by \f$\langle Ax, y \rangle = \langle x, A^* y \rangle\f$ over random
125/// probes. A transposed-but-unconjugated adjoint passes on real data and fails
126/// here on complex data, which is exactly where it would otherwise corrupt LSQR
127/// and the Arnoldi recurrence.
128template <class Op, class V = vec>
129requires adjointable_linear_operator<Op>
130[[nodiscard]] inline structured_op<Op, law::linear_map>
131assume_adjointable(Op op, std::source_location loc = std::source_location::current()) {
132 debug::verify_adjoint_sample<Op, V>(op, op.rows(), op.cols(), loc);
133 return structured_op<Op, law::linear_map>(std::move(op), loc);
134}
