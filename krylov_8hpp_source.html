42/// @brief Conjugate gradients for symmetric positive definite \f$A\f$.
43///
44/// Minimizes \f$\tfrac12 x^T A x - b^T x\f$ over the Krylov space
45/// \f$\mathcal{K}_k(A, r_0)\f$. Requires \f$A = A^T\f$ and \f$A \succ 0\f$: on an
46/// indefinite operator some search direction has \f$p^T A p \leq 0\f$, and the
47/// iteration is stopped rather than continued with a meaningless step length.
48///
49/// @param A Callable `A(const T *x, T *y)` writing \f$y = Ax\f$.
50/// @param x Solution, used as the initial guess on entry.
51/// @param b Right-hand side, length n.
52/// @param n System dimension.
53/// @param work Caller-supplied scratch of length 3n.
54/// @param tol Absolute tolerance on \f$\|r\|_2\f$.
55/// @param max_iter Iteration cap.
56/// @return `krylov_result`: `.iterations`, `.residual` (final residual norm), `.converged`.
57template <std::floating_point T, class MatVec>
58[[nodiscard]] inline krylov_result<T> cg(MatVec &&A, T *NUM_K_RESTRICT x, const T *b, idx n,
59 T *NUM_K_RESTRICT work, T tol = T(1e-10),
60 idx max_iter = 1000) {
61 T *r = work;
62 T *p = work + n;
63 T *Ap = work + (2 * n);
64
65 A(x, r);
66 for (idx i = 0; i < n; ++i) {
67 r[i] = b[i] - r[i];
68 p[i] = r[i];
69 }
70
71 T rs_old = dot(r, r, n);
72 krylov_result<T> result{0, std::sqrt(rs_old), false};
73 if (result.residual < tol) {
74 result.converged = true;
75 return result;
76 }
77
78 for (idx iter = 0; iter < max_iter; ++iter) {
79 result.iterations = iter + 1;
80 A(p, Ap);
81
82 const T pAp = dot(p, Ap, n);
83 // Positive definiteness is what makes this quotient a valid step length.
84 if (!(pAp > T(0)) || !std::isfinite(pAp)) {
85 break;
86 }
87 const T alpha = rs_old / pAp;
88
89 // x <- x + alpha*p
90 axpy(x, p, alpha, n);
91
92 // r <- r - alpha*A*p; rs_new <- r^T*r
93 const T rs_new = axpy_norm_sq(r, Ap, -alpha, n);
94 result.residual = std::sqrt(rs_new);
95 if (result.residual < tol) {
96 result.converged = true;
97 break;
98 }
99
100 const T beta = rs_new / rs_old;
101 // p <- r + beta*p
102 axpby(p, r, T(1), beta, n);
103 rs_old = rs_new;
104 }
105 return result;
106}

108/// @brief Preconditioned conjugate gradients.
109///
110/// Applies \f$M^{-1}\f$ once per iteration. \f$M\f$ must itself be symmetric
111/// positive definite: PCG is CG in the \f$M^{-1}\f$ inner product, and a
112/// non-symmetric preconditioner makes that form non-symmetric, so the short
113/// recurrence no longer generates a Krylov basis.
114///
115/// @param A Callable `A(const T *x, T *y)` writing \f$y = Ax\f$.
116/// @param M Callable `M(const T *r, T *z)` writing \f$z \approx M^{-1} r\f$.
117/// @param x Solution, used as the initial guess on entry.
118/// @param b Right-hand side, length n.
119/// @param n System dimension.
120/// @param work Caller-supplied scratch of length 4n.
121/// @param tol Absolute tolerance on \f$\|r\|_2\f$.
122/// @param max_iter Iteration cap.
123/// @return `krylov_result`: `.iterations`, `.residual` (final residual norm), `.converged`.
124template <std::floating_point T, class MatVec, class Precond>
125[[nodiscard]] inline krylov_result<T> pcg(MatVec &&A, Precond &&M, T *NUM_K_RESTRICT x, const T *b,
126 idx n, T *NUM_K_RESTRICT work, T tol = T(1e-10),
127 idx max_iter = 1000) {
128 T *r = work;
129 T *z = work + n;
130 T *p = work + (2 * n);
131 T *Ap = work + (3 * n);
132
133 A(x, r);
134 // r <- b - A*x
135 axpby(r, b, T(1), T(-1), n);
136
137 krylov_result<T> result{0, std::sqrt(dot(r, r, n)), false};
138 if (result.residual < tol) {
139 result.converged = true;
140 return result;
141 }
142
143 M(r, z);
144 // p <- M^-1*r
145 copy(p, z, n);
146 T rz_old = dot(r, z, n);
147
148 for (idx iter = 0; iter < max_iter; ++iter) {
149 result.iterations = iter + 1;
150 A(p, Ap);
151
152 const T pAp = dot(p, Ap, n);
153 if (!(pAp > T(0)) || !std::isfinite(pAp)) {
154 break;
155 }
156 const T alpha = rz_old / pAp;
157
158 // x <- x + alpha*p
159 axpy(x, p, alpha, n);
160
161 // r <- r - alpha*A*p; ||r||_2^2
162 result.residual = std::sqrt(axpy_norm_sq(r, Ap, -alpha, n));
163 if (result.residual < tol) {
164 result.converged = true;
165 break;
166 }
167
168 M(r, z);
169 const T rz_new = dot(r, z, n);
170 const T beta = rz_new / rz_old;
171 // p <- M^-1*r + beta*p
172 axpby(p, z, T(1), beta, n);
173 rz_old = rz_new;
174 }
175 return result;
176}
