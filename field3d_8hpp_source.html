108inline float scalar_field_3d::sample(float x, float y, float z) const {
109 const float gx = (x - ox()) / dx();
110 const float gy = (y - oy()) / dx();
111 const float gz = (z - oz()) / dx();
112
113 if (gx < 0 || gx >= nx() - 1 || gy < 0 || gy >= ny() - 1 || gz < 0 || gz >= nz() - 1) {
114 return 0.0f;
115 }
116
117 const int i0 = static_cast<int>(gx);
118 const int j0 = static_cast<int>(gy);
119 const int k0 = static_cast<int>(gz);
120 const float tx = gx - i0, ty = gy - j0, tz = gz - k0;
121
122 auto v = [&](int di, int dj, int dk) {
123 return static_cast<float>((*this)(i0 + di, j0 + dj, k0 + dk));
124 };
125 return ((1 - tz) * (((1 - ty) * (((1 - tx) * v(0, 0, 0)) + (tx * v(1, 0, 0)))) +
126 (ty * (((1 - tx) * v(0, 1, 0)) + (tx * v(1, 1, 0)))))) +
127 (tz * (((1 - ty) * (((1 - tx) * v(0, 0, 1)) + (tx * v(1, 0, 1)))) +
128 (ty * (((1 - tx) * v(0, 1, 1)) + (tx * v(1, 1, 1))))));
129}
