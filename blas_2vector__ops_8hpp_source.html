66/// @brief `z <- x + y`. No single BLAS call does an out-of-place add; a plain
67/// loop is exactly as fast as `dcopy`+`daxpy` and skips the extra pass.
68inline void add(const vec &x, const vec &y, vec &z) noexcept {
69 const idx n = x.size();
70 const real *xd = x.data();
71 const real *yd = y.data();
72 real *zd = z.data();
73 for (idx i = 0; i < n; ++i) {
74 zd[i] = xd[i] + yd[i];
75 }
76}
