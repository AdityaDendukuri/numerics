59/// @brief Sample the equivalence-relation axioms a union-find claims to maintain.
60///
61/// `find`, `unite` and `connected` are an *implementation* of an equivalence
62/// relation, and the properties that make them correct are reflexivity, symmetry
63/// and transitivity of \f$\sim\f$, plus agreement between `connected(u,v)` and
64/// \f$\mathrm{find}(u) = \mathrm{find}(v)\f$. None of that is decidable from the
65/// type, so it is sampled here.
66/// Takes a mutable reference because path compression rewrites the forest while
67/// answering queries: `find` is observationally pure but not physically const.
68template <class DS, std::integral Index = num::idx>
69inline void
70verify_equivalence_relation(DS &ds, Index n,
71 std::source_location loc = std::source_location::current()) {
72 if constexpr (!num::debug::sampling_compiled_in) {
73 return;
74 }
75 if (num::debug::get_level() != num::debug::diagnostic_level::full || n == 0) {
76 return;
77 }
78 const Index limit = n < Index(24) ? n : Index(24);
79
80 for (Index u = 0; u < limit; ++u) {
81 // Reflexivity: u ~ u.
82 if (!ds.connected(u, u)) {
83 num::debug::panic("PropertyError",
84 "union-find is not reflexive: connected(" + std::to_string(u) + "," +
85 std::to_string(u) + ") is false",
86 loc);
87 }
88 // find selects a canonical representative, so it is idempotent.
89 if (ds.find(ds.find(u)) != ds.find(u)) {
90 num::debug::panic("PropertyError",
91 "union-find representative is not canonical: find(find(" +
92 std::to_string(u) + ")) != find(" + std::to_string(u) + ")",
93 loc);
94 }
95 }
96
97 for (Index u = 0; u < limit; ++u) {
98 for (Index v = 0; v < limit; ++v) {
99 // Symmetry: u ~ v iff v ~ u.
100 if (ds.connected(u, v) != ds.connected(v, u)) {
101 num::debug::panic("PropertyError",
102 "union-find is not symmetric at (" + std::to_string(u) + "," +
103 std::to_string(v) + ")",
104 loc);
105 }
106 // connected must agree with equality of representatives.
107 if (ds.connected(u, v) != (ds.find(u) == ds.find(v))) {
108 num::debug::panic("PropertyError",
109 "union-find connected() disagrees with find() at (" +
110 std::to_string(u) + "," + std::to_string(v) + ")",
111 loc);
112 }
113 if (!ds.connected(u, v)) {
114 continue;
115 }
116 // Transitivity: u ~ v and v ~ w imply u ~ w.
117 for (Index w = 0; w < limit; ++w) {
118 if (ds.connected(v, w) && !ds.connected(u, w)) {
119 num::debug::panic("PropertyError",
120 "union-find is not transitive: " + std::to_string(u) + "~" +
121 std::to_string(v) + " and " + std::to_string(v) + "~" +
122 std::to_string(w) + " but not " + std::to_string(u) +
123 "~" + std::to_string(w),
124 loc);
125 }
126 }
127 }
128 }
129
130 if (ds.count() > ds.size()) {
131 num::debug::panic("PropertyError",
132 "union-find quotient is larger than the underlying set: count() = " +
133 std::to_string(ds.count()) +
134 " > size() = " + std::to_string(ds.size()),
135 loc);
136 }
137}

139/// @brief Verify the heap axiom: the reported top key is minimal over the queue.
140template <class PQ, class Key = double, std::integral Index = num::idx>
141inline void verify_heap_order(const PQ &pq, Index capacity,
142 std::source_location loc = std::source_location::current()) {
143 if (num::debug::get_level() != num::debug::diagnostic_level::full || pq.empty()) {

199/// @brief Verify that reported degrees agree with the enumerated adjacency.
200///
201/// `degree(u)` and `neighbors(u)` are two presentations of the same set
202/// \f$\{v : (u,v) \in E\}\f$; a graph whose degrees disagree with its adjacency
203/// silently corrupts every traversal built on it.
204template <class G, std::integral Index = num::idx>
205inline void verify_degree_consistency(const G &g,
206 std::source_location loc = std::source_location::current()) {
207 if (num::debug::get_level() == num::debug::diagnostic_level::off) {
208 return;
209 }
210 for (Index u = 0; u < g.n_vertices(); ++u) {
211 Index counted = 0;
212 for ([[maybe_unused]] const auto &edge : g.neighbors(u)) {
213 ++counted;
214 }
215 if (counted != g.degree(u)) {
216 num::debug::panic("GraphStructureError",
217 "degree(" + std::to_string(u) + ") = " + std::to_string(g.degree(u)) +
218 " but neighbors(" + std::to_string(u) + ") enumerates " +
219 std::to_string(counted) + " entries",
220 loc);
221 }
222 }
223}

225/// @brief Verify the handshake lemma \f$\sum_u \deg(u) = 2|E|\f$ for an undirected graph.
226template <class G, std::integral Index = num::idx>
227inline void verify_handshake_lemma(const G &g,
228 std::source_location loc = std::source_location::current()) {
229 if (num::debug::get_level() == num::debug::diagnostic_level::off) {
230 return;
231 }
232 Index degree_sum = 0;
233 for (Index u = 0; u < g.n_vertices(); ++u) {
234 degree_sum += g.degree(u);
235 }
236 if (degree_sum != Index(2) * g.n_edges()) {
237 num::debug::panic(
238 "GraphStructureError",
239 "handshake lemma violated: sum of degrees = " + std::to_string(degree_sum) +
240 " but 2|E| = " + std::to_string(Index(2) * g.n_edges()) +
241 ". The graph is not a consistent undirected incidence structure.",
242 loc);
243 }

246/// @brief Verify the defining structure of a graph Laplacian \f$\Delta = D - W\f$.
247///
248/// A Laplacian is symmetric with zero row sums, which is exactly the statement
249/// \f$\Delta \mathbf{1} = 0\f$: the constant vector lies in its null space. That
250/// null space is why a Laplacian is positive *semi*-definite and must be asserted
251/// `law::psd` rather than `law::spd` — asserting the latter claims an
252/// invertibility it does not have.
253///
254/// Accepts dense or CSR storage; for CSR only the stored entries are visited, so
255/// the check stays proportional to the number of non-zeros.
256template <class Mat>
257inline void verify_laplacian_structure(const Mat &L, double tol = 1e-9,
258 std::source_location loc = std::source_location::current()) {
259 if (num::debug::get_level() == num::debug::diagnostic_level::off) {
260 return;
261 }
262
263 num::idx n = 0;
264 num::idx m = 0;
265 if constexpr (requires { L.n_rows(); }) {
266 n = L.n_rows();
267 m = L.n_cols();
268 } else {
269 n = L.rows();
270 m = L.cols();
271 }
272 if (n != m) {
273 num::debug::panic("GraphStructureError", "Laplacian must be square", loc);
274 }
275
276 auto check_row = [&](num::idx i, double row_sum) {
277 if (std::abs(row_sum) > tol) {
278 num::debug::panic(
279 "GraphStructureError",
280 "Laplacian row " + std::to_string(i) + " sums to " + std::to_string(row_sum) +
281 " rather than 0, so the constant vector is not in its null space.",
282 loc);
283 }
284 };
285 auto check_symmetry = [&](num::idx i, num::idx j) {
286 const double a = static_cast<double>(L(i, j));
287 const double b = static_cast<double>(L(j, i));
288 if (std::abs(a - b) > tol) {
289 num::debug::panic("GraphStructureError",
290 "Laplacian is not symmetric at (" + std::to_string(i) + "," +
291 std::to_string(j) + ")",
292 loc);
293 }
294 };
295
296 if constexpr (requires {
297 L.row_ptr();
298 L.col_idx();
299 L.values();
300 }) {
301 const auto *row_ptr = L.row_ptr();
302 const auto *col_idx = L.col_idx();
303 const auto *values = L.values();
304 for (num::idx i = 0; i < n; ++i) {
305 double row_sum = 0.0;
306 for (num::idx k = row_ptr[i]; k < row_ptr[i + 1]; ++k) {
307 row_sum += static_cast<double>(values[k]);
308 check_symmetry(i, col_idx[k]);
309 }
310 check_row(i, row_sum);
311 }
312 } else {
313 for (num::idx i = 0; i < n; ++i) {
314 double row_sum = 0.0;
315 for (num::idx j = 0; j < n; ++j) {
316 row_sum += static_cast<double>(L(i, j));
317 check_symmetry(i, j);
318 }
319 check_row(i, row_sum);
320 }
321 }
322}
