258inline void banded_lu_solve_multi(const band_mat &A, const idx *ipiv, real *B, idx nrhs) {
259 const idx n = A.size(), kl = A.kl(), ku = A.ku(), ldab = A.ldab();
260 const real *ab = A.data();
261 const idx kv = ku + kl;
262
263#ifdef _OPENMP
264#pragma omp parallel for if (nrhs > 16)
265#endif
266 for (idx rhs = 0; rhs < nrhs; ++rhs) {
267 real *x = B + (rhs * n);
268 for (idx i = 0; i < n; ++i) {
269 if (ipiv[i] != i) {
270 std::swap(x[i], x[ipiv[i]]);
271 }
272 }
273 for (idx j = 0; j < n; ++j) {
274 if (x[j] != 0.0) {
275 const idx last = std::min(j + kl, n - 1);
276 real xj = x[j];
277 for (idx i = j + 1; i <= last; ++i) {
278 x[i] -= ab[kv + i - j + (j * ldab)] * xj;
279 }
280 }
281 }
282 for (idx j = n; j > 0; --j) {
283 const idx col = j - 1;
284 x[col] /= ab[kv + (col * ldab)];
285 if (x[col] != 0.0) {
286 const idx first = (col > ku) ? col - ku : 0;
287 real xc = x[col];
288 for (idx i = first; i < col; ++i) {
289 x[i] -= ab[kv + i - col + (col * ldab)] * xc;
290 }
291 }
292 }
293 }
294}

296inline banded_solver_result banded_solve(const band_mat &A, const vec &b, vec &x) {
297 const idx n = A.size();
298 if (b.size() != n || x.size() != n) {
299 throw std::invalid_argument("banded_solve: dimension mismatch");
300 }
301
302 band_mat a_work = A;
303 auto ipiv = std::make_unique<idx[]>(n);
304 banded_solver_result result = banded_lu(a_work, ipiv.get());
305 if (!result.success) {
306 return result;
307 }
308
309 for (idx i = 0; i < n; ++i) {
310 x[i] = b[i];
311 }
312 banded_lu_solve(a_work, ipiv.get(), x);
313 return result;
314}

338inline real banded_norm1(const band_mat &A) {
339 const idx n = A.size(), kl = A.kl(), ku = A.ku(), ldab = A.ldab();
340 const real *ab = A.data();
341 const idx kv = ku + kl;
342 real max_sum = 0.0;
343 for (idx j = 0; j < n; ++j) {
344 real col_sum = 0.0;
345 const idx i_start = (j > ku) ? j - ku : 0;
346 const idx i_end = std::min(j + kl, n - 1);
347 for (idx i = i_start; i <= i_end; ++i) {
348 col_sum += std::abs(ab[kv + i - j + (j * ldab)]);
349 }
350 max_sum = std::max(max_sum, col_sum);
351 }
352 return max_sum;
353}

355inline real banded_rcond(const band_mat &A, const idx *ipiv, real anorm) {
356 const idx n = A.size();
357 if (n == 0 || anorm == 0.0) {
358 return 0.0;
359 }
360 vec y(n, 1.0 / static_cast<real>(n));
361 const band_mat &a_copy = A;
362 banded_lu_solve(a_copy, ipiv, y);
363 real ainv_norm = 0.0;
364 for (idx i = 0; i < n; ++i) {
365 ainv_norm += std::abs(y[i]);
366 }
367 return 1.0 / (anorm * ainv_norm);
368}
