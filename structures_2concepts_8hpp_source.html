18/// @brief Equivalence relation \f$\sim\f$ on \f$\{0,\dots,N-1\}\f$ with its quotient set.
19///
20/// `find(u)` selects the canonical representative of \f$[u]\f$ and is therefore
21/// idempotent. `connected(u,v)` decides \f$u \sim v\f$ and must agree with
22/// \f$\mathrm{find}(u) = \mathrm{find}(v)\f$. `unite(u,v)` replaces \f$\sim\f$ by
23/// the finest relation containing it in which \f$u \sim v\f$. `count()` is the
24/// size of the quotient and drops by one per successful `unite`.
25template <typename T, typename Index = num::idx>
26concept equivalence_relation = requires(T ds, Index u, Index v) {
27 { ds.find(u) } -> std::same_as<Index>;
28 { ds.unite(u, v) } -> std::same_as<bool>;
29 { ds.connected(u, v) } -> std::same_as<bool>;
30 { ds.count() } -> std::same_as<Index>;
31 { ds.size() } -> std::same_as<Index>;
32};

34/// @brief Totally ordered key set with addressable elements and a retrievable minimum.
35///
36/// `top_key()` is the minimum over every contained key, maintained across `push`,
37/// `pop`, and `update`. `update` revises a key in place, which is what Dijkstra
38/// and Prim need and a plain priority queue cannot do.
39template <typename T, typename Key = double, typename Index = num::idx>
40concept addressable_priority_queue = std::totally_ordered<Key> &&
41 requires(T pq, Index index, const Key &key) {
42 { pq.push(index, key) } -> std::same_as<void>;
43 { pq.pop() } -> std::same_as<void>;
44 { pq.top_index() } -> std::same_as<Index>;
45 { pq.top_key() } -> std::convertible_to<Key>;
46 { pq.contains(index) } -> std::same_as<bool>;
47 { pq.update(index, key) } -> std::same_as<void>;
48 { pq.empty() } -> std::same_as<bool>;
49 { pq.size() } -> std::same_as<Index>;
50};

52/// @brief Directed incidence \f$u \to v\f$ carrying a weight in a scalar field.
53template <typename E, typename Index = num::idx, typename Weight = double>
54concept weighted_incidence = scalars::field<Weight> && requires(E e) {
55 { e.to } -> std::convertible_to<Index>;

59/// @brief Finite incidence structure \f$G = (V, E)\f$ with \f$E \subseteq V \times V\f$.
60///
61/// `neighbors(u)` enumerates \f$\{v : (u,v) \in E\}\f$ and `degree(u)` is that
62/// set's cardinality. The two must agree, and for an undirected graph the
63/// handshake lemma \f$\sum_u \deg(u) = 2|E|\f$ must hold.
64template <typename G, typename Index = num::idx>
65concept incidence_structure = requires(const G g, Index u) {
66 { g.n_vertices() } -> std::same_as<Index>;
67 { g.n_edges() } -> std::same_as<Index>;
68 { g.degree(u) } -> std::same_as<Index>;
69 { g.neighbors(u) };
70};
