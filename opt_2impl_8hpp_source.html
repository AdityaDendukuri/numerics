29/// Butterfly on hlen complex pairs using AVX2 (2 pairs per iteration).
30static inline void butterfly(num::cplx *__restrict__ u, num::cplx *__restrict__ v,
31 const num::cplx *__restrict__ tw, int hlen) {
32 auto *ud = reinterpret_cast<double *>(u);
33 auto *vd = reinterpret_cast<double *>(v);
34 auto *wd = reinterpret_cast<const double *>(tw);
35
36 int j = 0;
37 for (; j + 1 < hlen; j += 2) {
38 __m256d U = _mm256_loadu_pd(ud + 2 * j);
39 __m256d V = _mm256_loadu_pd(vd + 2 * j);
40 __m256d W = _mm256_loadu_pd(wd + 2 * j);
41 // complex multiply V * W
42 __m256d Wre = _mm256_unpacklo_pd(W, W); // [wr0,wr0,wr1,wr1]
43 __m256d Wim = _mm256_unpackhi_pd(W, W); // [wi0,wi0,wi1,wi1]
44 __m256d Vsw = _mm256_permute_pd(V, 0x5); // [vi0,vr0,vi1,vr1]
45 __m256d T = _mm256_addsub_pd(_mm256_mul_pd(V, Wre), _mm256_mul_pd(Vsw, Wim));
46 _mm256_storeu_pd(ud + 2 * j, _mm256_add_pd(U, T));
47 _mm256_storeu_pd(vd + 2 * j, _mm256_sub_pd(U, T));
48 }
49 // scalar tail for odd hlen
50 for (; j < hlen; ++j) {
51 num::cplx t = v[j] * tw[j];
52 num::cplx uu = u[j];
53 u[j] = uu + t;
54 v[j] = uu - t;
55 }
56}

60/// Butterfly on hlen complex pairs using NEON vld2/vst2 (SoA deinterleave).
61static inline void butterfly(num::cplx *__restrict__ u, num::cplx *__restrict__ v,
62 const num::cplx *__restrict__ tw, int hlen) {
63 auto *ud = reinterpret_cast<double *>(u);
64 auto *vd = reinterpret_cast<double *>(v);
65 auto *wd = reinterpret_cast<const double *>(tw);
66
67 int j = 0;
68 for (; j + 1 < hlen; j += 2) {
69 // Deinterleaved load: .val[0] = [re0,re1], .val[1] = [im0,im1]
70 float64x2x2_t U = vld2q_f64(ud + static_cast<ptrdiff_t>(2 * j));
71 float64x2x2_t V = vld2q_f64(vd + static_cast<ptrdiff_t>(2 * j));
72 float64x2x2_t W = vld2q_f64(wd + static_cast<ptrdiff_t>(2 * j));
73
74 // T = V * W (complex multiply, component-wise on SoA data).
75 float64x2_t tr = vfmsq_f64(vmulq_f64(V.val[0], W.val[0]), V.val[1], W.val[1]);
76 // Ti = Vr*Wi + Vi*Wr.
77 float64x2_t ti = vfmaq_f64(vmulq_f64(V.val[0], W.val[1]), V.val[1], W.val[0]);
78
79 float64x2x2_t ru, rv;
80 ru.val[0] = vaddq_f64(U.val[0], tr);
81 ru.val[1] = vaddq_f64(U.val[1], ti);
82 rv.val[0] = vsubq_f64(U.val[0], tr);
83 rv.val[1] = vsubq_f64(U.val[1], ti);
84 vst2q_f64(ud + 2 * j, ru);
85 vst2q_f64(vd + 2 * j, rv);
86 }
87 // scalar tail
88 for (; j < hlen; ++j) {
89 num::cplx t = v[j] * tw[j];
90 num::cplx uu = u[j];
91 u[j] = uu + t;
92 v[j] = uu - t;
93 }
94}

112/// Precomputed twiddle factors + SIMD butterfly execution.
113struct fft_plan_impl : public num::spectral::fft_plan_impl {
114 int n;
115 bool invert;
116 std::vector<std::vector<num::cplx>> twiddles;
117
118 fft_plan_impl(int n_, bool inv) : n(n_), invert(inv) {
119 if (n_ == 0 || (n_ & (n_ - 1))) {
120 throw std::invalid_argument("fft_plan: length must be a power of two");
121 }
122 for (int len = 2; len <= n_; len <<= 1) {
123 double ang = two_pi / static_cast<double>(len) * (inv ? 1.0 : -1.0);
124 num::cplx wlen{std::cos(ang), std::sin(ang)};
125 std::vector<num::cplx> tw(len / 2);
126 num::cplx w{1.0, 0.0};
127 for (int j = 0; j < len / 2; ++j) {
128 tw[j] = w;
129 w *= wlen;
130 }
131 twiddles.push_back(std::move(tw));
132 }
133 }
134
135 void execute(const num::cvec &in, num::cvec &out) const override {
136 for (num::idx i = 0; i < static_cast<num::idx>(n); ++i) {
137 out[i] = in[i];
138 }
139 execute_inplace(out);
140 }
141
142 void execute_inplace(num::cvec &out) const {
143 backends::seq::bit_reverse(out);
144 num::cplx *data = out.data();
145 int stage = 0;
146 for (int len = 2; len <= n; len <<= 1, ++stage) {
147 int hlen = len / 2;
148 const num::cplx *tw = twiddles[stage].data();
149 for (int i = 0; i < n; i += len) {
150 butterfly(data + i, data + i + hlen, tw, hlen);
151 }
152 }
153 }
154};

188 num::cvec tmp(static_cast<num::idx>(n), num::cplx{0, 0});
189 for (int k = 0; k < (n / 2) + 1; ++k) {
190 tmp[k] = in[k];

192 for (int k = 1; k < ((n - 1) / 2) + 1; ++k) {
193 tmp[n - k] = std::conj(in[k]);
194 }
195 fft_plan_impl plan(n, true);
196 plan.execute_inplace(tmp);
197 for (int i = 0; i < n; ++i) {
