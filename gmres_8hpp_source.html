13/// @brief Solve general unsymmetric linear systems \f$A x = b\f$ using restarted GMRES.
14///
15/// Constructs an orthogonal Arnoldi basis \f$V_m\f$ and solves the projected least-squares
16/// problem \f$\min \| \beta e_1 - \bar{H}_m y \|_2\f$ with Givens plane rotations.
17///
18/// @tparam Op Linear operator type satisfying `math::endomorphism_on<Op, vec>`.
19/// @param A General square linear operator (dense, sparse, or matrix-free).
20/// @param b Right-hand side vector.
21/// @param x Solution vector (serves as initial guess on input, updated in place).
22/// @param tolerance Relative convergence tolerance on Euclidean residual norm \f$\|b - A x\|_2\f$.
23/// @param max_iterations Maximum total Arnoldi iterations.
24/// @param restart Number of Krylov subspace vectors before restarting Arnoldi process (default: 30).
25/// @return `solver_result` containing iteration count, final residual norm, and convergence boolean.
26/// @throws std::invalid_argument If dimensions do not match.
27/// @see cg, minres, pcg

37/// @brief Solve sparse linear systems \f$A x = b\f$ using restarted GMRES.
38///
39/// @param A Square CSR sparse matrix.
40/// @param b Right-hand side vector.
41/// @param x Solution vector (serves as initial guess on input, updated in place).
42/// @param tolerance Convergence tolerance on Euclidean residual norm (default: 1e-6).
43/// @param max_iterations Maximum total iterations (default: 1000).
44/// @param restart Krylov subspace restart cycle (default: 30).
45/// @return `solver_result` containing iteration count, final residual norm, and convergence boolean.
46/// @throws std::invalid_argument If `A` is not square or dimensions mismatch.
47inline solver_result gmres(const spmat &A, const vec &b, vec &x, real tolerance = 1e-6,
48 idx max_iterations = 1000, idx restart = 30) {
49 if (A.n_rows() != A.n_cols()) {

51 }
52 return gmres(
