39/// BFS connected-component labelling with pre-allocated flat queue (no heap per
40/// call).
41///
42/// @param n_sites Total number of sites
43/// @param in_cluster bool(int i) -- include site i?
44/// @param neighbors void(int i, auto&& visit) -- call visit(nb) per neighbor
45/// of i
46template <typename InCluster, typename Neighbors>
47cluster_result connected_components(int n_sites, InCluster &&in_cluster, Neighbors &&neighbors) {
48 cluster_result res;
49 res.id.resize(n_sites);
50 res.sizes.reserve(64);
51
52 for (int i = 0; i < n_sites; ++i) {
53 res.id[i] = in_cluster(i) ? -1 : -2; // -1 = unvisited included, -2 = excluded
54 }
55
56 array<int> queue(n_sites);
57 int qhead = 0, qtail = 0;
58
59 for (int start = 0; start < n_sites; ++start) {
60 if (res.id[start] != -1) {
61 continue;
62 }
63
64 const int cid = static_cast<int>(res.sizes.size());
65 res.sizes.push_back(0);
66 res.id[start] = cid;
67 queue[qtail++] = start;
68
69 while (qhead < qtail) {
70 const int i = queue[qhead++];
71 ++res.sizes[cid];
72 neighbors(i, [&](int nb) {
73 if (res.id[nb] == -1) {
74 res.id[nb] = cid;
75 queue[qtail++] = nb;
76 }
77 });
78 }
79
80 if (res.sizes[cid] > res.largest_size) {
81 res.largest_size = res.sizes[cid];
82 res.largest_id = cid;
83 }
84 }
85 return res;
86}
