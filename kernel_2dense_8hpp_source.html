63/// @brief Applies a Givens rotation to columns \f$p\f$ and \f$q\f$ in place.
64template <std::floating_point T>
65NUM_K_AINLINE void rotate_columns(T *NUM_K_RESTRICT A, idx lda, idx rows, idx p, idx q, T c,
66 T s) noexcept {
67 for (idx i = 0; i < rows; ++i) {
68 T *row = A + (i * lda);
69 const T ap = row[p], aq = row[q];
70 row[p] = (c * ap) - (s * aq);
71 row[q] = (s * ap) + (c * aq);

75// Dense matrix product.
76//
77// The shape of this is forced by the arithmetic intensity of the operation. A
78// product does O(n^3) work over O(n^2) data, so it is compute-bound in
79// principle, but the textbook i-k-j triple loop does not get anywhere near the
80// machine's peak: each fused multiply-add reads a fresh element of C from
81// memory and writes it straight back, so the loop runs at the rate the store
82// unit and the L1 cache can retire traffic, not at the rate the FMA units can
83// issue. On this tree the plain version sustained ~13 GFLOP/s.
84//
85// Two nested levels of blocking fix that, and nothing else is needed.
86//
87// Register tile (`mr` x `nr`): the innermost loop holds a small block of C in
88// vector registers across the entire k sweep. Each element of A loaded is
89// reused across `nr` columns and each element of B across `mr` rows, so one
90// pair of loads feeds `mr*nr` FMAs instead of one. The tile is deliberately
91// sized to about half the architectural vector register file: large enough to
92// hide FMA latency, small enough that the accumulators are never spilled --
93// a spilled tile is slower than no tile at all.
94//
95// Cache panel (`kc`): the k range is cut so the slice of B the tile loop
96// sweeps (kc x n) stays resident while every row block streams past it.
97// Without this, a large product re-reads B from DRAM once per row block.
98//
99// Both bounds come from the target's own properties, not from a tuning
100// parameter, so there is nothing for a caller to get wrong. Together they take
101// the same computation to ~2.5x the plain loop, and the summation order per
102// output element is unchanged (still ascending in p), so results are
103// bit-identical to the naive triple loop.

116/// @brief The blocking `gemm` uses on this target, derived at compile time.
117///
118/// The structure is the Goto/BLIS one: a register-tiled microkernel computing an
119/// `mr x nr` block of C over `kc` inner products, fed from packed copies of A and
120/// B so that it streams contiguous, zero-padded panels rather than strided rows
121/// of the caller's matrices. Three loops around it block for the memory
122/// hierarchy: an `mc x kc` slab of A held in L2, a `kc x nc` panel of B held
123/// across one sweep of that slab, and the `mr x kc` / `kc x nr` slivers the
124/// microkernel touches held in L1.
125///
126/// Only five integers are target-specific, and every one is derived from the
127/// macros in `kernel/vector.hpp`:
128///
129/// | | NEON / SSE2 | AVX2 | AVX-512 |
130/// |---|---|---|---|
131/// | `mr x nr` (double) | 8 x 6 / 6 x 4 | 6 x 8 | 14 x 16 |
132///
133/// which are the shapes BLIS ships for those targets. `kc`, `mc` and `nc` come
134/// from `NUM_K_L1_BYTES`, `NUM_K_L2_BYTES` and `NUM_K_GEMM_PANEL_BYTES`.
135template <std::floating_point T>
136struct gemm_config {
137#if defined(NUM_K_VECTOR_EXT)
138 /// Elements per vector register.
139 static constexpr idx width = NUM_K_VECTOR_BYTES / sizeof(T);
140 /// B vectors held per microkernel: three when the file is wide and the
141 /// vectors narrow (NEON), else two.
142 static constexpr idx b_vectors = (NUM_K_VECTOR_REGISTERS >= 32 && width <= 2) ? 3 : 2;
143 /// Rows of the register tile: what remains of the file after the B vectors
144 /// and two broadcast registers, at most 14, rounded to even.
145 static constexpr idx mr =
146 std::min<idx>(14, ((NUM_K_VECTOR_REGISTERS - b_vectors - 2) / b_vectors) & ~idx{1});
147#else
148 static constexpr idx width = 4;
149 static constexpr idx b_vectors = 2;
150 static constexpr idx mr = 4;
151#endif
152 /// Columns of the register tile.
153 static constexpr idx nr = b_vectors * width;
154 /// Inner-product length per microkernel call. The `kc x nr` B sliver stays
155 /// in L1 across the whole sweep of A tiles, so it gets half of L1; the A
156 /// slivers stream from the L2-resident slab. Capped at 512: on a large L1
157 /// a longer `kc` only shrinks the other two blocks, measured here as a
158 /// loss at n = 256..512 and no gain above.
159 static constexpr idx kc =
160 std::clamp<idx>(detail::round_down((NUM_K_L1_BYTES / 2) / (sizeof(T) * nr), 8), 8, 512);
161 /// Rows of the packed A slab: half of L2, at most 1 MiB.
162 static constexpr idx mc = std::max<idx>(
163 mr,
164 detail::round_down(std::min<idx>(NUM_K_L2_BYTES / 2, idx{1} << 20) / (sizeof(T) * kc), mr));
165 /// Columns of the packed B panel.
166 static constexpr idx nc =
167 std::max<idx>(nr, detail::round_down(NUM_K_GEMM_PANEL_BYTES / (sizeof(T) * kc), nr));
168 /// Elements of workspace the packed panels need at their largest.
169 static constexpr idx workspace = (mc * kc) + (kc * nc);
170};

172/// @brief Workspace elements `gemm` needs for a product of the given shape.
173///
174/// Never more than `gemm_config<T>::workspace`; smaller products need less.
175template <std::floating_point T>
176[[nodiscard]] constexpr idx gemm_workspace(idx m, idx n, idx k) noexcept {
177 using cfg = gemm_config<T>;
178 const idx kb = std::min(k, cfg::kc);
179 const idx mb = detail::round_up(std::min(m, cfg::mc), cfg::mr);
180 const idx nb = detail::round_up(std::min(n, cfg::nc), cfg::nr);
181 return (mb * kb) + (kb * nb);
182}

186/// @brief Pack `rows x depth` of A, scaled by alpha, as `mr`-row tiles.
187///
188/// Element (i, p) of the operand is `A[i*row_stride + p*col_stride]`, so a
189/// transposed operand is the same call with the strides swapped. Tile `t`
190/// occupies `[t*mr*depth, (t+1)*mr*depth)`, laid out `depth` groups of `mr`
191/// consecutive rows; rows past `rows` in the last tile are zero.
192template <std::floating_point T>
193inline void gemm_pack_a(T *NUM_K_RESTRICT dst, const T *NUM_K_RESTRICT A, idx row_stride,
194 idx col_stride, T alpha, idx rows, idx depth) noexcept {
195 constexpr idx mr = gemm_config<T>::mr;
196 for (idx i0 = 0; i0 < rows; i0 += mr) {
197 const idx valid = std::min(mr, rows - i0);
198 if (valid == mr && row_stride == 1) {
199 // Transposed operand: each of the tile's rows is contiguous in
200 // memory, so sweep `p` along it and scatter into the tile.
201 for (idx a = 0; a < mr; ++a) {
202 const T *NUM_K_RESTRICT row = A + i0 + a;
203 for (idx p = 0; p < depth; ++p) {
204 dst[(p * mr) + a] = alpha * row[p * col_stride];
205 }
206 }
207 dst += mr * depth;
208 } else if (valid == mr) {
209 for (idx p = 0; p < depth; ++p) {
210 const T *NUM_K_RESTRICT column = A + (i0 * row_stride) + (p * col_stride);
211 for (idx a = 0; a < mr; ++a) {
212 dst[a] = alpha * column[a * row_stride];
213 }
214 dst += mr;
215 }
216 } else {
217 for (idx p = 0; p < depth; ++p) {
218 const T *NUM_K_RESTRICT column = A + (i0 * row_stride) + (p * col_stride);
219 for (idx a = 0; a < valid; ++a) {
220 dst[a] = alpha * column[a * row_stride];
221 }
222 for (idx a = valid; a < mr; ++a) {
223 dst[a] = T(0);
224 }
225 dst += mr;
226 }
227 }
228 }
229}

231/// @brief Pack `depth x cols` of B as `nr`-column tiles.
232///
233/// Element (p, j) of the operand is `B[p*row_stride + j*col_stride]`. Tile `t`
234/// occupies `[t*nr*depth, (t+1)*nr*depth)`, laid out `depth` groups of `nr`
235/// consecutive columns; columns past `cols` in the last tile are zero.
236template <std::floating_point T>
237inline void gemm_pack_b(T *NUM_K_RESTRICT dst, const T *NUM_K_RESTRICT B, idx row_stride,
238 idx col_stride, idx depth, idx cols) noexcept {
239 constexpr idx nr = gemm_config<T>::nr;
240 for (idx j0 = 0; j0 < cols; j0 += nr) {
241 const idx valid = std::min(nr, cols - j0);
242 if (valid == nr && col_stride == 1) {
243 for (idx p = 0; p < depth; ++p) {
244 const T *NUM_K_RESTRICT row = B + (p * row_stride) + j0;
245 NUM_K_IVDEP
246 for (idx b = 0; b < nr; ++b) {
247 dst[b] = row[b];
248 }
249 dst += nr;
250 }
251 } else if (row_stride == 1) {
252 // Transposed operand: each tile column is contiguous, so sweep
253 // `p` along it; columns past `cols` are zero.
254 for (idx b = 0; b < nr; ++b) {
255 if (b < valid) {
256 const T *NUM_K_RESTRICT column = B + ((j0 + b) * col_stride);
257 for (idx p = 0; p < depth; ++p) {
258 dst[(p * nr) + b] = column[p];
259 }
260 } else {
261 for (idx p = 0; p < depth; ++p) {
262 dst[(p * nr) + b] = T(0);
263 }
264 }
265 }
266 dst += nr * depth;
267 } else {
268 for (idx p = 0; p < depth; ++p) {
269 const T *NUM_K_RESTRICT row = B + (p * row_stride) + (j0 * col_stride);
270 for (idx b = 0; b < valid; ++b) {
271 dst[b] = row[b * col_stride];
272 }
273 for (idx b = valid; b < nr; ++b) {
274 dst[b] = T(0);
275 }
276 dst += nr;
277 }
278 }
279 }

282/// @brief `C(0:rows, 0:cols) += Ap * Bp` for one packed `mr x nr` tile.
283///
284/// `Ap` is one A tile from `gemm_pack_a` and `Bp` one B tile from
285/// `gemm_pack_b`, both of inner length `depth`. `rows` and `cols` are the
286/// valid extent when the tile overhangs C; the padded lanes are computed and
287/// discarded, so the microkernel never sees an edge.
288template <std::floating_point T>
289NUM_K_AINLINE void gemm_micro(T *NUM_K_RESTRICT C, idx ldc, const T *NUM_K_RESTRICT Ap,
290 const T *NUM_K_RESTRICT Bp, idx depth, idx rows, idx cols) noexcept {
291 using cfg = gemm_config<T>;
292 constexpr idx mr = cfg::mr;
293 constexpr idx nr = cfg::nr;
294#if defined(NUM_K_VECTOR_EXT)
295 using vector = typename vector_of<T, NUM_K_VECTOR_BYTES>::type;
296 constexpr idx width = cfg::width;
297 constexpr idx nv = cfg::b_vectors;
298
299 vector acc[mr][nv]{};
300 for (idx p = 0; p < depth; ++p) {
301 vector b[nv];
302 for (idx v = 0; v < nv; ++v) {
303 std::memcpy(&b[v], Bp + (p * nr) + (v * width), sizeof(vector));
304 }
305 const T *NUM_K_RESTRICT a = Ap + (p * mr);
306 for (idx r = 0; r < mr; ++r) {
307 vector broadcast;
308 for (idx k = 0; k < width; ++k) {
309 broadcast[k] = a[r];
310 }
311 for (idx v = 0; v < nv; ++v) {
312 acc[r][v] += broadcast * b[v];
313 }
314 }
315 }
316
317 if (rows == mr && cols == nr) {
318 for (idx r = 0; r < mr; ++r) {
319 T *NUM_K_RESTRICT c = C + (r * ldc);
320 for (idx v = 0; v < nv; ++v) {
321 vector current;
322 std::memcpy(&current, c + (v * width), sizeof(vector));
323 current += acc[r][v];
324 std::memcpy(c + (v * width), &current, sizeof(vector));
325 }
326 }
327 return;
328 }
329 for (idx r = 0; r < rows; ++r) {
330 T *NUM_K_RESTRICT c = C + (r * ldc);
331 for (idx j = 0; j < cols; ++j) {
332 c[j] += acc[r][j / width][j % width];
333 }
334 }
335#else
336 T acc[mr][nr]{};
337 for (idx p = 0; p < depth; ++p) {
338 const T *NUM_K_RESTRICT b = Bp + (p * nr);
339 const T *NUM_K_RESTRICT a = Ap + (p * mr);
340 for (idx r = 0; r < mr; ++r) {
341 NUM_K_IVDEP
342 for (idx j = 0; j < nr; ++j) {
343 acc[r][j] += a[r] * b[j];
344 }
345 }
346 }
347 for (idx r = 0; r < rows; ++r) {
348 T *NUM_K_RESTRICT c = C + (r * ldc);
349 for (idx j = 0; j < cols; ++j) {
350 c[j] += acc[r][j];
351 }
352 }
353#endif
354}

371/// @brief The blocked product over strided operands, into caller workspace.
372///
373/// Element (i, p) of A is `A[i*a_rows + p*a_cols]` and element (p, j) of B is
374/// `B[p*b_rows + j*b_cols]`, so every transposition variant is this routine
375/// with the strides swapped. `work` holds at least `gemm_workspace<T>(m, n, k)`
376/// elements.
377template <std::floating_point T>
378inline void gemm_strided(T *NUM_K_RESTRICT C, idx ldc, const T *NUM_K_RESTRICT A, idx a_rows,
379 idx a_cols, const T *NUM_K_RESTRICT B, idx b_rows, idx b_cols, T alpha,
380 T beta, idx m, idx n, idx k, T *NUM_K_RESTRICT work) noexcept {
381 using cfg = gemm_config<T>;
382 gemm_scale_c(C, ldc, beta, m, n);
383 if (m == 0 || n == 0 || k == 0 || alpha == T(0)) {
384 return;
385 }
386
387 const idx kc = std::min(k, cfg::kc);
388 T *NUM_K_RESTRICT Ap = work;
389 T *NUM_K_RESTRICT Bp = work + (round_up(std::min(m, cfg::mc), cfg::mr) * kc);
390
391 for (idx jc = 0; jc < n; jc += cfg::nc) {
392 const idx nb = std::min(cfg::nc, n - jc);
393 for (idx pc = 0; pc < k; pc += cfg::kc) {
394 const idx kb = std::min(cfg::kc, k - pc);
395 gemm_pack_b(Bp, B + (pc * b_rows) + (jc * b_cols), b_rows, b_cols, kb, nb);
396 for (idx ic = 0; ic < m; ic += cfg::mc) {
397 const idx mb = std::min(cfg::mc, m - ic);
398 gemm_pack_a(Ap, A + (ic * a_rows) + (pc * a_cols), a_rows, a_cols, alpha, mb, kb);
399 for (idx jr = 0; jr < nb; jr += cfg::nr) {
400 const idx cols = std::min(cfg::nr, nb - jr);
401 for (idx ir = 0; ir < mb; ir += cfg::mr) {
402 const idx rows = std::min(cfg::mr, mb - ir);
403 gemm_micro(C + ((ic + ir) * ldc) + jc + jr, ldc, Ap + (ir * kb),
404 Bp + (jr * kb), kb, rows, cols);
405 }
406 }
407 }
408 }

412/// @brief Per-thread packing storage for the overloads that take no workspace.
413///
414/// The one place in the kernel with storage of its own. It is static and
415/// thread-local rather than heap-allocated, so the routine still never calls
416/// an allocator and is still safe to call from any thread; the cost is
417/// `gemm_config<T>::workspace` elements of TLS per thread that uses it.
418/// @brief Packing storage for the overloads that take no workspace.
419///
420/// A per-thread static buffer of `gemm_config<T>::workspace` elements: the
421/// one place in the kernel with storage of its own. It never calls an
422/// allocator and is safe to call from any thread; the cost is that many
423/// elements of TLS per thread that uses it. `get()` is the buffer.
424///
425/// One toolchain is the exception. GCC on macOS implements `thread_local`
426/// through emulated TLS, whose control block is a weak symbol in `.data`, and
427/// its Darwin backend then emits other local data in that section (among
428/// them every `std::source_location` record) as an offset from that weak
429/// symbol. Once the linker coalesces the weak copies, each such offset lands
430/// in one translation unit's data, and every caller's source location comes
431/// out as the first unit's. No `thread_local` in a header survives that, so
432/// on that toolchain alone the workspace is allocated per call and freed
433/// when this object goes out of scope.
434template <std::floating_point T>
435class gemm_scratch {
436 public:
437#if defined(__APPLE__) && defined(__GNUC__) && !defined(__clang__)
438 gemm_scratch() noexcept
439 : buffer_(static_cast<T *>(std::malloc(gemm_config<T>::workspace * sizeof(T)))) {}
440 ~gemm_scratch() { std::free(buffer_); }
441 gemm_scratch(const gemm_scratch &) = delete;
442 gemm_scratch &operator=(const gemm_scratch &) = delete;
443 [[nodiscard]] T *get() const noexcept { return buffer_; }
444
445 private:
446 T *buffer_;
447#else
448 [[nodiscard]] T *get() const noexcept {
449 alignas(64) static thread_local T buffer[gemm_config<T>::workspace];
450 return buffer;
451 }
452#endif
453};

488// Packed LU without row pivoting. Suitable for matrices whose structure
489// guarantees nonzero pivots, including the M-matrices used by ELSE.
490/// @brief In-place LU without row pivoting. Returns false if a pivot fell below tolerance.
491template <std::floating_point T>
492[[nodiscard]] inline bool lu_no_pivot(T *A, idx n) noexcept {
493 constexpr T tolerance = T(1e-15);
494 bool nonsingular = true;
495 for (idx k = 0; k < n; ++k) {
496 if (std::abs(A[k * n + k]) < tolerance) {
497 nonsingular = false;
498 continue;
499 }
500 const T inverse_pivot = T(1) / A[k * n + k];
501 for (idx i = k + 1; i < n; ++i) {
502 A[i * n + k] *= inverse_pivot;
503 const T multiplier = A[i * n + k];
504 for (idx j = k + 1; j < n; ++j)
505 A[i * n + j] -= multiplier * A[k * n + j];
506 }
507 }
508 return nonsingular;
509}

511/// @brief Solves several right-hand sides from an unpivoted LU factor.
512template <std::floating_point T>
513inline void lu_no_pivot_solve_multiple(T *X, const T *LU, idx n, idx columns) noexcept {
514 for (idx i = 0; i < n; ++i)
515 for (idx j = 0; j < i; ++j)
516 for (idx c = 0; c < columns; ++c)
517 X[i * columns + c] -= LU[i * n + j] * X[j * columns + c];
518 for (idx i = n; i-- > 0;) {
519 for (idx j = i + 1; j < n; ++j)
520 for (idx c = 0; c < columns; ++c)
521 X[i * columns + c] -= LU[i * n + j] * X[j * columns + c];
522 for (idx c = 0; c < columns; ++c)
523 X[i * columns + c] /= LU[i * n + i];
524 }

528template <std::floating_point T>
529inline void lu_no_pivot_solve_transpose_multiple(T *X, const T *LU, idx n, idx columns) noexcept {

534 for (idx c = 0; c < columns; ++c)

543/// @brief Symmetric rank-k update of the lower triangle, `C <- alpha*A*A^T + beta*C`.
544///
545/// `A` is `rows x columns`; `C` is a row-major `rows x rows` matrix. Only
546/// `C(i,j)` for `j <= i` is touched, which is the update required by lower
547/// Cholesky and avoids doing work for the implied symmetric half.
548///
549/// Column strips of the triangle: each diagonal block is formed entry by entry
550/// so nothing above the diagonal is written, and the rectangle beneath it is
551/// one `gemm` with `A^T` as the right operand, so the bulk of the work runs at
552/// `gemm` speed.
553namespace detail {
554
555/// @brief `syrk_lower` over column strips of width `strip`.
556///
557/// Each diagonal block is formed entry by entry when `strip` is the narrow
558/// base width, else by recursion at the base width; the rectangle beneath it
559/// is one `gemm`. Two levels because the base width is what keeps the
560/// entry-wise part negligible, while wide strips are what let `gemm` reuse
561/// its packed A slab.
562template <std::floating_point T>
563inline void syrk_lower_strips(T *NUM_K_RESTRICT C, idx ldc, const T *NUM_K_RESTRICT A, idx lda,
564 T alpha, T beta, idx rows, idx columns, idx strip,
565 T *NUM_K_RESTRICT work) noexcept {
566 constexpr idx base = 4 * gemm_config<T>::nr;
567 for (idx j0 = 0; j0 < rows; j0 += strip) {
568 const idx j1 = std::min(rows, j0 + strip);
569 if (strip > base) {
570 syrk_lower_strips(C + (j0 * ldc) + j0, ldc, A + (j0 * lda), lda, alpha, beta, j1 - j0,
571 columns, base, work);
572 } else {
573 for (idx i = j0; i < j1; ++i) {
574 T *NUM_K_RESTRICT c_row = C + (i * ldc);
575 const T *NUM_K_RESTRICT a_i = A + (i * lda);
576 for (idx j = j0; j <= i; ++j) {
577 const T *NUM_K_RESTRICT a_j = A + (j * lda);
578 const T sum = reduce<T>(columns, [a_i, a_j](idx p) { return a_i[p] * a_j[p]; });
579 c_row[j] = (alpha * sum) + (beta * c_row[j]);
580 }
581 }
582 }
583 if (j1 < rows) {
584 gemm_strided(C + (j1 * ldc) + j0, ldc, A + (j1 * lda), lda, idx{1}, A + (j0 * lda),
585 idx{1}, lda, alpha, beta, rows - j1, j1 - j0, columns, work);
586 }
587 }
588}

590} // namespace detail

633/// @brief Block linear combination \f$y \leftarrow \alpha Vc + \beta y\f$.
634template <std::floating_point T>

636 const T *NUM_K_RESTRICT coefficients, T alpha, T beta, idx rows,
637 idx columns) noexcept {

646/// @brief Modified Gram--Schmidt against row-major basis columns.
647///
648/// This intentionally retains sequential projection/update ordering; callers
649/// requiring the faster classical block operation use `project_columns` followed
650/// by `combine_columns` and accept its different stability contract.
651template <std::floating_point T>
652inline void mgs_columns(T *NUM_K_RESTRICT v, const T *NUM_K_RESTRICT basis, idx ldb, idx rows,
653 idx columns, T *coefficients = nullptr) noexcept {
654 for (idx column = 0; column < columns; ++column) {
655 T projection = T(0);
656 for (idx row = 0; row < rows; ++row) {
657 projection += basis[(row * ldb) + column] * v[row];
658 }
659 if (coefficients != nullptr) {
660 coefficients[column] = projection;
661 }
662 for (idx row = 0; row < rows; ++row) {
663 v[row] -= projection * basis[(row * ldb) + column];
664 }
665 }

747/// @brief Rows or columns of the triangle solved per unblocked step in the
748/// `trsm` routines. Everything outside the diagonal block is a `gemm`.
749inline constexpr idx trsm_block = 64;
750
751/// @brief Unblocked `L*X=B` on one diagonal block.
752///
753/// Element (i, j) of the triangle is `L[i*row_stride + j*col_stride]`, so a
754/// transposed upper factor is the same call with the strides swapped; `unit`
755/// takes the diagonal as one without reading it.
756template <std::floating_point T>
757inline void trsm_lower_block(T *NUM_K_RESTRICT X, idx ldx, const T *NUM_K_RESTRICT L,
758 idx row_stride, idx col_stride, idx n, idx nrhs, bool unit) noexcept {
759 for (idx i = 0; i < n; ++i) {
760 T *NUM_K_RESTRICT x_row = X + (i * ldx);
761 for (idx j = 0; j < i; ++j) {
762 const T lij = L[(i * row_stride) + (j * col_stride)];
763 const T *NUM_K_RESTRICT solved_row = X + (j * ldx);
764 NUM_K_IVDEP
765 for (idx r = 0; r < nrhs; ++r) {
766 x_row[r] -= lij * solved_row[r];
767 }
768 }
769 if (!unit) {
770 const T inv_diag = T(1) / L[(i * row_stride) + (i * col_stride)];
771 NUM_K_IVDEP
772 for (idx r = 0; r < nrhs; ++r) {
773 x_row[r] *= inv_diag;
774 }
775 }
776 }
777}

779/// @brief Unblocked `L^T*X=B` on one diagonal block, strides as for `trsm_lower_block`.
780template <std::floating_point T>
781inline void trsm_lower_transpose_block(T *NUM_K_RESTRICT X, idx ldx, const T *NUM_K_RESTRICT L,
782 idx row_stride, idx col_stride, idx n, idx nrhs,
783 bool unit) noexcept {
784 for (idx i = n; i-- > 0;) {
785 T *NUM_K_RESTRICT row = X + (i * ldx);
786 for (idx k = i + 1; k < n; ++k) {
787 const T lki = L[(k * row_stride) + (i * col_stride)];
788 const T *NUM_K_RESTRICT solved = X + (k * ldx);
789 NUM_K_IVDEP
790 for (idx r = 0; r < nrhs; ++r) {
791 row[r] -= lki * solved[r];
792 }
793 }
794 if (!unit) {
795 const T inv = T(1) / L[(i * row_stride) + (i * col_stride)];
796 NUM_K_IVDEP
797 for (idx r = 0; r < nrhs; ++r) {
798 row[r] *= inv;
799 }
800 }
801 }
802}

804/// @brief Blocked `L*X=B`: solve a diagonal block, then subtract its
805/// contribution from every row below with one `gemm`.
806template <std::floating_point T>
807inline void trsm_lower_blocked(T *NUM_K_RESTRICT X, idx ldx, const T *NUM_K_RESTRICT L,
808 idx row_stride, idx col_stride, idx n, idx nrhs,
809 bool unit) noexcept {

811 T *NUM_K_RESTRICT work = scratch.get();
812 for (idx i0 = 0; i0 < n; i0 += trsm_block) {
813 const idx i1 = std::min(n, i0 + trsm_block);
814 trsm_lower_block(X + (i0 * ldx), ldx, L + (i0 * row_stride) + (i0 * col_stride), row_stride,
815 col_stride, i1 - i0, nrhs, unit);
816 if (i1 < n) {
817 gemm_strided(X + (i1 * ldx), ldx, L + (i1 * row_stride) + (i0 * col_stride), row_stride,
818 col_stride, X + (i0 * ldx), ldx, idx{1}, T(-1), T(1), n - i1, nrhs,
819 i1 - i0, work);
820 }
821 }

824/// @brief Blocked `L^T*X=B`, from the last diagonal block upward; the rows
825/// above each block are updated with `L^T` expressed as a stride swap.
826template <std::floating_point T>
827inline void trsm_lower_transpose_blocked(T *NUM_K_RESTRICT X, idx ldx, const T *NUM_K_RESTRICT L,
828 idx row_stride, idx col_stride, idx n, idx nrhs,
829 bool unit) noexcept {

832 idx i1 = n;

835 trsm_lower_transpose_block(X + (i0 * ldx), ldx, L + (i0 * row_stride) + (i0 * col_stride),
836 row_stride, col_stride, i1 - i0, nrhs, unit);
837 if (i0 > 0) {
838 gemm_strided(X, ldx, L + (i0 * row_stride), col_stride, row_stride, X + (i0 * ldx), ldx,
839 idx{1}, T(-1), T(1), i0, nrhs, i1 - i0, work);
840 }
841 i1 = i0;

845/// @brief Rows solved together by `trsm_lower_transpose_right_block`.
846inline constexpr idx trsm_row_batch = 16;
847
848/// @brief Unblocked `X*L^T=B` on one diagonal block of at most `trsm_block` columns.
849///
850/// Each row's solve is a chain of dependent updates, so a row at a time runs
851/// scalar. Instead a batch of rows is copied into a column-major tile on the
852/// stack, where the update of column `j` by column `k` is a contiguous axpy
853/// across the batch, and copied back once solved.
854template <std::floating_point T>
855inline void trsm_lower_transpose_right_block(T *NUM_K_RESTRICT X, idx ldx,
856 const T *NUM_K_RESTRICT L, idx ldl, idx rows,
857 idx n) noexcept {
858 constexpr idx batch = trsm_row_batch;
859 T tile[trsm_block][batch];
860 for (idx r0 = 0; r0 < rows; r0 += batch) {
861 const idx count = std::min(batch, rows - r0);
862 for (idx r = 0; r < count; ++r) {
863 const T *NUM_K_RESTRICT row = X + ((r0 + r) * ldx);
864 for (idx j = 0; j < n; ++j) {
865 tile[j][r] = row[j];
866 }
867 }
868 for (idx j = 0; j < n; ++j) {
869 // Column j accumulates in registers across the k loop; the tile
870 // row it came from is only touched at the ends.
871 T acc[batch];
872 for (idx r = 0; r < batch; ++r) {
873 acc[r] = tile[j][r];
874 }
875 const T *NUM_K_RESTRICT l_row = L + (j * ldl);
876 for (idx k = 0; k < j; ++k) {
877 const T ljk = l_row[k];
878 NUM_K_IVDEP
879 for (idx r = 0; r < batch; ++r) {
880 acc[r] -= ljk * tile[k][r];
881 }
882 }
883 const T inv = T(1) / l_row[j];
884 for (idx r = 0; r < batch; ++r) {
885 tile[j][r] = acc[r] * inv;
886 }
887 }
888 for (idx r = 0; r < count; ++r) {
889 T *NUM_K_RESTRICT row = X + ((r0 + r) * ldx);
890 for (idx j = 0; j < n; ++j) {
891 row[j] = tile[j][r];
892 }
893 }
894 }
895}
