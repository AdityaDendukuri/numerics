35/// @brief Dense matrix-vector multiplication \f$\mathbf{y} \leftarrow A \mathbf{x}\f$ for \f$A \in
36/// \mathbb{R}^{m \times n}\f$.
37template <std::floating_point T>
38NUM_K_AINLINE void matvec(T *NUM_K_RESTRICT y, const T *NUM_K_RESTRICT A, const T *NUM_K_RESTRICT x,
39 idx m, idx n) noexcept {
40 for (idx i = 0; i < m; ++i) {
41 const T *row = A + (i * n);
42 y[i] = detail::reduce<T>(n, [row, x](idx j) { return row[j] * x[j]; });
43 }
44}

46/// @brief Transposed dense matrix-vector multiplication \f$\mathbf{y} \leftarrow A^T \mathbf{x}\f$
47/// for \f$A \in \mathbb{R}^{m \times n}\f$.
48template <std::floating_point T>
49NUM_K_AINLINE void matvec_transpose(T *NUM_K_RESTRICT y, const T *NUM_K_RESTRICT A,

71/// @brief Applies a Givens rotation to columns \f$p\f$ and \f$q\f$ in place.
72template <std::floating_point T>
73NUM_K_AINLINE void rotate_columns(T *NUM_K_RESTRICT A, idx lda, idx rows, idx p, idx q, T c,
74 T s) noexcept {
75 for (idx i = 0; i < rows; ++i) {
76 T *row = A + (i * lda);
77 const T ap = row[p], aq = row[q];
78 row[p] = (c * ap) - (s * aq);
79 row[q] = (s * ap) + (c * aq);
80 }
81}

83// Dense matrix product.
84//
85// The shape of this is forced by the arithmetic intensity of the operation. A
86// product does O(n^3) work over O(n^2) data, so it is compute-bound in
87// principle, but the textbook i-k-j triple loop does not get anywhere near the
88// machine's peak: each fused multiply-add reads a fresh element of C from
89// memory and writes it straight back, so the loop runs at the rate the store
90// unit and the L1 cache can retire traffic, not at the rate the FMA units can
91// issue. On this tree the plain version sustained ~13 GFLOP/s.
92//
93// Two nested levels of blocking fix that, and nothing else is needed.
94//
95// Register tile (`mr` x `nr`): the innermost loop holds a small block of C in
96// vector registers across the entire k sweep. Each element of A loaded is
97// reused across `nr` columns and each element of B across `mr` rows, so one
98// pair of loads feeds `mr*nr` FMAs instead of one. The tile is deliberately
99// sized to about half the architectural vector register file: large enough to
100// hide FMA latency, small enough that the accumulators are never spilled --
101// a spilled tile is slower than no tile at all.
102//
103// Cache panel (`kc`): the k range is cut so the slice of B the tile loop
104// sweeps (kc x n) stays resident while every row block streams past it.
105// Without this, a large product re-reads B from DRAM once per row block.
106//
107// Both bounds come from the target's own properties, not from a tuning
108// parameter, so there is nothing for a caller to get wrong. Together they take
109// the same computation to ~2.5x the plain loop, and the summation order per
110// output element is unchanged (still ascending in p), so results are
111// bit-identical to the naive triple loop.

115/// @brief Rows in `gemm`'s register tile.
116inline constexpr idx gemm_tile_rows = 4;
117
118/// @brief Columns in `gemm`'s register tile: whole vectors, filling half the
119/// register file so the accumulators survive the inner loop without spilling.
120template <std::floating_point T>
121inline constexpr idx gemm_tile_cols =
122 (NUM_K_VECTOR_REGISTERS / 2 / gemm_tile_rows) * (NUM_K_VECTOR_BYTES / sizeof(T));
123
124/// @brief Bytes of the B panel to keep resident across one row sweep.
125///
126/// One mid-size L2 is the target. Measured on this tree at n = 512/1024/2048,
127/// a 1 MiB budget gave 34.5/31.0/24.9 GFLOP/s against 30.8/27.7/20.1 at 256 KiB
128/// and 35.1/28.4/23.4 at 4 MiB -- the smaller budget cuts reuse, the larger one
129/// stops fitting.
130inline constexpr idx gemm_panel_bytes = idx{1} << 20;
131
132/// @brief Accumulate `C += alpha*A*B` over `p in [p0, p1)`, register-tiled.
133template <std::floating_point T>
134inline void gemm_panel(T *NUM_K_RESTRICT C, idx ldc, const T *NUM_K_RESTRICT A, idx lda,
135 const T *NUM_K_RESTRICT B, idx ldb, T alpha, idx m, idx n, idx p0,
136 idx p1) noexcept {
137 constexpr idx mr = gemm_tile_rows;
138 constexpr idx nr = gemm_tile_cols<T>;
139
140 idx i = 0;
141 for (; i + mr <= m; i += mr) {
142 idx j = 0;
143 for (; j + nr <= n; j += nr) {
144 T acc[mr][nr];
145 for (idx a = 0; a < mr; ++a) {
146 NUM_K_IVDEP
147 for (idx b = 0; b < nr; ++b) {
148 acc[a][b] = C[((i + a) * ldc) + j + b];
149 }
150 }
151 for (idx p = p0; p < p1; ++p) {
152 const T *NUM_K_RESTRICT b_row = B + (p * ldb) + j;
153 for (idx a = 0; a < mr; ++a) {
154 const T a_val = alpha * A[((i + a) * lda) + p];
155 NUM_K_IVDEP
156 for (idx b = 0; b < nr; ++b) {
157 acc[a][b] += a_val * b_row[b];
158 }
159 }
160 }
161 for (idx a = 0; a < mr; ++a) {
162 NUM_K_IVDEP
163 for (idx b = 0; b < nr; ++b) {
164 C[((i + a) * ldc) + j + b] = acc[a][b];
165 }
166 }
167 }
168 // Columns past the last whole tile: same row block, one column at a time.
169 for (; j < n; ++j) {
170 for (idx a = 0; a < mr; ++a) {
171 T sum = C[((i + a) * ldc) + j];
172 for (idx p = p0; p < p1; ++p) {
173 sum += alpha * A[((i + a) * lda) + p] * B[(p * ldb) + j];
174 }
175 C[((i + a) * ldc) + j] = sum;
176 }
177 }
178 }
179 // Rows past the last whole tile: the plain i-k-j loop, still vectorized
180 // along j but without register reuse. At most `mr - 1` rows land here.
181 for (; i < m; ++i) {
182 T *NUM_K_RESTRICT c_row = C + (i * ldc);
183 for (idx p = p0; p < p1; ++p) {
184 const T a_val = alpha * A[(i * lda) + p];
185 const T *NUM_K_RESTRICT b_row = B + (p * ldb);
186 NUM_K_IVDEP
187 for (idx j = 0; j < n; ++j) {
188 c_row[j] += a_val * b_row[j];
189 }
190 }
191 }
192}

196/// @brief Dense matrix product `C <- alpha*A*B + beta*C`, with row strides.
197///
198/// Register-tiled and cache-panelled (see above). A, B, and C must not overlap.
199template <std::floating_point T>
200inline void gemm(T *NUM_K_RESTRICT C, idx ldc, const T *NUM_K_RESTRICT A, idx lda,
201 const T *NUM_K_RESTRICT B, idx ldb, T alpha, T beta, idx m, idx n,
202 idx k) noexcept {
203 if (beta == T(0)) {
204 for (idx i = 0; i < m; ++i) {
205 std::fill_n(C + (i * ldc), n, T(0));
206 }
207 } else if (beta != T(1)) {
208 for (idx i = 0; i < m; ++i) {
209 T *NUM_K_RESTRICT c_row = C + (i * ldc);
210 NUM_K_IVDEP
211 for (idx j = 0; j < n; ++j) {
212 c_row[j] *= beta;
213 }
214 }
215 }
216
217 const idx kc = std::max<idx>(1, detail::gemm_panel_bytes / std::max<idx>(1, n * sizeof(T)));
218 for (idx p0 = 0; p0 < k; p0 += kc) {
219 detail::gemm_panel(C, ldc, A, lda, B, ldb, alpha, m, n, p0, std::min(p0 + kc, k));
220 }
221}

230// Packed LU without row pivoting. Suitable for matrices whose structure
231// guarantees nonzero pivots, including the M-matrices used by ELSE.
232/// @brief In-place LU without row pivoting. Returns false if a pivot fell below tolerance.
233template <std::floating_point T>
234[[nodiscard]] inline bool lu_no_pivot(T *A, idx n) noexcept {
235 constexpr T tolerance = T(1e-15);
236 bool nonsingular = true;
237 for (idx k = 0; k < n; ++k) {
238 if (std::abs(A[k * n + k]) < tolerance) {
239 nonsingular = false;
240 continue;
241 }
242 const T inverse_pivot = T(1) / A[k * n + k];
243 for (idx i = k + 1; i < n; ++i) {
244 A[i * n + k] *= inverse_pivot;
245 const T multiplier = A[i * n + k];
246 for (idx j = k + 1; j < n; ++j)
247 A[i * n + j] -= multiplier * A[k * n + j];
248 }
249 }
250 return nonsingular;
251}

253/// @brief Solves several right-hand sides from an unpivoted LU factor.
254template <std::floating_point T>
255inline void lu_no_pivot_solve_multiple(T *X, const T *LU, idx n, idx columns) noexcept {
256 for (idx i = 0; i < n; ++i)
257 for (idx j = 0; j < i; ++j)
258 for (idx c = 0; c < columns; ++c)
259 X[i * columns + c] -= LU[i * n + j] * X[j * columns + c];
260 for (idx i = n; i-- > 0;) {
261 for (idx j = i + 1; j < n; ++j)
262 for (idx c = 0; c < columns; ++c)
263 X[i * columns + c] -= LU[i * n + j] * X[j * columns + c];
264 for (idx c = 0; c < columns; ++c)
265 X[i * columns + c] /= LU[i * n + i];
266 }
267}

269/// @brief Solves \f$A^T x = b\f$ for several right-hand sides from an unpivoted LU factor.
270template <std::floating_point T>
271inline void lu_no_pivot_solve_transpose_multiple(T *X, const T *LU, idx n, idx columns) noexcept {
272 for (idx i = 0; i < n; ++i) {
273 for (idx j = 0; j < i; ++j)
274 for (idx c = 0; c < columns; ++c)
275 X[i * columns + c] -= LU[j * n + i] * X[j * columns + c];
276 for (idx c = 0; c < columns; ++c)
277 X[i * columns + c] /= LU[i * n + i];
278 }
279 for (idx i = n; i-- > 0;)
280 for (idx j = i + 1; j < n; ++j)
281 for (idx c = 0; c < columns; ++c)

283}

285/// @brief Symmetric rank-k update of the lower triangle, `C <- alpha*A*A^T + beta*C`.
286///
287/// `A` is `rows x columns`; `C` is a row-major `rows x rows` matrix. Only
288/// `C(i,j)` for `j <= i` is touched, which is the update required by lower
289/// Cholesky and avoids doing work for the implied symmetric half.
290template <std::floating_point T>
291inline void syrk_lower(T *NUM_K_RESTRICT C, idx ldc, const T *NUM_K_RESTRICT A, idx lda, T alpha,
292 T beta, idx rows, idx columns) noexcept {
293 for (idx i = 0; i < rows; ++i) {
294 T *NUM_K_RESTRICT c_row = C + (i * ldc);
295 for (idx j = 0; j <= i; ++j) {
296 const T *NUM_K_RESTRICT a_i = A + (i * lda);
297 const T *NUM_K_RESTRICT a_j = A + (j * lda);
298 const T sum = detail::reduce<T>(columns, [a_i, a_j](idx p) { return a_i[p] * a_j[p]; });
299 c_row[j] = (alpha * sum) + (beta * c_row[j]);
300 }
301 }
302}

304template <std::floating_point T>
305inline void syrk_lower(T *NUM_K_RESTRICT C, const T *NUM_K_RESTRICT A, T alpha, T beta, idx rows,
306 idx columns) noexcept {
307 syrk_lower(C, rows, A, columns, alpha, beta, rows, columns);

310/// @brief Dense product `C <- alpha*A^T*B + beta*C`, with row strides.
311///
312/// A is `rows x a_cols`, B is `rows x b_cols`, and C is `a_cols x b_cols`.
313template <std::floating_point T>
314inline void gemm_transpose_left(T *NUM_K_RESTRICT C, idx ldc, const T *NUM_K_RESTRICT A, idx lda,
315 const T *NUM_K_RESTRICT B, idx ldb, T alpha, T beta, idx rows,
316 idx a_cols, idx b_cols) noexcept {
317 for (idx i = 0; i < a_cols; ++i) {
318 T *NUM_K_RESTRICT c_row = C + (i * ldc);
319 if (beta == T(0)) {
320 fill(c_row, T(0), b_cols);
321 } else if (beta != T(1)) {
322 scale(c_row, beta, b_cols);
323 }
324 for (idx r = 0; r < rows; ++r) {
325 const T a = alpha * A[(r * lda) + i];
326 const T *NUM_K_RESTRICT b_row = B + (r * ldb);
327 NUM_K_IVDEP
328 for (idx j = 0; j < b_cols; ++j) {
329 c_row[j] += a * b_row[j];
330 }
331 }
332 }
333}

351template <std::floating_point T>

353 const T *NUM_K_RESTRICT coefficients, T alpha, T beta, idx rows,
354 idx columns) noexcept {

363/// @brief Modified Gram--Schmidt against row-major basis columns.
364///
365/// This intentionally retains sequential projection/update ordering; callers
366/// requiring the faster classical block operation use `project_columns` followed
367/// by `combine_columns` and accept its different stability contract.
368template <std::floating_point T>
369inline void mgs_columns(T *NUM_K_RESTRICT v, const T *NUM_K_RESTRICT basis, idx ldb, idx rows,
370 idx columns, T *coefficients = nullptr) noexcept {
371 for (idx column = 0; column < columns; ++column) {
372 T projection = T(0);
373 for (idx row = 0; row < rows; ++row) {
374 projection += basis[(row * ldb) + column] * v[row];
375 }
376 if (coefficients != nullptr) {
377 coefficients[column] = projection;
378 }
379 for (idx row = 0; row < rows; ++row) {
380 v[row] -= projection * basis[(row * ldb) + column];
381 }
382 }
383}

419/// @brief Forward substitution solving lower triangular system \f$L \mathbf{x} = \mathbf{b}\f$
420/// (\f$L \in \mathbb{R}^{n \times n}\f$).
421template <std::floating_point T>
422NUM_K_AINLINE void trsv_lower(T *NUM_K_RESTRICT x, const T *NUM_K_RESTRICT L,
423 const T *NUM_K_RESTRICT b, idx n) noexcept {
424 for (idx i = 0; i < n; ++i) {
425 T s = b[i];
426 const T *row = L + (i * n);
427 for (idx j = 0; j < i; ++j) {
428 s -= row[j] * x[j];
429 }
430 x[i] = s / row[i];
431 }
432}

462/// @brief Solves `L*X=B` in-place for row-major `X` with `nrhs` columns.
463///
464/// Vectorization is across independent right-hand sides, making this the
465/// natural primitive for block methods and batches sharing one factor.
466template <std::floating_point T>
467inline void trsm_lower_inplace(T *NUM_K_RESTRICT X, idx ldx, const T *NUM_K_RESTRICT L, idx n,
468 idx nrhs) noexcept {
469 for (idx i = 0; i < n; ++i) {
470 T *NUM_K_RESTRICT x_row = X + (i * ldx);
471 for (idx j = 0; j < i; ++j) {
472 const T lij = L[(i * n) + j];
473 const T *NUM_K_RESTRICT solved_row = X + (j * ldx);
474 NUM_K_IVDEP
475 for (idx r = 0; r < nrhs; ++r) {
476 x_row[r] -= lij * solved_row[r];
477 }
478 }
479 const T inv_diag = T(1) / L[(i * n) + i];
480 NUM_K_IVDEP
481 for (idx r = 0; r < nrhs; ++r) {
482 x_row[r] *= inv_diag;
483 }
484 }
485}

507/// @brief Solves `X*L^T=B` in-place for row-major panel rows.
508///
509/// `X` is `rows x n`, `L` is lower triangular `n x n`; each row is an
510/// independent right-side solve. This is the panel solve in blocked
511/// Cholesky: `L21 <- A21 * L11^{-T}`.
512template <std::floating_point T>
513inline void trsm_lower_transpose_right_inplace(T *NUM_K_RESTRICT X, idx ldx,
514 const T *NUM_K_RESTRICT L, idx ldl, idx rows,
515 idx n) noexcept {
516 for (idx r = 0; r < rows; ++r) {
517 T *NUM_K_RESTRICT row = X + (r * ldx);
518 for (idx j = 0; j < n; ++j) {
519 T s = row[j];
520 for (idx k = 0; k < j; ++k) {
521 s -= row[k] * L[(j * ldl) + k];
522 }
523 row[j] = s / L[(j * ldl) + j];
