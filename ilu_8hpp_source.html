1/// @file linear/solvers/ilu.hpp
2/// @brief Incomplete LU preconditioner with zero fill-in, ILU(0).
3///
4/// The workhorse preconditioner for nonsymmetric sparse systems, and the reason
5/// GMRES is usable on them at all. It factors \f$A \approx LU\f$ subject to the
6/// constraint that \f$L\f$ and \f$U\f$ occupy exactly \f$A\f$'s own sparsity
7/// pattern — every entry the elimination would create outside that pattern is
8/// discarded. So the factors cost the same memory as the matrix, the setup is a
9/// single sweep, and applying the preconditioner is two sparse triangular
10/// solves.
11///
12/// ### What it does not promise
13///
14/// Dropping fill-in is a genuine approximation, and how good an approximation
15/// depends on the matrix. ILU(0) is reliable on diagonally dominant and M-matrix
16/// systems, which covers most convection-diffusion discretizations. It can
17/// break down outright on an indefinite or strongly non-diagonally-dominant
18/// matrix — a pivot reaches zero — and the constructor reports that as an error
19/// rather than producing a preconditioner that quietly amplifies the residual.
20/// It is also *not* symmetric even when \f$A\f$ is, so it carries no
21/// self-adjointness claim and must not be handed to PCG or MINRES; those need
22/// `approx_chol_preconditioner`, `chebyshev_preconditioner`, or Jacobi.
23///
24/// All arithmetic lives in `kernel/sparse.hpp` (`ilu0_factor`, `csr_lu_solve`);
25/// this class owns the storage, the evidence, and the error reporting.

38/// @brief ILU(0) preconditioner: \f$M^{-1} r\f$ by two sparse triangular solves.
39///
40/// Holds its own copy of the factored values, so the source matrix need not
41/// outlive it. Allocates nothing after construction.
42class ilu0_preconditioner final {
43 public:
44 using domain_type = vec;
45 using codomain_type = vec;
46 // Deliberately no property claims. An incomplete LU is not self-adjoint even
47 // for a symmetric A, so PCG and MINRES will not accept it -- which is the
48 // correct outcome, enforced by the type system rather than by documentation.
49
50 /// @brief Factor `A` in place over its own pattern.
51 /// @throws std::invalid_argument If `A` is not square, or a row has no diagonal entry.
52 /// @throws std::runtime_error If the factorization reaches a zero or non-finite pivot.
53 explicit ilu0_preconditioner(const spmat &A)
54 : n_(A.n_rows()), values_(A.values(), A.values() + A.nnz()),
55 col_idx_(A.col_idx(), A.col_idx() + A.nnz()),
56 row_ptr_(A.row_ptr(), A.row_ptr() + A.n_rows() + 1), diagonal_(A.n_rows(), 0),
57 scratch_(A.n_rows(), 0) {
58 if (A.n_rows() != A.n_cols()) {
59 throw std::invalid_argument("ilu0: matrix must be square");
60 }
61 if (n_ == 0) {
62 return;
63 }
64 if (!kernel::csr_diagonal_positions(diagonal_.data(), row_ptr_.data(),
65 col_idx_.data(), n_)) {
66 throw std::invalid_argument(
67 "ilu0: every row must have a stored diagonal entry, including explicit zeros");
68 }
69 if (!kernel::ilu0_factor(values_.data(), row_ptr_.data(), col_idx_.data(),
70 diagonal_.data(), scratch_.data(), n_)) {
71 throw std::runtime_error(
72 "ilu0: zero or non-finite pivot; the matrix is too far from diagonally dominant "
73 "for zero fill-in");
74 }
75 }
76
77 [[nodiscard]] idx rows() const noexcept { return n_; }
78 [[nodiscard]] idx cols() const noexcept { return n_; }
79 [[nodiscard]] idx nnz() const noexcept { return values_.size(); }
80
81 /// @brief Apply \f$z \leftarrow (LU)^{-1} r\f$.
82 void apply(const vec &r, vec &z) const {
83 if (r.size() != n_) {
84 throw std::invalid_argument("ilu0: dimension mismatch");
85 }
86 if (z.size() != n_) {
87 z = vec(n_, 0.0);
88 }
89 kernel::csr_lu_solve(z.data(), values_.data(), row_ptr_.data(), col_idx_.data(),
90 diagonal_.data(), r.data(), n_);
91 }
92
93 private:
94 idx n_ = 0;
95 array<real> values_;
96 array<idx> col_idx_;
97 array<idx> row_ptr_;
98 array<idx> diagonal_;
99 array<idx> scratch_;
100};

102/// @brief Build an ILU(0) preconditioner for a sparse matrix.
103[[nodiscard]] inline ilu0_preconditioner make_ilu0_preconditioner(const spmat &A) {
104 return ilu0_preconditioner(A);
105}
