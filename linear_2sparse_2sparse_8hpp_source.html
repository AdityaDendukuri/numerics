57 /// Operator protocol application: y <- A * x
58 template <class X = vec, class Y = vec>
59 void apply(const X &x, Y &y) const {
60 if constexpr (std::is_same_v<X, vec> && std::is_same_v<Y, vec>) {
61 sparse_matvec(*this, x, y);
62 } else {
63 for (idx i = 0; i < n_rows_; ++i) {
64 real sum = 0.0;
65 for (idx k = row_ptr_[i]; k < row_ptr_[i + 1]; ++k) {
66 sum += vals_[k] * x[col_idx_[k]];
67 }
68 y[i] = sum;
69 }
70 }
71 }
72
73 private:
74 idx n_rows_ = 0, n_cols_ = 0;
75 array<real> vals_;
76 array<idx> col_idx_;
77 array<idx> row_ptr_; // size n_rows_ + 1
78};

130inline spmat spmat::from_triplets(idx n_rows, idx n_cols, const array<idx> &rows,
131 const array<idx> &cols,
132 const array<real> &vals) {
133 if (rows.size() != cols.size() || rows.size() != vals.size()) {
134 throw std::invalid_argument("spmat::from_triplets: inconsistent input sizes");
135 }
136
137 // Count entries per row
138 array<idx> row_count(n_rows, 0);
139 for (idx k = 0; k < rows.size(); ++k) {
140 if (rows[k] >= n_rows || cols[k] >= n_cols) {
141 throw std::out_of_range("spmat::from_triplets: index out of range");
142 }
143 ++row_count[rows[k]];
144 }
145
146 // Build row_ptr
147 array<idx> row_ptr(n_rows + 1, 0);
148 for (idx i = 0; i < n_rows; ++i) {
149 row_ptr[i + 1] = row_ptr[i] + row_count[i];
150 }
151
152 idx nnz = row_ptr[n_rows];
153 array<real> out_vals(nnz, 0.0);
154 array<idx> out_col(nnz);
155
156 // Fill entries (stable insertion within each row)
157 array<idx> fill_pos = row_ptr;
158 for (idx k = 0; k < rows.size(); ++k) {
159 idx pos = fill_pos[rows[k]]++;
160 out_col[pos] = cols[k];
161 out_vals[pos] = vals[k];
162 }
163
164 // Sort each row by column and sum duplicates
165 for (idx i = 0; i < n_rows; ++i) {
166 idx start = row_ptr[i], end = row_ptr[i + 1];
167 // Sort by column index
168 array<idx> order(end - start);
169 std::iota(order.begin(), order.end(), 0);
170 std::sort(order.begin(), order.end(),
171 [&](idx a, idx b) { return out_col[start + a] < out_col[start + b]; });
172
173 array<real> sv(end - start);
174 array<idx> sc(end - start);
175 for (idx k = 0; k < order.size(); ++k) {
176 sv[k] = out_vals[start + order[k]];
177 sc[k] = out_col[start + order[k]];
178 }
179 for (idx k = 0; k < order.size(); ++k) {
180 out_vals[start + k] = sv[k];
181 out_col[start + k] = sc[k];
182 }
183
184 // Sum duplicates in-place
185 idx write = start;
186 for (idx k = start; k < end;) {
187 idx cur_col = out_col[k];
188 real sum = 0.0;
189 while (k < end && out_col[k] == cur_col) {
190 sum += out_vals[k++];
191 }
192 out_col[write] = cur_col;
193 out_vals[write++] = sum;
194 }
195 // Compact row_ptr if duplicates were merged
196 row_ptr[i + 1] = write;
197 // Shift remaining rows' data (rare; only matters if duplicates exist)
198 if (write < end) {
199 for (idx k = end; k < nnz; ++k) {
200 out_vals[write + (k - end)] = out_vals[k];
201 out_col[write + (k - end)] = out_col[k];
202 }
203 nnz -= (end - write);
204 out_vals.resize(nnz);
205 out_col.resize(nnz);
206 // Fix subsequent row_ptr entries
207 idx delta = end - write;
208 for (idx r = i + 2; r <= n_rows; ++r) {
209 row_ptr[r] -= delta;
210 }
211 }
212 }
213
214 return spmat(n_rows, n_cols, std::move(out_vals), std::move(out_col),
215 std::move(row_ptr));
216}

218inline spmat spmat::from_csc(idx n_rows, idx n_cols, const array<real> &vals,
219 const array<idx> &row_indices,
220 const array<idx> &col_ptrs) {
221 if (col_ptrs.size() != n_cols + 1) {
222 throw std::invalid_argument("spmat::from_csc: col_ptrs must have length n_cols+1");
223 }
224 if (col_ptrs.empty() || col_ptrs.front() != 0) {
225 throw std::invalid_argument("spmat::from_csc: col_ptrs must start at zero");
226 }
227 for (idx j = 0; j < n_cols; ++j) {
228 if (col_ptrs[j] > col_ptrs[j + 1]) {
229 throw std::invalid_argument("spmat::from_csc: col_ptrs must be nondecreasing");
230 }
231 }
232
233 // Armadillo's serializer historically emitted n_nonzero+1 payload
234 // entries while the final column pointer still reports n_nonzero. Use
235 // the pointer, rather than payload length, as the authoritative nnz.
236 const idx nnz = col_ptrs.back();
237 if (nnz > vals.size() || nnz > row_indices.size()) {
238 throw std::invalid_argument("spmat::from_csc: payload shorter than col_ptrs.back()");
239 }
240
241 array<idx> row_ptr(n_rows + 1, 0);
242 for (idx k = 0; k < nnz; ++k) {
243 if (row_indices[k] >= n_rows) {
244 throw std::out_of_range("spmat::from_csc: row index out of range");
245 }
246 ++row_ptr[row_indices[k] + 1];
247 }
248 for (idx i = 0; i < n_rows; ++i) {
249 row_ptr[i + 1] += row_ptr[i];
250 }
251
252 array<real> out_vals(nnz);
253 array<idx> out_col(nnz);
254 array<idx> next = row_ptr;
255 for (idx j = 0; j < n_cols; ++j) {
256 for (idx k = col_ptrs[j]; k < col_ptrs[j + 1]; ++k) {
257 const idx i = row_indices[k];
258 const idx p = next[i]++;
259 out_vals[p] = vals[k];
260 out_col[p] = j;
261 }
262 }
263
264 return spmat(n_rows, n_cols, std::move(out_vals), std::move(out_col),
265 std::move(row_ptr));
266}

294inline spmat transpose(const spmat &A) {
295 array<idx> column_ptr(A.n_cols() + 1, 0);
296 for (idx entry = 0; entry < A.nnz(); ++entry) {
297 ++column_ptr[A.col_idx()[entry] + 1];
298 }
299 for (idx column = 0; column < A.n_cols(); ++column) {
300 column_ptr[column + 1] += column_ptr[column];
301 }
302
303 array<real> values(A.nnz());
304 array<idx> columns(A.nnz());
305 array<idx> next = column_ptr;
306 for (idx row = 0; row < A.n_rows(); ++row) {
307 for (idx entry = A.row_ptr()[row]; entry < A.row_ptr()[row + 1]; ++entry) {
308 const idx destination = next[A.col_idx()[entry]]++;
309 values[destination] = A.values()[entry];
310 columns[destination] = row;
311 }
312 }
313 return {A.n_cols(), A.n_rows(), std::move(values), std::move(columns), std::move(column_ptr)};
314}

340inline mat diagonal_similarity(const spmat &A, view<const real> weights) {
341 if (A.n_rows() != A.n_cols() || weights.size() != A.n_rows()) {
342 throw std::invalid_argument("diagonal_similarity: dimensions must match");
343 }
344 if (!std::all_of(weights.begin(), weights.end(), [](real value) { return value > 0.0; })) {
345 throw std::invalid_argument("diagonal_similarity: weights must be positive");
346 }
347 mat result(A.n_rows(), A.n_cols(), 0.0);
348 for (idx row = 0; row < A.n_rows(); ++row) {
349 for (idx entry = A.row_ptr()[row]; entry < A.row_ptr()[row + 1]; ++entry) {
350 const idx column = A.col_idx()[entry];
351 result(row, column) = A.values()[entry] * weights[column] / weights[row];
352 }
353 }
354 return result;
355}
