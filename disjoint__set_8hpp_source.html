14/// @brief Disjoint-Set forest data structure supporting near O(1) amortized operations.
15/// @tparam Index Integer type used for vertex/element indices (e.g. num::idx, uint32_t, int).
16template <std::integral Index = num::idx>
17class basic_disjoint_set {
18 public:
19 using index_type = Index;

21 /// Construct a disjoint-set structure with n singleton sets: {0}, {1}, ..., {n-1}.
22 explicit basic_disjoint_set(Index n = 0)
23 : parent_(n), rank_(n, 0), size_(n, 1), count_(n) {
24 std::iota(parent_.begin(), parent_.end(), static_cast<Index>(0));
25 }
26
27 /// Find the representative root of the set containing element u with path compression.
28 Index find(Index u) {
29 structures::debug::check_index_bounds(u, static_cast<Index>(parent_.size()), "disjoint_set::find");
30 Index root = u;
31 while (root != parent_[root]) {
32 root = parent_[root];
33 }
34 // Two-pass path compression
35 Index curr = u;
36 while (curr != root) {
37 Index nxt = parent_[curr];
38 parent_[curr] = root;
39 curr = nxt;
40 }
41 return root;
42 }
43
44 /// Find representative root (const view without mutation).
45 [[nodiscard]] Index find(Index u) const {
46 structures::debug::check_index_bounds(u, static_cast<Index>(parent_.size()), "disjoint_set::find");
47 while (u != parent_[u]) {
48 u = parent_[u];
49 }
50 return u;
51 }
52
53 /// Unite the sets containing elements u and v.
54 /// @return true if u and v were in different sets, false if already in the same set.
55 bool unite(Index u, Index v) {
56 structures::debug::check_index_bounds(u, static_cast<Index>(parent_.size()), "disjoint_set::unite u");
57 structures::debug::check_index_bounds(v, static_cast<Index>(parent_.size()), "disjoint_set::unite v");
58
59 Index root_u = find(u);
60 Index root_v = find(v);
61
62 if (root_u == root_v) {
63 return false;
64 }
65
66 // Union by rank
67 if (rank_[root_u] < rank_[root_v]) {
68 parent_[root_u] = root_v;
69 size_[root_v] += size_[root_u];
70 } else {
71 parent_[root_v] = root_u;
72 size_[root_u] += size_[root_v];
73 if (rank_[root_u] == rank_[root_v]) {
74 rank_[root_u]++;
75 }
76 }
77 count_--;
78 return true;
79 }
80
81 /// Check whether elements u and v belong to the same connected component.
82 [[nodiscard]] bool connected(Index u, Index v) {
83 return find(u) == find(v);
84 }
85

87 [[nodiscard]] Index component_size(Index u) {

89 }

101 /// Reset the structure back to n singleton sets.
102 void reset(Index n = 0) {
103 if (n == 0) n = static_cast<Index>(parent_.size());
104 parent_.resize(n);
105 std::iota(parent_.begin(), parent_.end(), static_cast<Index>(0));
106 rank_.assign(n, 0);
107 size_.assign(n, 1);
108 count_ = n;
109 }
110
111 /// Return all connected components as lists of element indices.
112 [[nodiscard]] array<array<Index>> components() {
113 array<array<Index>> comp_map(parent_.size());
114 for (Index i = 0; i < static_cast<Index>(parent_.size()); ++i) {
115 comp_map[find(i)].push_back(i);
116 }
117 array<array<Index>> result;
118 result.reserve(count_);
119 for (auto &c : comp_map) {
120 if (!c.empty()) {
121 result.push_back(std::move(c));
122 }
123 }
124 return result;
125 }
