34/// @brief Compute \f$\exp(tA)v\f$ for any \f$y=A x\f$ adapter.
35template <class Op>
36requires linear_operator<Op, vec, vec> vec expv(real t, const Op &A, const vec &v,
37 int m_max = 30, real tol = 1e-8) {
38 const idx n = A.rows();
39 if (A.cols() != n || v.size() != n) {
40 throw std::invalid_argument("expv: dimension mismatch");
41 }
42
43 real beta = norm(v);
44 if (beta < 1e-300) {
45 return vec(n, 0.0);
46 }
47
48 array<vec> V;
49 V.reserve(m_max + 1);
50
51 vec v0(n);
52 for (idx i = 0; i < n; i++) {
53 v0[i] = v[i] / beta;
54 }
55 V.push_back(std::move(v0));
56
57 mat H(m_max + 1, m_max, 0.0);
58 int m_actual = m_max;
59 array<real> h_col(m_max + 1, 0.0);
60
61 for (int j = 0; j < m_max; j++) {
62 vec w(n, 0.0);
63 A.apply(V[j], w);
64
65 const real h_next = dispatch::subspace::mgs_orthogonalize(V, w, h_col, j + 1);
66 for (int i = 0; i <= j; i++) {
67 H(i, j) = h_col[i];
68 }
69 H(j + 1, j) = h_next;
70
71 if (h_next < tol) {
72 m_actual = j + 1;
73 break;
74 }
75
76 scale(w, real(1) / h_next);
77 V.push_back(std::move(w));
78 }
79
80 mat projected(m_actual, m_actual, 0.0);
81 for (int i = 0; i < m_actual; i++) {
82 for (int j = 0; j < m_actual; j++) {
83 projected(i, j) = t * H(i, j);
84 }
85 }
86
87 mat E = detail::dense_expm_pade6(projected);
88
89 vec result(n, 0.0);
90 for (int j = 0; j < m_actual; j++) {
91 axpy(beta * E(j, 0), V[j], result);
92 }
93
94 return result;
95}

101inline mat dense_expm_pade6(const mat &A) {
102 const idx m = A.rows();
103
104 static constexpr double c[7] = {1.0, 0.5, 5.0 / 44.0, 1.0 / 66.0,
105 1.0 / 792.0, 1.0 / 15840.0, 1.0 / 665280.0};
106
107 double norm_inf = 0.0;
108 for (idx i = 0; i < m; i++) {
109 double row_sum = 0.0;
110 for (idx j = 0; j < m; j++) {
111 row_sum += std::abs(A(i, j));
112 }
113 norm_inf = std::max(norm_inf, row_sum);
114 }
115
116 int s = 0;
117 if (norm_inf > 0.5) {
118 s = (int)std::max(0.0, std::ceil(std::log2(norm_inf / 0.5)));
119 }
120
121 double scale = std::ldexp(1.0, -s);
122 mat as(m, m, 0.0);
123 // A_s <- 2^{-s} A.
124 kernel::scale_copy_strided(as.data(), 1, A.data(), 1, scale, m * m);
125
126 mat B(m, m, 0.0);
127 matmul(as, as, B);
128
129 mat B2(m, m, 0.0);
130 matmul(B, B, B2);
131
132 mat B3(m, m, 0.0);
133 matmul(B2, B, B3);
134
135 mat v_mat(m, m, 0.0);
136 for (idx i = 0; i < m; i++) {
137 for (idx j = 0; j < m; j++) {
138 v_mat(i, j) = (c[6] * B3(i, j)) + (c[4] * B2(i, j)) + (c[2] * B(i, j));
139 }
140 v_mat(i, i) += c[0];
141 }
142
143 mat W(m, m, 0.0);
144 for (idx i = 0; i < m; i++) {
145 for (idx j = 0; j < m; j++) {
146 W(i, j) = (c[5] * B2(i, j)) + (c[3] * B(i, j));
147 }
148 W(i, i) += c[1];
149 }
150
151 mat U(m, m, 0.0);
152 matmul(as, W, U);
153
154 mat vp_u(m, m, 0.0);
155 mat vm_u(m, m, 0.0);
156 // V_plus <- V + U, V_minus <- V - U.
157 kernel::axpbyz(vp_u.data(), v_mat.data(), U.data(), real(1), real(1), m * m);
158 kernel::axpbyz(vm_u.data(), v_mat.data(), U.data(), real(1), real(-1), m * m);
159
160 // vm_u is constructed m-by-m.
161 lu_result fac = lu(assume_square(vm_u));
162 mat E(m, m, 0.0);
163 lu_solve(fac, vp_u, E);
164
165 for (int i = 0; i < s; i++) {
166 mat E2(m, m, 0.0);
167 matmul(E, E, E2);
168 E = std::move(E2);
169 }
170
171 return E;
172}
