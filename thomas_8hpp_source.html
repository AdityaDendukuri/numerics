27template <typename Float = double>
28inline void thomas(const basic_vec<Float> &a, const basic_vec<Float> &b,
29 const basic_vec<Float> &c, const basic_vec<Float> &d,
30 basic_vec<Float> &x) {
31 const idx n = b.size();
32 basic_vec<Float> b_work = b;
33 basic_vec<Float> d_work = d;
34
35 for (idx i = 1; i < n; ++i) {
36 Float w = a[i - 1] / b_work[i - 1];
37 b_work[i] -= w * c[i - 1];
38 d_work[i] -= w * d_work[i - 1];
39 }
40
41 x[n - 1] = d_work[n - 1] / b_work[n - 1];
42 for (idx i = n - 1; i > 0; --i) {
43 x[i - 1] = (d_work[i - 1] - (c[i - 1] * x[i])) / b_work[i - 1];
44 }
45}

91/// @brief Solve tridiagonal linear system \f$a_{i-1} x_{i-1} + b_i x_i + c_i x_{i+1} = d_i\f$ in \f$\mathcal{O}(n)\f$ time.
92///
93/// Executes the Thomas algorithm (specialized Gaussian elimination for tridiagonal systems) with zero heap allocations.
94/// Picks LAPACK (`dgtsv`) if configured, else the in-tree sequential elimination.
95/// To force a specific one (including the CUDA batched kernel when built with
96/// CUDA), call `num::lapack::thomas`/`num::seq::thomas`/`num::cuda::thomas`.
97///
98/// @tparam Float Floating-point scalar type (`double`, `float`).
99/// @param a Subdiagonal entries (\f$n-1\f$ elements: \f$a_0, \dots, a_{n-2}\f$).
100/// @param b Main diagonal entries (\f$n\f$ elements: \f$b_0, \dots, b_{n-1}\f$).
101/// @param c Superdiagonal entries (\f$n-1\f$ elements: \f$c_0, \dots, c_{n-2}\f$).
102/// @param d Right-hand side vector (\f$n\f$ elements).
103/// @param x Output solution vector (\f$n\f$ elements).
104/// @throws std::invalid_argument If dimensions do not match (\f$a, c\f$ size \f$n-1\f$, \f$b, d, x\f$ size \f$n\f$).
105/// @see banded_solve, lu_solve
106template <typename Float = double>
107inline void thomas(const basic_vec<Float> &a, const basic_vec<Float> &b,
108 const basic_vec<Float> &c, const basic_vec<Float> &d,
109 basic_vec<Float> &x) {
110 const idx n = b.size();
111 if (a.size() != n - 1 || c.size() != n - 1 || d.size() != n || x.size() != n) {
112 throw std::invalid_argument("Dimension mismatch in Thomas solver");
113 }
114
115 if constexpr (std::is_same_v<Float, double>) {
116#if defined(NUMERICS_HAS_LAPACK)
117 lapack::thomas(a, b, c, d, x);
118#else
119 seq::thomas(a, b, c, d, x);
120#endif
121 } else {
122 seq::thomas(a, b, c, d, x);
123 }
124}
