17/// Provides O(1) insert, erase, rekey, and O(1) amortized pop_min for bounded integer degrees.
18template <std::integral Index = num::idx>
19class basic_degree_queue {
20 public:
21 using index_type = Index;

24 /// Construct an empty queue with capacity for n vertices.
25 explicit basic_degree_queue(Index n = 0) {
26 head_.assign(static_cast<std::size_t>(n) + 1, none);
27 next_.assign(static_cast<std::size_t>(n), none);
28 prev_.assign(static_cast<std::size_t>(n), none);
29 degree_.assign(static_cast<std::size_t>(n), 0);
30 low_ = 0;
31 size_ = 0;
32 }

51 /// Insert vertex v with degree d.
52 void insert(Index v, Index d) {
53 debug::check_index_bounds(v, static_cast<Index>(degree_.size()), "degree_queue::insert");
54 if (d >= head_.size()) {
55 head_.resize(std::max(static_cast<std::size_t>(d) + 1, 2 * head_.size()), none);
56 }
57 next_[v] = head_[d];
58 prev_[v] = none;
59 if (head_[d] != none) {
60 prev_[head_[d]] = v;
61 }
62 head_[d] = v;
63 degree_[v] = d;
64 if (d < low_) {
65 low_ = d;
66 }
67 ++size_;
68 }
69
70 /// Erase vertex v from the queue.
71 void erase(Index v) {
72 debug::check_index_bounds(v, static_cast<Index>(degree_.size()), "degree_queue::erase");
73 const Index d = degree_[v];
74 if (prev_[v] == none) {
75 head_[d] = next_[v];
76 } else {
77 next_[prev_[v]] = next_[v];
78 }
79 if (next_[v] != none) {
80 prev_[next_[v]] = prev_[v];
81 }
82 next_[v] = none;
83 prev_[v] = none;
84 --size_;
85 }

87 /// Update the degree of vertex v.
88 void rekey(Index v, Index d) {
89 erase(v);

91 }

93 /// Pop vertex with minimum degree.
94 Index pop_min() {
95 if (empty()) {
96 throw std::runtime_error("degree_queue::pop_min: queue is empty");

100 }

102 throw std::runtime_error("degree_queue::pop_min: corrupted bucket state");
103 }
104 const Index v = head_[low_];
105 erase(v);
106 return v;
107 }
