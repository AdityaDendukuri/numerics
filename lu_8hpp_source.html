86/// Blocked partial-pivoting LU through `kernel::lu_factor_blocked`: panel
87/// factorization, then a `trsm` and a `gemm` per panel, so the bulk of the
88/// work runs at `gemm` speed. A pivot below `singular_tol` marks the result
89/// singular; the factorization still completes so the caller can inspect it.
90inline lu_result lu(const mat &A) {
91 constexpr real singular_tol = 1e-14;
92 const idx n = A.rows();
93 lu_result f;
94 f.LU = A;
95 f.piv.resize(n);
96 f.singular = !kernel::lu_factor_blocked(f.LU.data(), f.piv.data(), n);
97 for (idx k = 0; k < n && !f.singular; ++k) {
98 f.singular = std::abs(f.LU(k, k)) < singular_tol;
99 }
100 return f;
101}
102} // namespace seq

105inline lu_result lu(const mat &A) {
106#if defined(NUMERICS_HAS_LAPACK)
107 const idx n = A.rows();
108 lu_result f;
109 f.LU = A;
110 f.piv.resize(n);
111 f.singular = false;
112
113 array<lapack_int> ipiv(n);
114 int info =
115 LAPACKE_dgetrf(LAPACK_ROW_MAJOR, static_cast<lapack_int>(n), static_cast<lapack_int>(n),
116 f.LU.data(), static_cast<lapack_int>(n), ipiv.data());
117 if (info < 0) {
118 throw std::runtime_error("lu (lapack): dgetrf argument error, info=" +
119 std::to_string(info));
120 }
121 if (info > 0) {
122 f.singular = true;
123 }
124
125 for (idx k = 0; k < n; ++k) {
126 f.piv[k] = static_cast<idx>(ipiv[k] - 1);
127 }
128
129 return f;
130#else
131 return seq::lu(A);
132#endif
133}

147inline void lu_solve(const lu_result &f, const vec &b, vec &x) {
148 const idx n = f.LU.rows();
149 const mat &M = f.LU;
150 vec y = b;
151
152 for (idx k = 0; k < n; ++k) {
153 if (f.piv[k] != k) {
154 std::swap(y[k], y[f.piv[k]]);
155 }
156 }
157
158 for (idx i = 1; i < n; ++i) {
159 for (idx j = 0; j < i; ++j) {
160 y[i] -= M(i, j) * y[j];
161 }
162 }
163
164 for (idx i = n; i-- > 0;) {
165 for (idx j = i + 1; j < n; ++j) {
166 y[i] -= M(i, j) * y[j];
167 }
168 y[i] /= M(i, i);
169 }
170
171 x = std::move(y);
172}

174inline void lu_solve(const lu_result &f, const mat &B, mat &X) {
175 const idx n = B.rows();
176 if (f.LU.rows() != n || f.LU.cols() != n) {
177 throw std::invalid_argument("lu_solve: dimension mismatch");
178 }
179 X = B;
180 // P B, then L Y = P B and U X = Y, both as blocked triangular solves.
181 // Never dgetrs: the kernel solve measured faster at every size, and on a
182 // threaded BLAS the LAPACK call pays a thread wake-up per solve.
183 const idx nrhs = B.cols();
184 for (idx k = 0; k < n; ++k) {
185 if (f.piv[k] != k) {
186 kernel::swap_rows(X.data(), nrhs, k, f.piv[k], nrhs);
187 }
188 }
189 kernel::trsm_unit_lower_inplace(X.data(), nrhs, f.LU.data(), n, n, nrhs);
190 kernel::trsm_upper_inplace(X.data(), nrhs, f.LU.data(), n, n, nrhs);
191}

193/// Solve \f$A^T X = B\f$ for several right-hand sides from \f$PA = LU\f$:
194/// \f$A^T = U^T L^T P\f$, so \f$U^T Q = B\f$, \f$L^T Y = Q\f$, \f$X = P^T Y\f$.
195inline void lu_solve_transpose(const lu_result &f, const mat &B, mat &X) {
196 const idx n = f.LU.rows();
197 if (f.LU.cols() != n || B.rows() != n) {
198 throw std::invalid_argument("lu_solve_transpose: dimension mismatch");
199 }
200 X = B;
201 const idx nrhs = B.cols();
202 kernel::trsm_upper_transpose_inplace(X.data(), nrhs, f.LU.data(), n, n, nrhs);
203 kernel::trsm_unit_lower_transpose_inplace(X.data(), nrhs, f.LU.data(), n, n, nrhs);
204 // X = P^T Y: undo the row interchanges in reverse order.
205 for (idx step = n; step-- > 0;) {
206 if (f.piv[step] != step) {
207 kernel::swap_rows(X.data(), nrhs, step, f.piv[step], nrhs);
208 }
209 }
210}

212inline void lu_solve_transpose(const lu_result &f, const vec &b, vec &x) {
213 const idx n = f.LU.rows();
214 if (f.LU.cols() != n || b.size() != n) {
215 throw std::invalid_argument("lu_solve_transpose: dimension mismatch");
216 }
217 mat column(n, 1, 0.0);
218 for (idx row = 0; row < n; ++row) {
219 column(row, 0) = b[row];
220 }
221 mat solution;
222 lu_solve_transpose(f, column, solution);
223 x = vec(n, 0.0);
224 for (idx row = 0; row < n; ++row) {
225 x[row] = solution(row, 0);
226 }
227}

241inline real lu_det(const lu_result &f) {
242 const idx n = f.LU.rows();
243 real det = real(1);
244 for (idx i = 0; i < n; ++i) {
245 det *= f.LU(i, i);
246 }
247 idx swaps = 0;
248 for (idx k = 0; k < n; ++k) {
249 if (f.piv[k] != k) {
250 ++swaps;
251 }
252 }
253 return (swaps % 2 == 0) ? det : -det;
254}

256inline mat lu_inv(const lu_result &f) {
257 const idx n = f.LU.rows();
258 mat inv = f.LU;
259#if defined(NUMERICS_LAPACK_DEFAULT)
260 array<lapack_int> ipiv(n);
261 for (idx i = 0; i < n; ++i)
262 ipiv[i] = static_cast<lapack_int>(f.piv[i] + 1);
263 LAPACKE_dgetri(LAPACK_ROW_MAJOR, static_cast<lapack_int>(n), inv.data(),
264 static_cast<lapack_int>(n), ipiv.data());
265#else
266 array<real> work(n);
267 kernel::lu_invert(inv.data(), f.LU.data(), f.piv.data(), n, work.data());
268#endif
269 return inv;
270}
