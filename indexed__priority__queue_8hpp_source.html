15/// @brief Addressable priority queue where each item is identified by a unique index in [0, capacity).
16/// @tparam Key The priority key type (e.g. double, float, int).
17/// @tparam Index The integer type for element indices (e.g. num::idx, uint32_t).
18/// @tparam Compare Comparator where comp(a, b) == true means a has higher priority (default std::less for min-heap).
19template <typename Key = double, std::integral Index = num::idx, typename Compare = std::less<Key>>
20class indexed_priority_queue {

26 /// Construct an indexed priority queue for indices up to capacity.
27 explicit indexed_priority_queue(Index capacity = 0, Compare comp = Compare{})

31 /// Insert index with associated priority key.
32 void push(Index index, const Key &key) {
33 structures::debug::check_index_bounds(index, capacity_, "indexed_priority_queue::push");
34 structures::debug::check_not_contains(contains(index), index, "indexed_priority_queue::push");
35
36 size_++;
37 heap_[size_] = index;
38 pos_[index] = size_;
39 keys_[index] = key;
40 swim(size_);
41 }

43 /// Remove the top (highest-priority) element from the queue.
44 void pop() {
45 if (empty()) {
46 structures::debug::check_index_bounds(static_cast<Index>(0), static_cast<Index>(0),
47 "indexed_priority_queue::pop empty queue");
48 return;

55 }

88 /// Update the key of element index (re-heaps up or down as needed).
89 void update(Index index, const Key &new_key) {
90 structures::debug::check_index_bounds(index, capacity_, "indexed_priority_queue::update");
91 if (!contains(index)) {
92 push(index, new_key);
93 return;
94 }
95 keys_[index] = new_key;
96 swim(pos_[index]);

98 }

100 /// Decrease/increase key with priority improvement.
101 void improve_key(Index index, const Key &new_key) {
102 structures::debug::check_index_bounds(index, capacity_, "indexed_priority_queue::improve_key");
103 structures::debug::check_contains(contains(index), index, "indexed_priority_queue::improve_key");
104 if (comp_(new_key, keys_[index])) {
105 keys_[index] = new_key;
106 swim(pos_[index]);
107 }
108 }

110 /// Erase element index from the priority queue.
111 void erase(Index index) {
112 structures::debug::check_index_bounds(index, capacity_, "indexed_priority_queue::erase");
113 structures::debug::check_contains(contains(index), index, "indexed_priority_queue::erase");
114 Index heap_idx = pos_[index];
115 swap_nodes(heap_idx, size_);
116 pos_[index] = 0;
117 size_--;
118 if (heap_idx <= size_) {
119 swim(heap_idx);
120 sink(heap_idx);
121 }
122 }

125 [[nodiscard]] bool empty() const noexcept { return size_ == 0; }
126
127 /// Return the number of elements in the priority queue.
128 [[nodiscard]] Index size() const noexcept { return size_; }
129

131 [[nodiscard]] Index capacity() const noexcept { return capacity_; }
132
133 /// Clear all elements.
134 void clear() noexcept {
135 for (Index i = 1; i <= size_; ++i) {
136 pos_[heap_[i]] = 0;
137 }
138 size_ = 0;
139 }
140
141 private:
142 void swap_nodes(Index i, Index j) {
143 std::swap(heap_[i], heap_[j]);
144 pos_[heap_[i]] = i;
145 pos_[heap_[j]] = j;
146 }
147
148 void swim(Index k) {
149 while (k > 1 && comp_(keys_[heap_[k]], keys_[heap_[k / 2]])) {
150 swap_nodes(k, k / 2);
151 k = k / 2;
152 }
153 }
154
155 void sink(Index k) {
156 while (2 * k <= size_) {
157 Index j = 2 * k;
158 if (j < size_ && comp_(keys_[heap_[j + 1]], keys_[heap_[j]])) {
159 j++;
160 }
161 if (!comp_(keys_[heap_[j]], keys_[heap_[k]])) {
162 break;
163 }
164 swap_nodes(k, j);
165 k = j;
166 }
167 }
168
169 Compare comp_;
170 Index capacity_ = 0;
171 Index size_ = 0;
172 array<Index> heap_;
173 array<Index> pos_;
174 array<Key> keys_;
175};
