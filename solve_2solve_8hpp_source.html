31/// @brief Solve an ODE initial value problem with adaptive Dormand-Prince (RK45) integration.
32/// @tparam P ODE problem type satisfying `is_ode_problem`.
33/// @param prob ODE problem instance holding RHS `f`, initial state `u0`, and time bounds `[t0, tf]`.
34/// @param alg Algorithm configuration (`rk45_method{ .rtol=1e-6, .atol=1e-8 }`).
35/// @param obs Optional step observer callback `void(real t, const vec& u)`.
36/// @return `ode_result` containing final state `u`, time `t`, step counts, and convergence boolean.
37template <is_ode_problem P>
38ode_result solve(const P &prob, const rk45_method &alg, observer_fn obs = nullptr) {
39 ode_params p{.t0 = prob.t0,
40 .tf = prob.tf,
41 .h = alg.h,
42 .rtol = alg.rtol,
43 .atol = alg.atol,
44 .max_steps = alg.max_steps};
45 return ode_rk45(prob.f, prob.u0, p, obs);
46}

48/// @brief Solve an ODE initial value problem with fixed-step classical 4th-order Runge-Kutta (RK4).

50/// @param prob ODE problem instance.

52/// @param obs Optional step observer callback.
53/// @return `ode_result` containing final solution and stats.

126/// @brief Reusable linear-solve cache (CommonSolve `init`/`solve!`): a view of
127/// the problem plus the algorithm and the current iterate. Re-solving warm-starts
128/// from cache.u. @note A and b are held by reference and must outlive the cache.
129template <class Op, class Alg>
130struct linear_cache {
131 const Op &A;
132 const vec &b;
133 Alg alg;
134 vec u; ///< warm-start on entry, solution on exit
135};
136
137/// @brief Build a solve cache; the iterate starts at zero.
138template <class Op, class Alg>
139linear_cache<Op, Alg> init(const linear_problem<Op> &prob, const Alg &alg) {
140 return {prob.A, prob.b, alg, vec(prob.b.size(), real(0))};
141}
142
143/// @brief Build a solve cache seeded with an initial guess u0 (warm start).
144template <class Op, class Alg>
145linear_cache<Op, Alg> init(const linear_problem<Op> &prob, const Alg &alg, vec u0) {
146 return {prob.A, prob.b, alg, std::move(u0)};
147}
148
149/// @brief Re-solve from a cache, warm-starting from its current iterate; cache.u
150/// is updated in place (the C++ spelling of CommonSolve `solve!`).
151template <class Op, class Alg>
152linear_solution solve(linear_cache<Op, Alg> &cache) {
153 const solver_result r = detail::run(cache.A, cache.b, cache.u, cache.alg);
154 return {cache.u, r.iterations, r.residual, r.converged};
155}
156
157/// @brief One-shot linear solve: solve(problem, algorithm) == solve(init(...)).
158template <class Op, class Alg>
159linear_solution solve(const linear_problem<Op> &prob, const Alg &alg) {
160 vec u(prob.b.size(), real(0));
161 const solver_result r = detail::run(prob.A, prob.b, u, alg);
162 return {std::move(u), r.iterations, r.residual, r.converged};
163}
