15/// @brief Welford updates for mean and variance.
16template <typename Float = double, std::integral Index = num::idx>
17struct basic_running_stats {
18 Float mean = Float{0};
19 Float M2 = Float{0};
20 Index count = 0;
21
22 /// Incorporate one observation without storing the sample history.
23 void update(Float x) {
24 ++count;
25 Float delta = x - mean;
26 mean += delta / static_cast<Float>(count);
27 Float delta2 = x - mean;
28 M2 += delta * delta2;
29 }
30
31 /// Return the unbiased sample variance, or zero with fewer than two samples.
32 [[nodiscard]] Float variance() const {
33 return (count < 2) ? Float{0} : M2 / static_cast<Float>(count - 1);
34 }
35
36 /// Return the sample standard deviation.
37 [[nodiscard]] Float std_dev() const { return std::sqrt(variance()); }
38
39 /// Return the uncorrelated standard error of the mean.
40 [[nodiscard]] Float stderr_mean() const {
41 return (count < 2) ? Float{0} : std_dev() / std::sqrt(static_cast<Float>(count));
42 }
43
44 /// Discard all accumulated observations.
45 void reset() {
46 mean = M2 = Float{0};
47 count = 0;
48 }
49};

53/// @brief Fixed-bin histogram over \f$[\ell,h)\f$.
54template <typename Float = double, std::integral Index = num::idx>
55struct basic_histogram {
56 array<Float> counts;
57 Float lo = Float{0};
58 Float hi = Float{0};
59 Index nbins = 0;
60
61 /// Divide [lo,hi) into equally sized bins.
62 basic_histogram(Index nbins, Float lo, Float hi)
63 : counts(static_cast<std::size_t>(nbins), Float{0}), lo(lo), hi(hi), nbins(nbins) {}
64
65 /// Return the containing bin, or nbins when x lies outside the interval.
66 [[nodiscard]] Index bin(Float x) const {
67 if (x < lo || x >= hi) {
68 return nbins;
69 }
70 return static_cast<Index>((x - lo) / (hi - lo) * static_cast<Float>(nbins));
71 }
72
73 /// Return the center coordinate of bin b.
74 [[nodiscard]] Float bin_centre(Index b) const {
75 return lo + ((static_cast<Float>(b) + Float{0.5}) * (hi - lo) / static_cast<Float>(nbins));
76 }
77
78 [[nodiscard]] Float bin_width() const { return (hi - lo) / static_cast<Float>(nbins); }
79
80 /// Accumulate a weighted observation when it lies inside the interval.
81 void fill(Float x, Float weight = Float{1.0}) {
82 Index b = bin(x);
83 if (b < nbins) {
84 counts[static_cast<std::size_t>(b)] += weight;
85 }
86 }
87
88 void reset() { std::fill(counts.begin(), counts.end(), Float{0}); }
89
90 /// Return the total accumulated bin weight.
91 [[nodiscard]] Float total() const {
92 Float s = Float{0};
93 for (Float c : counts) {
94 s += c;
95 }
96 return s;
97 }
98
99 /// Normalise so that the histogram integrates to 1 (probability density).
100 [[nodiscard]] array<Float> pdf() const {
101 Float norm = total() * bin_width();
102 array<Float> p(static_cast<std::size_t>(nbins));
103 if (norm > Float{0}) {
104 for (Index b = 0; b < nbins; ++b) {
105 p[static_cast<std::size_t>(b)] = counts[static_cast<std::size_t>(b)] / norm;
106 }
107 }
108 return p;
109 }
110};

114// Autocorrelation time
115/// Integrated autocorrelation time tau_int from a stored time series.
116template <typename Float = double, std::integral Index = num::idx>
117inline Float autocorr_time(const Float *data, Index n, Float c = Float{6.0}) {
118 if (n < 4) {
119 return Float{0.5};
120 }
121
122 Float mean = Float{0};
123 for (Index i = 0; i < n; ++i) {
124 mean += data[i];
125 }
126 mean /= static_cast<Float>(n);
127
128 Float c0 = Float{0};
129 for (Index i = 0; i < n; ++i) {
130 Float d = data[i] - mean;
131 c0 += d * d;
132 }
133 c0 /= static_cast<Float>(n);
134 if (c0 < Float{1e-15}) {
135 return Float{0.5};
136 }
137
138 Float tau = Float{0.5};
139 for (Index t = 1; t < n / 2; ++t) {
140 Float ct = Float{0};
141 for (Index i = 0; i < n - t; ++i) {
142 ct += (data[i] - mean) * (data[i + t] - mean);
143 }
144 ct /= static_cast<Float>(n - t);
145 tau += ct / c0;
146 if (static_cast<Float>(t) >= c * tau) {
147 break;
148 }
149 }
150 return (tau < Float{0.5}) ? Float{0.5} : tau;
151}
