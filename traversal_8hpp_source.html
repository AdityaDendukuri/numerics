19/// Check if graph is connected.
20template <typename Weight, std::integral Index>
21[[nodiscard]] inline bool is_connected(const basic_graph<Weight, Index> &G) {
22 const Index n = G.n_vertices();
23 if (n <= 1) return true;
24
25 basic_disjoint_set<Index> ds(n);
26 for (Index u = 0; u < n; ++u) {
27 for (const auto &e : G.neighbors(u)) {
28 ds.unite(u, e.to);
29 }
30 }
31 return ds.count() == 1;
32}

47/// Compute single-source shortest path distances using Dijkstra with an Indexed Priority Queue.
48template <typename Weight, std::integral Index, std::integral Source = Index>
49[[nodiscard]] inline array<Weight> dijkstra(const basic_graph<Weight, Index> &G, Source source_in) {
50 const Index n = G.n_vertices();
51 const Index source = static_cast<Index>(source_in);
52 debug::check_vertex_bounds(source, n, "structures::dijkstra source");
53
54 constexpr Weight inf = std::numeric_limits<Weight>::infinity();
55 array<Weight> dist(n, inf);
56 dist[source] = Weight{0};
57
58 min_indexed_pq<Weight, Index> pq(n);
59 pq.push(source, Weight{0});
60
61 while (!pq.empty()) {
62 Index u = pq.top_index();
63 Weight d = pq.top_key();
64 pq.pop();
65
66 for (const auto &e : G.neighbors(u)) {
67 debug::check_positive_weight(e.weight, "structures::dijkstra edge weight");
68 Weight new_dist = d + e.weight;
69 if (new_dist < dist[e.to]) {
70 dist[e.to] = new_dist;
71 pq.update(e.to, new_dist);
72 }
73 }
74 }
75 return dist;
76}

78/// Breadth-first search traversal order starting from source.
79template <typename Weight, std::integral Index, std::integral Source = Index>
80[[nodiscard]] inline array<Index> bfs(const basic_graph<Weight, Index> &G, Source source_in) {
81 const Index n = G.n_vertices();
82 const Index source = static_cast<Index>(source_in);
83 debug::check_vertex_bounds(source, n, "structures::bfs source");
84
85 array<bool> visited(n, false);
86 array<Index> order;
87 order.reserve(n);
88
89 std::queue<Index> q;
90 q.push(source);
91 visited[source] = true;
92
93 while (!q.empty()) {
94 Index u = q.front();
95 q.pop();
96 order.push_back(u);
97
98 for (const auto &e : G.neighbors(u)) {
99 if (!visited[e.to]) {
100 visited[e.to] = true;
101 q.push(e.to);
102 }
103 }
104 }
105 return order;
106}

108/// Depth-first search traversal order starting from source.
109template <typename Weight, std::integral Index, std::integral Source = Index>
110[[nodiscard]] inline array<Index> dfs(const basic_graph<Weight, Index> &G, Source source_in) {
111 const Index n = G.n_vertices();
112 const Index source = static_cast<Index>(source_in);
113 debug::check_vertex_bounds(source, n, "structures::dfs source");
114
115 array<bool> visited(n, false);
116 array<Index> order;
117 order.reserve(n);
118
119 array<Index> stack = {source};
120
121 while (!stack.empty()) {
122 Index u = stack.back();
123 stack.pop_back();
124
125 if (visited[u]) continue;
126 visited[u] = true;
127 order.push_back(u);
128
129 for (const auto &e : G.neighbors(u)) {
130 if (!visited[e.to]) {
131 stack.push_back(e.to);
132 }
133 }
134 }
135 return order;
136}

138/// Compute Minimum Spanning Tree (MST) using Kruskal's algorithm with disjoint_set.
139template <typename Weight, std::integral Index>
140[[nodiscard]] inline basic_graph<Weight, Index> minimum_spanning_tree(const basic_graph<Weight, Index> &G) {
141 const Index n = G.n_vertices();
142 struct kruskal_edge {
143 Index u, v;
144 Weight weight;
145 bool operator<(const kruskal_edge &o) const { return weight < o.weight; }
146 };
147
148 array<kruskal_edge> edges;
149 for (Index u = 0; u < n; ++u) {
150 for (const auto &e : G.neighbors(u)) {
151 if (u < e.to || G.is_directed()) {
152 edges.push_back({u, e.to, e.weight});
153 }
154 }
155 }
156 std::sort(edges.begin(), edges.end());
157
158 basic_disjoint_set<Index> ds(n);
159 basic_graph<Weight, Index> mst(n, G.is_directed());
160
161 for (const auto &e : edges) {
162 if (ds.unite(e.u, e.v)) {
163 mst.add_edge(e.u, e.v, e.weight);
164 if (mst.n_edges() == n - 1) {
165 break;
166 }
167 }
168 }
169 return mst;
170}
