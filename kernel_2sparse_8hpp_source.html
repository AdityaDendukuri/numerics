25/// @brief Row length below which `reduce`'s blocked accumulation costs more than
26/// it saves, so a plain running sum wins.
27///
28/// `reduce` splits a range across several vector accumulators and combines them
29/// pairwise at the end. That is a large win on a long run and a pure loss on a
30/// row too short to fill even one accumulator block: the setup and the final
31/// combine still happen, but no lane is ever reused. A five-point stencil — the
32/// single most common sparse pattern there is — has five entries per row, and
33/// measured ~15% slower through `reduce` than through a plain loop. One
34/// accumulator block is the natural cutoff.
35template <std::floating_point T>
36inline constexpr idx short_row_cutoff = 4 * (NUM_K_VECTOR_BYTES / sizeof(T));
37
38/// @brief One CSR row: plain running sum when short, blocked reduction when long.
39template <std::floating_point T, std::integral Index>
40[[nodiscard]] NUM_K_AINLINE T csr_row_dot(const T *NUM_K_RESTRICT val,
41 const Index *NUM_K_RESTRICT col_idx,
42 const T *NUM_K_RESTRICT x, Index start,
43 idx length) noexcept {
44 if (length < short_row_cutoff<T>) {
45 T sum = T(0);
46 for (idx k = 0; k < length; ++k) {
47 const Index p = start + static_cast<Index>(k);
48 sum += val[p] * x[col_idx[p]];
49 }
50 return sum;
51 }
52 return reduce<T>(length, [val, col_idx, x, start](idx k) {
53 const Index p = start + static_cast<Index>(k);
54 return val[p] * x[col_idx[p]];
55 });
56}

60/// @brief Compressed Sparse Row (CSR) matrix-vector multiplication \f$\mathbf{y} \leftarrow A
61/// \mathbf{x}\f$.
62template <std::floating_point T, std::integral Index>
63NUM_K_AINLINE void spmv(T *NUM_K_RESTRICT y, const T *NUM_K_RESTRICT val,
64 const Index *NUM_K_RESTRICT row_ptr, const Index *NUM_K_RESTRICT col_idx,
65 const T *NUM_K_RESTRICT x, std::type_identity_t<Index> m) noexcept {
66 for (Index i = 0; i < m; ++i) {
67 const Index start = row_ptr[i];
68 const idx length = static_cast<idx>(row_ptr[i + 1] - start);
69 y[i] = detail::csr_row_dot<T, Index>(val, col_idx, x, start, length);
70 }
71}

73// Sparse incomplete factorization
74//
75// ILU(0) computes L and U whose product approximates A while reusing A's own
76// sparsity pattern exactly: no fill-in is admitted, so the factors cost the same
77// storage as the matrix and the whole computation is a rewrite of the value
78// array in place. The unit diagonal of L is implicit, so L's strict lower part
79// and all of U share one array, distinguished by the position of each row's
80// diagonal.
81//
82// These take a caller-supplied `diagonal` index and scratch buffer rather than
83// allocating, in keeping with the rest of this file.
84
85/// @brief Locate each row's diagonal entry in a CSR pattern.
86///
87/// Requires column indices sorted within each row.
88///
89/// @param diagonal Output, size n: index into `col_idx` of entry (i,i).
90/// @param row_ptr CSR row offsets, size n+1.
91/// @param col_idx CSR column indices, sorted within each row.
92/// @param n Number of rows.
93/// @return False if some row has no diagonal entry, which ILU(0) cannot proceed without.
94template <std::integral Index>
95[[nodiscard]] NUM_K_AINLINE bool csr_diagonal_positions(Index *NUM_K_RESTRICT diagonal,
96 const Index *NUM_K_RESTRICT row_ptr,
97 const Index *NUM_K_RESTRICT col_idx,
98 std::type_identity_t<Index> n) noexcept {
99 for (Index i = 0; i < n; ++i) {
100 const Index end = row_ptr[i + 1];
101 Index found = end;
102 for (Index k = row_ptr[i]; k < end; ++k) {
103 if (col_idx[k] == i) {
104 found = k;
105 break;
106 }
107 }
108 if (found == end) {
109 return false;
110 }
111 diagonal[i] = found;
112 }
113 return true;
114}

116/// @brief In-place ILU(0) factorization of a CSR value array.
117///
118/// On success `val` holds the strict lower part of L and the whole of U, sharing
119/// the original pattern; L's diagonal is unit and not stored. The pattern arrays
120/// are untouched.
121///
122/// The inner update needs, for each entry of the pivot row, the matching column
123/// in the current row. Searching for it would make the factorization quadratic
124/// in the row length, so `scratch` holds a column-to-position map for the row
125/// being eliminated: written on entry to the row, cleared on exit, so its state
126/// never leaks between rows.
127///
128/// @param val CSR values, overwritten with the combined factors.
129/// @param row_ptr CSR row offsets, size n+1. Not modified.
130/// @param col_idx CSR column indices, sorted within each row. Not modified.
131/// @param diagonal Diagonal positions from `csr_diagonal_positions`.
132/// @param scratch Workspace of size n; contents on entry and exit are irrelevant.
133/// @param n Number of rows.
134/// @return False if a pivot was zero or non-finite, leaving `val` partially overwritten.
135template <std::floating_point T, std::integral Index>
136[[nodiscard]] inline bool
137ilu0_factor(T *NUM_K_RESTRICT val, const Index *NUM_K_RESTRICT row_ptr,
138 const Index *NUM_K_RESTRICT col_idx, const Index *NUM_K_RESTRICT diagonal,
139 Index *NUM_K_RESTRICT scratch, std::type_identity_t<Index> n) noexcept {
140 constexpr Index unmarked = static_cast<Index>(-1);
141 for (Index i = 0; i < n; ++i) {
142 scratch[i] = unmarked;
143 }
144
145 for (Index i = 0; i < n; ++i) {
146 const Index row_begin = row_ptr[i];
147 const Index row_end = row_ptr[i + 1];
148 for (Index k = row_begin; k < row_end; ++k) {
149 scratch[col_idx[k]] = k;
150 }
151
152 // Columns strictly left of the diagonal are the L part of this row.
153 for (Index k = row_begin; k < diagonal[i]; ++k) {
154 const Index j = col_idx[k];
155 const T pivot = val[diagonal[j]];
156 if (pivot == T(0) || !std::isfinite(pivot)) {
157 return false;
158 }
159 const T multiplier = val[k] / pivot;
160 val[k] = multiplier;
161 // Subtract multiplier times the U part of row j, but only where the
162 // column already exists in row i: that restriction is what makes
163 // this ILU(0) rather than a complete factorization.
164 for (Index p = diagonal[j] + 1; p < row_ptr[j + 1]; ++p) {
165 const Index target = scratch[col_idx[p]];
166 if (target != unmarked) {
167 val[target] -= multiplier * val[p];
168 }
169 }
170 }
171
172 const T pivot = val[diagonal[i]];
173 if (pivot == T(0) || !std::isfinite(pivot)) {
174 return false;
175 }
176 for (Index k = row_begin; k < row_end; ++k) {
177 scratch[col_idx[k]] = unmarked;
178 }
179 }
180 return true;
181}

183/// @brief Solve \f$LUx = b\f$ for factors packed by `ilu0_factor`.
184///
185/// Forward substitution against the implicit unit-diagonal L, then backward
186/// substitution against U. `x` may alias `b`.
187template <std::floating_point T, std::integral Index>
188NUM_K_AINLINE void
189csr_lu_solve(T *NUM_K_RESTRICT x, const T *NUM_K_RESTRICT val, const Index *NUM_K_RESTRICT row_ptr,
190 const Index *NUM_K_RESTRICT col_idx, const Index *NUM_K_RESTRICT diagonal, const T *b,
191 std::type_identity_t<Index> n) noexcept {
192 for (Index i = 0; i < n; ++i) {
193 T sum = b[i];
194 for (Index k = row_ptr[i]; k < diagonal[i]; ++k) {
195 sum -= val[k] * x[col_idx[k]];
196 }
197 x[i] = sum; // L has a unit diagonal, so no division here
198 }
199 for (Index i = n; i-- > 0;) {
200 T sum = x[i];
201 for (Index k = diagonal[i] + 1; k < row_ptr[i + 1]; ++k) {
202 sum -= val[k] * x[col_idx[k]];
203 }
204 x[i] = sum / val[diagonal[i]];
205 }
206}

208/// @brief Fused CSR SpMV and vector accumulation \f$\mathbf{y} \leftarrow \alpha A \mathbf{x} +
209/// \beta \mathbf{y}\f$.
210template <std::floating_point T, std::integral Index>
211NUM_K_AINLINE void spmv_axpy(T *NUM_K_RESTRICT y, T alpha, const T *NUM_K_RESTRICT val,
212 const Index *NUM_K_RESTRICT row_ptr,
213 const Index *NUM_K_RESTRICT col_idx, const T *NUM_K_RESTRICT x, T beta,
214 std::type_identity_t<Index> m) noexcept {
215 for (Index i = 0; i < m; ++i) {
216 const Index start = row_ptr[i];
217 const idx length = static_cast<idx>(row_ptr[i + 1] - start);
218 const T s = detail::csr_row_dot<T, Index>(val, col_idx, x, start, length);
219 y[i] = (alpha * s) + (beta * y[i]);
220 }
221}

223/// @brief CSR sparse matrix times a row-major dense block, `Y <- A*X`.
224///
225/// `X` and `Y` contain `nrhs` contiguous values per matrix row. Traversing the
226/// right-hand-side dimension innermost amortizes CSR index/value loads and gives
227/// the compiler a regular SIMD loop even though the sparse row itself is irregular.
228template <std::floating_point T, std::integral Index>
229inline void spmm(T *NUM_K_RESTRICT Y, idx ldy, const T *NUM_K_RESTRICT val,
230 const Index *NUM_K_RESTRICT row_ptr, const Index *NUM_K_RESTRICT col_idx,
231 const T *NUM_K_RESTRICT X, idx ldx, std::type_identity_t<Index> m,
232 idx nrhs) noexcept {
233 for (Index i = 0; i < m; ++i) {
234 T *NUM_K_RESTRICT y_row = Y + (static_cast<idx>(i) * ldy);
235 NUM_K_IVDEP
236 for (idx r = 0; r < nrhs; ++r) {
237 y_row[r] = T(0);
238 }
239 for (Index p = row_ptr[i]; p < row_ptr[i + 1]; ++p) {
240 const T a = val[p];
241 const T *NUM_K_RESTRICT x_row = X + (static_cast<idx>(col_idx[p]) * ldx);
242 NUM_K_IVDEP
243 for (idx r = 0; r < nrhs; ++r) {
244 y_row[r] += a * x_row[r];
245 }
246 }
247 }
248}
