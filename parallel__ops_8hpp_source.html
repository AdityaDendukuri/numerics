1/// @file omp/parallel_ops.hpp
2/// @brief Threaded block decomposition and reduction over the raw kernels.
3///
4/// Two rules shape everything here.
5///
6/// **A thread must not be handed a scalar loop.** Splitting a reduction across
7/// threads and giving each one a single accumulator reintroduces exactly the
8/// loop-carried dependency the raw kernels exist to break: every core then runs
9/// at the latency of the adder, and eight of them reach roughly what one core
10/// reaches running at throughput. Threads take a block each and call the raw
11/// kernel on it, so the two levels of parallelism compose instead of cancelling.
12///
13/// **The answer must not depend on how many threads happened to run.** An
14/// OpenMP `reduction(+:s)` clause combines the per-thread partials in an
15/// unspecified order, so the same binary would return different bits under
16/// `OMP_NUM_THREADS=4` and `=8`. Instead each block writes its partial to a
17/// slot, and the slots are summed in index order afterwards. The block
18/// decomposition is a function of `n` alone, so the result depends only on the
19/// input and the build — never on the scheduler.
20///
21/// Independent of `vec`/`mat`, unlike the rest of `num::omp`: kept this
22/// way so container-tier headers can use the block helpers without a circular
23/// include back through `omp/vector_ops.hpp`.

31/// @brief Elements handled by one thread's call into a raw kernel.
32inline constexpr idx parallel_block = idx{1} << 14;
33
34/// @brief Below this element count an operation stays on one thread.
35///
36/// Entering an OpenMP parallel region is not free: measured at roughly 34 us on
37/// macOS libomp with eight threads, against about 6 us of actual work for a
38/// 32k-element dot product. Threading below the crossover is not a small loss,
39/// it is a large one — an order of magnitude at 16k elements — so the default
40/// errs high. Tunable at configure time with `-DNUMERICS_PARALLEL_THRESHOLD=<n>`;
41/// runtimes with cheaper team startup (typically libgomp on Linux) should set a
42/// considerably lower value.
43#ifndef NUMERICS_PARALLEL_THRESHOLD
44#define NUMERICS_PARALLEL_THRESHOLD (1 << 18)
45#endif
46inline constexpr idx parallel_threshold = idx{NUMERICS_PARALLEL_THRESHOLD};
47
48/// @brief Upper bound on blocks, so the partial-sum buffer can live on the stack.
49inline constexpr idx max_parallel_blocks = 256;
50
51/// @brief Element count per block for a problem of size `n`.
52///
53/// At least `parallel_block`, and large enough that no more than
54/// `max_parallel_blocks` blocks are produced.
55[[nodiscard]] inline constexpr idx block_size_for(idx n) noexcept {
56 const idx even_split = (n + max_parallel_blocks - 1) / max_parallel_blocks;
57 return even_split > parallel_block ? even_split : parallel_block;
58}
59
60/// @brief Number of blocks covering `n` elements.
61[[nodiscard]] inline constexpr idx block_count_for(idx n) noexcept {
62 const idx size = block_size_for(n);
63 return (n + size - 1) / size;
64}
65
66/// @brief Sum `block(offset, length)` over a blocked decomposition of `[0, n)`.
67///
68/// `block` is expected to be a raw kernel call over the slice, returning its
69/// partial. Runs on one thread below `parallel_threshold`, or when the build has
70/// no OpenMP. The summation order of the partials is fixed regardless.
71template <class T, class Block>
72[[nodiscard]] inline T parallel_reduce(idx n, Block block) {
73 if (n == 0) {
74 return T(0);
75 }
76#if defined(NUMERICS_HAS_OMP)
77 if (n >= parallel_threshold) {
78 const idx size = block_size_for(n);
79 const idx blocks = (n + size - 1) / size;
80 T partial[max_parallel_blocks]{};
81#pragma omp parallel for schedule(static)
82 for (idx b = 0; b < blocks; ++b) {
83 const idx offset = b * size;
84 partial[b] = block(offset, std::min(size, n - offset));
85 }
86 // Fixed order, so the result does not vary with the thread count.
87 T total = T(0);
88 for (idx b = 0; b < blocks; ++b) {
89 total += partial[b];
90 }
91 return total;
92 }
93#endif
94 return block(idx{0}, n);
95}
96
97/// @brief Apply `block(offset, length)` over a blocked decomposition of `[0, n)`.
98///
99/// For elementwise work, where blocks are independent and no combination step is
100/// needed. Deterministic by construction.
101template <class Block>
102inline void parallel_apply(idx n, Block block) {
103 if (n == 0) {
104 return;
105 }
106#if defined(NUMERICS_HAS_OMP)
107 if (n >= parallel_threshold) {
108 const idx size = block_size_for(n);
109 const idx blocks = (n + size - 1) / size;
110#pragma omp parallel for schedule(static)
111 for (idx b = 0; b < blocks; ++b) {
112 const idx offset = b * size;
113 block(offset, std::min(size, n - offset));
114 }
115 return;
116 }
117#endif
118 block(idx{0}, n);
119}
