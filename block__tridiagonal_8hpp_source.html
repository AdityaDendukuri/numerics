1/// @file linear/factorization/block_tridiagonal.hpp
2/// @brief Block LU and block Cholesky for block-tridiagonal sparse matrices.
3///
4/// A matrix whose rows carry a *level* label is block tridiagonal when every
5/// nonzero couples rows of the same level or of adjacent levels. Reordering by
6/// level makes that structure explicit, and the factorization then proceeds one
7/// block row at a time: each step touches only a dense diagonal block and its two
8/// neighbours, so the work is \f$O(\sum_k n_k^3)\f$ rather than \f$O(n^3)\f$, and
9/// no fill appears outside the band.
10///
11/// The level labels are the caller's. This header takes them as given, compresses
12/// them to contiguous block indices, and never interprets them: what a level
13/// *means* — a copy number, a mesh layer, a time slice — belongs to the caller,
14/// as does the choice between LU and Cholesky.
15///
16/// ### Index convention
17///
18/// With `nb` blocks, `offsets` has `nb + 1` entries and block `k` spans the
19/// reordered rows `[offsets[k], offsets[k+1])`. Both `upper` and `lower` have
20/// `nb - 1` entries and are indexed by the *coupling*, not the block row:
21///
22/// - `upper[k]` is the block at row `k`, column `k+1`;
23/// - `lower[k]` is the block at row `k+1`, column `k`.
24///
25/// So the elimination reads `lower[k] <- lower[k] D_k^{-1}` followed by
26/// `D_{k+1} <- D_{k+1} - lower[k] upper[k]`.
27///
28/// ### The factors are public
29///
30/// `block_lu_factor` and `block_cholesky_factor` are plain aggregates rather than the
31/// internals of a solver object, so a later low-rank layer can reach the block
32/// factors directly: apply the factor to a tall `U`, apply its transpose to `V`,
33/// form the small Woodbury matrix \f$K = I + V^{T}A_0^{-1}U\f$, and keep reusing
34/// the same block factor until the update rank or residual makes that
35/// unprofitable. Nothing here needs to change for that to work.
36///
37/// All dense arithmetic lowers to the existing kernels — `num::factor_no_pivot`,
38/// `num::cholesky`, `num::matmul`, `num::transpose`, and the raw triangular
39/// solves. No GEMM or triangular solve is reimplemented.

60/// @brief Block LU factors of a block-tridiagonal matrix.
61///
62/// After factorization the matrix reads \f$PA P^{T} = \mathcal{L}\,\mathcal{U}\f$
63/// with \f$\mathcal{L}\f$ block unit lower bidiagonal (its off-diagonal blocks in
64/// `lower`) and \f$\mathcal{U}\f$ block upper bidiagonal (`diagonal` on the
65/// diagonal, `upper` off it).
66struct block_lu_factor {
67 idx size = 0; ///< Order of the original matrix.
68 array<idx> offsets; ///< `nb + 1` block boundaries in reordered indexing.
69 array<idx> order; ///< `order[p]` is the original row at reordered position `p`.
70 array<no_pivot_lu> diagonal; ///< No-pivot factors of the diagonal Schur blocks.
71 array<mat> upper; ///< `upper[k]`: block (k, k+1). Size `nb - 1`.
72 array<mat> lower; ///< `lower[k]`: block (k+1, k), already scaled. Size `nb - 1`.
73
74 /// @brief Number of blocks.
75 [[nodiscard]] idx blocks() const noexcept {
76 return offsets.empty() ? 0 : offsets.size() - 1;
77 }
78 /// @brief Rows in block `k`.
79 [[nodiscard]] idx block_size(idx k) const { return offsets[k + 1] - offsets[k]; }
80 /// @brief True when some diagonal block was singular.
81 [[nodiscard]] bool singular() const noexcept {
82 return std::any_of(diagonal.begin(), diagonal.end(),
83 [](const no_pivot_lu &f) { return f.singular; });
84 }
85};

87/// @brief Block Cholesky factors of a symmetric positive-definite block-tridiagonal matrix.
88///
89/// \f$PAP^{T} = \mathcal{L}\mathcal{L}^{T}\f$ with \f$\mathcal{L}\f$ block lower
90/// bidiagonal: `diagonal[k]` holds the dense Cholesky factor of block `k`, and
91/// `lower[k]` the block at row `k+1`, column `k`, already right-scaled by
92/// \f$L_k^{-T}\f$.
93struct block_cholesky_factor {
94 idx size = 0;
95 array<idx> offsets;
96 array<idx> order;
97 array<cholesky_result> diagonal;
98 array<mat> lower; ///< `lower[k]`: block (k+1, k). Size `nb - 1`.
99
100 [[nodiscard]] idx blocks() const noexcept {
101 return offsets.empty() ? 0 : offsets.size() - 1;
102 }
103 [[nodiscard]] idx block_size(idx k) const { return offsets[k + 1] - offsets[k]; }
104 /// @brief True when some diagonal block was not positive definite.
105 [[nodiscard]] bool failed() const noexcept {
106 return std::any_of(diagonal.begin(), diagonal.end(),
107 [](const cholesky_result &f) { return !f.success; });
108 }
109};

113/// @brief Reordering induced by a level labelling.
114struct block_layout {
115 array<idx> offsets; ///< `nb + 1` boundaries.
116 array<idx> order; ///< reordered position -> original row.
117 array<idx> position; ///< original row -> reordered position.
118 array<idx> block_of; ///< original row -> block index.
119};

121/// @brief Compress arbitrary level labels into contiguous blocks and build the permutation.
122///
123/// Labels need not be contiguous, zero-based, or dense; they are sorted, and the
124/// distinct values become blocks `0, 1, ...` in that order. Rows keep their
125/// original relative order within a block, so the permutation is deterministic
126/// and a run of already-sorted labels is the identity.
127[[nodiscard]] inline block_layout build_block_order(view<const idx> levels) {
128 const idx n = levels.size();
129 array<idx> distinct(levels.begin(), levels.end());
130 std::sort(distinct.begin(), distinct.end());
131 distinct.erase(std::unique(distinct.begin(), distinct.end()), distinct.end());
132
133 block_layout layout;
134 const idx blocks = distinct.size();
135 layout.block_of.resize(n);
136 layout.position.resize(n);
137 layout.order.resize(n);
138 layout.offsets.assign(blocks + 1, 0);
139
140 for (idx row = 0; row < n; ++row) {
141 const auto found = std::lower_bound(distinct.begin(), distinct.end(), levels[row]);
142 layout.block_of[row] = static_cast<idx>(found - distinct.begin());
143 layout.offsets[layout.block_of[row] + 1] += 1;
144 }
145 for (idx k = 0; k < blocks; ++k) {
146 layout.offsets[k + 1] += layout.offsets[k];
147 }
148
149 array<idx> cursor(layout.offsets.begin(), layout.offsets.end() - 1);
150 for (idx row = 0; row < n; ++row) {
151 const idx slot = cursor[layout.block_of[row]]++;
152 layout.order[slot] = row;
153 layout.position[row] = slot;
154 }
155 return layout;
156}

158/// @brief Reject any nonzero coupling non-adjacent blocks.
159///
160/// Structural entries that store an exact zero are skipped: they contribute
161/// nothing to the factorization, and rejecting them would make an explicitly
162/// padded sparsity pattern unusable for no numerical reason.
163inline void validate_block_structure(const spmat &A, const block_layout &layout) {
164 const idx n = A.n_rows();
165 const idx *row_ptr = A.row_ptr();
166 const idx *col_idx = A.col_idx();
167 const real *values = A.values();
168 for (idx row = 0; row < n; ++row) {
169 const idx block_row = layout.block_of[row];
170 for (idx k = row_ptr[row]; k < row_ptr[row + 1]; ++k) {
171 if (values[k] == 0.0) {
172 continue;
173 }
174 const idx block_col = layout.block_of[col_idx[k]];
175 const idx distance =
176 block_row > block_col ? block_row - block_col : block_col - block_row;
177 if (distance > 1) {
178 throw std::invalid_argument(
179 "block_tridiagonal: entry (" + std::to_string(row) + "," +
180 std::to_string(col_idx[k]) + ") couples blocks " +
181 std::to_string(block_row) + " and " + std::to_string(block_col) +
182 ", which are not adjacent");
183 }
184 }
185 }
186}

195/// @brief Scatter the sparse entries into dense blocks, scaling as it goes.
196[[nodiscard]] inline assembled_blocks gather_blocks(const spmat &A, const block_layout &layout,
197 real scale) {
198 const idx blocks = layout.offsets.size() - 1;
199 assembled_blocks out;
200 out.diagonal.reserve(blocks);
201 for (idx k = 0; k < blocks; ++k) {
202 const idx rows = layout.offsets[k + 1] - layout.offsets[k];
203 out.diagonal.emplace_back(rows, rows, 0.0);
204 }
205 for (idx k = 0; k + 1 < blocks; ++k) {
206 const idx rows = layout.offsets[k + 1] - layout.offsets[k];
207 const idx cols = layout.offsets[k + 2] - layout.offsets[k + 1];
208 out.upper.emplace_back(rows, cols, 0.0);
209 out.lower.emplace_back(cols, rows, 0.0);
210 }
211
212 const idx n = A.n_rows();
213 const idx *row_ptr = A.row_ptr();
214 const idx *col_idx = A.col_idx();
215 const real *values = A.values();
216 for (idx row = 0; row < n; ++row) {
217 const idx block_row = layout.block_of[row];
218 const idx local_row = layout.position[row] - layout.offsets[block_row];
219 for (idx k = row_ptr[row]; k < row_ptr[row + 1]; ++k) {
220 const real value = values[k] * scale;
221 if (values[k] == 0.0) {
222 continue;
223 }
224 const idx column = col_idx[k];
225 const idx block_col = layout.block_of[column];
226 const idx local_col = layout.position[column] - layout.offsets[block_col];
227 if (block_col == block_row) {
228 out.diagonal[block_row](local_row, local_col) += value;
229 } else if (block_col == block_row + 1) {
230 out.upper[block_row](local_row, local_col) += value;
231 } else {
232 out.lower[block_col](local_row, local_col) += value;
233 }
234 }
235 }
236 return out;
237}

331/// @brief Common validation and layout construction for both factorizations.
332[[nodiscard]] inline block_layout prepare(const spmat &A, view<const idx> levels) {
333 if (A.n_rows() != A.n_cols()) {
334 throw std::invalid_argument("block_tridiagonal: matrix must be square");
335 }
336 if (levels.size() != A.n_rows()) {
337 throw std::invalid_argument(
338 "block_tridiagonal: one level is required per row, got " +
339 std::to_string(levels.size()) + " for " + std::to_string(A.n_rows()) + " rows");
340 }
341 block_layout layout = build_block_order(levels);
342 validate_block_structure(A, layout);
343 return layout;
344}

346/// @brief Solve `L z = rhs` for lower-triangular `L`, column by column.
347[[nodiscard]] inline mat forward_substitute(const mat &L, const mat &rhs) {
348 const idx n = L.rows();
349 mat out(n, rhs.cols(), 0.0);
350 vec column(n, 0.0);
351 vec solution(n, 0.0);
352 for (idx c = 0; c < rhs.cols(); ++c) {
353 for (idx r = 0; r < n; ++r) {
354 column[r] = rhs(r, c);
355 }
356 kernel::trsv_lower(solution.data(), L.data(), column.data(), n);
357 for (idx r = 0; r < n; ++r) {
358 out(r, c) = solution[r];
359 }
360 }
361 return out;
362}

399/// @brief Factor a block-tridiagonal matrix by block LU without row pivoting.
400///
401/// @param A Square sparse matrix whose nonzeros stay within adjacent level blocks.
402/// @param levels One level label per row; arbitrary values, compressed internally.
403/// @param scale Uniform scaling applied while assembling, so `factor(scale*A)`
404/// needs no separate scaled copy of `A`.
405/// The caller must guarantee that Gaussian elimination in the supplied block
406/// ordering has nonzero pivots. Nonsingular M-matrices satisfy this condition,
407/// and their diagonal Schur blocks remain nonsingular M-matrices.
408///
409/// @throws std::invalid_argument If `A` is not square, the level count is wrong,
410/// or a nonzero couples non-adjacent blocks.
411/// @throws std::runtime_error If a diagonal Schur block has a zero pivot.
412[[nodiscard]] inline block_lu_factor factor_block_lu(const spmat &A,
413 view<const idx> levels,
414 real scale = 1.0) {
415 const detail::block_layout layout = detail::prepare(A, levels);
416 detail::assembled_blocks blocks = detail::gather_blocks(A, layout, scale);
417 const idx count = layout.offsets.size() - 1;
418
419 block_lu_factor factor;
420 factor.size = A.n_rows();
421 factor.offsets = layout.offsets;
422 factor.order = layout.order;
423 factor.upper = std::move(blocks.upper);
424 factor.lower = std::move(blocks.lower);
425 factor.diagonal.reserve(count);
426
427 for (idx k = 0; k < count; ++k) {
428 factor.diagonal.push_back(factor_no_pivot(assume_square(blocks.diagonal[k])));
429 if (factor.diagonal.back().singular) {
430 throw std::runtime_error("block_tridiagonal: zero pivot in diagonal block " +
431 std::to_string(k));
432 }
433 if (k + 1 >= count) {
434 break;
435 }
436 // lower[k] <- lower[k] D_k^{-1}. Right-side solves are not exposed
437 // directly, so use Z D = L <=> D^T Z^T = L^T.
438 mat transposed = transpose(factor.lower[k]);
439 mat scaled_transposed;
440 solve_transpose(factor.diagonal[k], transposed, scaled_transposed);
441 factor.lower[k] = transpose(scaled_transposed);
442
443 // D_{k+1} <- D_{k+1} - lower[k] upper[k]
444 detail::subtract_from(blocks.diagonal[k + 1],
445 detail::product(factor.lower[k], factor.upper[k]));
446 }
447 return factor;
448}

450/// @brief Solve \f$AX = B\f$ using stored block LU factors.
451inline void solve(const block_lu_factor &factor, const mat &B, mat &X) {
452 if (B.rows() != factor.size) {
453 throw std::invalid_argument("block_tridiagonal: right-hand side row count mismatch");
454 }
455 const idx count = factor.blocks();
456 mat work = detail::gather_rows(B, factor.order);
457
458 // Forward: Y_0 = B_0 ; Y_k = B_k - lower[k-1] Y_{k-1}
459 for (idx k = 1; k < count; ++k) {
460 const mat previous =
461 detail::block_rows(work, factor.offsets[k - 1], factor.block_size(k - 1));
462 mat current = detail::block_rows(work, factor.offsets[k], factor.block_size(k));
463 detail::subtract_from(current, detail::product(factor.lower[k - 1], previous));
464 detail::set_block_rows(work, factor.offsets[k], current);
465 }
466
467 // Backward: X_k = D_k^{-1} Y_k ; Y_{k-1} -= upper[k-1] X_k
468 for (idx k = count; k-- > 0;) {
469 const mat current = detail::block_rows(work, factor.offsets[k], factor.block_size(k));
470 mat solved;
471 solve(factor.diagonal[k], current, solved);
472 detail::set_block_rows(work, factor.offsets[k], solved);
473 if (k > 0) {
474 mat previous =
475 detail::block_rows(work, factor.offsets[k - 1], factor.block_size(k - 1));
476 detail::subtract_from(previous, detail::product(factor.upper[k - 1], solved));
477 detail::set_block_rows(work, factor.offsets[k - 1], previous);
478 }
479 }
480
481 X = mat(factor.size, B.cols(), 0.0);
482 detail::scatter_rows(work, factor.order, X);
483}

485/// @brief Solve \f$Ax = b\f$ using stored block LU factors.
486inline void solve(const block_lu_factor &factor, const vec &b, vec &x) {
487 if (b.size() != factor.size) {
488 throw std::invalid_argument("block_tridiagonal: right-hand side size mismatch");
489 }
490 const idx count = factor.blocks();
491 vec work = detail::gather_rows(b, factor.order);
492
493 for (idx k = 1; k < count; ++k) {
494 const vec previous =
495 detail::block_rows(work, factor.offsets[k - 1], factor.block_size(k - 1));
496 vec current = detail::block_rows(work, factor.offsets[k], factor.block_size(k));
497 detail::subtract_from(current, detail::product(factor.lower[k - 1], previous));
498 detail::set_block_rows(work, factor.offsets[k], current);
499 }
500
501 for (idx k = count; k-- > 0;) {
502 const vec current = detail::block_rows(work, factor.offsets[k], factor.block_size(k));
503 vec solved(current.size(), 0.0);
504 solve(factor.diagonal[k], current, solved);
505 detail::set_block_rows(work, factor.offsets[k], solved);
506 if (k > 0) {
507 vec previous =
508 detail::block_rows(work, factor.offsets[k - 1], factor.block_size(k - 1));
509 detail::subtract_from(previous, detail::product(factor.upper[k - 1], solved));
510 detail::set_block_rows(work, factor.offsets[k - 1], previous);
511 }
512 }
513
514 x = vec(factor.size, 0.0);
515 detail::scatter_rows(work, factor.order, x);
516}

518/// @brief Solve \f$A^{T}X = B\f$ from the same factors, without refactorizing.
519///
520/// \f$A = \mathcal{L}\mathcal{U}\f$ gives \f$A^{T} = \mathcal{U}^{T}\mathcal{L}^{T}\f$,
521/// so the block dependency runs the other way: a forward sweep through
522/// \f$\mathcal{U}^{T}\f$ (block lower bidiagonal, `upper[k]^T` below the
523/// diagonal) followed by a backward sweep through \f$\mathcal{L}^{T}\f$ (block
524/// unit upper bidiagonal, `lower[k]^T` above it).
525inline void solve_transpose(const block_lu_factor &factor, const mat &B, mat &X) {
526 if (B.rows() != factor.size) {
527 throw std::invalid_argument("block_tridiagonal: right-hand side row count mismatch");
528 }
529 const idx count = factor.blocks();
530 mat work = detail::gather_rows(B, factor.order);
531
532 // U^T Z = B : Z_k = D_k^{-T}(B_k - upper[k-1]^T Z_{k-1})
533 for (idx k = 0; k < count; ++k) {
534 mat current = detail::block_rows(work, factor.offsets[k], factor.block_size(k));
535 if (k > 0) {
536 const mat previous =
537 detail::block_rows(work, factor.offsets[k - 1], factor.block_size(k - 1));
538 detail::subtract_from(current,
539 detail::product(transpose(factor.upper[k - 1]), previous));
540 }
541 mat solved;
542 solve_transpose(factor.diagonal[k], current, solved);
543 detail::set_block_rows(work, factor.offsets[k], solved);
544 }
545
546 // L^T X = Z : X_k = Z_k - lower[k]^T X_{k+1}
547 for (idx k = count - 1; k-- > 0;) {
548 const mat next =
549 detail::block_rows(work, factor.offsets[k + 1], factor.block_size(k + 1));
550 mat current = detail::block_rows(work, factor.offsets[k], factor.block_size(k));
551 detail::subtract_from(current, detail::product(transpose(factor.lower[k]), next));
552 detail::set_block_rows(work, factor.offsets[k], current);
553 }
554
555 X = mat(factor.size, B.cols(), 0.0);
556 detail::scatter_rows(work, factor.order, X);
557}

559/// @brief Solve \f$A^{T}x = b\f$ from the same factors, without refactorizing.
560inline void solve_transpose(const block_lu_factor &factor, const vec &b, vec &x) {
561 if (b.size() != factor.size) {
562 throw std::invalid_argument("block_tridiagonal: right-hand side size mismatch");
563 }
564 const idx count = factor.blocks();
565 vec work = detail::gather_rows(b, factor.order);
566
567 for (idx k = 0; k < count; ++k) {
568 vec current = detail::block_rows(work, factor.offsets[k], factor.block_size(k));
569 if (k > 0) {
570 const vec previous =
571 detail::block_rows(work, factor.offsets[k - 1], factor.block_size(k - 1));
572 detail::subtract_from(current,
573 detail::product(transpose(factor.upper[k - 1]), previous));
574 }
575 vec solved(current.size(), 0.0);
576 solve_transpose(factor.diagonal[k], current, solved);
577 detail::set_block_rows(work, factor.offsets[k], solved);
578 }
579
580 for (idx k = count - 1; k-- > 0;) {
581 const vec next =
582 detail::block_rows(work, factor.offsets[k + 1], factor.block_size(k + 1));
583 vec current = detail::block_rows(work, factor.offsets[k], factor.block_size(k));
584 detail::subtract_from(current, detail::product(transpose(factor.lower[k]), next));
585 detail::set_block_rows(work, factor.offsets[k], current);
586 }
587
588 x = vec(factor.size, 0.0);
589 detail::scatter_rows(work, factor.order, x);
590}

596/// @brief Factor a symmetric positive-definite block-tridiagonal matrix.
597///
598/// Only the diagonal and sub-diagonal blocks are used; the super-diagonal blocks
599/// are taken to be their transposes, so an input storing both halves and one
600/// storing only the lower half factor identically.
601///
602/// @throws std::invalid_argument If `A` is not square, the level count is wrong,
603/// or a nonzero couples non-adjacent blocks.
604/// @throws std::runtime_error If a diagonal block is not positive definite.
605[[nodiscard]] inline block_cholesky_factor factor_block_cholesky(const spmat &A,
606 view<const idx> levels) {
607 const detail::block_layout layout = detail::prepare(A, levels);
608 detail::assembled_blocks blocks = detail::gather_blocks(A, layout, real(1));
609 const idx count = layout.offsets.size() - 1;
610
611 block_cholesky_factor factor;
612 factor.size = A.n_rows();
613 factor.offsets = layout.offsets;
614 factor.order = layout.order;
615 factor.lower = std::move(blocks.lower);
616 factor.diagonal.reserve(count);
617
618 for (idx k = 0; k < count; ++k) {
619 auto result = cholesky(assume_spd(blocks.diagonal[k]));
620 if (!result.success) {
621 throw std::runtime_error("block_tridiagonal: diagonal block " + std::to_string(k) +
622 " is not positive definite");
623 }
624 factor.diagonal.push_back(std::move(result));
625 if (k + 1 >= count) {
626 break;
627 }
628 // C <- C L_k^{-T}, i.e. L_k Z^T = C^T: forward-substitute each row of C.
629 const mat &chol = factor.diagonal[k].L;
630 factor.lower[k] = transpose(detail::forward_substitute(chol, transpose(factor.lower[k])));
631
632 // A_{k+1} <- A_{k+1} - C C^T
633 detail::subtract_from(blocks.diagonal[k + 1],
634 detail::product(factor.lower[k], transpose(factor.lower[k])));
635 }
636 return factor;
637}

639/// @brief Solve \f$AX = B\f$ using stored block Cholesky factors.
640inline void solve(const block_cholesky_factor &factor, const mat &B, mat &X) {
641 if (B.rows() != factor.size) {
642 throw std::invalid_argument("block_tridiagonal: right-hand side row count mismatch");
643 }
644 const idx count = factor.blocks();
645 mat work = detail::gather_rows(B, factor.order);
646
647 // L Y = B
648 for (idx k = 0; k < count; ++k) {
649 mat current = detail::block_rows(work, factor.offsets[k], factor.block_size(k));
650 if (k > 0) {
651 const mat previous =
652 detail::block_rows(work, factor.offsets[k - 1], factor.block_size(k - 1));
653 detail::subtract_from(current, detail::product(factor.lower[k - 1], previous));
654 }
655 detail::set_block_rows(work, factor.offsets[k],
656 detail::forward_substitute(factor.diagonal[k].L, current));
657 }
658
659 // L^T X = Y
660 for (idx k = count; k-- > 0;) {
661 mat current = detail::block_rows(work, factor.offsets[k], factor.block_size(k));
662 if (k + 1 < count) {
663 const mat next =
664 detail::block_rows(work, factor.offsets[k + 1], factor.block_size(k + 1));
665 detail::subtract_from(current, detail::product(transpose(factor.lower[k]), next));
666 }
667 detail::set_block_rows(work, factor.offsets[k],
668 detail::backward_substitute(factor.diagonal[k].L, current));
669 }
670
671 X = mat(factor.size, B.cols(), 0.0);
672 detail::scatter_rows(work, factor.order, X);
673}

675/// @brief Solve \f$Ax = b\f$ using stored block Cholesky factors.
676inline void solve(const block_cholesky_factor &factor, const vec &b, vec &x) {
677 if (b.size() != factor.size) {
678 throw std::invalid_argument("block_tridiagonal: right-hand side size mismatch");
679 }
680 const idx count = factor.blocks();
681 vec work = detail::gather_rows(b, factor.order);
682
683 for (idx k = 0; k < count; ++k) {
684 vec current = detail::block_rows(work, factor.offsets[k], factor.block_size(k));
685 if (k > 0) {
686 const vec previous =
687 detail::block_rows(work, factor.offsets[k - 1], factor.block_size(k - 1));
688 detail::subtract_from(current, detail::product(factor.lower[k - 1], previous));
689 }
690 detail::set_block_rows(work, factor.offsets[k],
691 detail::forward_substitute(factor.diagonal[k].L, current));
692 }
693
694 for (idx k = count; k-- > 0;) {
695 vec current = detail::block_rows(work, factor.offsets[k], factor.block_size(k));
696 if (k + 1 < count) {
697 const vec next =
698 detail::block_rows(work, factor.offsets[k + 1], factor.block_size(k + 1));
699 detail::subtract_from(current, detail::product(transpose(factor.lower[k]), next));
700 }
701 detail::set_block_rows(work, factor.offsets[k],
702 detail::backward_substitute(factor.diagonal[k].L, current));
703 }
704
705 x = vec(factor.size, 0.0);
706 detail::scatter_rows(work, factor.order, x);
707}
