1/// @file linear/solvers/chebyshev.hpp
2/// @brief mat-free Chebyshev polynomial preconditioner.
3///
4/// Every other preconditioner here needs the matrix entries: Jacobi reads the
5/// diagonal, ApproxChol eliminates a graph, an incomplete factorization walks
6/// the sparsity pattern. None of them can precondition an operator built with
7/// `num::operators::make_op`, which is the library's own headline pattern — a
8/// discrete Laplacian written as a stencil has no entries to read.
9///
10/// A polynomial preconditioner needs nothing but the operator's action. It
11/// approximates \f$A^{-1}\f$ by \f$p_m(A)\f$, where \f$p_m\f$ is the degree-m
12/// polynomial minimising \f$\max_{\lambda \in [\ell, h]} |1 - \lambda p(\lambda)|\f$ —
13/// the Chebyshev problem, whose solution is available as a three-term
14/// recurrence costing one operator application per degree and no setup at all.
15///
16/// ### The precondition, and what happens when it is wrong
17///
18/// \f$p_m(A)\f$ is positive definite exactly when \f$p_m > 0\f$ across the
19/// spectrum, which holds when \f$0 < \ell \le \lambda_{\min}\f$ and
20/// \f$\lambda_{\max} \le h\f$. bounds that exclude part of the spectrum produce
21/// an *indefinite* preconditioner, and preconditioned CG built on one silently
22/// loses the property that makes its error monotone.
23///
24/// This is why both bounds are the caller's to supply: they are a mathematical
25/// claim, so they go through the same channel as every other one in this
26/// library. `chebyshev_preconditioner` asserts positive definiteness, `num::pcg`
27/// requires that assertion, and the runtime property sampler rejects the
28/// operator if the claim was false. `estimate_largest_eigenvalue` will find
29/// \f$h\f$ for you; there is no equally cheap way to find \f$\ell\f$, and
30/// guessing it from an assumed condition number is how this preconditioner is
31/// most often gotten wrong.
32///
33/// ### What this is, and is not, good for
34///
35/// A degree-m polynomial can only improve a condition number by roughly a factor
36/// of m. On a second-order elliptic operator, where \f$\kappa\f$ grows with the
37/// mesh, that is not a competitive preconditioner and no degree makes it one —
38/// use ApproxChol for SDD systems, or an algebraic multigrid for the rest.
39///
40/// It earns its place in three situations a factorization cannot serve:
41/// - **matrix-free operators**, where there are no entries to factor at all;
42/// - **moderately conditioned systems** — mass matrices, shifted or damped
43/// operators, regularized least squares — where a factor of m is the whole
44/// problem;
45/// - as a **smoother**, damping the upper spectrum inside a multigrid cycle.
46///
47/// It also uses no global reductions, which is why polynomial preconditioning
48/// keeps reappearing in distributed solvers where those dominate the cost.

62/// @brief Estimate \f$\lambda_{\max}\f$ of a self-adjoint operator by power iteration.
63///
64/// Needs only the operator's action, so it works on a matrix-free operator. The
65/// starting vector is fixed, so repeated calls on the same operator agree.
66///
67/// Power iteration approaches the dominant eigenvalue **from below**, so the
68/// raw Rayleigh quotient is an under-estimate and unsafe as an upper bound.
69/// `safety` inflates the result; the default leaves 10% of headroom.
70///
71/// @param A Self-adjoint linear operator.
72/// @param iterations Power iterations to run.
73/// @param safety Multiplier applied to the converged Rayleigh quotient.
74/// @return An estimate of \f$\lambda_{\max}\f$, or 0 for an empty operator.
75template <class Op>
76[[nodiscard]] inline real estimate_largest_eigenvalue(const Op &A, idx iterations = 25,
77 real safety = 1.1) {
78 const idx n = static_cast<idx>(A.rows());
79 if (n == 0) {
80 return real(0);
81 }
82 if (iterations == 0 || !(safety > 0.0)) {
83 throw std::invalid_argument("estimate_largest_eigenvalue: invalid iteration or safety");
84 }
85
86 vec v(n);
87 vec image(n, 0.0);
88 // A fixed, non-symmetric starting vector: a constant one is an eigenvector
89 // of too many operators of interest (any Laplacian) to be a safe default.
90 for (idx i = 0; i < n; ++i) {
91 v[i] = real(1) + std::sin(real(i) * real(0.7));
92 }
93
94 real lambda = 0.0;
95 for (idx step = 0; step < iterations; ++step) {
96 const real norm = math::norm(v);
97 if (!(norm > 0.0) || !std::isfinite(norm)) {
98 return real(0);
99 }
100 math::scale(real(1) / norm, v);
101 math::apply(A, v, image);
102 lambda = math::inner(v, image); // Rayleigh quotient, v already unit norm
103 v = image;
104 }
105 return std::abs(lambda) * safety;
106}

108/// @brief Chebyshev polynomial preconditioner \f$M^{-1} = p_m(A)\f$.
109///
110/// Applies `degree` operator applications per invocation and allocates nothing
111/// after construction. Holds a reference to the operator, which must outlive it.
112///
113/// @tparam Op Self-adjoint linear operator type.
114template <class Op>
115class chebyshev_preconditioner final {
116 public:
117 using domain_type = vec;
118 using codomain_type = vec;
119 /// Valid only for bounds that enclose the spectrum; see the file comment.
120 using math_laws = math::type_list<law::spd>;
121
122 /// @brief Build a degree-`degree` Chebyshev preconditioner for the spectrum \f$[lo, hi]\f$.
123 /// @throws std::invalid_argument If the interval is not a positive, non-degenerate range.
124 chebyshev_preconditioner(const Op &A, real lo, real hi, idx degree)
125 : op_(&A), lo_(lo), hi_(hi), degree_(degree), n_(static_cast<idx>(A.rows())),
126 residual_(n_, 0.0), direction_(n_, 0.0), work_(n_, 0.0) {
127 if (A.rows() != A.cols()) {
128 throw std::invalid_argument("chebyshev: operator must be square");
129 }
130 if (degree_ == 0) {
131 throw std::invalid_argument("chebyshev: degree must be at least 1");
132 }
133 if (!(lo_ > 0.0) || !(hi_ > lo_) || !std::isfinite(hi_)) {
134 throw std::invalid_argument(
135 "chebyshev: spectral bounds must satisfy 0 < lo < hi and be finite");
136 }
137 }
138
139 [[nodiscard]] idx rows() const noexcept { return n_; }
140 [[nodiscard]] idx cols() const noexcept { return n_; }
141 [[nodiscard]] idx degree() const noexcept { return degree_; }
142
143 /// @brief Apply \f$z \leftarrow p_m(A)\, r\f$.
144 void apply(const vec &r, vec &z) const {
145 if (r.size() != n_) {
146 throw std::invalid_argument("chebyshev: dimension mismatch");
147 }
148 if (z.size() != n_) {
149 z = vec(n_, 0.0);
150 }
151
152 const real centre = 0.5 * (hi_ + lo_);
153 const real half_width = 0.5 * (hi_ - lo_);
154 const real sigma = centre / half_width;
155 real rho = real(1) / sigma;
156
157 // Degree 1: the scaled Richardson step z = r / centre.
158 z = r;
159 math::scale(real(1) / centre, z);
160 if (degree_ == 1) {
161 return;
162 }
163
164 // residual <- r - A z
165 math::apply(*op_, z, work_);
166 residual_ = r;
167 math::axpy(real(-1), work_, residual_);
168
169 // The first direction is the first iterate itself (Saad, Alg. 12.1:
170 // d_0 = r/theta), not zero. Starting it at zero drops a term from the
171 // recurrence and yields a polynomial that is negative on part of the
172 // interval — an indefinite preconditioner rather than an SPD one.
173 kernel::copy(direction_.data(), z.data(), n_);
174
175 for (idx k = 1; k < degree_; ++k) {
176 const real rho_next = real(1) / ((real(2) * sigma) - rho);
177 // direction <- (rho * rho_next) direction + (2 rho_next / half_width) residual
178 math::linear_combination(real(2) * rho_next / half_width, residual_, rho * rho_next,
179 direction_);
180 math::axpy(real(1), direction_, z);
181 if (k + 1 < degree_) {
182 math::apply(*op_, direction_, work_);
183 math::axpy(real(-1), work_, residual_);
184 }
185 rho = rho_next;
186 }
187 }
188
189 private:
190 const Op *op_;
191 real lo_;
192 real hi_;
193 idx degree_;
194 idx n_;
195 mutable vec residual_;
196 mutable vec direction_;
197 mutable vec work_;
198};

211/// @brief Build a Chebyshev preconditioner from a known \f$\lambda_{\min}\f$,
212/// estimating only the upper bound.
213///
214/// Power iteration gives \f$\lambda_{\max}\f$ cheaply and reliably. There is no
215/// equally cheap estimate of \f$\lambda_{\min}\f$, and it is deliberately *not*
216/// inferred here from an assumed condition number: too large a value excludes
217/// the bottom of the spectrum, which makes the polynomial indefinite and
218/// destroys the property PCG depends on. Supply a genuine lower bound — from the
219/// problem's physics, from a shift or regularization parameter, or from
220/// `num::lanczos` — or use the two-bound overload directly.
221///
222/// @param A Self-adjoint linear operator.
223/// @param lambda_min A true lower bound on the spectrum, greater than zero.
224/// @param degree Polynomial degree, and operator applications per solve.
225template <class Op>
226[[nodiscard]] inline chebyshev_preconditioner<Op>
227chebyshev_preconditioner_from_below(const Op &A, real lambda_min, idx degree = 4) {
228 const real hi = estimate_largest_eigenvalue(A);
229 if (!(hi > 0.0)) {
230 throw std::invalid_argument("chebyshev: could not estimate the spectral radius");
231 }
232 if (!(lambda_min > 0.0) || lambda_min >= hi) {
233 throw std::invalid_argument(
234 "chebyshev: lambda_min must be positive and below the estimated spectral radius");
235 }
236 return chebyshev_preconditioner<Op>(A, lambda_min, hi, degree);
237}
