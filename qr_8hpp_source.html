45inline qr_result qr(const mat &A) {
46 const idx m = A.rows();
47 const idx n = A.cols();
48 const idx r = (m > n) ? n : m - 1;
49
50 mat R = A;
51 array<array<real>> vs(r);
52 array<real> betas(r, 0.0);
53 array<real> tau(r), v(m), work(n);
54 // A <- compact Householder QR; reflector tails remain below R's diagonal.
55 kernel::qr_factor_blocked(R.data(), n, m, n, tau.data(), v.data(), work.data());
56 for (idx k = 0; k < r; ++k) {
57 const idx len = m - k;
58 betas[k] = tau[k];
59 vs[k].assign(len, real(0));
60 vs[k][0] = real(1);
61 for (idx i = 1; i < len; ++i) vs[k][i] = R((k + i), k);
62 }
63
64 mat Q(m, m, real(0));
65 for (idx i = 0; i < m; ++i) {
66 Q(i, i) = real(1);
67 }
68
69 for (idx k = r; k-- > 0;) {
70 if (betas[k] == 0.0) {
71 continue;
72 }
73 const array<real> &v = vs[k];
74 const idx len = static_cast<idx>(v.size());
75 array<real> work(m - k);
76 kernel::householder_left(&Q(k, k), m, v.data(), betas[k], len, m - k, work.data());
77 }
78
79 for (idx i = 1; i < m; ++i) {
80 for (idx j = 0; j < std::min(i, n); ++j) {
81 R(i, j) = real(0);
82 }
83 }
84
85 return {std::move(Q), std::move(R)};
86}

103 }

108 Rmat(i, j) = 0.0;
109 }
110 }
111
112 mat Q(m, m, 0.0);
113 for (idx j = 0; j < k; ++j) {
114 for (idx i = 0; i < m; ++i) {
115 Q(i, j) = R(i, j);
116 }
117 }

121 tau.data());

141inline void qr_solve(const qr_result &f, const vec &b, vec &x) {
142 const idx m = f.Q.rows();
143 const idx n = f.R.cols();
144
145 vec y(m, real(0));
146 for (idx i = 0; i < m; ++i) {
147 for (idx j = 0; j < m; ++j) {
148 y[i] += f.Q(j, i) * b[j];
149 }
150 }
151
152 vec xv(n, real(0));
153 for (idx i = n; i-- > 0;) {
154 xv[i] = y[i];
155 for (idx j = i + 1; j < n; ++j) {
156 xv[i] -= f.R(i, j) * xv[j];
157 }
158 xv[i] /= f.R(i, i);
159 }
160
161 x = std::move(xv);
162}
