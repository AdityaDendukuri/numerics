33/// @brief Cholesky factorization \f$A = L L^T\f$ for symmetric positive definite \f$A\f$.
34///
35/// Writes the lower triangular factor into `L` (row-major, \f$n \times n\f$) and
36/// zeroes the strict upper triangle. `L` and `A` may alias.
37///
38/// Returns false as soon as a non-positive pivot appears, which is the exact point
39/// at which \f$A\f$ is shown not to be positive definite — the factorization is the
40/// definitive test, where sampling the quadratic form is only evidence.
41///
42/// @param L Output lower triangular factor, n*n.
43/// @param A Input symmetric matrix, n*n; only the lower triangle is read.
44/// @param n mat dimension.
45template <std::floating_point T>
46[[nodiscard]] inline bool cholesky(T *L, const T *A, idx n) noexcept {
47 for (idx i = 0; i < n; ++i) {
48 for (idx j = 0; j <= i; ++j) {
49 T sum = A[(i * n) + j];
50 for (idx k = 0; k < j; ++k) {
51 sum -= L[(i * n) + k] * L[(j * n) + k];
52 }
53 if (i == j) {
54 if (!(sum > T(0))) {
55 return false;
56 }
57 L[(i * n) + j] = std::sqrt(sum);
58 } else {
59 L[(i * n) + j] = sum / L[(j * n) + j];
60 }
61 }
62 for (idx j = i + 1; j < n; ++j) {
63 L[(i * n) + j] = T(0);
64 }
65 }
66 return true;
67}

71/// @brief Unblocked lower Cholesky of an `n x n` block with row stride `lda`,
72/// writing zeros above the diagonal.
73template <std::floating_point T>
74[[nodiscard]] inline bool cholesky_block(T *A, idx lda, idx n) noexcept {
75 for (idx i = 0; i < n; ++i) {
76 for (idx j = 0; j <= i; ++j) {
77 T sum = A[(i * lda) + j];
78 for (idx p = 0; p < j; ++p) {
79 sum -= A[(i * lda) + p] * A[(j * lda) + p];
80 }
81 if (i == j) {
82 if (!(sum > T(0))) {
83 return false;
84 }
85 A[(i * lda) + j] = std::sqrt(sum);
86 } else {
87 A[(i * lda) + j] = sum / A[(j * lda) + j];
88 }
89 }
90 for (idx j = i + 1; j < n; ++j) {
91 A[(i * lda) + j] = T(0);
92 }
93 }
94 return true;
95}

97/// @brief Blocked lower Cholesky of an `n x n` block with row stride `lda`.
98///
99/// Diagonal blocks wider than the `trsm`/`syrk` block are factored by
100/// recursion, so the panel solve and trailing update always see blocks long
101/// enough for their internal `gemm` to matter.
102template <std::floating_point T>
103[[nodiscard]] inline bool cholesky_blocked(T *A, idx lda, idx n, idx block_size) noexcept {
104 for (idx kk = 0; kk < n; kk += block_size) {
105 const idx kb = std::min(block_size, n - kk);
106 T *diagonal = A + (kk * lda) + kk;
107
108 // L11 <- chol(A11), retaining the parent row stride.
109 const bool ok = kb > trsm_block ? cholesky_blocked(diagonal, lda, kb, trsm_block)
110 : cholesky_block(diagonal, lda, kb);
111 if (!ok) {
112 return false;
113 }
114
115 const idx trailing = n - (kk + kb);
116 if (trailing == 0) {
117 continue;
118 }
119
120 // L21 <- A21 * L11^{-T}.
121 trsm_lower_transpose_right_inplace(A + ((kk + kb) * lda) + kk, lda, diagonal, lda, trailing,
122 kb);
123
124 // A22 <- A22 - L21*L21^T (lower triangle only).
125 syrk_lower(A + (((kk + kb) * lda) + (kk + kb)), lda, A + (((kk + kb) * lda) + kk), lda,
126 T(-1), T(1), trailing, kb);
127 }
128 return true;
129}

133/// @brief In-place blocked Cholesky factorization, `A <- L` with `A = L*L^T`.
134///
135/// The lower triangle is factored in panels of `block_size` columns. The panel
136/// solve and trailing update are the blocked `trsm` and `syrk` kernels, so past
137/// a few hundred rows the work runs at `gemm` speed; a diagonal block wider
138/// than their internal block is factored recursively. The default panel is
139/// wide so those updates are long; the panel factorization itself is
140/// \f$O(n b^2)\f$ and stays a small fraction for `b <= n/4`.
141template <std::floating_point T>
142[[nodiscard]] inline bool cholesky_blocked(T *A, idx n, idx block_size = 256) noexcept {
143 if (n == 0) {
144 return true;
145 }
146 if (block_size == 0) {
147 block_size = 1;
148 }
149 if (!detail::cholesky_blocked(A, n, n, block_size)) {
150 return false;
151 }
152 // The packed lower factor is the public result; make the unused triangle explicit.
153 for (idx i = 0; i < n; ++i) {
154 for (idx j = i + 1; j < n; ++j) {
155 A[(i * n) + j] = T(0);
156 }
157 }
158 return true;
159}

161/// @brief Solve \f$A x = b\f$ from a Cholesky factor, via \f$L y = b\f$ then \f$L^T x = y\f$.
162///
163/// `x` and `b` may alias.
164template <std::floating_point T>
165inline void cholesky_solve(T *x, const T *L, const T *b, idx n) noexcept {
166 // The public contract permits x == b, so select the explicitly alias-safe
167 // raw path rather than making a false restrict promise to the compiler.
168 trsv_lower(contract::alias_safe, x, L, b, n);
169 trsv_transpose_lower(x, L, n, n);
170}

172/// @brief Factors a batch of independent small SPD matrices.
173///
174/// Each matrix is row-major and begins `matrix_stride` elements after the
175/// previous one. `L` and `A` may be the same batch. Per-system status avoids
176/// throwing away successful factorizations when one member is not SPD.
177template <std::floating_point T>
178[[nodiscard]] inline bool cholesky_batched(T *L, const T *A, idx n, idx batch_count,
179 idx matrix_stride, bool *success = nullptr) noexcept {
180 bool all_ok = true;
181 for (idx batch = 0; batch < batch_count; ++batch) {
182 const bool ok = cholesky(L + (batch * matrix_stride), A + (batch * matrix_stride), n);
183 if (success != nullptr) {
184 success[batch] = ok;
185 }
186 all_ok = all_ok && ok;
187 }
188 return all_ok;
189}

191/// @brief Solves a batch of independent systems from batched Cholesky factors.
192/// `x` and `b` may be the same batch.
193template <std::floating_point T>
194inline void cholesky_solve_batched(T *x, const T *L, const T *b, idx n, idx batch_count,
195 idx matrix_stride, idx vector_stride) noexcept {
196 for (idx batch = 0; batch < batch_count; ++batch) {
197 cholesky_solve(x + (batch * vector_stride), L + (batch * matrix_stride),

202/// @brief LU factorization \f$PA = LU\f$ with partial pivoting, packed in place.
203///
204/// On return `LU` holds the unit-lower factor below the diagonal and the upper
205/// factor on and above it; `piv[i]` is the row swapped with row i. Returns false
206/// when an exactly zero pivot column is found, i.e. when A is singular.
207///
208/// @param LU In/out packed factors, n*n row-major; pass a copy of A.
209/// @param piv Output pivot sequence, length n.
210/// @param n mat dimension.
211template <std::floating_point T, class Index>
212[[nodiscard]] inline bool lu_factor(T *NUM_K_RESTRICT LU, Index *NUM_K_RESTRICT piv,
213 idx n) noexcept {
214 bool nonsingular = true;
215 for (idx k = 0; k < n; ++k) {
216 idx pivot_row = k;
217 T best = std::abs(LU[(k * n) + k]);
218 for (idx i = k + 1; i < n; ++i) {
219 const T candidate = std::abs(LU[(i * n) + k]);
220 if (candidate > best) {
221 best = candidate;
222 pivot_row = i;
223 }
224 }
225 piv[k] = static_cast<Index>(pivot_row);
226
227 if (pivot_row != k) {
228 swap_rows(LU, n, k, pivot_row, n);
229 }
230 const T pivot = LU[(k * n) + k];
231 if (pivot == T(0)) {
232 nonsingular = false;
233 continue;
234 }
235 for (idx i = k + 1; i < n; ++i) {
236 const T factor = LU[(i * n) + k] / pivot;
237 LU[(i * n) + k] = factor;
238 for (idx j = k + 1; j < n; ++j) {
239 LU[(i * n) + j] -= factor * LU[(k * n) + j];
240 }
241 }
242 }
243 return nonsingular;
244}

246/// @brief Blocked variant of `lu_factor`: same packed \f$PA = LU\f$ contract, panel-at-a-time.
247template <std::floating_point T, class Index>
248[[nodiscard]] inline bool lu_factor_blocked(T *NUM_K_RESTRICT LU, Index *NUM_K_RESTRICT piv, idx n,
249 idx block_size = 64) noexcept {
250 if (block_size == 0)
251 block_size = 1;
252 bool nonsingular = true;
253 for (idx kk = 0; kk < n; kk += block_size) {
254 const idx kb = std::min(block_size, n - kk), panel_end = kk + kb;
255 for (idx k = kk; k < panel_end; ++k) {
256 idx pivot_row = k;
257 T best = std::abs(LU[(k * n) + k]);
258 for (idx i = k + 1; i < n; ++i) {
259 const T candidate = std::abs(LU[(i * n) + k]);
260 if (candidate > best) {
261 best = candidate;
262 pivot_row = i;
263 }
264 }
265 piv[k] = static_cast<Index>(pivot_row);
266 if (pivot_row != k)
267 swap_rows(LU, n, k, pivot_row, n);
268 const T pivot = LU[(k * n) + k];
269 if (pivot == T(0)) {
270 nonsingular = false;
271 continue;
272 }
273 for (idx i = k + 1; i < n; ++i) {
274 const T factor = LU[(i * n) + k] / pivot;
275 LU[(i * n) + k] = factor;
276 for (idx j = k + 1; j < panel_end; ++j)
277 LU[(i * n) + j] -= factor * LU[(k * n) + j];
278 }
279 }
280 const idx trailing = n - panel_end;
281 if (trailing == 0)
282 continue;
283 // U12 <- L11^{-1} A12.
284 trsm_unit_lower_inplace(LU + (kk * n) + panel_end, n, LU + (kk * n) + kk, n, kb, trailing);
285 // A22 <- A22 - L21*U12.
286 gemm(LU + (panel_end * n) + panel_end, n, LU + (panel_end * n) + kk, n,
287 LU + (kk * n) + panel_end, n, T(-1), T(1), trailing, trailing, kb);
288 }
289 return nonsingular;
290}

292/// @brief Solve \f$A x = b\f$ from a packed \f$PA = LU\f$ factorization.
293template <std::floating_point T, class Index>
294inline void lu_solve(T *NUM_K_RESTRICT x, const T *NUM_K_RESTRICT LU,
295 const Index *NUM_K_RESTRICT piv, const T *b, idx n) noexcept {
296 for (idx i = 0; i < n; ++i) {
297 x[i] = b[i];
298 }
299 for (idx k = 0; k < n; ++k) {
300 const idx p = static_cast<idx>(piv[k]);
301 if (p != k) {
302 const T tmp = x[k];
303 x[k] = x[p];
304 x[p] = tmp;
305 }
306 }
307 // Unit-lower forward substitution, then upper back substitution.
308 for (idx i = 1; i < n; ++i) {
309 T sum = x[i];
310 for (idx j = 0; j < i; ++j) {
311 sum -= LU[(i * n) + j] * x[j];
312 }
313 x[i] = sum;
314 }
315 for (idx i = n; i-- > 0;) {
316 T sum = x[i];
317 for (idx j = i + 1; j < n; ++j) {
318 sum -= LU[(i * n) + j] * x[j];
319 }
320 x[i] = sum / LU[(i * n) + i];
321 }
322}

324/// @brief Inverse from packed LU factors: `inverse <- A^{-1}`, one column at a time.
325///
326/// Column `j` of the inverse is the solve `A x = e_j`, so this is `n` calls
327/// of `lu_solve` written without the per-column copy. Out of place: every
328/// column's solve reads the whole factor, so the result cannot overwrite it.
329/// `work` holds `n` elements.
330template <std::floating_point T, class Index>
331inline void lu_invert(T *NUM_K_RESTRICT inverse, const T *NUM_K_RESTRICT LU,
332 const Index *NUM_K_RESTRICT piv, idx n, T *NUM_K_RESTRICT work) noexcept {
333 for (idx col = 0; col < n; ++col) {
334 for (idx i = 0; i < n; ++i)
335 work[i] = (i == col) ? T(1) : T(0);
336 for (idx k = 0; k < n; ++k) {
337 const idx p = static_cast<idx>(piv[k]);
338 if (p != k)
339 std::swap(work[k], work[p]);
340 }
341 for (idx i = 1; i < n; ++i) {
342 T sum = work[i];
343 for (idx j = 0; j < i; ++j)
344 sum -= LU[(i * n) + j] * work[j];
345 work[i] = sum;
346 }
347 for (idx i = n; i-- > 0;) {
348 T sum = work[i];
349 for (idx j = i + 1; j < n; ++j)
350 sum -= LU[(i * n) + j] * work[j];
351 work[i] = sum / LU[(i * n) + i];
352 }
353 for (idx i = 0; i < n; ++i) {
354 inverse[(i * n) + col] = work[i];
355 }
356 }
357}

359/// @brief Invert lower Cholesky factor in place to produce A^{-1} = L^{-T} L^{-1}.
360template <std::floating_point T>
361inline void cholesky_invert(T *NUM_K_RESTRICT L, idx n, T *NUM_K_RESTRICT work) noexcept {
362 for (idx j = 0; j < n; ++j) {
363 L[(j * n) + j] = T(1) / L[(j * n) + j];
364 for (idx i = j + 1; i < n; ++i) {
365 T sum = T(0);
366 for (idx k = j; k < i; ++k) {
367 sum -= L[(i * n) + k] * L[(k * n) + j];
368 }
369 L[(i * n) + j] = sum / L[(i * n) + i];
370 }
371 }
372 for (idx i = 0; i < n; ++i) {
373 for (idx j = 0; j <= i; ++j) {
374 T sum = T(0);
375 for (idx k = i; k < n; ++k) {
376 sum += L[(k * n) + i] * L[(k * n) + j];
377 }
378 work[(i * n) + j] = sum;
379 work[(j * n) + i] = sum;
380 }
381 }
382 std::copy_n(work, n * n, L);
383}

385/// @brief banded LU factorization with partial pivoting over LAPACK-compatible band storage.
386template <std::floating_point T, class Index>
387[[nodiscard]] inline bool banded_factor(T *NUM_K_RESTRICT ab, idx ldab, idx n, idx kl, idx ku,
388 Index *NUM_K_RESTRICT ipiv) noexcept {
389 for (idx i = 0; i < n; ++i)
390 ipiv[i] = static_cast<Index>(i);
391 const idx kv = ku + kl;
392 for (idx j = 0; j < n; ++j) {
393 const idx last_row = std::min(j + kl, n - 1);
394 T max_val = std::abs(ab[kv + (j * ldab)]);
395 idx pivot = j;
396 for (idx i = j + 1; i <= last_row; ++i) {
397 T val = std::abs(ab[kv + i - j + (j * ldab)]);
398 if (val > max_val) {
399 max_val = val;
400 pivot = i;
401 }
402 }
403 if (max_val == T(0))
404 return false;
405 ipiv[j] = static_cast<Index>(pivot);
406 if (pivot != j) {
407 const idx col_start = (j > ku) ? j - ku : 0;
408 const idx col_end = std::min(j + kl, n - 1);
409 for (idx col = col_start; col <= col_end; ++col) {
410 idx band_j = kl + ku + j - col;
411 idx band_pivot = kl + ku + pivot - col;
412 if (band_j < ldab && band_pivot < ldab) {
413 std::swap(ab[band_j + (col * ldab)], ab[band_pivot + (col * ldab)]);
414 }
415 }
416 }
417 T pivot_val = ab[kv + (j * ldab)];
418 const idx num_rows = std::min(kl, n - j - 1);
419 if (num_rows > 0 && pivot_val != T(0)) {
420 T inv_pivot = T(1) / pivot_val;
421 for (idx i = 1; i <= num_rows; ++i) {
422 ab[kv + i + (j * ldab)] *= inv_pivot;
423 }
424 const idx col_end = std::min(j + ku, n - 1);
425 for (idx k = j + 1; k <= col_end; ++k) {
426 T akj = ab[kv + j - k + (k * ldab)];
427 if (akj != T(0)) {
428 for (idx i = 1; i <= num_rows; ++i) {
429 idx band_row = kv + j + i - k;
430 if (band_row < ldab) {
431 ab[band_row + (k * ldab)] -= ab[kv + i + (j * ldab)] * akj;
432 }
433 }
434 }
435 }
436 }
437 }
438 return true;
439}

441/// @brief banded LU solve for x = b over LAPACK-compatible factored band storage.
442template <std::floating_point T, class Index>
443inline void banded_solve(T *x, const T *NUM_K_RESTRICT ab, idx ldab, idx n, idx kl, idx ku,
444 const Index *NUM_K_RESTRICT ipiv) noexcept {
445 const idx kv = ku + kl;
446 for (idx i = 0; i < n; ++i) {
447 if (static_cast<idx>(ipiv[i]) != i) {
448 std::swap(x[i], x[static_cast<idx>(ipiv[i])]);
449 }
450 }
451 for (idx j = 0; j < n; ++j) {
452 if (x[j] != T(0)) {
453 const idx last = std::min(j + kl, n - 1);
454 T xj = x[j];
455 for (idx i = j + 1; i <= last; ++i) {
456 x[i] -= ab[kv + i - j + (j * ldab)] * xj;
457 }
458 }
459 }
460 for (idx j = n; j > 0; --j) {
461 const idx col = j - 1;
462 x[col] /= ab[kv + (col * ldab)];
463 if (x[col] != T(0)) {
464 const idx first = (col > ku) ? col - ku : 0;
465 T xc = x[col];
466 for (idx i = first; i < col; ++i) {
467 x[i] -= ab[kv + i - col + (col * ldab)] * xc;
468 }
469 }
470 }
