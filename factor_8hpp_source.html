33/// @brief Cholesky factorization \f$A = L L^T\f$ for symmetric positive definite \f$A\f$.
34///
35/// Writes the lower triangular factor into `L` (row-major, \f$n \times n\f$) and
36/// zeroes the strict upper triangle. `L` and `A` may alias.
37///
38/// Returns false as soon as a non-positive pivot appears, which is the exact point
39/// at which \f$A\f$ is shown not to be positive definite — the factorization is the
40/// definitive test, where sampling the quadratic form is only evidence.
41///
42/// @param L Output lower triangular factor, n*n.
43/// @param A Input symmetric matrix, n*n; only the lower triangle is read.
44/// @param n mat dimension.
45template <std::floating_point T>
46[[nodiscard]] inline bool cholesky(T *L, const T *A, idx n) noexcept {
47 for (idx i = 0; i < n; ++i) {
48 for (idx j = 0; j <= i; ++j) {
49 T sum = A[(i * n) + j];
50 for (idx k = 0; k < j; ++k) {
51 sum -= L[(i * n) + k] * L[(j * n) + k];
52 }
53 if (i == j) {
54 if (!(sum > T(0))) {
55 return false;
56 }
57 L[(i * n) + j] = std::sqrt(sum);
58 } else {
59 L[(i * n) + j] = sum / L[(j * n) + j];
60 }
61 }
62 for (idx j = i + 1; j < n; ++j) {
63 L[(i * n) + j] = T(0);
64 }
65 }
66 return true;
67}

69/// @brief In-place blocked Cholesky factorization, `A <- L` with `A = L*L^T`.
70///
71/// The lower triangle is factored in panels. The panel solve and trailing
72/// update are delegated to the raw TRSM/SYRK kernels so larger problems expose
73/// the same computational spine as the standalone dense primitives.
74template <std::floating_point T>
75[[nodiscard]] inline bool cholesky_blocked(T *A, idx n, idx block_size = 64) noexcept {
76 if (n == 0) {
77 return true;
78 }
79 if (block_size == 0) {
80 block_size = 1;
81 }
82 for (idx kk = 0; kk < n; kk += block_size) {
83 const idx kb = std::min(block_size, n - kk);
84
85 // L11 <- chol(A11), retaining the parent row stride.
86 for (idx i = 0; i < kb; ++i) {
87 for (idx j = 0; j <= i; ++j) {
88 T sum = A[((kk + i) * n) + (kk + j)];
89 for (idx p = 0; p < j; ++p) {
90 sum -= A[((kk + i) * n) + (kk + p)] * A[((kk + j) * n) + (kk + p)];
91 }
92 if (i == j) {
93 if (!(sum > T(0))) {
94 return false;
95 }
96 A[((kk + i) * n) + (kk + j)] = std::sqrt(sum);
97 } else {
98 A[((kk + i) * n) + (kk + j)] = sum / A[((kk + j) * n) + (kk + j)];
99 }
100 }
101 for (idx j = i + 1; j < kb; ++j) {
102 A[((kk + i) * n) + (kk + j)] = T(0);
103 }
104 }
105
106 const idx trailing = n - (kk + kb);
107 if (trailing == 0) {
108 continue;
109 }
110
111 // L21 <- A21 * L11^{-T}.
112 trsm_lower_transpose_right_inplace(A + ((kk + kb) * n) + kk, n,
113 A + (kk * n) + kk, n, trailing, kb);
114
115 // A22 <- A22 - L21*L21^T (lower triangle only).
116 syrk_lower(A + (((kk + kb) * n) + (kk + kb)), n,
117 A + (((kk + kb) * n) + kk), n, T(-1), T(1), trailing, kb);
118 }
119 // The packed lower factor is the public result; make the unused triangle explicit.
120 for (idx i = 0; i < n; ++i) {
121 for (idx j = i + 1; j < n; ++j) {
122 A[(i * n) + j] = T(0);
123 }
124 }
125 return true;
126}

128/// @brief Solve \f$A x = b\f$ from a Cholesky factor, via \f$L y = b\f$ then \f$L^T x = y\f$.
129///
130/// `x` and `b` may alias.
131template <std::floating_point T>
132inline void cholesky_solve(T *x, const T *L, const T *b, idx n) noexcept {
133 // The public contract permits x == b, so select the explicitly alias-safe
134 // raw path rather than making a false restrict promise to the compiler.
135 trsv_lower(contract::alias_safe, x, L, b, n);
136 trsv_transpose_lower(x, L, n, n);
137}

139/// @brief Factors a batch of independent small SPD matrices.
140///
141/// Each matrix is row-major and begins `matrix_stride` elements after the
142/// previous one. `L` and `A` may be the same batch. Per-system status avoids
143/// throwing away successful factorizations when one member is not SPD.
144template <std::floating_point T>
145[[nodiscard]] inline bool cholesky_batched(T *L, const T *A, idx n, idx batch_count,
146 idx matrix_stride, bool *success = nullptr) noexcept {
147 bool all_ok = true;
148 for (idx batch = 0; batch < batch_count; ++batch) {
149 const bool ok = cholesky(L + (batch * matrix_stride), A + (batch * matrix_stride), n);
150 if (success != nullptr) {
151 success[batch] = ok;
152 }
153 all_ok = all_ok && ok;
154 }
155 return all_ok;
156}

158/// @brief Solves a batch of independent systems from batched Cholesky factors.
159/// `x` and `b` may be the same batch.
160template <std::floating_point T>
161inline void cholesky_solve_batched(T *x, const T *L, const T *b, idx n, idx batch_count,
162 idx matrix_stride, idx vector_stride) noexcept {
163 for (idx batch = 0; batch < batch_count; ++batch) {
164 cholesky_solve(x + (batch * vector_stride), L + (batch * matrix_stride),
165 b + (batch * vector_stride), n);
166 }

169/// @brief LU factorization \f$PA = LU\f$ with partial pivoting, packed in place.
170///
171/// On return `LU` holds the unit-lower factor below the diagonal and the upper
172/// factor on and above it; `piv[i]` is the row swapped with row i. Returns false
173/// when an exactly zero pivot column is found, i.e. when A is singular.
174///
175/// @param LU In/out packed factors, n*n row-major; pass a copy of A.
176/// @param piv Output pivot sequence, length n.
177/// @param n mat dimension.
178template <std::floating_point T, class Index>
179[[nodiscard]] inline bool lu_factor(T *NUM_K_RESTRICT LU, Index *NUM_K_RESTRICT piv,
180 idx n) noexcept {
181 bool nonsingular = true;
182 for (idx k = 0; k < n; ++k) {
183 idx pivot_row = k;
184 T best = std::abs(LU[(k * n) + k]);
185 for (idx i = k + 1; i < n; ++i) {
186 const T candidate = std::abs(LU[(i * n) + k]);
187 if (candidate > best) {
188 best = candidate;
189 pivot_row = i;
190 }
191 }
192 piv[k] = static_cast<Index>(pivot_row);
193
194 if (pivot_row != k) {
195 swap_rows(LU, n, k, pivot_row, n);
196 }
197 const T pivot = LU[(k * n) + k];
198 if (pivot == T(0)) {
199 nonsingular = false;
200 continue;
201 }
202 for (idx i = k + 1; i < n; ++i) {
203 const T factor = LU[(i * n) + k] / pivot;
204 LU[(i * n) + k] = factor;
205 for (idx j = k + 1; j < n; ++j) {
206 LU[(i * n) + j] -= factor * LU[(k * n) + j];
207 }
208 }
209 }
210 return nonsingular;
211}

213/// @brief Blocked variant of `lu_factor`: same packed \f$PA = LU\f$ contract, panel-at-a-time.
214template <std::floating_point T, class Index>
215[[nodiscard]] inline bool lu_factor_blocked(T *NUM_K_RESTRICT LU, Index *NUM_K_RESTRICT piv, idx n,
216 idx block_size = 64) noexcept {
217 if (block_size == 0) block_size = 1;
218 bool nonsingular = true;
219 for (idx kk = 0; kk < n; kk += block_size) {
220 const idx kb = std::min(block_size, n - kk), panel_end = kk + kb;
221 for (idx k = kk; k < panel_end; ++k) {
222 idx pivot_row = k;
223 T best = std::abs(LU[(k * n) + k]);
224 for (idx i = k + 1; i < n; ++i) {
225 const T candidate = std::abs(LU[(i * n) + k]);
226 if (candidate > best) { best = candidate; pivot_row = i; }
227 }
228 piv[k] = static_cast<Index>(pivot_row);
229 if (pivot_row != k) swap_rows(LU, n, k, pivot_row, n);
230 const T pivot = LU[(k * n) + k];
231 if (pivot == T(0)) { nonsingular = false; continue; }
232 for (idx i = k + 1; i < n; ++i) {
233 const T factor = LU[(i * n) + k] / pivot;
234 LU[(i * n) + k] = factor;
235 for (idx j = k + 1; j < panel_end; ++j)
236 LU[(i * n) + j] -= factor * LU[(k * n) + j];
237 }
238 }
239 const idx trailing = n - panel_end;
240 if (trailing == 0) continue;
241 // U12 <- L11^{-1} A12.
242 trsm_unit_lower_inplace(LU + (kk * n) + panel_end, n, LU + (kk * n) + kk, n, kb, trailing);
243 // A22 <- A22 - L21*U12.
244 gemm(LU + (panel_end * n) + panel_end, n, LU + (panel_end * n) + kk, n,
245 LU + (kk * n) + panel_end, n, T(-1), T(1), trailing, trailing, kb);
246 }
247 return nonsingular;
248}

250/// @brief Solve \f$A x = b\f$ from a packed \f$PA = LU\f$ factorization.
251template <std::floating_point T, class Index>
252inline void lu_solve(T *NUM_K_RESTRICT x, const T *NUM_K_RESTRICT LU,
253 const Index *NUM_K_RESTRICT piv, const T *b, idx n) noexcept {
254 for (idx i = 0; i < n; ++i) {
255 x[i] = b[i];
256 }
257 for (idx k = 0; k < n; ++k) {
258 const idx p = static_cast<idx>(piv[k]);
259 if (p != k) {
260 const T tmp = x[k];
261 x[k] = x[p];
262 x[p] = tmp;
263 }
264 }
265 // Unit-lower forward substitution, then upper back substitution.
266 for (idx i = 1; i < n; ++i) {
267 T sum = x[i];
268 for (idx j = 0; j < i; ++j) {
269 sum -= LU[(i * n) + j] * x[j];
270 }
271 x[i] = sum;
272 }
273 for (idx i = n; i-- > 0;) {
274 T sum = x[i];
275 for (idx j = i + 1; j < n; ++j) {
276 sum -= LU[(i * n) + j] * x[j];
277 }
278 x[i] = sum / LU[(i * n) + i];
279 }
280}

282/// @brief Invert packed LU factors in place: A <- A^{-1}.
283template <std::floating_point T, class Index>
284inline void lu_invert(T *NUM_K_RESTRICT LU, const Index *NUM_K_RESTRICT piv, idx n,
285 T *NUM_K_RESTRICT work) noexcept {
286 for (idx col = 0; col < n; ++col) {
287 for (idx i = 0; i < n; ++i) work[i] = (i == col) ? T(1) : T(0);
288 for (idx k = 0; k < n; ++k) {
289 const idx p = static_cast<idx>(piv[k]);
290 if (p != k) std::swap(work[k], work[p]);
291 }
292 for (idx i = 1; i < n; ++i) {
293 T sum = work[i];
294 for (idx j = 0; j < i; ++j) sum -= LU[(i * n) + j] * work[j];
295 work[i] = sum;
296 }
297 for (idx i = n; i-- > 0;) {
298 T sum = work[i];
299 for (idx j = i + 1; j < n; ++j) sum -= LU[(i * n) + j] * work[j];
300 work[i] = sum / LU[(i * n) + i];
301 }
302 for (idx i = 0; i < n; ++i) {
303 LU[(i * n) + col] = work[i];
304 }
305 }
306}

308/// @brief Invert lower Cholesky factor in place to produce A^{-1} = L^{-T} L^{-1}.
309template <std::floating_point T>
310inline void cholesky_invert(T *NUM_K_RESTRICT L, idx n, T *NUM_K_RESTRICT work) noexcept {
311 for (idx j = 0; j < n; ++j) {
312 L[(j * n) + j] = T(1) / L[(j * n) + j];
313 for (idx i = j + 1; i < n; ++i) {
314 T sum = T(0);
315 for (idx k = j; k < i; ++k) {
316 sum -= L[(i * n) + k] * L[(k * n) + j];
317 }
318 L[(i * n) + j] = sum / L[(i * n) + i];
319 }
320 }
321 for (idx i = 0; i < n; ++i) {
322 for (idx j = 0; j <= i; ++j) {
323 T sum = T(0);
324 for (idx k = i; k < n; ++k) {
325 sum += L[(k * n) + i] * L[(k * n) + j];
326 }
327 work[(i * n) + j] = sum;
328 work[(j * n) + i] = sum;
329 }
330 }
331 std::copy_n(work, n * n, L);
332}

334/// @brief banded LU factorization with partial pivoting over LAPACK-compatible band storage.
335template <std::floating_point T, class Index>
336[[nodiscard]] inline bool banded_factor(T *NUM_K_RESTRICT ab, idx ldab, idx n, idx kl, idx ku,
337 Index *NUM_K_RESTRICT ipiv) noexcept {
338 for (idx i = 0; i < n; ++i) ipiv[i] = static_cast<Index>(i);
339 const idx kv = ku + kl;
340 for (idx j = 0; j < n; ++j) {
341 const idx last_row = std::min(j + kl, n - 1);
342 T max_val = std::abs(ab[kv + (j * ldab)]);
343 idx pivot = j;
344 for (idx i = j + 1; i <= last_row; ++i) {
345 T val = std::abs(ab[kv + i - j + (j * ldab)]);
346 if (val > max_val) { max_val = val; pivot = i; }
347 }
348 if (max_val == T(0)) return false;
349 ipiv[j] = static_cast<Index>(pivot);
350 if (pivot != j) {
351 const idx col_start = (j > ku) ? j - ku : 0;
352 const idx col_end = std::min(j + kl, n - 1);
353 for (idx col = col_start; col <= col_end; ++col) {
354 idx band_j = kl + ku + j - col;
355 idx band_pivot = kl + ku + pivot - col;
356 if (band_j < ldab && band_pivot < ldab) {
357 std::swap(ab[band_j + (col * ldab)], ab[band_pivot + (col * ldab)]);
358 }
359 }
360 }
361 T pivot_val = ab[kv + (j * ldab)];
362 const idx num_rows = std::min(kl, n - j - 1);
363 if (num_rows > 0 && pivot_val != T(0)) {
364 T inv_pivot = T(1) / pivot_val;
365 for (idx i = 1; i <= num_rows; ++i) {
366 ab[kv + i + (j * ldab)] *= inv_pivot;
367 }
368 const idx col_end = std::min(j + ku, n - 1);
369 for (idx k = j + 1; k <= col_end; ++k) {
370 T akj = ab[kv + j - k + (k * ldab)];
371 if (akj != T(0)) {
372 for (idx i = 1; i <= num_rows; ++i) {
373 idx band_row = kv + j + i - k;
374 if (band_row < ldab) {
375 ab[band_row + (k * ldab)] -= ab[kv + i + (j * ldab)] * akj;
376 }
377 }
378 }
379 }
380 }
381 }
382 return true;
383}

385/// @brief banded LU solve for x = b over LAPACK-compatible factored band storage.
386template <std::floating_point T, class Index>
387inline void banded_solve(T *x, const T *NUM_K_RESTRICT ab, idx ldab, idx n, idx kl, idx ku,
388 const Index *NUM_K_RESTRICT ipiv) noexcept {
389 const idx kv = ku + kl;
390 for (idx i = 0; i < n; ++i) {
391 if (static_cast<idx>(ipiv[i]) != i) {
392 std::swap(x[i], x[static_cast<idx>(ipiv[i])]);
393 }
394 }
395 for (idx j = 0; j < n; ++j) {
396 if (x[j] != T(0)) {
397 const idx last = std::min(j + kl, n - 1);
398 T xj = x[j];
399 for (idx i = j + 1; i <= last; ++i) {
400 x[i] -= ab[kv + i - j + (j * ldab)] * xj;
401 }
402 }
403 }
404 for (idx j = n; j > 0; --j) {
405 const idx col = j - 1;
406 x[col] /= ab[kv + (col * ldab)];
407 if (x[col] != T(0)) {
408 const idx first = (col > ku) ? col - ku : 0;
409 T xc = x[col];
410 for (idx i = first; i < col; ++i) {
411 x[i] -= ab[kv + i - col + (col * ldab)] * xc;
412 }
413 }
414 }
415}
