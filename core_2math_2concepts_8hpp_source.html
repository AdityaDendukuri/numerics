1/// @file concepts.hpp
2/// @brief The concept hierarchy: a scalar field, successively equipped with more structure.
3///
4/// Every concept here refines the one above it, adding one operation the compiler can check
5/// or one law the caller asserts. A normed space is a vector space equipped with a norm;
6/// an inner product space is a normed space equipped with an inner product; and so on.
7///
8/// No concept in this library asserts a mathematical property with a bare `requires`
9/// block. It names the concept it refines and adds only what is new.
10///
11/// ```
12/// field<T> + - * /, 0, 1
13/// └ additive_group<V> dimension, zero_like, copy
14/// └ vector_space<V> + scale, axpy ── law::vector_space
15/// └ normed_space<V> + norm ── law::normed_space
16/// └ inner_product_space<V> + inner ── law::inner_product_space
17/// └ hilbert_space<V> ‖x‖² = ⟨x,x⟩ ── law::hilbert_space
18///
19/// linear_map<Op> domain and codomain are spaces ── law::linear_map
20/// └ linear_operator<Op> + apply, rows, cols
21/// └ endomorphism<Op> domain == codomain ── law::endomorphism
22/// └ normal_operator AA* = A*A ── law::normal
23/// ├ self_adjoint_operator A = A* ── law::self_adjoint
24/// │ └ psd_operator ⟨x,Ax⟩ ≥ 0 ── law::psd
25/// │ ├ spd_operator ⟨x,Ax⟩ > 0 ── law::spd
26/// │ └ projection_operator P = P* = P² ── law::projection
27/// ├ skew_adjoint_operator A = −A* ── law::skew_adjoint
28/// └ unitary_operator A*A = I ── law::unitary
29/// ```
30///
31/// The `requires` clauses decide structure. Laws come from `claims<T, L>` and are never
32/// inferred from syntax. A type therefore cannot become SPD by accident. It must declare
33/// the law, and declaring it runs the probe bound to that law.
34///
35/// The operator concepts default their domain and codomain to the type's own associated
36/// types. `linear_operator<Op>` and `linear_operator<Op, vec, vec>` are the same
37/// concept with the spaces left implicit or stated.
38///
39/// This header depends on nothing but the standard library, so the whole hierarchy is
40/// available to any tier, including code that never touches `num::vec`.

55/// @brief scalar field \f$\mathbb{K}\f$: closed under `+ - * /` with identities 0 and 1.
56template <class T>
57concept field = claims<T, law::field> && requires(T a, T b) {
58 {a + b}->std::convertible_to<T>;
59 {a - b}->std::convertible_to<T>;
60 {a * b}->std::convertible_to<T>;
61 {a / b}->std::convertible_to<T>;
62 T{0};
63 T{1};
64};

66// -----------------------------------------------------------------------------
67// Spaces
68// -----------------------------------------------------------------------------

70/// @brief Abelian group under addition, over a scalar field: sized, copyable, zeroable.
71///
72/// The structural floor of the hierarchy. It does not require scalar multiplication, which
73/// `vector_space` adds. A type that can be added and measured but not scaled still has a
74/// level to stand on.
75template <class V>
76concept additive_group = field<scalar_t<V>> && std::copy_constructible<V> &&
77 requires(const V &x) {
78 {dimension(x)}->std::integral;
79 {zero_like(x)}->std::same_as<V>;
80};
81
82/// @brief vec space over \f$\mathbb{K}\f$: an additive group with a compatible scalar action.
83template <class V>
84concept vector_space = additive_group<V> && claims<V, law::vector_space> &&
85 requires(V &v, const V &x, scalar_t<V> a) {
86 scale(a, v);
87 axpy(a, x, v);
88};
89
90/// @brief vec space carrying a norm \f$\|\cdot\|\f$.
91template <class V>
92concept normed_space = vector_space<V> && claims<V, law::normed_space> && requires(const V &v) {
93 norm(v);
94};
95
96/// @brief Normed space carrying an inner product \f$\langle\cdot,\cdot\rangle\f$.
97template <class V>
98concept inner_product_space =
99 normed_space<V> && claims<V, law::inner_product_space> && requires(const V &x, const V &y) {
100 {inner(x, y)}->std::convertible_to<scalar_t<V>>;
101};
102
103/// @brief Inner product space whose norm is the induced one: \f$\|x\|^2 = \langle x,x\rangle\f$.
104///
105/// The setting every Krylov method needs. CG, MINRES and GMRES all require that the norm
106/// they minimize is the one the inner product induces.
107template <class V>
108concept hilbert_space = inner_product_space<V> && claims<V, law::hilbert_space>;
109
110/// @brief vec space whose coordinates can be written individually.
111template <class V>
112concept mutable_vector_space = vector_space<V> && requires(V &v, scalar_t<V> a) {
113 { v[std::size_t{0}] = a };
114};
115
116/// @brief vec space backed by one contiguous run of scalars.
117///
118/// A statement about storage rather than mathematics. It allows an algorithm to drop
119/// through to `num::kernel`, which operates only on pointers.
120template <class V>
121concept contiguous_vector = vector_space<V> && requires(V & v, const V &cv) {
122 {v.data()}->std::convertible_to<scalar_t<V> *>;
123 {cv.data()}->std::convertible_to<const scalar_t<V> *>;
124};

130/// @brief Linear map \f$A: X \to Y\f$ between vector spaces.
131template <class Op, class X = domain_t<Op>, class Y = codomain_t<Op>>
132concept linear_map = claims<Op, law::linear_map> && vector_space<X> && vector_space<Y>;
133
134/// @brief Linear map that can be evaluated and knows its shape.
135template <class Op, class X = domain_t<Op>, class Y = codomain_t<Op>>
136concept linear_operator = linear_map<Op, X, Y> && requires(const Op &A, const X &x, Y &y) {
137 // The CPO rather than `A.apply(x, y)`. It also resolves a `tag_invoke` overload and
138 // unwraps evidence wrappers such as `certified_ref`, which carry a law about a matrix
139 // without re-exposing its interface.
140 apply(A, x, y);
141 {A.rows()}->std::integral;
142 {A.cols()}->std::integral;
143};
144
145/// @brief Linear operator exposing its adjoint \f$A^*\f$.
146template <class Op, class X = domain_t<Op>, class Y = codomain_t<Op>>
147concept adjointable_linear_operator =
148 linear_operator<Op, X, Y> && requires(const Op &A, const Y &y, X &x) {
149 A.apply_adjoint(y, x);
150};
151
152/// @brief Linear operator of a space into itself, so `rows() == cols()`.
153template <class Op, class V = domain_t<Op>>
154concept endomorphism =
155 linear_operator<Op, V, V> && claims<Op, law::endomorphism> && std::same_as<domain_t<Op>, V>;
156
157/// @brief An operator that maps `V` into itself, whether or not it claims `law::endomorphism`.
158///
159/// The structural half of `endomorphism`. It is kept separate so a solver signature can
160/// require that an operator acts on a given space without also requiring the squareness
161/// claim.
162template <class Op, class V>
163concept endomorphism_on =
164 linear_operator<Op, V, V> && std::same_as<domain_t<Op>, V> && std::same_as<codomain_t<Op>, V>;
165
166// -----------------------------------------------------------------------------
167// Operator laws
168// -----------------------------------------------------------------------------
169//
170// A subsumption chain, so overload resolution orders correctly: an SPD operator offered
171// to a set containing CG, MINRES and GMRES selects CG rather than being ambiguous.
172
173/// @brief \f$AA^* = A^*A\f$: unitarily diagonalizable, with an orthogonal eigenbasis.
174template <class Op, class X = domain_t<Op>, class Y = codomain_t<Op>>
175concept normal_operator = linear_operator<Op, X, Y> && claims<Op, law::normal>;
176
177/// @brief \f$A = A^*\f$: real spectrum. The precondition for MINRES and Lanczos.
178template <class Op, class X = domain_t<Op>, class Y = codomain_t<Op>>
179concept self_adjoint_operator = normal_operator<Op, X, Y> && claims<Op, law::self_adjoint>;
180
181/// @brief \f$\langle x, Ax \rangle \ge 0\f$: graph Laplacians and Gram matrices, possibly singular.
182template <class Op, class X = domain_t<Op>, class Y = codomain_t<Op>>
183concept psd_operator = self_adjoint_operator<Op, X, Y> && claims<Op, law::psd>;
184
185/// @brief \f$\langle x, Ax \rangle > 0\f$: invertible, Cholesky-factorable. The precondition for CG.
186template <class Op, class X = domain_t<Op>, class Y = codomain_t<Op>>
187concept spd_operator = psd_operator<Op, X, Y> && claims<Op, law::spd>;
188
189/// @brief \f$P = P^* = P^2\f$: the orthogonal projector onto its range.
190///
191/// Refines `psd_operator` rather than `self_adjoint_operator`. A projector is always positive
192/// semidefinite, since \f$\langle x, Px \rangle = \langle Px, Px \rangle \ge 0\f$.
193template <class Op, class X = domain_t<Op>, class Y = codomain_t<Op>>
194concept projection_operator = psd_operator<Op, X, Y> && claims<Op, law::projection>;
195
196/// @brief \f$A = -A^*\f$: imaginary spectrum. Conservative advection, Hamiltonian fields.
197template <class Op, class X = domain_t<Op>, class Y = codomain_t<Op>>
198concept skew_adjoint_operator = normal_operator<Op, X, Y> && claims<Op, law::skew_adjoint>;
199
200/// @brief \f$A^*A = I\f$: an isometry. Givens rotations, Householder reflectors, the DFT.
201template <class Op, class X = domain_t<Op>, class Y = codomain_t<Op>>
202concept unitary_operator = normal_operator<Op, X, Y> && claims<Op, law::unitary>;
203
204// -----------------------------------------------------------------------------
205// Maps carrying no linearity claim
206// -----------------------------------------------------------------------------
207
208/// @brief \f$F: X \to Y\f$ between vector spaces, with no claim of linearity.
209template <class Op, class X = domain_t<Op>, class Y = codomain_t<Op>>
210concept nonlinear_operator =
211 vector_space<X> && mutable_vector_space<Y> && requires(const Op &F, const X &x, Y &y) {
212 apply(F, x, y);
213 {F.rows()}->std::integral;
214 {F.cols()}->std::integral;
215};
