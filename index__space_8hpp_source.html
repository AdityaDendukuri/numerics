1/// @file core/index_space.hpp
2/// @brief Discrete index spaces: the root the grid and lattice families are built on.
3///
4/// A structured grid and a periodic lattice are not vector spaces. They carry no addition
5/// and no scalar action, so they have no place in the algebraic hierarchy in
6/// `core/math/concepts.hpp`. They are the same kind of object as each other: a finite set
7/// of sites addressed by a linear index. This header names that, so both families refine a
8/// common concept instead of asserting their shape independently.
9///
10/// A field sampled on a grid does live in a vector space. The grid is the index set and
11/// the field over it is the space. Separating the two allows one discretization to carry a
12/// real field, a complex field, or several at once.
13#pragma once

19/// @brief A square \f$N \times N\f$ extent.
20///
21/// A structured grid and a periodic lattice share exactly this much structure: both are
22/// square and know their side length. Naming it makes the two families siblings rather
23/// than independent shapes that happen to spell `N` the same way.
24template <class G>
25concept square_extent_2d = requires(const G &g) {
26 { g.N } -> std::convertible_to<int>;
27};
28
29/// @brief A square extent carrying a flattening \f$\mathbb{Z}^2 \to \mathbb{Z}\f$.
30///
31/// `flat(i, j)` puts a site at a linear position, which is what lets a field over the grid
32/// be stored as one contiguous vector and handed to `num::kernel` unchanged.
33template <class G>
34concept cartesian_index_space_2d = square_extent_2d<G> && requires(const G &g, int i, int j) {
35 { g.flat(i, j) } -> std::convertible_to<int>;
36};
37
38/// @brief A square extent carrying precomputed periodic neighbour maps.
39///
40/// The wrap-around is folded into the four index arrays. A stencil sweep therefore needs
41/// no modulo and no boundary branch, which is why these are stored rather than computed.
42template <class P>
43concept periodic_neighbourhood_2d = square_extent_2d<P> && requires(const P &lattice, int i) {
44 { lattice.up[i] } -> std::convertible_to<int>;
45 { lattice.dn[i] } -> std::convertible_to<int>;
46 { lattice.lt[i] } -> std::convertible_to<int>;
47 { lattice.rt[i] } -> std::convertible_to<int>;
48};
