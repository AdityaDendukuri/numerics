53inline auto_resolvent_solver::auto_resolvent_solver(const spmat &matrix, auto_resolvent_options options)
54 : impl_(std::make_unique<Impl>()) {
55 if (matrix.n_rows() <= options.dense_limit) {
56 impl_->dense = std::make_unique<dense_resolvent_solver>(matrix);
57 return;
58 }
59 if (!sparse_resolvent_available()) {
60 throw std::runtime_error("large shifted systems require the sparse complex backend");
61 }
62 impl_->sparse = std::make_unique<sparse_resolvent_solver>(
63 matrix, sparse_resolvent_options{.symmetric_pattern = options.symmetric_pattern});
64}
