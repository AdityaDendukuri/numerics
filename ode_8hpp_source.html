104/// @brief Integrate first-order ODE \f$\dot{y} = f(t, y)\f$ using fixed-step forward Euler.
105///
106/// @tparam RHS Derivative callable.
107/// @tparam State State vector type.
108/// @param f Right-hand side function.
109/// @param y0 Initial state \f$y(t_0)\f$.
110/// @param p Integration parameters (`t0`, `tf`, `dt`).
111/// @param observer Optional callback `void(real t, const vec& u)` invoked at each step.
112/// @return `ode_result` with final state `u`, final time `t`, total `steps`, and convergence flag.
113template <typename RHS = ode_rhs_fn, typename State = vec>
114inline ode_result ode_euler(RHS f, State y0, ode_params p = {}, const observer_fn &observer = {}) {
115 auto s = euler(std::move(f), std::move(y0), p);
116 if (!observer) {
117 return s.run();
118 }
119 for (auto step : s) {
120 observer(step.t, step.u);
121 }
122 return s.run();
123}
