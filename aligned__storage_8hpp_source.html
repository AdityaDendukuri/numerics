1/// @file container/util/aligned_storage.hpp
2/// @brief Over-aligned owning storage for the dense containers.
3///
4/// `new T[n]` guarantees only `__STDCPP_DEFAULT_NEW_ALIGNMENT__` — 16 bytes on
5/// every platform this library targets. The dense containers instead allocate
6/// through `make_aligned` below, which uses the aligned `operator new` (C++17)
7/// to place the first element on a `storage_alignment` boundary.
8///
9/// ### What this is and is not worth
10///
11/// Measured on AArch64/NEON and cross-compiled to x86-64/AVX-512, clang emits
12/// *identical* code for the level-1 kernels with and without the guarantee: it
13/// already prefers the unaligned move forms, which carry no penalty on an
14/// aligned address on any current microarchitecture. Do not expect the
15/// `std::assume_aligned` in `data()` to speed up a loop today; it is there so
16/// the annotation is correct and in place for the kernels that will need it.
17///
18/// What the alignment does buy is structural:
19/// - a vector load never straddles a cache line, so no access is split across
20/// two L1 lookups;
21/// - two containers never share a line at their boundaries, which is what
22/// makes a threaded reduction over adjacent buffers free of false sharing;
23/// - it is a hard *precondition* for aligned SIMD intrinsics
24/// (`_mm512_load_pd`) and non-temporal stores (`_mm512_stream_pd`), and for
25/// efficient pinned-host-memory registration on the device paths.
26///
27/// It also costs something: the aligned `operator new` bypasses the small-size
28/// fast path in at least libc++'s allocator, which on macOS measured roughly
29/// 4x the latency of plain `new` for buffers under a kilobyte. That is only
30/// visible to code that allocates inside a loop — which is a defect in that
31/// code, not in this header.
32///
33/// ### Scope of the guarantee
34///
35/// It covers the *base* pointer. An interior pointer such as `A.data() + j` is
36/// aligned only when `j * sizeof(T)` is a multiple of `storage_alignment`, and
37/// nothing here claims otherwise; the assumption is attached to the base, and
38/// the compiler reasons about the offset itself. A matrix row therefore starts
39/// on a boundary only when its stride happens to; padding the stride would
40/// change what `data()` means and is deliberately not done here.

50/// @brief Storage alignment in bytes. Override at configure time with

62/// @brief Alignment, in bytes, of the storage owned by every dense container.
63///
64/// This value participates in the deallocation call, so it must be identical in
65/// every translation unit of a build. The CMake package sets it on the `numerics`
66/// target as a `PUBLIC` definition for exactly that reason; headers copied out of
67/// the tree fall back to the default above.
68inline constexpr std::size_t storage_alignment = NUMERICS_STORAGE_ALIGNMENT;
69
70static_assert(storage_alignment >= alignof(std::max_align_t),
71 "NUMERICS_STORAGE_ALIGNMENT must be at least the default new alignment");
72static_assert((storage_alignment & (storage_alignment - 1)) == 0,
73 "NUMERICS_STORAGE_ALIGNMENT must be a power of two");

77/// @brief Releases storage obtained from `allocate_aligned`.
78///
79/// Carries the element count for two reasons: over-aligned storage must be
80/// returned through the matching aligned `operator delete`, and a non-trivially
81/// destructible element type needs its destructors run explicitly. The array
82/// form of `new` would track both, but it cannot be given an alignment for an
83/// element type whose own `alignof` is weaker.
84template <class T>
85class aligned_deleter {
86 public:
87 constexpr aligned_deleter() noexcept = default;
88 constexpr explicit aligned_deleter(idx count) noexcept : count_(count) {}
89
90 void operator()(T *pointer) const noexcept {
91 if (pointer == nullptr) {
92 return;
93 }
94 std::destroy_n(pointer, count_);
95 ::operator delete(static_cast<void *>(pointer), count_ * sizeof(T),
96 std::align_val_t{storage_alignment});
97 }
98
99 private:
100 idx count_ = 0;
101};
102
103/// @brief Obtain raw `storage_alignment`-aligned storage for `count` elements.
104/// @throws std::bad_alloc If the byte count overflows `idx`, or on allocation failure.
105template <class T>
106[[nodiscard]] inline T *allocate_aligned(idx count) {
107 static_assert(storage_alignment >= alignof(T),
108 "NUMERICS_STORAGE_ALIGNMENT is weaker than this element type requires");
109 if (count > std::numeric_limits<idx>::max() / sizeof(T)) {
110 throw std::bad_alloc();
111 }
112 return static_cast<T *>(::operator new(count * sizeof(T), std::align_val_t{storage_alignment}));
113}
114
115/// @brief Release raw storage whose elements were never constructed.
116template <class T>
117inline void deallocate_aligned(T *pointer, idx count) noexcept {
118 ::operator delete(static_cast<void *>(pointer), count * sizeof(T),
119 std::align_val_t{storage_alignment});
120}

124/// @brief Owning handle to `storage_alignment`-aligned storage for `T`.
125template <class T>
126using aligned_array = std::unique_ptr<T[], detail::aligned_deleter<T>>;
127
128/// @brief Allocate `count` value-initialized elements on an aligned boundary.
129///
130/// Equivalent to `new T[count]()`, except for the alignment and the overflow check.
131/// @throws std::bad_alloc On overflow or allocation failure.
132template <class T>
133[[nodiscard]] inline aligned_array<T> make_aligned(idx count) {
134 if (count == 0) {
135 return aligned_array<T>(nullptr, detail::aligned_deleter<T>(0));
136 }
137 T *storage = detail::allocate_aligned<T>(count);
138 try {
139 std::uninitialized_value_construct_n(storage, count);
140 } catch (...) {
141 detail::deallocate_aligned(storage, count);
142 throw;
143 }
144 return aligned_array<T>(storage, detail::aligned_deleter<T>(count));
145}
146
147/// @brief Allocate `count` default-initialized elements on an aligned boundary.
148///
149/// Equivalent to `new T[count]`: for a scalar element type the contents are
150/// indeterminate, so every element must be written before it is read. Use this
151/// where the caller immediately fills or copies over the whole buffer, to avoid
152/// paying for a zero fill that is about to be overwritten.
153/// @throws std::bad_alloc On overflow or allocation failure.
154template <class T>
155[[nodiscard]] inline aligned_array<T> make_aligned_for_overwrite(idx count) {
156 if (count == 0) {
157 return aligned_array<T>(nullptr, detail::aligned_deleter<T>(0));
158 }
159 T *storage = detail::allocate_aligned<T>(count);
160 try {
161 std::uninitialized_default_construct_n(storage, count);
162 } catch (...) {
163 detail::deallocate_aligned(storage, count);
164 throw;
165 }
166 return aligned_array<T>(storage, detail::aligned_deleter<T>(count));
167}

169/// @brief Restate a container's storage alignment for the optimizer.
170///
171/// Valid only on the base pointer of storage obtained above. Null is passed
172/// through untouched: `std::assume_aligned` requires a pointer to an object, and
173/// an empty container owns none.
174template <class T>
175[[nodiscard]] inline T *assume_storage_aligned(T *pointer) noexcept {
176 if (pointer == nullptr) {
177 return nullptr;
178 }
179 return std::assume_aligned<storage_alignment>(pointer);
180}

182/// @brief True when `pointer` sits on a `storage_alignment` boundary.
183///
184/// Diagnostic only; the containers guarantee this by construction.
185template <class T>
186[[nodiscard]] inline bool is_storage_aligned(const T *pointer) noexcept {
187 return (reinterpret_cast<std::uintptr_t>(pointer) % storage_alignment) == 0;
188}
