28template <typename Float, std::integral Index>
29inline void write_column(Index v, Index step, Float total_weight,
30 const std::vector<neighbor<Float, Index>> &nbr,
31 cholesky_factor<Float, Index> &factor) {
32 const Float sqrt_total = std::sqrt(total_weight);
33
34 auto &col_entries = factor.columns[step].entries;
35 col_entries.reserve(nbr.size() + 1);
36 col_entries.push_back({v, sqrt_total});
37 for (const auto &n_info : nbr) {
38 col_entries.push_back({n_info.to, -n_info.weight / sqrt_total});
39 }
40}

42template <typename Float, std::integral Index>
43inline void permute_rows(cholesky_factor<Float, Index> &factor) {
44 const Index n = static_cast<Index>(factor.order.size());
45 std::vector<Index> position(n);
46
47 for (Index step = 0; step < n; ++step) {
48 position[factor.order[step]] = step;
49 }
50
51 for (Index step = 0; step < n; ++step) {
52 for (auto &entry : factor.columns[step].entries) {
53 entry.row = position[entry.row];

67/// Factorize a multigraph Laplacian into approximate or exact Cholesky factor \f$L L^T\f$.
68template <typename Float = double, std::integral Index = num::idx, typename Rng = rng64,
69 typename Queue = structures::basic_degree_queue<Index>>
70inline cholesky_factor<Float, Index> factorize(const graph<Float, Index> &input_G,
71 std::type_identity_t<Index> samples = 1,
72 bool exact_mode = false, Rng *rng = nullptr,
73 clique_sampler sampler = clique_sampler::independent,
74 std::optional<Index> pinned_vertex = std::nullopt) {
75 if (exact_mode) {
76 sampler = clique_sampler::exact;
77 }
78 const Index n = static_cast<Index>(input_G.size());
79 if (pinned_vertex && *pinned_vertex >= n) {
80 throw std::out_of_range("factorize: pinned vertex is outside the graph");
81 }
82 graph<Float, Index> G = input_G;
83
84 if (samples > 1 && !exact_mode) {
85 for (auto &row : G) {
86 for (auto &e : row) {
87 e.count = static_cast<std::uint8_t>(samples);
88 }
89 }
90 }
91
92 Queue q(n);
93 for (Index i = 0; i < n; ++i) {
94 Index d = 0;
95 for (const auto &e : G[i]) {
96 d += e.count;
97 }
98 q.insert(i, d);
99 }
100 std::vector<std::uint8_t> done(n, 0);
101
102 cholesky_factor<Float, Index> factor;
103 factor.columns.resize(n);
104 factor.order.reserve(n);
105
106 std::vector<graph_edge<Float, Index>> star_buf;
107 star_buf.reserve(128);
108
109 std::vector<neighbor<Float, Index>> nbr_buf;
110 nbr_buf.reserve(128);
111
112 Rng local_rng(42);
113 Rng &active_rng = rng ? *rng : local_rng;
114
115 for (Index step = 0; step + 1 < n; ++step) {
116 Index v = q.pop_min();
117 if (pinned_vertex && v == *pinned_vertex) {
118 const Index pinned_degree = q.degree_of(v);
119 v = q.pop_min();
120 q.insert(*pinned_vertex, pinned_degree);
121 }
122 factor.order.push_back(v);
123
124 Float total_weight = structures::collect_neighbors(v, G, done, q, star_buf, nbr_buf);
125 done[v] = 1;
126
127 if (total_weight <= static_cast<Float>(0)) {
128 continue;
129 }
130
131 detail::write_column(v, step, total_weight, nbr_buf, factor);
132
133 switch (sampler) {
134 case clique_sampler::exact:
135 structures::add_exact_clique(G, q, nbr_buf, total_weight);
136 break;
137 case clique_sampler::tree:
138 structures::sample_clique_tree(G, q, nbr_buf, total_weight, active_rng,
139 static_cast<std::size_t>(samples));
140 break;
141 case clique_sampler::independent:
142 structures::sample_clique(G, q, nbr_buf, total_weight, samples, active_rng);
143 break;
144 }
145 }
146
147 // Nullspace vertex
148 const Index last_v = q.pop_min();
149 if (pinned_vertex && last_v != *pinned_vertex) {
150 throw std::runtime_error("factorize: failed to retain pinned vertex");
151 }
152 factor.order.push_back(last_v);
153
154 detail::permute_rows(factor);
155
156 return factor;
157}

185/// @brief Spanning-tree clique sampler: one reweighted random tree per elimination.
186///
187/// Keeps \f$d-1\f$ edges per eliminated vertex instead of \f$d\f$, drawn from a
188/// negatively dependent distribution rather than independently. See
189/// `num::structures::sample_clique_tree` for why the reweighting needs no
190/// effective-resistance solve.
191/// @return `cholesky_factor`: the sparse approximate factor, consumed by `randommat::solve`.
192template <typename Float = double, std::integral Index = num::idx, typename Rng = rng64>
193inline cholesky_factor<Float, Index> act(const graph<Float, Index> &G, Rng &rng,
194 std::type_identity_t<Index> trees = 1) {
195 return factorize<Float, Index, Rng>(G, trees, false, &rng, clique_sampler::tree);
196}
