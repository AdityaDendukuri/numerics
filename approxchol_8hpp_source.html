31template <typename Float, std::integral Index>
32inline void write_column(Index v, Index step, Float total_weight,
33 const std::vector<neighbor<Float, Index>> &nbr,
34 cholesky_factor<Float, Index> &factor) {
35 const Float sqrt_total = std::sqrt(total_weight);
36
37 auto &col_entries = factor.columns[step].entries;
38 col_entries.reserve(nbr.size() + 1);
39 col_entries.push_back({v, sqrt_total});
40 for (const auto &n_info : nbr) {
41 col_entries.push_back({n_info.to, -n_info.weight / sqrt_total});
42 }
43}

45template <typename Float, std::integral Index>
46inline void permute_rows(cholesky_factor<Float, Index> &factor) {
47 const Index n = static_cast<Index>(factor.order.size());
48 std::vector<Index> position(n);
49
50 for (Index step = 0; step < n; ++step) {
51 position[factor.order[step]] = step;
52 }
53
54 for (Index step = 0; step < n; ++step) {
55 for (auto &entry : factor.columns[step].entries) {
56 entry.row = position[entry.row];
57 }
58 }
59}

70/// Factorize a multigraph Laplacian into approximate or exact Cholesky factor \f$L L^T\f$.
71template <typename Float = double, std::integral Index = num::idx, typename Rng = std::mt19937_64,
72 typename Queue = structures::basic_degree_queue<Index>>
73inline cholesky_factor<Float, Index> factorize(const graph<Float, Index> &input_G,
74 std::type_identity_t<Index> samples = 1,
75 bool exact_mode = false, Rng *rng = nullptr,
76 clique_sampler sampler = clique_sampler::independent) {
77 if (exact_mode) {
78 sampler = clique_sampler::exact;
79 }
80 const Index n = static_cast<Index>(input_G.size());
81 graph<Float, Index> G = input_G;
82
83 if (samples > 1 && !exact_mode) {
84 for (auto &row : G) {
85 for (auto &e : row) {
86 e.count = static_cast<std::uint8_t>(samples);
87 }
88 }
89 }
90
91 Queue q(n);
92 for (Index i = 0; i < n; ++i) {
93 Index d = 0;
94 for (const auto &e : G[i]) {
95 d += e.count;
96 }
97 q.insert(i, d);
98 }
99 std::vector<std::uint8_t> done(n, 0);
100
101 cholesky_factor<Float, Index> factor;
102 factor.columns.resize(n);
103 factor.order.reserve(n);
104
105 std::vector<graph_edge<Float, Index>> star_buf;
106 star_buf.reserve(128);
107
108 std::vector<neighbor<Float, Index>> nbr_buf;
109 nbr_buf.reserve(128);
110
111 Rng local_rng(42);
112 Rng &active_rng = rng ? *rng : local_rng;
113
114 for (Index step = 0; step + 1 < n; ++step) {
115 const Index v = q.pop_min();
116 factor.order.push_back(v);
117
118 Float total_weight = structures::collect_neighbors(v, G, done, q, star_buf, nbr_buf);
119 done[v] = 1;
120
121 if (total_weight <= static_cast<Float>(0)) {
122 continue;
123 }
124
125 detail::write_column(v, step, total_weight, nbr_buf, factor);
126
127 switch (sampler) {
128 case clique_sampler::exact:
129 structures::add_exact_clique(G, q, nbr_buf, total_weight);
130 break;
131 case clique_sampler::tree:
132 structures::sample_clique_tree(G, q, nbr_buf, total_weight, active_rng,
133 static_cast<std::size_t>(samples));
134 break;
135 case clique_sampler::independent:
136 structures::sample_clique(G, q, nbr_buf, total_weight, samples, active_rng);
137 break;
138 }
139 }
140
141 // Nullspace vertex
142 const Index last_v = q.pop_min();
143 factor.order.push_back(last_v);
144
145 detail::permute_rows(factor);
146
147 return factor;
148}

174/// @brief Spanning-tree clique sampler: one reweighted random tree per elimination.
175///
176/// Keeps \f$d-1\f$ edges per eliminated vertex instead of \f$d\f$, drawn from a
177/// negatively dependent distribution rather than independently. See
178/// `num::structures::sample_clique_tree` for why the reweighting needs no
179/// effective-resistance solve.
180template <typename Float = double, std::integral Index = num::idx, typename Rng = std::mt19937_64>
181inline cholesky_factor<Float, Index> act(const graph<Float, Index> &G, Rng &rng,
182 std::type_identity_t<Index> trees = 1) {
183 return factorize<Float, Index, Rng>(G, trees, false, &rng, clique_sampler::tree);
184}

192/// @brief preconditioner adapter backed by Randomized Approximate Cholesky (ApproxChol).
193/// Satisfies the num::preconditioner concept for use with num::pcg and Krylov solvers.
194class approx_chol_preconditioner final {
195 public:
196 using domain_type = vec;
197 using codomain_type = vec;
198 // A graph-Laplacian factor is singular on the constant-vector nullspace.
199 // Callers using PCG on a compatible subspace must attach that stronger,
200 // problem-specific evidence explicitly.
201 using math_laws = math::type_list<law::psd>;
202
203 /// Construct from an existing factor.
204 explicit approx_chol_preconditioner(cholesky_factor<real, idx> factor)
205 : factor_(std::move(factor)), n_(factor_.order.size()), scratch_(n_, 0.0) {}
206
207 /// Number of rows.
208 [[nodiscard]] idx rows() const noexcept { return n_; }
209
210 /// Number of columns.
211 [[nodiscard]] idx cols() const noexcept { return n_; }
212
213 /// Apply preconditioner z = M^-1 r via forward and backward substitution.
214 void apply(const vec &r, vec &z) const {
215 if (r.size() != n_) {
216 throw std::invalid_argument("approx_chol_preconditioner: dimension mismatch");
217 }
218 if (z.size() != n_) {
219 z = vec(n_, 0.0);
220 }
221 randommat::solve(factor_, r.data(), z.data(), scratch_);
222 }
223
224 /// Access underlying factor.
225 [[nodiscard]] const cholesky_factor<real, idx> &factor() const noexcept { return factor_; }
226
227 private:
228 cholesky_factor<real, idx> factor_;
229 idx n_ = 0;
230 mutable std::vector<real> scratch_;
231};
