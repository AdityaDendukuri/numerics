2/// @brief Runtime sampling of algebraic laws and operator properties.
3///
4/// A property quantified over all vectors cannot be proved by evaluating it a
5/// finite number of times. These routines are built to reject violations. Basis

7/// property away from the coordinate axes with a fixed seed, so a reported
8/// violation reproduces.

25// ---------------------------------------------------------------------------
26// Randomized probing for sampled property tests
27// ---------------------------------------------------------------------------
28//
29// A property such as symmetry or positive definiteness is a statement about
30// *all* vectors. It cannot be established by evaluating a single fixed probe,
31// which is why each sampler below combines deterministic basis probes (testing
32// a necessary condition exactly) with several randomized probes.
33
34/// @brief Number of random probe vectors drawn per sampled property test.
35inline idx g_probe_count = 6;
36
37/// @brief Relative tolerance override for sampled property tests; 0 selects sqrt(eps).
38inline double g_property_tol = 0.0;
39
40/// @brief Relative tolerance used when comparing sampled quantities over field T.
41template <class T>
42[[nodiscard]] inline scalars::real_t<T> property_tol() noexcept {
43 return g_property_tol > 0.0 ? static_cast<scalars::real_t<T>>(g_property_tol)
44 : scalars::sampling_tol<T>();
45}

47/// @brief Deterministic xorshift generator, so a reported violation reproduces exactly.
48struct probe_rng {
49 std::uint64_t state;
50
51 explicit constexpr probe_rng(std::uint64_t seed = 0x9E3779B97F4A7C15ULL) noexcept
52 : state(seed == 0 ? 1 : seed) {}
53
54 constexpr std::uint64_t next() noexcept {
55 state ^= state << 13;
56 state ^= state >> 7;
57 state ^= state << 17;
58 return state;
59 }
60
61 /// Uniform sample in [-1, 1).
62 constexpr double uniform() noexcept {
63 return ((static_cast<double>(next() >> 11) / 9007199254740992.0) * 2.0) - 1.0;
64 }
65};

67/// @brief Overwrite v with a reproducible random probe over its own scalar field.
68template <class VectorType>
69inline void fill_probe(VectorType &v, probe_rng &rng) {
70 using T = num::scalar_t<VectorType>;
71 using R = scalars::real_t<T>;
72 for (idx i = 0; i < v.size(); ++i) {
73 if constexpr (scalars::is_complex_v<T>) {
74 v[i] = T(static_cast<R>(rng.uniform()), static_cast<R>(rng.uniform()));
75 } else {
76 v[i] = static_cast<T>(rng.uniform());
77 }
78 }
79}

81/// @brief Overwrite v with the i-th basis vector e_i.
82template <class VectorType>
83inline void fill_basis(VectorType &v, idx i) {
84 using T = num::scalar_t<VectorType>;
85 for (idx k = 0; k < v.size(); ++k) {
86 v[k] = T(0);
87 }
88 v[i] = T(1);
89}

91/// @brief Hermitian inner product \f$\langle x,y \rangle = \sum_i \overline{x_i} y_i\f$.
92template <class VectorType>
93[[nodiscard]] inline num::scalar_t<VectorType> probe_inner(const VectorType &x,
94 const VectorType &y) {
95 using T = num::scalar_t<VectorType>;
96 if constexpr (algebra::detail::raw_reducible<VectorType>) {
97 return kernel::dot(x.data(), y.data(), x.size());
98 } else {
99 T sum = T(0);
100 for (idx i = 0; i < x.size(); ++i) {
101 sum += scalars::conj(x[i]) * y[i];
102 }
103 return sum;
104 }
105}

107/// @brief Induced norm \f$\|x\| = \sqrt{\langle x,x \rangle}\f$.
108template <class VectorType>
109[[nodiscard]] inline scalars::real_t<num::scalar_t<VectorType>> probe_norm(const VectorType &x) {
110 return std::sqrt(scalars::re(probe_inner(x, x)));

113/// @brief Stride giving at most `cap` basis probes across dimension n.
114[[nodiscard]] inline idx probe_stride(idx n, idx cap = 8) noexcept {
115 const idx s = n / cap;
116 return s == 0 ? idx(1) : s;

119// ---------------------------------------------------------------------------
120// Algebraic law sampling
121// ---------------------------------------------------------------------------
122//
123// The concepts in container/algebra.hpp establish that a type *supplies* the
124// vector space operations and is closed under them. That it obeys the axioms --
125// associativity, distributivity, conjugate symmetry, the triangle inequality --
126// is not decidable from a type and is sampled here.
127//
128// These probes route through num::algebra, which resolves to the type's own
129// operations. Checking num::vec therefore checks the shipped `dot`, `norm`,
130// `axpy` and `scale`, backend dispatch included: a BLAS path that returns an
131// unconjugated inner product for complex data fails conjugate symmetry here.
132
133/// @brief Sample the abelian group axioms of vector addition.
134///
135/// Associativity, commutativity, the additive identity, and additive inverses.
136template <class V>
137inline void
138verify_additive_group_axioms(idx n, std::source_location loc = std::source_location::current()) {
139 if constexpr (!num::debug::sampling_compiled_in) {
140 return;
141 }
142 if (num::debug::get_level() != num::debug::diagnostic_level::full || n == 0) {
143 return;
144 }
145 using T = num::scalar_t<V>;
146 using R = scalars::real_t<T>;
147 const R tol = property_tol<T>();
148 probe_rng rng;
149
150 auto rel_diff = [&](const V &a, const V &b) {
151 R num_ = R(0), den = R(0);
152 for (idx i = 0; i < n; ++i) {
153 const T d = a[i] - b[i];
154 num_ += scalars::re(scalars::conj(d) * d);
155 den += scalars::re(scalars::conj(a[i]) * a[i]);
156 }
157 return std::sqrt(num_) / (std::sqrt(den) + std::numeric_limits<R>::min());
158 };
159
160 for (idx probe = 0; probe < g_probe_count; ++probe) {
161 V x(n), y(n), z(n);
162 fill_probe(x, rng);
163 fill_probe(y, rng);
164 fill_probe(z, rng);
165
166 // (x + y) + z == x + (y + z)
167 V left = x;
168 algebra::axpy_into(T(1), y, left);
169 algebra::axpy_into(T(1), z, left);
170 V right = y;
171 algebra::axpy_into(T(1), z, right);
172 V right_total = x;
173 algebra::axpy_into(T(1), right, right_total);
174 if (rel_diff(left, right_total) > tol) {
175 panic("AlgebraError",
176 "vector addition is not associative: (x+y)+z != x+(y+z) on probe " +
177 std::to_string(probe),
178 loc);
179 }
180
181 // x + y == y + x
182 V xy = x;
183 algebra::axpy_into(T(1), y, xy);
184 V yx = y;
185 algebra::axpy_into(T(1), x, yx);
186 if (rel_diff(xy, yx) > tol) {
187 panic("AlgebraError",
188 "vector addition is not commutative: x+y != y+x on probe " +
189 std::to_string(probe),
190 loc);
191 }
192
193 // x + 0 == x
194 V zero_v = algebra::zero<V>(n);
195 V x_plus_zero = x;
196 algebra::axpy_into(T(1), zero_v, x_plus_zero);
197 if (rel_diff(x_plus_zero, x) > tol) {
198 panic("AlgebraError", "the zero vector is not an additive identity: x+0 != x", loc);
199 }
200
201 // x + (-x) == 0
202 V cancel = x;
203 algebra::axpy_into(T(-1), x, cancel);
204 R residual = R(0);
205 for (idx i = 0; i < n; ++i) {
206 residual += scalars::re(scalars::conj(cancel[i]) * cancel[i]);
207 }
208 if (std::sqrt(residual) > tol) {
209 panic("AlgebraError", "additive inverses do not cancel: x + (-x) != 0", loc);
210 }
211 }
212}

214/// @brief Sample the vector space axioms relating scalar multiplication to addition.
215///
216/// Distributivity over vector and over scalar addition, compatibility of scalar
217/// multiplication with field multiplication, and the unit scalar acting as identity.
218template <class V>
219inline void verify_vector_space_axioms(idx n,
220 std::source_location loc = std::source_location::current()) {
221 if constexpr (!num::debug::sampling_compiled_in) {
222 return;
223 }
224 if (num::debug::get_level() != num::debug::diagnostic_level::full || n == 0) {
225 return;
226 }
227 using T = num::scalar_t<V>;
228 using R = scalars::real_t<T>;
229 const R tol = property_tol<T>();
230 probe_rng rng(0xC0FFEE123456789ULL);
231
232 auto rel_diff = [&](const V &a, const V &b) {
233 R num_ = R(0), den = R(0);
234 for (idx i = 0; i < n; ++i) {
235 const T d = a[i] - b[i];
236 num_ += scalars::re(scalars::conj(d) * d);
237 den += scalars::re(scalars::conj(a[i]) * a[i]);
238 }
239 return std::sqrt(num_) / (std::sqrt(den) + std::numeric_limits<R>::min());
240 };
241
242 verify_additive_group_axioms<V>(n, loc);
243
244 for (idx probe = 0; probe < g_probe_count; ++probe) {
245 V x(n), y(n);
246 fill_probe(x, rng);
247 fill_probe(y, rng);
248 const T a = static_cast<T>(R(0.7));
249 const T b = static_cast<T>(R(-1.3));
250
251 // a(x + y) == ax + ay
252 V sum = x;
253 algebra::axpy_into(T(1), y, sum);
254 algebra::scale_inplace(sum, a);
255 V ax = x;
256 algebra::scale_inplace(ax, a);
257 V ay = y;
258 algebra::scale_inplace(ay, a);
259 V ax_plus_ay = ax;
260 algebra::axpy_into(T(1), ay, ax_plus_ay);
261 if (rel_diff(sum, ax_plus_ay) > tol) {
262 panic("AlgebraError",
263 "scalar multiplication does not distribute over vector addition: "
264 "a(x+y) != ax+ay on probe " +
265 std::to_string(probe),
266 loc);
267 }
268
269 // (a + b)x == ax + bx
270 V ab_x = x;
271 algebra::scale_inplace(ab_x, a + b);
272 V bx = x;
273 algebra::scale_inplace(bx, b);
274 V ax2 = x;
275 algebra::scale_inplace(ax2, a);
276 V ax_plus_bx = ax2;
277 algebra::axpy_into(T(1), bx, ax_plus_bx);
278 if (rel_diff(ab_x, ax_plus_bx) > tol) {
279 panic("AlgebraError",
280 "scalar multiplication does not distribute over scalar addition: "
281 "(a+b)x != ax+bx on probe " +
282 std::to_string(probe),
283 loc);
284 }
285
286 // a(bx) == (ab)x
287 V abx = x;
288 algebra::scale_inplace(abx, b);
289 algebra::scale_inplace(abx, a);
290 V ab_together = x;
291 algebra::scale_inplace(ab_together, a * b);
292 if (rel_diff(abx, ab_together) > tol) {
293 panic("AlgebraError",
294 "scalar action is not compatible with field multiplication: "
295 "a(bx) != (ab)x",
296 loc);
297 }
298
299 // 1x == x
300 V one_x = x;
301 algebra::scale_inplace(one_x, T(1));
302 if (rel_diff(one_x, x) > tol) {
303 panic("AlgebraError", "the unit scalar is not an identity: 1x != x", loc);
304 }
305 }
306}

308/// @brief Sample the inner product axioms.
309///
310/// Conjugate symmetry \f$\langle x,y \rangle = \overline{\langle y,x \rangle}\f$,
311/// linearity in the second argument, and positive definiteness
312/// \f$\langle x,x \rangle > 0\f$ for \f$x \neq 0\f$. Conjugate symmetry is the one
313/// that separates a genuine Hermitian form from a transpose-only implementation,
314/// and it is invisible on real data.
315template <class V>
316inline void
317verify_inner_product_axioms(idx n, std::source_location loc = std::source_location::current()) {
318 if constexpr (!num::debug::sampling_compiled_in) {
319 return;
320 }
321 if (num::debug::get_level() != num::debug::diagnostic_level::full || n == 0) {
322 return;
323 }
324 using T = num::scalar_t<V>;
325 using R = scalars::real_t<T>;
326 const R tol = property_tol<T>();
327 probe_rng rng(0xABCDEF0123456789ULL);
328
329 for (idx probe = 0; probe < g_probe_count; ++probe) {
330 V x(n), y(n), z(n);
331 fill_probe(x, rng);
332 fill_probe(y, rng);
333 fill_probe(z, rng);
334
335 const T xy = algebra::inner(x, y);
336 const T yx = algebra::inner(y, x);
337 const R sym_err = scalars::mag(xy - scalars::conj(yx));
338 const R sym_scale = scalars::mag(xy) + std::numeric_limits<R>::min();
339 if (sym_err / sym_scale > tol) {
340 panic("AlgebraError",
341 "inner product is not conjugate symmetric: <x,y> != conj(<y,x>) "
342 "(relative error " +
343 std::to_string(static_cast<double>(sym_err / sym_scale)) + ") on probe " +
344 std::to_string(probe),
345 loc);
346 }
347
348 // Linearity in the second argument: <x, ay + z> == a<x,y> + <x,z>
349 const T a = static_cast<T>(R(0.6));
350 V combo = z;
351 algebra::axpy_into(a, y, combo);
352 const T lhs = algebra::inner(x, combo);
353 const T rhs = (a * algebra::inner(x, y)) + algebra::inner(x, z);
354 const R lin_err = scalars::mag(lhs - rhs);
355 const R lin_scale = scalars::mag(lhs) + std::numeric_limits<R>::min();
356 if (lin_err / lin_scale > tol) {
357 panic("AlgebraError",
358 "inner product is not linear in its second argument: "
359 "<x,ay+z> != a<x,y>+<x,z> on probe " +
360 std::to_string(probe),
361 loc);
362 }
363
364 // Positive definiteness on a nonzero vector.
365 const R xx = scalars::re(algebra::inner(x, x));
366 if (!(xx > R(0))) {
367 panic("AlgebraError",
368 "inner product is not positive definite: <x,x> = " +
369 std::to_string(static_cast<double>(xx)) + " for a nonzero x",
370 loc);
371 }
372 }
373}

375/// @brief Sample the norm axioms, and that the norm is the one the inner product induces.
376///
377/// Absolute homogeneity \f$\|ax\| = |a|\,\|x\|\f$, the triangle inequality, and
378/// \f$\|x\|^2 = \langle x,x \rangle\f$ -- the compatibility every Krylov method
379/// assumes when it uses a residual norm to reason about an inner product.
380template <class V>
381inline void verify_norm_axioms(idx n, std::source_location loc = std::source_location::current()) {
382 if constexpr (!num::debug::sampling_compiled_in) {
383 return;
384 }
385 if (num::debug::get_level() != num::debug::diagnostic_level::full || n == 0) {
386 return;
387 }
388 using T = num::scalar_t<V>;
389 using R = scalars::real_t<T>;
390 const R tol = property_tol<T>();
391 probe_rng rng(0x13579BDF02468ACEULL);
392
393 for (idx probe = 0; probe < g_probe_count; ++probe) {
394 V x(n), y(n);
395 fill_probe(x, rng);
396 fill_probe(y, rng);
397
398 const R nx = algebra::norm_of(x);
399 const R ny = algebra::norm_of(y);
400
401 // ||a x|| == |a| ||x||
402 const T a = static_cast<T>(R(-2.5));
403 V ax = x;
404 algebra::scale_inplace(ax, a);
405 const R nax = algebra::norm_of(ax);
406 const R expect = scalars::mag(a) * nx;
407 if (scalars::mag(nax - expect) / (expect + std::numeric_limits<R>::min()) > tol) {
408 panic("AlgebraError",
409 "norm is not absolutely homogeneous: ||ax|| != |a| ||x|| on probe " +
410 std::to_string(probe),
411 loc);
412 }
413
414 // ||x + y|| <= ||x|| + ||y||
415 V sum = x;
416 algebra::axpy_into(T(1), y, sum);
417 const R nsum = algebra::norm_of(sum);
418 if (nsum > (nx + ny) * (R(1) + tol)) {
419 panic("AlgebraError",
420 "norm violates the triangle inequality: ||x+y|| = " +
421 std::to_string(static_cast<double>(nsum)) +
422 " > ||x||+||y|| = " + std::to_string(static_cast<double>(nx + ny)) +
423 " on probe " + std::to_string(probe),
424 loc);
425 }
426
427 // ||x||^2 == <x,x>
428 const R induced = std::sqrt(scalars::re(algebra::inner(x, x)));
429 if (scalars::mag(nx - induced) / (induced + std::numeric_limits<R>::min()) > tol) {
430 panic("AlgebraError",
431 "norm is not induced by the inner product: ||x|| = " +
432 std::to_string(static_cast<double>(nx)) +
433 " but sqrt(<x,x>) = " + std::to_string(static_cast<double>(induced)),
434 loc);
435 }
436 }

449// Operator property sampling
450// ---------------------------------------------------------------------------
451//
452// Generic over anything exposing apply(x, y) and cols(); nothing here knows what
453// an "operator" is beyond that. They live in core because the property hierarchy binds
454// to them, and the hierarchy is vocabulary the whole library speaks.

456/// @brief Estimate the extreme eigenvalues of a self-adjoint operator by power iteration.
457///
458/// Returns {lambda_min, lambda_max}. The maximum is found directly; the minimum
459/// comes from power-iterating the shifted operator \f$\lambda_{max} I - A\f$, whose
460/// dominant eigenvalue is \f$\lambda_{max} - \lambda_{min}\f$. This is what separates a
461/// definite operator from a merely semi-definite one: a null space has measure
462/// zero, so no amount of random probing will land on it.
463template <class Op, class VectorType>
464[[nodiscard]] inline auto estimate_spectrum_bounds(const Op &A, idx n, idx iterations = 64) {
465 using T = num::scalar_t<VectorType>;
466 using R = scalars::real_t<T>;
467
468 VectorType v(n), w(n);
469 probe_rng rng(0xD1B54A32D192ED03ULL);
470
471 auto normalize = [&](VectorType &u) {
472 const R nrm = probe_norm(u);
473 if (nrm > R(0)) {
474 for (idx i = 0; i < n; ++i) {
475 u[i] = u[i] / static_cast<T>(nrm);
476 }
477 }
478 };
479
480 fill_probe(v, rng);
481 normalize(v);
482 R lambda_max = R(0);
483 for (idx it = 0; it < iterations; ++it) {
484 A.apply(v, w);
485 lambda_max = scalars::re(probe_inner(v, w));
486 normalize(w);
487 v = w;
488 }
489
490 fill_probe(v, rng);
491 normalize(v);
492 R shifted = R(0);
493 for (idx it = 0; it < iterations; ++it) {
494 A.apply(v, w);
495 for (idx i = 0; i < n; ++i) {
496 w[i] = (static_cast<T>(lambda_max) * v[i]) - w[i];
497 }
498 shifted = scalars::re(probe_inner(v, w));
499 normalize(w);
500 v = w;
501 }
502
503 struct bounds {
504 R min;
505 R max;
506 };
507 return bounds{lambda_max - shifted, lambda_max};
508}

510/// @brief Sampled test for positive definiteness \f$\langle x, A x \rangle > 0\ \forall x \neq
511/// 0\f$.
512///
513/// Basis probes check the necessary condition \f$A_{ii} > 0\f$ exactly; randomized
514/// probes then sample the quadratic form away from the axes.
515template <class Op, class VectorType>
516inline void verify_spd_sample(const Op &A, idx n,
517 std::source_location loc = std::source_location::current()) {
518 if constexpr (!num::debug::sampling_compiled_in) {
519 return;
520 }
521 if (num::debug::get_level() != num::debug::diagnostic_level::full || n == 0) {
522 return;
523 }
524 using T = num::scalar_t<VectorType>;
525 using R = scalars::real_t<T>;
526
527 VectorType x(n), Ax(n);
528
529 const idx stride = probe_stride(n);
530 for (idx i = 0; i < n; i += stride) {
531 fill_basis(x, i);
532 A.apply(x, Ax);
533 const R diagonal = scalars::re(Ax[i]);
534 if (!(diagonal > R(0))) {
535 panic("PropertyError",
536 "assume_spd() assertion failed: diagonal entry A(" + std::to_string(i) + "," +
537 std::to_string(i) + ") = " + std::to_string(static_cast<double>(diagonal)) +
538 " is not positive, so the operator is NOT positive definite.",
539 loc);
540 }
541 }
542
543 probe_rng rng;
544 for (idx probe = 0; probe < g_probe_count; ++probe) {
545 fill_probe(x, rng);
546 A.apply(x, Ax);
547 const R quadratic_form = scalars::re(probe_inner(x, Ax));
548 if (!(quadratic_form > R(0))) {
549 panic("PropertyError",
550 "assume_spd() assertion failed: sampled quadratic form Re<x,Ax> = " +
551 std::to_string(static_cast<double>(quadratic_form)) + " on probe " +
552 std::to_string(probe) +
553 " is not positive, so the operator is NOT positive definite.",
554 loc);
555 }
556 }
557
558 // Definiteness is a statement about the smallest eigenvalue, which random
559 // probing cannot reach: a singular positive *semi*-definite operator has a
560 // null space of measure zero and passes every probe above.
561 const auto bounds = estimate_spectrum_bounds<Op, VectorType>(A, n);
562 const R floor_value = bounds.max * property_tol<T>();
563 if (!(bounds.min > floor_value)) {
564 panic("PropertyError",
565 "assume_spd() assertion failed: estimated smallest eigenvalue " +
566 std::to_string(static_cast<double>(bounds.min)) + " against largest " +
567 std::to_string(static_cast<double>(bounds.max)) +
568 " indicates the operator is singular or indefinite, NOT positive definite.",
569 loc);
570 }
571}

573/// @brief Sampled test for positive semi-definiteness \f$\langle x, A x \rangle \geq 0\f$.
574///
575/// Identical to the definite test but admitting a null space, as required by
576/// Gram matrices and graph Laplacians.
577template <class Op, class VectorType>
578inline void verify_psd_sample(const Op &A, idx n,
579 std::source_location loc = std::source_location::current()) {
580 if constexpr (!num::debug::sampling_compiled_in) {
581 return;
582 }
583 if (num::debug::get_level() != num::debug::diagnostic_level::full || n == 0) {
584 return;
585 }
586 using T = num::scalar_t<VectorType>;
587 using R = scalars::real_t<T>;
588
589 VectorType x(n), Ax(n);
590 const R tol = property_tol<T>();
591
592 const idx stride = probe_stride(n);
593 for (idx i = 0; i < n; i += stride) {
594 fill_basis(x, i);
595 A.apply(x, Ax);
596 const R diagonal = scalars::re(Ax[i]);
597 const R scale = probe_norm(Ax) + std::numeric_limits<R>::min();
598 if (diagonal / scale < -tol) {
599 panic("PropertyError",
600 "assume_psd() assertion failed: diagonal entry A(" + std::to_string(i) + "," +
601 std::to_string(i) + ") = " + std::to_string(static_cast<double>(diagonal)) +
602 " is negative, so the operator is NOT positive semi-definite.",
603 loc);
604 }
605 }
606
607 probe_rng rng;
608 for (idx probe = 0; probe < g_probe_count; ++probe) {
609 fill_probe(x, rng);
610 A.apply(x, Ax);
611 const R quadratic_form = scalars::re(probe_inner(x, Ax));
612 const R scale = (probe_norm(x) * probe_norm(Ax)) + std::numeric_limits<R>::min();
613 if (quadratic_form / scale < -tol) {
614 panic("PropertyError",
615 "assume_psd() assertion failed: sampled quadratic form Re<x,Ax> = " +
616 std::to_string(static_cast<double>(quadratic_form)) + " on probe " +
617 std::to_string(probe) +
618 " is negative, so the operator is NOT positive semi-definite.",
619 loc);
620 }
621 }

624/// @brief Sampled test for self-adjointness \f$\langle x, A y \rangle = \overline{\langle y, A x
625/// \rangle}\f$.
626///
627/// On a real field this is symmetry \f$A = A^T\f$; on a complex field it is the
628/// Hermitian condition \f$A = A^*\f$, which conjugate-free comparison would miss.
629template <class Op, class VectorType>
630inline void verify_symmetry_sample(const Op &A, idx n,
631 std::source_location loc = std::source_location::current()) {
632 if constexpr (!num::debug::sampling_compiled_in) {
633 return;
634 }
635 if (num::debug::get_level() != num::debug::diagnostic_level::full || n <= 1) {
636 return;
637 }
638 using T = num::scalar_t<VectorType>;
639 using R = scalars::real_t<T>;
640
641 VectorType x(n), y(n), Ax(n), Ay(n);
642 probe_rng rng;
643 const R tol = property_tol<T>();
644
645 for (idx probe = 0; probe < g_probe_count; ++probe) {
646 fill_probe(x, rng);
647 fill_probe(y, rng);
648 A.apply(x, Ax);
649 A.apply(y, Ay);
650
651 const T x_A_y = probe_inner(x, Ay);
652 const T y_A_x = probe_inner(y, Ax);
653 const R difference = scalars::mag(x_A_y - scalars::conj(y_A_x));
654 const R scale =
655 std::max(scalars::mag(x_A_y), scalars::mag(y_A_x)) + std::numeric_limits<R>::min();
656
657 if (difference / scale > tol) {
658 panic("PropertyError",
659 "assume_symmetric() assertion failed: relative |<x,Ay> - conj(<y,Ax>)| = " +
660 std::to_string(static_cast<double>(difference / scale)) + " on probe " +
661 std::to_string(probe) + " exceeds tolerance " +
662 std::to_string(static_cast<double>(tol)) +
663 ", so the operator is NOT self-adjoint.",
664 loc);
665 }
666 }
667}

710/// @brief Sampled test for isometry \f$\|A x\| = \|x\|\f$ (equivalently \f$A^* A = I\f$).
711///
712/// Basis probes check the necessary condition \f$\|A e_i\| = 1\f$ exactly, which a
713/// single all-ones probe cannot detect: any A whose column norms happen to average
714/// correctly would otherwise pass.
715template <class Op, class VectorType>
716inline void verify_orthogonal_sample(const Op &A, idx n,
717 std::source_location loc = std::source_location::current()) {
718 if constexpr (!num::debug::sampling_compiled_in) {
719 return;
720 }
721 if (num::debug::get_level() != num::debug::diagnostic_level::full || n == 0) {
722 return;
723 }
724 using T = num::scalar_t<VectorType>;
725 using R = scalars::real_t<T>;
726
727 VectorType x(n), Ax(n);
728 const R tol = property_tol<T>();
729
730 const idx stride = probe_stride(n);
731 for (idx i = 0; i < n; i += stride) {
732 fill_basis(x, i);
733 A.apply(x, Ax);
734 const R column_norm = probe_norm(Ax);
735 if (scalars::mag(column_norm - R(1)) > tol) {
736 panic("PropertyError",
737 "assume_orthogonal() assertion failed: column " + std::to_string(i) +
738 " has norm " + std::to_string(static_cast<double>(column_norm)) +
739 " rather than 1, so the operator is NOT an isometry.",
740 loc);
741 }
742 }
743
744 probe_rng rng;
745 for (idx probe = 0; probe < g_probe_count; ++probe) {
746 fill_probe(x, rng);
747 A.apply(x, Ax);
748 const R norm_x = probe_norm(x);
749 const R norm_Ax = probe_norm(Ax);
750 const R scale = norm_x + std::numeric_limits<R>::min();
751 if (scalars::mag(norm_Ax - norm_x) / scale > tol) {
752 panic("PropertyError",
753 "assume_orthogonal() assertion failed: relative | ||Ax|| - ||x|| | = " +
754 std::to_string(static_cast<double>(scalars::mag(norm_Ax - norm_x) / scale)) +
755 " on probe " + std::to_string(probe) +
756 ", so the operator is NOT an isometry.",
757 loc);
758 }
759 }

802/// @brief Sampled test for skew-adjointness \f$A = -A^*\f$, i.e. \f$\mathrm{Re}\langle x, A x
803/// \rangle = 0\ \forall x\f$.
804///
805/// The comparison is relative to \f$\|x\|\,\|Ax\|\f$ (the Cauchy-Schwarz bound on
806/// the quantity being tested), so the test does not weaken as the operator scale
807/// grows. Basis probes additionally pin every sampled diagonal entry to zero,
808/// which is what distinguishes a skew operator from a symmetric one that merely
809/// annihilates a particular probe vector.
810template <class Op, class VectorType>
811inline void
812verify_skew_symmetry_sample(const Op &A, idx n,
813 std::source_location loc = std::source_location::current()) {
814 if constexpr (!num::debug::sampling_compiled_in) {
815 return;
816 }
817 if (num::debug::get_level() != num::debug::diagnostic_level::full || n == 0) {
818 return;
819 }
820 using T = num::scalar_t<VectorType>;
821 using R = scalars::real_t<T>;
822
823 VectorType x(n), Ax(n);
824 const R tol = property_tol<T>();
825
826 const idx stride = probe_stride(n);
827 for (idx i = 0; i < n; i += stride) {
828 fill_basis(x, i);
829 A.apply(x, Ax);
830 const R diagonal = scalars::mag(scalars::re(Ax[i]));
831 const R scale = probe_norm(Ax) + std::numeric_limits<R>::min();
832 if (diagonal / scale > tol) {
833 panic("PropertyError",
834 "assume_skew_symmetric() assertion failed: diagonal entry A(" +
835 std::to_string(i) + "," + std::to_string(i) +
836 ") = " + std::to_string(static_cast<double>(scalars::re(Ax[i]))) +
837 " is nonzero, so the operator is NOT skew-adjoint.",
838 loc);
839 }
840 }
841
842 probe_rng rng;
843 for (idx probe = 0; probe < g_probe_count; ++probe) {
844 fill_probe(x, rng);
845 A.apply(x, Ax);
846 const R quadratic_form = scalars::mag(scalars::re(probe_inner(x, Ax)));
847 const R scale = (probe_norm(x) * probe_norm(Ax)) + std::numeric_limits<R>::min();
848 if (quadratic_form / scale > tol) {
849 panic("PropertyError",
850 "assume_skew_symmetric() assertion failed: relative Re<x,Ax> = " +
851 std::to_string(static_cast<double>(quadratic_form / scale)) + " on probe " +
852 std::to_string(probe) + " is nonzero, so the operator is NOT skew-adjoint.",
853 loc);
854 }
855 }

858/// @brief Sampled test for linearity \f$A(\alpha x + \beta y) = \alpha A x + \beta A y\f$.
859///
860/// Every other operator property presupposes this one, yet a stateful or
861/// accidentally affine `apply` satisfies none of the tests above by contradiction.
862template <class Op, class VectorType>
863inline void verify_linearity_sample(const Op &A, idx n,
864 std::source_location loc = std::source_location::current()) {
865 if constexpr (!num::debug::sampling_compiled_in) {
866 return;
867 }
868 if (num::debug::get_level() != num::debug::diagnostic_level::full || n == 0) {
869 return;
870 }
871 using T = num::scalar_t<VectorType>;
872 using R = scalars::real_t<T>;
873
874 VectorType x(n), y(n), combination(n), Ax(n), Ay(n), A_combination(n);
875 probe_rng rng;
876 const R tol = property_tol<T>();
877
878 for (idx probe = 0; probe < g_probe_count; ++probe) {
879 fill_probe(x, rng);
880 fill_probe(y, rng);
881 const T alpha = static_cast<T>(R(0.75));
882 const T beta = static_cast<T>(R(-1.25));
883 for (idx i = 0; i < n; ++i) {
884 combination[i] = (alpha * x[i]) + (beta * y[i]);
885 }
886
887 A.apply(x, Ax);
888 A.apply(y, Ay);
889 A.apply(combination, A_combination);
890
891 R residual_sq = R(0);
892 R scale_sq = R(0);
893 for (idx i = 0; i < n; ++i) {
894 const T expected = (alpha * Ax[i]) + (beta * Ay[i]);
895 const T d = A_combination[i] - expected;
896 residual_sq += scalars::re(scalars::conj(d) * d);
897 scale_sq += scalars::re(scalars::conj(expected) * expected);
898 }
899 const R relative =
900 std::sqrt(residual_sq) / (std::sqrt(scale_sq) + std::numeric_limits<R>::min());
901
902 if (relative > tol) {
903 panic("PropertyError",
904 "Linearity check failed: relative ||A(ax+by) - (aAx+bAy)|| = " +
905 std::to_string(static_cast<double>(relative)) + " on probe " +
906 std::to_string(probe) + ", so the operator is NOT linear.",
907 loc);
908 }
909 }
