15/// @brief Finite-difference stencil applied over a structured grid.
16///
17/// A stencil writes \f$(Lu)_i\f$ from a neighbourhood of \f$u_i\f$ without
18/// assembling a matrix, which is what keeps the work \f$O(N)\f$ per step for a
19/// grid of \f$N\f$ points.
20template <class S, class V = vec>
21concept grid_stencil = vector_space<V> && requires(const S &stencil, const V &u, V &out, int n) {
22 stencil.apply(u, out, n);
23};

25/// @brief Grid operator that can also materialize itself as a sparse matrix.
26///
27/// Krylov methods need only the action. A direct solve needs the matrix. An
28/// operator satisfying this supports both, so the choice of solver does not
29/// change how the discretization is written.
30template <class Op>
31concept assemblable_grid_operator = linear_operator<Op> && requires(const Op &A) {
32 { A.to_sparse() };
33};
34
35/// @brief Operator arising from an implicit step, \f$(I - \Delta t\, L)\f$.
36///
37/// For a diffusion operator \f$L\f$ this is symmetric positive definite for every
38/// \f$\Delta t > 0\f$, which is what allows conjugate gradients rather than a
39/// general solver. The property is declared through the hierarchy, so a stepper
40/// constrained here cannot be handed an operator that lacks it.
41template <class Op>
42concept implicit_step_operator = spd_operator<Op>;
43
44/// @brief Stepper advancing a field from \f$t\f$ to \f$t + \Delta t\f$.
45///
46/// The field is required to expose a vector space, which is what an implicit step needs
47/// to solve in — the same requirement `num::vec_field` states for ODE state.
48template <class S, class F, class V = vec>
49concept field_stepper = vec_field<F, V> && requires(S &stepper, F &u, real dt) {
50 stepper.step(u, dt);
51};
