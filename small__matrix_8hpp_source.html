140/// Plane rotation used to eliminate one component of a two-vector.
141struct givens_rotation {
142 real c = 1;
143 real s = 0;
144
145 /// @brief Construct \f$G=\begin{bmatrix}c&s\\-s&c\end{bmatrix}\f$.
146 static constexpr givens_rotation from(real a, real b) noexcept {
147 if (b == 0) {
148 return {};
149 }
150 const real r = std::sqrt((a * a) + (b * b));
151 return {a / r, b / r};
152 }
153
154 /// Apply G to the pair (x,y) in place.
155 constexpr void apply(real &x, real &y) const noexcept {
156 const real tmp = (c * x) + (s * y);
157 y = (-s * x) + (c * y);
158 x = tmp;
159 }
160
161 /// Apply G^T to the pair (x,y) in place.
162 constexpr void apply_t(real &x, real &y) const noexcept {
163 const real tmp = (c * x) - (s * y);
164 y = (s * x) + (c * y);
165 x = tmp;
166 }
167
168 /// Materialize the rotation as a 2x2 matrix.
169 constexpr small_matrix<2, 2> matrix() const noexcept {
170 small_matrix<2, 2> G{};
171 G(0, 0) = c;
172 G(0, 1) = s;
173 G(1, 0) = -s;
174 G(1, 1) = c;
175 return G;
176 }
