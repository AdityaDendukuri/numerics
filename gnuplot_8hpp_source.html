17/// Ordered (x,y) samples accepted by the plotting helpers.
18struct series : array<plot_point> {
19 using array<plot_point>::vector;
20 /// Append one sample.
21 void store(double x, double y) { emplace_back(x, y); }
22};

24/// Extract one row as (x_j,u_j) plot data.
25inline series row_slice(const vec &u, int N, double h, int row) {
26 series result;
27 result.reserve(static_cast<std::size_t>(N));
28 for (int column = 0; column < N; ++column) {
29 result.store((column + 1) * h, u[(static_cast<std::size_t>(row) * N) + column]);
30 }
31 return result;
32}

56/// Owning pipe for issuing low-level commands directly to gnuplot.
57class gnuplot {
58 public:
59 /// Launch gnuplot with optional command-line arguments.
60 explicit gnuplot(const std::string &args = "") {
61 const std::string command = "gnuplot " + args;
62 pipe_ = popen(command.c_str(), "w");
63 if (!pipe_) {
64 throw std::runtime_error("could not open gnuplot -- is it installed?");
65 }
66 }
67
68 ~gnuplot() {
69 if (pipe_) {
70 pclose(pipe_);
71 }
72 }
73
74 gnuplot(const gnuplot &) = delete;
75 gnuplot &operator=(const gnuplot &) = delete;
76
77 /// Send a raw gnuplot command.
78 gnuplot &operator<<(const std::string &command) {
79 fputs(command.c_str(), pipe_);
80 return *this;
81 }
82
83 /// Send one inline two-column data block and flush it.
84 void send1d(const series &data) {
85 for (const auto &[x, y] : data) {
86 fprintf(pipe_, "%.15g %.15g\n", x, y);
87 }
88 fputs("e\n", pipe_);
89 fflush(pipe_);
90 }
91
92 /// Flush pending commands to gnuplot.
93 void flush() { fflush(pipe_); }
94
95 private:
96 FILE *pipe_ = nullptr;
97};
