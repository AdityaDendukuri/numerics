13/// Dense storage participates in the map protocol without pretending that every
14/// matrix value is positive definite or even square.
15template <std::floating_point T>
16inline void tag_invoke(math::apply_t, const basic_mat<T> &matrix, const basic_vec<T> &x,
17 basic_vec<T> &y) {
18 if (x.size() != matrix.cols()) {
19 throw std::invalid_argument("math::apply: dense matrix input dimension mismatch");
20 }
21 if (y.size() != matrix.rows()) {
22 y = basic_vec<T>(matrix.rows());
23 }
24 kernel::matvec(y.data(), matrix.data(), x.data(), matrix.rows(), matrix.cols());
25}
