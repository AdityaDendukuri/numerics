12/// @brief Reusable plan executing a fixed-length transform.
13///
14/// \f[ X_k = \sum_{n=0}^{N-1} x_n \, e^{-2\pi i k n / N} \f]
15///
16/// A plan is built once for a length and reused, which is what lets a backend
17/// precompute twiddle factors or hand the length to FFTW. Transform length is
18/// fixed at construction, so a plan and a vector must agree on it.
19template <class P, class V = cvec>
20concept transform_plan = inner_product_space<V> && requires(const P &plan, const V &in, V &out) {
21 { plan.size() } -> std::convertible_to<int>;
22 plan.execute(in, out);
23};

25/// @brief Transform preserving the inner product up to a scale factor.
26///
27/// \f[ \sum_n |x_n|^2 = \frac{1}{N} \sum_k |X_k|^2 \f]
28///
29/// Parseval's identity is a statement about the inner product, which is why the
30/// domain is required to be an inner product space rather than merely indexable.
31/// It is the property that makes a spectral method conserve energy, and
32/// `num::spectral::debug::verify_parseval` samples it.
33template <class P, class V = cvec>
34concept unitary_transform = transform_plan<P, V>;
