1/// @file container/matrix_expr.hpp
2/// @brief Value-returning dense arithmetic, and opt-in operators built on it.
3///
4/// Numerics' primary kernels write into caller-provided buffers so that no
5/// allocation happens inside a simulation loop. That remains the default. The
6/// functions here are the convenience tier above them: each returns its result
7/// and checks conformance, which the out-parameter forms cannot do because the
8/// caller has already sized the output.
9///
10/// Operators live in the nested `num::ops` namespace and are therefore opt-in.
11/// Ordinary lookup does not find them until a translation unit asks:
12///
13/// ```cpp
14/// using namespace num::ops;
15/// const vec r = y - Z * q;
16/// ```
17///
18/// Prefer the out-parameter forms in hot loops; prefer these where the code is
19/// stating a formula and clarity is worth one temporary.

30/// Return A*B. Allocates; prefer the out-param form in hot loops.
31[[nodiscard]] inline mat matmul(const mat &A, const mat &B) {
32 if (A.cols() != B.rows()) {
33 throw std::invalid_argument("matmul: inner dimensions do not agree");
34 }
35 mat C(A.rows(), B.cols(), 0.0);
36 matmul(A, B, C);
37 return C;
38}

40/// Return A*x. Allocates; prefer the out-param form in hot loops.

44 }

50/// Return A+B. Allocates; prefer the out-param form in hot loops.

90/// Return alpha*A, matching the value-returning `scaled` for sparse matrices.
91/// Allocates; prefer in-place scaling in hot loops.
92[[nodiscard]] inline mat scaled(const mat &A, real alpha) {
93 mat result(A.rows(), A.cols(), 0.0);
94 matadd(alpha, A, 0.0, A, result);
95 return result;
96}

98/// Return alpha*x for vectors.
99[[nodiscard]] inline vec scaled(const vec &x, real alpha) {
100 vec result = x;
