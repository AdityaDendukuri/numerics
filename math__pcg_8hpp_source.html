22template <class Op, class M, class V, class Invariant>
23requires math::inner_product_space<V> &&math::endomorphism_on<Op, V> &&math::endomorphism_on<M, V> &&
24 std::floating_point<math::scalar_t<V>> [[nodiscard]] solver_result
25 pcg_recurrence(const Op &A, const M &preconditioner, const V &b, V &x, pcg_options options,
26 const Invariant &check_invariant) {
27 using S = math::scalar_t<V>;
28 const auto n = math::dimension(b);
29 if (math::dimension(x) != n || A.rows() != n || A.cols() != n || preconditioner.rows() != n ||
30 preconditioner.cols() != n) {
31 throw std::invalid_argument(
32 "pcg: incompatible operator, preconditioner, and vector dimensions");
33 }
34 if (!(options.tolerance > 0.0)) {
35 throw std::invalid_argument("pcg: invalid convergence options");
36 }
37 V residual = math::zero_like(b);
38 V preconditioned = math::zero_like(b);
39 V direction = math::zero_like(b);
40 V applied = math::zero_like(b);
41
42 math::apply(A, x, residual);
43 check_invariant(residual, "pcg: operator did not preserve the certified subspace");
44 // r <- b - A*x
45 math::linear_combination(S(1), b, S(-1), residual);
46 check_invariant(residual, "pcg: residual left the certified subspace");
47
48 solver_result result{0, static_cast<real>(math::norm(residual)), false};
49 if (result.residual < options.tolerance) {
50 result.converged = true;
51 return result;
52 }
53
54 math::apply(preconditioner, residual, preconditioned);
55 check_invariant(preconditioned, "pcg: preconditioner did not preserve the certified subspace");
56 S weighted_residual = math::inner(residual, preconditioned);
57 if (!(weighted_residual > S(0)) || !std::isfinite(weighted_residual)) {
58 throw std::runtime_error("pcg: positive-definite preconditioner invariant was violated");
59 }
60 direction = preconditioned;
61
62 for (idx iteration = 0; iteration < options.max_iterations; ++iteration) {
63 result.iterations = iteration + 1;
64 math::apply(A, direction, applied);
65 check_invariant(applied, "pcg: operator did not preserve the certified subspace");
66 const S curvature = math::inner(direction, applied);
67 if (!(curvature > S(0)) || !std::isfinite(curvature)) {
68 throw std::runtime_error("pcg: positive-definite operator invariant was violated");
69 }
70
71 const S alpha = weighted_residual / curvature;
72 // x <- x + alpha*p
73 math::axpy(alpha, direction, x);
74 // r <- r - alpha*A*p; ||r||_2^2
75 const S residual_square = math::axpy_norm_sq(-alpha, applied, residual);
76 check_invariant(x, "pcg: iterate left the certified subspace");
77 check_invariant(residual, "pcg: residual left the certified subspace");
78 result.residual = static_cast<real>(std::sqrt(residual_square));
79 if (!std::isfinite(result.residual)) {
80 throw std::runtime_error("pcg: inner-product norm invariant was violated");
81 }
82 if (result.residual < options.tolerance) {
83 result.converged = true;
84 break;
85 }
86
87 math::apply(preconditioner, residual, preconditioned);
88 check_invariant(preconditioned,
89 "pcg: preconditioner did not preserve the certified subspace");
90 const S next_weighted_residual = math::inner(residual, preconditioned);
91 if (!(next_weighted_residual > S(0)) || !std::isfinite(next_weighted_residual)) {
92 throw std::runtime_error(
93 "pcg: positive-definite preconditioner invariant was violated");
94 }
95 const S beta = next_weighted_residual / weighted_residual;
96 // p <- M^-1*r + beta*p
97 math::linear_combination(S(1), preconditioned, beta, direction);
98 check_invariant(direction, "pcg: search direction left the certified subspace");
99 weighted_residual = next_weighted_residual;
100 }
101 return result;
102}

117/// PCG on a named invariant subspace. Evidence for A and M is tied to the exact
118/// same Subspace type, and membership is checked throughout the recurrence.
119template <class Subspace, class Op, class M, class V>
120requires math::inner_product_space<V> &&math::linear_subspace_of<Subspace, V> &&
121 math::endomorphism_on<Op, V> &&math::endomorphism_on<M, V> &&
122 claims<Op, law::spd_on<Subspace>> &&
123 claims<M, law::spd_on<Subspace>> &&
124 std::floating_point<math::scalar_t<V>> [[nodiscard]] solver_result
125 pcg(const Op &A, const M &preconditioner, const V &b, V &x,
126 const Subspace &subspace, pcg_options options = {}) {
127 if (!math::contains(subspace, b)) {
128 throw std::invalid_argument("pcg: right-hand side is outside the certified subspace");
129 }
130 if (!math::contains(subspace, x)) {
131 throw std::invalid_argument("pcg: initial iterate is outside the certified subspace");
132 }
133 const auto enforce_subspace = [&subspace](const V &value, const char *message) {
134 if (!math::contains(subspace, value)) {
135 throw std::runtime_error(message);
136 }
137 };
138 return math_krylov_detail::pcg_recurrence(A, preconditioner, b, x, options, enforce_subspace);
139}
