18/// @brief Verify that \f$f\f$ changes sign on \f$[a,b]\f$.
19///
20/// Bisection and Brent's method converge because a sign change forces a root
21/// inside the interval. Without one they return a value that is not a root and
22/// report no error, so the bracket is checked rather than assumed.
23template <class F, class T = real>
24requires scalar_function<F, T> inline void
25verify_bracket(F &&f, T a, T b, std::source_location loc = std::source_location::current()) {
26 if (get_level() == diagnostic_level::off) {
27 return;
28 }
29 if (!(a < b)) {
30 panic("BracketError", "root bracket requires a < b", loc);
31 }
32 const T fa = f(a);
33 const T fb = f(b);
34 if (fa == T(0) || fb == T(0)) {
35 return;
36 }
37 if ((fa > T(0)) == (fb > T(0))) {
38 panic("BracketError",
39 "f does not change sign on the bracket: f(a) = " +
40 std::to_string(static_cast<double>(fa)) +
41 ", f(b) = " + std::to_string(static_cast<double>(fb)) +
42 ". A bracketing method cannot converge to a root here.",
43 loc);
44 }
45}

47/// @brief Verify a supplied derivative against a central difference.
48///
49/// A wrong derivative makes Newton's method converge slowly, to the wrong point,
50/// or not at all, and none of those report an error on their own.
51template <class F, class D, class T = real>
52requires differentiable_function<F, D, T> inline void
53verify_derivative(F &&f, D &&df, T x, T tol = T(1e-5),
54 std::source_location loc = std::source_location::current()) {
55 if constexpr (!num::debug::sampling_compiled_in) {
56 return;
57 }
58 if (num::debug::get_level() != num::debug::diagnostic_level::full) {
59 return;
60 }
61 const T h = std::cbrt(std::numeric_limits<T>::epsilon()) * std::max(T(1), std::abs(x));
62 const T numeric = (f(x + h) - f(x - h)) / (T(2) * h);
63 const T supplied = df(x);
64 const T scale = std::max(std::abs(numeric), std::abs(supplied)) + std::numeric_limits<T>::min();
65 if (std::abs(numeric - supplied) / scale > tol) {
66 panic("DerivativeError",
67 "supplied derivative disagrees with a central difference at x = " +
68 std::to_string(static_cast<double>(x)) + ": got " +
69 std::to_string(static_cast<double>(supplied)) + ", expected about " +
70 std::to_string(static_cast<double>(numeric)),
71 loc);
72 }
73}
