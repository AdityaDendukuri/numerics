231// Native mathematical operations lower to the dependency-free raw kernels, one
232// block per thread. This is the path every generic solver takes: `math_cg` and
233// friends call these CPOs and nothing else, so threading has to arrive here
234// rather than through an explicit `num::omp::dot(x, y)` call, which no
235// algorithm uses. Foreign vectors use the coordinate fallbacks or provide their
236// own tag_invoke overloads without inheriting from a numerics type.
237template <std::floating_point T>
238inline void tag_invoke(math::scale_t, T alpha, basic_vec<T> &vector) noexcept {
239 T *data = vector.data();
240 omp::parallel_apply(vector.size(), [data, alpha](idx offset, idx length) {
241 kernel::scale(data + offset, alpha, length);
242 });
243}

272[[nodiscard]] inline T tag_invoke(math::inner_t, const basic_vec<T> &x, const basic_vec<T> &y) {

283template <std::floating_point T>
284[[nodiscard]] inline T tag_invoke(math::axpy_norm_sq_t, T alpha, const basic_vec<T> &x,
285 basic_vec<T> &y) {
286 if (x.size() != y.size()) {
287 throw std::invalid_argument("math::axpy_norm_sq: vector dimensions must match");
288 }
289 // y <- y + alpha*x; return ||y||_2^2. Blocks are disjoint, so the update and
290 // the reduction stay correct when they are split across threads.
291 const T *xd = x.data();
292 T *yd = y.data();
293 return omp::parallel_reduce<T>(x.size(), [xd, yd, alpha](idx offset, idx length) {
294 return kernel::axpy_norm_sq(yd + offset, xd + offset, alpha, length);
295 });
296}

298template <std::floating_point T>
299[[nodiscard]] inline T tag_invoke(math::norm_t, const basic_vec<T> &vector) noexcept {
300 const T *data = vector.data();
301 const idx n = vector.size();
302 const T squared = omp::parallel_reduce<T>(n, [data](idx offset, idx length) {
303 return kernel::norm_sq(data + offset, length);
304 });
305 if (squared > T(0) && std::isfinite(squared)) {
306 return std::sqrt(squared);
307 }
308 // Overflowed, underflowed, or not a number: fall back to the rescaled
309 // single-threaded form, which handles all three.
310 return kernel::norm(data, n);
311}
