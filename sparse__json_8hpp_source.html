21inline spmat sparse_matrix(const json &value) {
22 if (!value.is_object()) {
23 throw std::invalid_argument("sparse matrix JSON must be an object");
24 }
25 if (!value.contains("shape") || !value.contains("values")) {
26 throw std::invalid_argument("sparse matrix JSON missing required fields");
27 }
28 const auto shape = value["shape"].get<array<idx>>();
29 if (shape.size() != 2) {
30 throw std::invalid_argument("sparse matrix shape must have two entries");
31 }
32 const auto values = value["values"].get<array<real>>();
33 const std::string storage = value.value("format", "csc");
34
35 if (storage == "csr") {
36 if (!value.contains("col_indices") || !value.contains("row_ptrs")) {
37 throw std::invalid_argument("csr sparse matrix JSON missing col_indices or row_ptrs");
38 }
39 const auto columns = value["col_indices"].get<array<idx>>();
40 const auto pointers = value["row_ptrs"].get<array<idx>>();
41 if (pointers.size() != shape[0] + 1 || pointers.back() != values.size() ||
42 columns.size() != values.size()) {
43 throw std::invalid_argument("invalid sparse matrix csr arrays");
44 }
45 return spmat(shape[0], shape[1], values, columns, pointers);
46 }
47 if (storage != "csc") {
48 throw std::invalid_argument("sparse matrix format must be 'csr' or 'csc'");
49 }
50 if (!value.contains("row_indices") || !value.contains("col_ptrs")) {
51 throw std::invalid_argument("CSC sparse matrix JSON missing row_indices or col_ptrs");
52 }
53 const auto rows = value["row_indices"].get<array<idx>>();
54 const auto pointers = value["col_ptrs"].get<array<idx>>();
55 if (pointers.empty() || pointers.back() > values.size() || pointers.back() > rows.size()) {
56 throw std::invalid_argument("invalid sparse matrix CSC pointers");
57 }
58 return spmat::from_csc(shape[0], shape[1], values, rows, pointers);
59}

61inline json sparse_matrix_json(const spmat &matrix) {
62 json result;
63 result["shape"] = {matrix.n_rows(), matrix.n_cols()};
64 result["format"] = "csr";
65
66 array<real> values;
67 array<idx> columns;
68 array<idx> pointers;
69 pointers.reserve(matrix.n_rows() + 1);
70 pointers.push_back(0);
71
72 for (idx row = 0; row < matrix.n_rows(); ++row) {
73 for (idx k = matrix.row_ptr()[row]; k < matrix.row_ptr()[row + 1]; ++k) {
74 values.push_back(matrix.values()[k]);
75 columns.push_back(matrix.col_idx()[k]);
76 }
77 pointers.push_back(values.size());
78 }
79
80 result["values"] = std::move(values);
81 result["col_indices"] = std::move(columns);
82 result["row_ptrs"] = std::move(pointers);
83 return result;
84}
