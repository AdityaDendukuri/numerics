25/// @brief One Arnoldi iteration step: expands orthonormal Krylov basis \f$V_k \to V_{k+1}\f$.
26template <class Op>
27requires requires(const Op &A, const vec &x, vec &y) {
28 A.apply(x, y);
29}
30[[nodiscard]] real arnoldi_step(const Op &A, array<vec> &basis, array<real> &h,
31 idx k, vec &scratch, real breakdown_tol = real(1e-14)) {
32 // w <- A*v_k
33 A.apply(basis[k], scratch);
34
35 // (h_{0:k,k}, h_{k+1,k}) <- Arnoldi orthogonalization of w
36 const real beta = mgs_orthogonalize(basis, scratch, h, k + 1);
37 h[k + 1] = beta;
38
39 if (beta > breakdown_tol) {
40 // v_{k+1} <- w/h_{k+1,k}
41 scale(scratch, real(1) / beta);
42 basis.push_back(scratch);
43 }
44
45 return beta;
46}

48inline real mgs_orthogonalize(const array<vec> &basis, vec &v, array<real> &h,
49 idx k) {
50 for (idx i = 0; i < k; ++i) {
51 // h_i <- v_i^T*v
52 h[i] = dot(v, basis[i]);
53 // v <- v - h_i*v_i
54 axpy(-h[i], basis[i], v);
55 }
56 return norm(v);
57}

59inline real mgs_orthogonalize(const mat &basis, idx k, vec &v) {
60 const idx n = basis.rows();
61 // v <- (I - V_k*V_k^T)v, in modified Gram--Schmidt order
62 kernel::mgs_columns(v.data(), basis.data(), basis.cols(), n, k);
63 return norm(v);
64}
