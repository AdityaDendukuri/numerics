1/// @file linear/concepts.hpp
2/// @brief Structural and property contracts for linear algebra objects and solvers.
3///
4/// Matrices and matrix-free operators state their properties in one vocabulary:
5/// both record a position in the `num::law` hierarchy, so `spd_matrix_like` and
6/// `num::spd_operator` mean the same mathematical thing about different
7/// representations.
8///

39/// @brief mat carrying a square-dimension guarantee \f$A: V \to V\f$.
40///
41/// Squareness of a dynamically sized matrix is a runtime fact, so this asks
42/// whether a guarantee has been *attached* — by `assume_square`, `make_square`, or
43/// by a property that implies it. It deliberately does not ask whether
44/// `rows() == cols()` is a well-formed expression, which is true of every matrix
45/// and therefore says nothing.
46template <class M>
47concept square_matrix_like =
48 matrix_space<M> && (detail::has_square_tag<M> || claims<M, law::self_adjoint>);
49
50/// @brief mat asserted self-adjoint: \f$A = A^T\f$, or \f$A = A^*\f$ over \f$\mathbb{C}\f$.
51template <class M>
52concept symmetric_matrix_like =
53 square_matrix_like<M> && claims<M, law::self_adjoint>;
54
55/// @brief mat asserted positive semi-definite: \f$x^T A x \geq 0\f$; may be singular.
56template <class M>
57concept psd_matrix_like =
58 symmetric_matrix_like<M> && claims<M, law::psd>;
59
60/// @brief mat asserted positive definite: \f$x^T A x > 0\f$ for \f$x \neq 0\f$; admits a Cholesky factor.
61template <class M>
62concept spd_matrix_like =
63 psd_matrix_like<M> && claims<M, law::spd>;

88/// @brief Factor admitting a substitution solve for a single right-hand side.
89///
90/// The library's factorizations are plain results paired with free functions, so
91/// the contract is `solve_in_place(factor, rhs)` found by argument-dependent
92/// lookup. Requiring a `.solve()` member, as this once did, made the concept
93/// unsatisfiable by `cholesky_result` and `lu_result` alike.
94template <class F, class Vec = vec>
95concept triangular_factor = vector_space<Vec> && requires(const F &factor, Vec &rhs) {
96 solve_in_place(factor, rhs);
97};
98
99/// @brief Reusable factorization of \f$A\f$ solving single and multiple right-hand sides.
100///
101/// The defining property is reuse: the decomposition is computed once and applied
102/// to any number of right-hand sides, which is what distinguishes a factorization
103/// from a one-shot solve. That is expressed by also accepting a matrix of
104/// right-hand sides.
105template <class F, class Vec = vec, class Mat = mat>
106concept direct_factorization = triangular_factor<F, Vec> && requires(const F &factor, Mat &rhs) {
107 solve_in_place(factor, rhs);
108};

114/// @brief Approximate inverse \f$z \approx M^{-1} r\f$ applied once per iteration.
115///
116/// Structurally a linear operator; what makes it a preconditioner is intent, not
117/// interface, so it is stated as such rather than given a distinguishing method.
118template <class M, class X = vec, class Y = vec>
119concept preconditioner = linear_operator<M, X, Y>;
120
121/// @brief preconditioner asserted self-adjoint, as PCG and MINRES require.
122///
123/// A non-symmetric preconditioner silently destroys the Krylov space those methods
124/// build; this is the constraint that should gate them.
125template <class M, class X = vec, class Y = vec>
126concept symmetric_preconditioner = preconditioner<M, X, Y> && self_adjoint_operator<M, X, Y>;
127
128/// @brief preconditioner asserted SPD, the precondition for PCG's error norm to be a norm.
129template <class M, class X = vec, class Y = vec>
130concept spd_preconditioner = symmetric_preconditioner<M, X, Y> && spd_operator<M, X, Y>;
