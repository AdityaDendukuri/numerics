2/// @brief Raw-pointer kernels: BLAS-1 vector ops and fused reductions.
3///
4/// SPDX-License-Identifier: MIT
5/// Part of numerics, (c) 2026 Aditya Dendukuri.
6/// https://github.com/AdityaDendukuri/numerics
7///
8/// This file has no dependencies outside the standard library: copy it into
9/// another project as-is, or lift a single routine out of it together with the
10/// NUM_K_* macro block below. Please keep the two attribution lines above with
11/// whatever you take.
12///
13/// Kernels assume non-owning, caller-sized buffers and do not allocate. This is
14/// the base file of the `num::kernel` tier: `dense.hpp`, `sparse.hpp`, and
15/// `rotations.hpp` all build on the macros, the `contract` tags, and
16/// `detail::reduce` defined here.
17///
18/// @section kernel_contract The contract every kernel in this tier assumes
19///
20/// These rules hold for **every** function in `num::kernel` and are not repeated
21/// on each one. A per-function `@pre` states only what is additional to this.
22///
23/// - **Nothing is checked.** No dimension is validated, no pointer is tested for
24/// null, no divisor is tested for zero. Violating a precondition is undefined
25/// behaviour, not a thrown exception. Proving the preconditions is the job of
26/// the typed layers above (`num::vec`, `num::mat`, the `algebra`
27/// concepts); this tier assumes the proof already happened.
28/// - **Every buffer is caller-allocated and caller-sized.** A kernel never
29/// allocates, never frees, never grows a buffer, and never keeps a pointer past
30/// the call. A parameter documented as length `n` must be readable (and, if an
31/// output, writable) for exactly `n` elements.
32/// - **Pointers marked `NUM_K_RESTRICT` must not overlap.** That is nearly all of
33/// them, and it is load-bearing: it is what lets the compiler vectorize these
34/// loops. Passing the same buffer as two restrict-qualified parameters is
35/// undefined behaviour and will silently produce wrong results at `-O2`, not a
36/// diagnostic. Where in-place operation *is* permitted (`x` and `y` may be the
37/// same pointer), the function says so explicitly under `@aliasing`.
38/// - **`noexcept` throughout.** A kernel has no failure mode it can report by
39/// throwing. Routines that can fail numerically return `bool` or a result
40/// struct instead.
41/// - **Reductions are not in source order** unless the name or a
42/// `contract::ordered` overload says they are; see `detail::reduce` for the
43/// summation-order guarantee and its limits.
44///
45/// Complexity is quoted in elements touched, not in FLOPs, since every routine
46/// here is bandwidth-bound at realistic sizes.

102/// @brief Independent accumulation chains used by every reduction here.
103///
104/// A reduction written as one accumulator carries a loop-carried floating-point
105/// dependency. Because addition is not associative the compiler may not break
106/// it, so the loop runs at the latency of the adder rather than its throughput —
107/// measured at three to four times slower here, and the gap persists past cache
108/// size because the loop is latency bound, not bandwidth bound.
109inline constexpr idx reduction_lanes = 8;
110
111#if defined(NUM_K_VECTOR_EXT)
112
113/// @brief Independent *vector* accumulators on the vector-extension path.
114inline constexpr idx reduction_accumulators = 4;
115
116/// @brief A native vector of `Bytes` bytes holding elements of type T.
117///
118/// An array of scalar lanes is not enough on its own: clang packs such an array
119/// into 128-bit registers and never widens it, whatever the lane count, so on
120/// AVX2 it would use half the register width and on AVX-512 an eighth. Naming
121/// the vector type explicitly is what makes the accumulator as wide as the
122/// hardware.
123template <class T, idx Bytes>
124struct vector_of {
125 typedef T type __attribute__((vector_size(Bytes)));
126};
127
128#endif

130/// @brief Reduce `element(i)` over `[0, n)` with several accumulators in flight.
131///
132/// Each index is visited exactly once, so `element` may carry a side effect —
133/// that is how the fused update-and-reduce kernels work.
134///
135/// ### Summation order
136///
137/// The visiting order is a permutation of the source order: indices are spread
138/// across accumulators and combined pairwise at the end. Bounding the chain
139/// length this way makes the error grow like \f$O(n/K)\f$ instead of
140/// \f$O(n)\f$, so on unstructured data this is more accurate than source order
141/// as well as faster. It is *less* accurate on data whose sign pattern is
142/// periodic in the accumulator count, where source order cancels adjacent terms
143/// immediately.
144///
145/// The grouping is fixed for a given build but **depends on the target's vector
146/// width**, so a result computed under AVX-512 need not match the same source
147/// built for SSE bit for bit. Callers who need a result reproducible across
148/// machines, or exact source-order semantics, select `contract::ordered`.
149template <std::floating_point T, class Element>
150NUM_K_AINLINE T reduce(idx n, Element element) noexcept {
151 idx i = 0;
152 T total = T(0);
153
154#if defined(NUM_K_VECTOR_EXT)
155 using vector = typename vector_of<T, NUM_K_VECTOR_BYTES>::type;
156 constexpr idx width = NUM_K_VECTOR_BYTES / sizeof(T);
157 constexpr idx accumulators = reduction_accumulators;
158 constexpr idx block = width * accumulators;
159
160 vector lane[accumulators]{};
161 for (; i + block <= n; i += block) {
162 for (idx a = 0; a < accumulators; ++a) {
163 vector value{};
164 for (idx k = 0; k < width; ++k) {
165 value[k] = element(i + (a * width) + k);
166 }
167 lane[a] += value;
168 }
169 }
170 // One vector at a time, so the scalar tail is shorter than a vector rather
171 // than shorter than a whole block. Without this a vector of 20 elements
172 // would be reduced entirely serially on a 512-bit target.
173 for (; i + width <= n; i += width) {
174 vector value{};
175 for (idx k = 0; k < width; ++k) {
176 value[k] = element(i + k);
177 }
178 lane[0] += value;
179 }
180 for (idx w = accumulators / 2; w > 0; w /= 2) {
181 for (idx a = 0; a < w; ++a) {
182 lane[a] += lane[a + w];
183 }
184 }
185 for (idx k = 0; k < width; ++k) {
186 total += lane[0][k];
187 }
188#else
189 // Portable fallback: independent scalar chains. Breaks the dependency, but
190 // leaves the compiler to choose the vector width on its own.
191 constexpr idx lanes = reduction_lanes;
192 T lane[lanes]{};
193 for (; i + lanes <= n; i += lanes) {
194 NUM_K_IVDEP
195 for (idx k = 0; k < lanes; ++k) {
196 lane[k] += element(i + k);
197 }
198 }
199 for (idx w = lanes / 2; w > 0; w /= 2) {
200 for (idx k = 0; k < w; ++k) {
201 lane[k] += lane[k + w];
202 }
203 }
204 total = lane[0];
205#endif
206
207 for (; i < n; ++i) {
208 total += element(i);
209 }
210 return total;
211}

316/// @brief Computes vector dot product \f$\mathbf{x} \cdot \mathbf{y} = \sum_{i=0}^{n-1} x_i y_i\f$.
317///

354/// @brief Computes two inner products sharing the same left operand in one pass.
355///
356/// This is the primitive needed by fused Krylov recurrences: `x` is loaded once
357/// instead of once per reduction. The result members are deliberately named so
358/// their mathematical meaning remains clear at call sites.
359template <std::floating_point T>
360struct dot2_result {
361 T xy;
362 T xz;
363};
364
365template <std::floating_point T>
366[[nodiscard]] NUM_K_AINLINE dot2_result<T> dot2(const T *NUM_K_RESTRICT x,
367 const T *NUM_K_RESTRICT y,
368 const T *NUM_K_RESTRICT z, idx n) noexcept {
369 // Half the usual chain count per result: two reductions share the register
370 // budget, and the shared load of x already supplies part of the parallelism.
371 constexpr idx lanes = detail::reduction_lanes / 2;
372 T xy[lanes]{};
373 T xz[lanes]{};
374 idx i = 0;
375 for (; i + lanes <= n; i += lanes) {
376 NUM_K_IVDEP
377 for (idx k = 0; k < lanes; ++k) {
378 const T xi = x[i + k];
379 xy[k] += xi * y[i + k];
380 xz[k] += xi * z[i + k];
381 }
382 }
383 for (idx width = lanes / 2; width > 0; width /= 2) {
384 for (idx k = 0; k < width; ++k) {
385 xy[k] += xy[k + width];
386 xz[k] += xz[k + width];
387 }
388 }
389 T sum_xy = xy[0];
390 T sum_xz = xz[0];
391 for (; i < n; ++i) {
392 const T xi = x[i];
393 sum_xy += xi * y[i];
394 sum_xz += xi * z[i];
395 }
396 return {sum_xy, sum_xz};
397}

399/// @brief Computes an inner product and squared norm in one memory pass.
400template <std::floating_point T>
401struct dot_norm_result {
402 T dot;
403 T norm_sq;
404};
405
406template <std::floating_point T>
407[[nodiscard]] NUM_K_AINLINE dot_norm_result<T>
408dot_norm_sq(const T *NUM_K_RESTRICT x, const T *NUM_K_RESTRICT y, idx n) noexcept {
409 constexpr idx lanes = detail::reduction_lanes / 2;
410 T xy[lanes]{};
411 T yy[lanes]{};
412 idx i = 0;
413 for (; i + lanes <= n; i += lanes) {
414 NUM_K_IVDEP
415 for (idx k = 0; k < lanes; ++k) {
416 const T yi = y[i + k];
417 xy[k] += x[i + k] * yi;
418 yy[k] += yi * yi;
419 }
420 }
421 for (idx width = lanes / 2; width > 0; width /= 2) {
422 for (idx k = 0; k < width; ++k) {
423 xy[k] += xy[k + width];
424 yy[k] += yy[k + width];
425 }
426 }
427 T sum_xy = xy[0];
428 T sum_yy = yy[0];
429 for (; i < n; ++i) {
430 const T yi = y[i];
431 sum_xy += x[i] * yi;
432 sum_yy += yi * yi;
433 }
434 return {sum_xy, sum_yy};
435}

437/// @brief Updates `y <- y + alpha*x` and returns the new squared norm of `y`.
438///
439/// The update and convergence statistic share one traversal. This is generally
440/// bandwidth-optimal for iterative methods because the updated values remain in

474/// @brief Computes Euclidean \f$L_2\f$ norm \f$\|\mathbf{x}\|_2 = \sqrt{\sum_{i=0}^{n-1} x_i^2}\f$.
475///
476/// The direct form `sqrt(norm_sq(x, n))` squares before it sums, so it overflows
477/// to infinity once \f$\|x\|_2\f$ passes about `1.3e154` in double precision,
478/// and flushes to zero below about `1.5e-154` — in both cases returning a wrong
479/// answer for a vector every element of which is a perfectly ordinary finite
480/// number. The direct form is taken whenever its result is usable, and only a
481/// non-finite or zero sum pays for the rescaled second pass.
482template <std::floating_point T>
483[[nodiscard]] NUM_K_AINLINE T norm(const T *NUM_K_RESTRICT x, idx n) noexcept {
484 const T squared = norm_sq(x, n);
485 if (squared > T(0) && std::isfinite(squared)) {
486 return std::sqrt(squared);
487 }
488 if (std::isnan(squared)) {
489 return squared;
490 }
491 // Either the sum overflowed, or every term underflowed to zero. Both are
492 // fixed by factoring out the largest magnitude before squaring.
493 const T scale = linf_norm(x, n);
494 if (scale == T(0) || !std::isfinite(scale)) {
495 return scale;
496 }
497 const T scaled = detail::reduce<T>(n, [x, scale](idx i) {
498 const T v = x[i] / scale;
499 return v * v;
500 });
501 return scale * std::sqrt(scaled);
502}

589/// @brief Returns index of element with maximum absolute value \f$\arg\max_i |x_i|\f$ in vector
590/// \f$\mathbf{x}\f$.
591template <std::floating_point T>
592[[nodiscard]] NUM_K_AINLINE idx argmax_abs(const T *NUM_K_RESTRICT x, idx n) noexcept {
593 if (n == 0) {
594 return 0;
595 }
596 idx best_idx = 0;
597 T best_val = std::abs(x[0]);
598 for (idx i = 1; i < n; ++i) {
599 const T val = std::abs(x[i]);
600 if (val > best_val) {
601 best_val = val;
602 best_idx = i;
603 }
604 }
605 return best_idx;
606}
