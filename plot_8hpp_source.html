62// Write all datablocks for a panel, then emit the plot command for that panel.
63inline void write_panel(FILE *pipe, const plot_panel &p, int block_offset) {
64 if (p.series.empty() && p.heatmaps.empty()) {
65 return;
66 }
67
68 // Common decorators
69 if (!p.title_.empty()) {
70 fprintf(pipe, "set title '%s'\n", p.title_.c_str());
71 } else {
72 fputs("unset title\n", pipe);
73 }
74 if (!p.xlabel_.empty()) {
75 fprintf(pipe, "set xlabel '%s'\n", p.xlabel_.c_str());
76 } else {
77 fputs("unset xlabel\n", pipe);
78 }
79 if (!p.ylabel_.empty()) {
80 fprintf(pipe, "set ylabel '%s'\n", p.ylabel_.c_str());
81 } else {
82 fputs("unset ylabel\n", pipe);
83 }
84
85 if (!p.heatmaps.empty()) {
86 const auto &hm = p.heatmaps[0];
87 if (!p.palette_.empty()) {
88 fprintf(pipe, "set palette %s\n", p.palette_.c_str());
89 } else {
90 fputs("set palette defined "
91 "(0 'white', 0.35 '#ffffb2', 0.65 '#fd8d3c', 1 '#bd0026')\n",
92 pipe);
93 }
94 fprintf(pipe, "set cbrange [%g:%g]\n", hm.vmin, hm.vmax);
95 fputs("set pm3d map\n", pipe);
96 fputs("set size ratio 1\n", pipe);
97 if (!p.xrange_.empty()) {
98 fprintf(pipe, "set xrange %s\n", p.xrange_.c_str());
99 } else {
100 fputs("set xrange [*:*]\n", pipe);
101 }
102 if (!p.yrange_.empty()) {
103 fprintf(pipe, "set yrange %s\n", p.yrange_.c_str());
104 } else {
105 fputs("set yrange [*:*]\n", pipe);
106 }
107 fputs("unset key\n", pipe);
108 fprintf(pipe, "splot $d_%d with pm3d notitle\n", block_offset);
109 } else {
110 // Line-plot panel
111 fputs("unset pm3d\n", pipe);
112 if (!p.xrange_.empty()) {
113 fprintf(pipe, "set xrange %s\n", p.xrange_.c_str());
114 } else {
115 fputs("set xrange [*:*]\n", pipe);
116 }
117 if (!p.yrange_.empty()) {
118 fprintf(pipe, "set yrange %s\n", p.yrange_.c_str());
119 } else {
120 fputs("set yrange [*:*]\n", pipe);
121 }
122
123 if (p.logx_ && p.logy_) {
124 fputs("set logscale xy\nset format x '10^{%L}'\nset format y '10^{%L}'\n", pipe);
125 } else if (p.logx_) {
126 fputs("unset logscale y\nset logscale x\nset format x '10^{%L}'\nset format y '%g'\n", pipe);
127 } else if (p.logy_) {
128 fputs("unset logscale x\nset logscale y\nset format y '10^{%L}'\nset format x '%g'\n", pipe);
129 } else {
130 fputs("unset logscale\nset format x '%g'\nset format y '%g'\n", pipe);
131 }
132
133 if (p.legend_) {
134 fprintf(pipe, "set key %s Left reverse samplen 3 spacing 1.2\n"
135 "set key box lt 1 lw 0.5\n",
136 p.legend_pos_.c_str());
137 } else {
138 fputs("unset key\n", pipe);
139 }
140
141 fputs("plot ", pipe);
142 for (std::size_t i = 0; i < p.series.size(); ++i) {
143 if (i) {
144 fputs(", ", pipe);
145 }
146 const auto &e = p.series[i];
147 if (e.style.find("lw") != std::string::npos ||
148 e.style.find("lc") != std::string::npos ||
149 e.style.find("ps") != std::string::npos) {
150 fprintf(pipe, "$d_%d with %s", block_offset + (int)i, e.style.c_str());
151 } else {
152 fprintf(pipe, "$d_%d with %s ls %zu", block_offset + (int)i, e.style.c_str(),
153 i + 1);
154 }
155
156 if (!e.label.empty()) {
157 fprintf(pipe, " title '%s'", e.label.c_str());
158 } else {
159 fputs(" notitle", pipe);
160 }
161 }
162 fputc('\n', pipe);
163 }

166inline void flush_to(FILE *pipe, const std::string &outfile) {
167 auto &s = state();
168
169 array<plot_panel> all = s.panels;
170 all.push_back(s.current);
171
172 bool multiplot = (s.mp_rows_ > 0);
173
174 // Terminal
175 if (s.term_override_ == "dumb") {
176 fprintf(pipe, "set terminal dumb size %d,%d\nset autoscale\n", s.term_w_, s.term_h_);
177 } else if (outfile.empty()) {
178 int h = multiplot ? 300 * s.mp_rows_ : 600;
179 fprintf(pipe, "set terminal qt size 900,%d\n", h);
180 } else {
181 std::string ext = outfile.size() > 4 ? outfile.substr(outfile.size() - 4) : "";
182 if (ext == ".pdf") {
183 double h = multiplot ? 3.0 * s.mp_rows_ : 4.0;
184 fprintf(pipe, "set terminal pdfcairo size 6,%.0f font 'Arial,11'\n", h);
185 } else {
186 int w = multiplot ? std::max(900, 300 * s.mp_cols_) : 900;
187 int h = multiplot ? 350 * s.mp_rows_ : 600;
188 fprintf(pipe, "set terminal pngcairo size %d,%d enhanced font 'Arial,11'\n", w, h);
189 }
190 fprintf(pipe, "set output '%s'\n", outfile.c_str());
191 }
192
193 // Global theme
194 if (s.term_override_ != "dumb") {
195 fputs("set style line 1 lt 1 lw 2 pt 7 ps 0.7 lc rgb '#2c3e50'\n", pipe);
196 fputs("set style line 2 lt 2 lw 2 pt 5 ps 0.7 lc rgb '#c0392b'\n", pipe);
197 fputs("set style line 3 lt 3 lw 2 pt 9 ps 0.7 lc rgb '#2980b9'\n", pipe);
198 fputs("set style line 4 lt 4 lw 2 pt 13 ps 0.7 lc rgb '#27ae60'\n", pipe);
199 fputs("set style line 5 lt 5 lw 2 pt 11 ps 0.7 lc rgb '#8e44ad'\n", pipe);
200 fputs("set style line 100 lt 1 lw 0.5 lc rgb '#cccccc'\n", pipe);
201 fputs("set grid back ls 100\n", pipe);
202 fputs("set border 3 lw 1.5\n", pipe);
203 fputs("set tics nomirror\n", pipe);
204 }
205
206 // Write datablocks
207 int block = 0;
208 for (const auto &p : all) {
209 for (const auto &e : p.series) {
210 fprintf(pipe, "$d_%d << EOD\n", block++);
211 for (const auto &[x, y] : e.data) {
212 fprintf(pipe, "%.15g %.15g\n", x, y);
213 }
214 fputs("EOD\n", pipe);
215 }
216 for (const auto &hm : p.heatmaps) {
217 fprintf(pipe, "$d_%d << EOD\n", block++);
218 for (int i = 0; i < hm.N; ++i) {
219 double xi = (i + 1) * hm.h;
220 for (int j = 0; j < hm.N; ++j) {
221 fprintf(pipe, "%.8g %.8g %.8g\n", xi, (j + 1) * hm.h,
222 hm.data[(static_cast<std::size_t>(i) * hm.N) + j]);
223 }
224 fputs("\n", pipe);
225 }
226 fputs("EOD\n", pipe);
227 }
228 }
229
230 if (multiplot) {
231 fprintf(pipe, "set multiplot layout %d,%d spacing 0.08,0.12\n", s.mp_rows_, s.mp_cols_);
232 int off = 0;
233 for (const auto &p : all) {
234 write_panel(pipe, p, off);
235 off += (int)p.series.size() + (int)p.heatmaps.size();
236 }
237 fputs("unset multiplot\n", pipe);
238 } else {
239 write_panel(pipe, all.back(), 0);
240 }
241
242 fflush(pipe);
243}

255/// Append parallel x and y vectors to the current panel (supports std::vector,
256/// num::vec, etc.).
257template <typename ContainerX, typename ContainerY>
258inline void plot(const ContainerX &x, const ContainerY &y, const std::string &label = "",
259 const std::string &style = "lines") {
260 series s;
261 s.reserve(x.size());
262 for (std::size_t i = 0; i < x.size() && i < y.size(); ++i) {
263 s.emplace_back(static_cast<double>(x[i]), static_cast<double>(y[i]));
264 }
265 detail::state().current.series.push_back({std::move(s), label, style});
266}

298/// Plot component-wise available-path means and their min-max envelope.
299///
300/// The sampling grid extends to the latest path endpoint. A path contributes
301/// only through its own endpoint, so the active sample count may decrease near
302/// the right edge instead of holding shorter paths at their final states.
303template <class Paths, class Labels, class Colors>
304inline void plot_paths(const Paths &paths, const Labels &labels, const Colors &colors,
305 const std::string &plot_title, std::size_t samples = 1000) {
306 if (paths.empty() || labels.empty() || labels.size() != colors.size() || samples < 2) {
307 throw std::invalid_argument("invalid path plot data");
308 }
309
310 double first_end = std::numeric_limits<double>::infinity();
311 double last_end = 0.0;
312 for (const auto &path : paths) {
313 if (path.times.empty() || path.states.empty()) {
314 throw std::invalid_argument("path plot requires nonempty trajectories");
315 }
316 first_end = std::min(first_end, path.times.back());
317 last_end = std::max(last_end, path.times.back());
318 }
319
320 array<double> times(samples);
321 for (std::size_t i = 0; i < samples; ++i) {
322 times[i] = last_end * static_cast<double>(i) / static_cast<double>(samples - 1);
323 }
324
325 const std::size_t components = labels.size();
326 array<array<double>> lower(
327 components, array<double>(samples, std::numeric_limits<double>::max()));
328 array<array<double>> upper(
329 components, array<double>(samples, std::numeric_limits<double>::lowest()));
330 array<array<double>> mean(components, array<double>(samples, 0.0));
331 array<std::size_t> active(samples, 0);
332
333 for (const auto &path : paths) {
334 std::size_t state = 0;
335 for (std::size_t i = 0; i < samples; ++i) {
336 if (times[i] > path.times.back()) {
337 continue;
338 }
339 while (state + 1 < path.times.size() && path.times[state + 1] <= times[i]) {
340 ++state;
341 }
342 ++active[i];
343 for (std::size_t component = 0; component < components; ++component) {
344 const double value = path.states[state][component];
345 lower[component][i] = std::min(lower[component][i], value);
346 upper[component][i] = std::max(upper[component][i], value);
347 mean[component][i] += value;
348 }
349 }
350 }
351
352 for (std::size_t i = 0; i < samples; ++i) {
353 if (active[i] == 0) {
354 throw std::runtime_error("path plot has no active trajectory at a sample time");
355 }
356 for (std::size_t component = 0; component < components; ++component) {
357 mean[component][i] /= static_cast<double>(active[i]);
358 }
359 }
360
361 if (paths.size() > 1) {
362 for (std::size_t component = 0; component < components; ++component) {
363 series envelope;
364 envelope.reserve(2 * samples);
365 for (std::size_t i = 0; i < samples; ++i) {
366 envelope.emplace_back(times[i], upper[component][i]);
367 }
368 for (std::size_t i = samples; i-- > 0;) {
369 envelope.emplace_back(times[i], lower[component][i]);
370 }
371 plot(envelope, std::string{},
372 "filledcurves closed fs transparent solid 0.18 noborder lc rgb '" +
373 std::string(colors[component]) + "'");
374 }
375 }
376
377 for (std::size_t component = 0; component < components; ++component) {
378 plot(times, mean[component], std::string(labels[component]),
379 "lines lw 2 lc rgb '" + std::string(colors[component]) + "'");
380 }
381
382 std::ostringstream endpoint;
383 endpoint << std::setprecision(3) << " endpoint=[" << first_end << ", " << last_end << "]";
384 if (paths.size() > 1) {
385 endpoint << " active=" << paths.size() << "->" << active.back();
386 }
387 title(plot_title + endpoint.str());
388 xlabel("t");
389 ylabel("count");
390 legend();
391}

423template <typename Container>
424/// Add an N-by-N row-major scalar array as a heatmap.
425inline void heatmap(const Container &u, int N, double h, double vmin = 0.0, double vmax = 1.0) {
426 detail::heatmap_entry e;
427 e.data.assign(u.data(), u.data() + u.size());
428 e.N = N;
429 e.h = h;
430 e.vmin = vmin;
431 e.vmax = vmax;
432 detail::state().current.heatmaps.push_back(std::move(e));
433}
