21/// Diagonal inverse preconditioner.
22class jacobi_preconditioner final {
23 public:
24 using domain_type = vec;
25 using codomain_type = vec;
26 using math_laws = math::type_list<law::spd>;

28 /// Take ownership of a precomputed inverse diagonal.
29 explicit jacobi_preconditioner(vec inv_diag) : inv_diag_(std::move(inv_diag)) {
30 for (const real value : inv_diag_) {
31 if (!(value > 0.0) || !std::isfinite(value)) {
32 throw std::invalid_argument(
33 "jacobi_preconditioner: inverse diagonal must be positive and finite");
34 }
35 }
36 }
37
38 [[nodiscard]] idx rows() const noexcept { return inv_diag_.size(); }
39 [[nodiscard]] idx cols() const noexcept { return inv_diag_.size(); }
40
41 /// Compute z=D^-1 r.
42 void apply(const vec &r, vec &z) const {
43 const idx n = inv_diag_.size();
44 if (r.size() != n) {
45 throw std::invalid_argument("jacobi_preconditioner: dimension mismatch");
46 }
47 if (z.size() != n) {
48 z = vec(n, 0.0);
49 }
50 kernel::hadamard_mul(z.data(), inv_diag_.data(), r.data(), n);
51 }
52
53 private:
54 vec inv_diag_;
55};

57/// Construct a Jacobi preconditioner from a dense matrix diagonal.
58[[nodiscard]] inline jacobi_preconditioner make_jacobi_preconditioner(const mat &A) {
59 if (A.rows() != A.cols()) {
60 throw std::invalid_argument("jacobi_preconditioner: matrix must be square");
61 }
62 vec inv(A.rows());
63 for (idx i = 0; i < A.rows(); ++i) {
64 if (std::abs(A(i, i)) < real(1e-15)) {
65 throw std::invalid_argument("jacobi_preconditioner: zero diagonal");
66 }
67 inv[i] = real(1) / A(i, i);
68 }
69 return jacobi_preconditioner(std::move(inv));
70}

72/// Construct a Jacobi preconditioner from a sparse matrix diagonal.
73[[nodiscard]] inline jacobi_preconditioner make_jacobi_preconditioner(const spmat &A) {
74 if (A.n_rows() != A.n_cols()) {
75 throw std::invalid_argument("jacobi_preconditioner: matrix must be square");
76 }
77 vec inv(A.n_rows(), 0.0);
78 for (idx i = 0; i < A.n_rows(); ++i) {
79 const idx row_begin = A.row_ptr()[i];
80 const idx row_end = A.row_ptr()[i + 1];
81 for (idx p = row_begin; p < row_end; ++p) {
82 if (A.col_idx()[p] == i) {
83 inv[i] = A.values()[p];
84 break;
85 }
86 }
87 if (std::abs(inv[i]) < real(1e-15)) {
88 throw std::invalid_argument("jacobi_preconditioner: zero diagonal");
89 }
90 inv[i] = real(1) / inv[i];
91 }
92 return jacobi_preconditioner(std::move(inv));
93}
